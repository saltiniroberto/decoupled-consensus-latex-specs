\section{Model and definitions}

We consider a system of~$n$ validators of which at most~$f$ are adversarial, where $n \geq 3f + 1$.
A \emph{quorum} is any set of at least $2n/3$ validators.
Any two quorums overlap in at least $n/3 + 1 > f$ validators, so their intersection contains at least one honest validator.

\begin{definition}[Height]\label{def:height}
\emph{Height} is the finality-gadget counter, separate from slot.
The genesis height is~$0$; height advances by exactly~$1$ via fixed-target justification (Definition~\ref{def:justification}) or prefix notarization (Definition~\ref{def:notarization}).
\end{definition}

\begin{definition}[Block]\label{def:block}
A \emph{block} has a \emph{parent} block, a \emph{slot} field, and a set of votes (Definition~\ref{def:vote}).
The \emph{genesis block}~$\genesis$ has no parent and slot~$0$.
Slots strictly increase along parent links: $B.\slot > B.\parent.\slot$ for every non-genesis~$B$.
\end{definition}

\begin{definition}[Chain]\label{def:chain}
A \emph{chain} is a path from~$\genesis$ to some block~$B$ via parent links.
Chains are in bijection with their tips, and we identify the two.
Blocks form a tree rooted at~$\genesis$; we write $B \preceq B'$ when $B$ is an ancestor of~$B'$ (or $B = B'$), and $B \sim B'$ (\emph{compatible}) when $B \preceq B'$ or $B' \preceq B$.
Two chains \emph{conflict} when their tips are not compatible.
\end{definition}

\begin{definition}[Vote]\label{def:vote}
A \emph{vote} is a signed tuple
\[
  (\validator,\; \height,\; \slot,\; \target,\; \kind,\; \finalizeHeight,\; \finalizeTarget),
\]
where $\validator$ is the signer, $\height$ is a height, $\slot$ a slot, $\target$ a block (never~$\bot$), and $\kind \in \{\just,\, \notar\}$.
The pair $(\finalizeHeight,\, \finalizeTarget)$ is either both~$\bot$ or a commitment to finalize a block at a height.
A vote included on a chain must reference a target (and finalize target, when present) that exists on that chain.
\end{definition}

\paragraph{State.}
The \emph{state after block~$B$}, written $\sigma[B]$, is the tuple
\[
  (L,\; s,\; h,\; s_h,\; \jtargets,\; \ntargets,\; \nslots,\; J,\; h_j,\; N,\; h_n,\; s_n,\; F,\; P),
\]
with components:
\begin{itemize}
  \item $L$: the \emph{chain head} ($L = B$ after processing~$B$).
  \item $s$: the \emph{current slot}, incremented once per slot by \processSlot (Figure~\ref{alg:state-machine}).
  \item $h$: the \emph{current height}.
  \item $s_h$: the slot of the block whose processing last advanced~$h$ on this chain ($0$ if no advance has occurred).
  \item $\jtargets[1..n]$: for each validator~$i$, the target of~$i$'s \just vote at the current height ($\bot$ if none processed).
  \item $\ntargets[1..n]$: for each~$i$, the target of the latest-slot fresh vote of~$i$ at the current height, across both kinds ($\bot$ if none processed). Both \just and \notar votes can set $\ntargets[i]$; a later-slot vote overwrites any earlier entry.
  \item $\nslots[1..n]$: for each~$i$, the $\slot$ of the vote that set $\ntargets[i]$ ($0$ if none).
  \item $J$, $h_j$: the most recently justified block and its height.
  \item $N$, $h_n$: the most recently notarized (or justified) block and its height; $J$ is itself a notarization (Definition~\ref{def:notarization}).
  \item $s_n$: the \emph{notarization slot} of the most recent prefix-notarization-only advance --- the common slot~$s$ shared by the $\geq 2n/3$ contributors of the firing quorum, i.e., a slot such that $\geq 2n/3$ of the contributors whose $\ntargets$ covers~$N$ satisfy $\nslots = s$. Justifications do not set $s_n$; the field is reset to $0$ on a justification advance, and initializes to $0$.
  \item $F$: the most recently finalized block.
  \item $P \subseteq \{1, \ldots, n\}$: validators who have cast a vote with $\finalizeHeight = h_j$ and $\finalizeTarget = J$.
\end{itemize}
The genesis state is
\[
  \sigma_{\mathrm{gen}} \coloneqq (\genesis,\; 0,\; 0,\; 0,\; \bot^n,\; \bot^n,\; 0^n,\; \genesis,\; 0,\; \genesis,\; 0,\; 0,\; \genesis,\; \emptyset),
\]
with $\sigma[\genesis] \coloneqq \sigma_{\mathrm{gen}}$.
When unambiguous we drop the chain prefix and write $L$, $s$, $h$, etc.\ for the components of $\sigma[L]$.

\begin{definition}[State-height]\label{def:sheight}
The \emph{state-height} of a block~$B$ is $\sigma[B].h$: the height after processing~$B$ on its own chain.
\end{definition}

\paragraph{Height freshness.}
A vote~$v$ with target~$T$ is processed on a chain with head state $\sigma[L]$ only if
\[
  \sigma[L].h = v.\height \;\wedge\; T \preceq L \;\wedge\; T.\slot \geq \sigma[L].s_h:
\]
the chain is at~$v$'s height, $T$ lies on this chain, and $T$'s slot is no earlier than where the chain's current height began.
Intuitively, $v$ attests that its signer saw~$T$ as a chain of height~$v.\height$.

\begin{remark}[Freshness via $T$'s own chain]\label{rem:fresh-equiv}
Given $T \preceq L$, the constraint $T.\slot \geq \sigma[L].s_h$ is equivalent to $\sigma[T].h = \sigma[L].h$.
State-height is $\preceq$-monotone (Lemma~\ref{lem:slotmono}), equals $\sigma[L].h$ on~$L$, and advanced to its current value at the block whose slot is $\sigma[L].s_h$; so the ancestors of~$L$ at that state-height are exactly those with slot $\geq \sigma[L].s_h$.
\end{remark}

Freshness guarantees that every justification or notarization at height~$h$ is witnessed by targets whose own chains reached state-height~$h$.

\begin{definition}[Fixed-target justification]\label{def:justification}
A block~$T$ is \emph{justified at height~$h$} on a chain with state $\sigma[L]$ with $\sigma[L].h = h$ iff
\[
  \bigl|\{i \,:\, \jtargets[i] = T\}\bigr| \;\geq\; 2n/3.
\]
Only votes with target exactly~$T$ contribute.
\end{definition}

\begin{definition}[Prefix notarization]\label{def:notarization}
On a chain with head state $\sigma[L]$ at height~$h$, a block~$T$ is \emph{notarized at height~$h$ with slot~$s$} iff~$T$ is the $\preceq$-maximum block for which some slot~$s$ satisfies
\[
  \bigl|\{i : T \preceq \ntargets[i] \preceq L \;\wedge\; \nslots[i] = s\}\bigr| \;\geq\; 2n/3.
\]
If no such pair exists, nothing is notarized; when~$T$ exists the witnessing~$s$ is unique, since the sets indexed by distinct~$s$ are disjoint and cannot both exceed~$n/3$.
Notarization uses \emph{prefix counting}: \just or \notar votes targeting descendants of~$T$ contribute via $\ntargets$, but only the latest-slot entry per validator counts, and the firing quorum must share a single $\nslots$ value~$s$.
A justification of~$T$ is paired with a notarization cert event at~$T$: the justification branch of \processHeight sets $(J, N) \gets (T, T)$ jointly (Figure~\ref{alg:state-machine}).
\end{definition}

\begin{definition}[Finality]\label{def:finality}
The justified block~$J$ is \emph{finalized at height~$h_j$} once $|P| \geq 2n/3$.
\end{definition}

\begin{definition}[Slashing, rule E1]\label{def:slashing}
A validator is \emph{slashable} if it signed two votes~$a, b$ (possibly $a = b$) with
\[
  b.\finalizeTarget \neq \bot \;\wedge\; a.\height = b.\finalizeHeight \;\wedge\; a.\target \neq b.\finalizeTarget.
\]
That is, a validator committing $(\finalizeHeight,\, \finalizeTarget) = (h_f, T_f)$ must not sign any vote at height~$h_f$ with target $\neq T_f$.
\end{definition}

Two \just (or two \notar) votes at the same height are not by themselves slashable; per-height uniqueness of honest \just votes is imposed behaviorally (Definition~\ref{def:voting-rule}).
A validator who never sets $\finalizeTarget \neq \bot$ trivially avoids slashing.

\paragraph{State transitions.}
The state machine (Figure~\ref{alg:state-machine}) exposes two per-chain routines: \processSlot, invoked once per slot on every maintained chain, and \processBlock, invoked additionally whenever a block~$B$ is processed.

\processSlot increments~$s$ and calls \processHeight, which in turn checks three branches in order: (i) finality, $F \gets J$ if $|P| \geq 2n/3$; (ii) justification, if some~$T$ has $\geq 2n/3$ entries in $\jtargets$; (iii) prefix notarization, at the firing block~$T^*$ located by the walk of Definition~\ref{def:notarization}.
Branches (ii) and (iii) both advance height and reset $\jtargets, \ntargets$, so at most one height-advance fires per invocation.

$\processBlock(\sigma, B)$ has precondition $\sigma.s = B.\slot$: the caller runs \processSlot up to $B.\slot$ first.
Then $L \gets B$, and each vote~$v$ in~$B$ is fed through \processVote.
A fresh vote updates $\ntargets[i] \gets \max_{\preceq}(\ntargets[i],\, v.\target)$ (well-defined as both arguments lie on~$L$ at height $\sigma.h$, treating~$\bot$ as~$\genesis$); if also $v.\kind = \just$, then $\jtargets[i] \gets v.\target$.
The $P$-gate adds~$i$ to~$P$ iff $v.\finalizeHeight = \sigma.h_j$ and $v.\finalizeTarget = \sigma.J$; this is independent of freshness.
\processBlock never advances~$h$: height advances happen only in \processSlot, which runs on every slot (including empty ones).

\begin{algorithmic}[1]

\Function{\processSlot}{$\sigma$} \Comment{Runs once per slot on every chain, even without a block}
  \State $\sigma.s \gets \sigma.s + 1$
  \State $\sigma \gets \processHeight(\sigma)$
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\processBlock}{$\sigma,\; B$}
  \State \textbf{assert} $\sigma.s = B.\slot$ \Comment{Precondition: caller advanced $\sigma.s$ via \processSlot}
  \State $\sigma.L \gets B$
  \For{each vote $v$ in $B$} $\sigma \gets \processVote(\sigma,\, v)$ \EndFor
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\processVote}{$\sigma,\; v$}
  \State $i \gets v.\validator$
  \State $\fresh \gets (v.\height = \sigma.h) \land (v.\target \preceq \sigma.L) \land (v.\target.\slot \geq \sigma.s_h)$
  \If{$\fresh$ \textbf{and} $v.\slot > \sigma.\nslots[i]$}
    \State $\sigma.\ntargets[i] \gets v.\target$ \Comment{both kinds bump $\ntargets$, slot-gated}
    \State $\sigma.\nslots[i]  \gets v.\slot$
  \EndIf
  \If{$\fresh$ \textbf{and} $v.\kind = \just$}
    \State $\sigma.\jtargets[i] \gets v.\target$
  \EndIf
  \If{$v.\finalizeHeight = \sigma.h_j$ \textbf{and} $v.\finalizeTarget = \sigma.J$}
    \State $\sigma.P \gets \sigma.P \cup \{i\}$
  \EndIf
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\processHeight}{$\sigma$}
  \If{$|\sigma.P| \geq 2n/3$} \Comment{Finality}
    \State $\sigma.F \gets \sigma.J$
  \EndIf
  \State $T \gets \arg\max_{T' \in \{\sigma.\jtargets[i] \,:\, \sigma.\jtargets[i] \neq \bot\}} |\{i : \sigma.\jtargets[i] = T'\}|$ \Comment{$T \gets \bot$ if the set is empty;}
  \If{$T \neq \bot$ \textbf{and} $|\{i : \sigma.\jtargets[i] = T\}| \geq 2n/3$} \Comment{Justification (also a notarization)}
    \State $\sigma.J \gets T$, \; $\sigma.h_j \gets \sigma.h$, \; $\sigma.P \gets \emptyset$
    \State $\sigma.N \gets T$, \; $\sigma.h_n \gets \sigma.h$, \; $\sigma.s_n \gets 0$
    \State $\sigma.h \gets \sigma.h + 1$, \; $\sigma.s_h \gets \sigma.L.\slot$
    \State reset $\jtargets, \ntargets, \nslots$
    \State \Return $\sigma$
  \EndIf
  \State $(B^*, s^*) \gets \arg\max_{\preceq}\,\bigl\{(B, s) : B \preceq \sigma.L \;\wedge\; |\{i : B \preceq \sigma.\ntargets[i] \preceq \sigma.L \wedge \sigma.\nslots[i] = s\}| \geq 2n/3\bigr\}$
  \Statex \hspace{6.1em} \Comment{$\arg\max$ by $\preceq$ on~$B$; $s$ given $B$ is unique; $(B^*, s^*) \gets \bot$ if the set is empty}
  \If{$(B^*, s^*) \neq \bot$} \Comment{Prefix notarization fires}
    \State $\sigma.N \gets B^*$, \; $\sigma.h_n \gets \sigma.h$, \; $\sigma.s_n \gets s^*$
    \State $\sigma.h \gets \sigma.h + 1$, \; $\sigma.s_h \gets \sigma.L.\slot$
    \State reset $\jtargets, \ntargets, \nslots$
  \EndIf
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\finalizeConflict}{$v_1, v_2$} \Comment{Does $v_1$ conflict with $v_2$'s finalize commitment?}
  \State \Return $v_2.\finalizeTarget \neq \bot \;\wedge\; v_1.\height = v_2.\finalizeHeight \;\wedge\; v_1.\target \neq v_2.\finalizeTarget$
\EndFunction
\vspace{3pt}

\Function{\isSlashable}{$v_1, v_2$}
  \State \Return $v_1.\validator = v_2.\validator$
  \Statex \hspace{3em}$\land \;(\finalizeConflict(v_1, v_2) \lor \finalizeConflict(v_2, v_1))$
\EndFunction

\end{algorithmic}

\section{Accountable safety}

\begin{lemma}[Height progression]\label{lem:heightprog}
\processHeight increments~$h$ by at most~$1$ per invocation, and each increment is gated by a $\geq 2n/3$ justification or prefix-notarization quorum.
\end{lemma}

\begin{proof}
The only assignments to~$h$ are $h \gets h + 1$ in the justification and notarization branches of \processHeight, each guarded by its $\geq 2n/3$ quorum.
\end{proof}

\begin{lemma}[Checkpoint monotonicity]\label{lem:slotmono}
On any chain, the fields $s$, $h$, $h_j$, $h_n$, $s_h$, $F$, $J$, $N$ are non-decreasing along~$\preceq$, with $F \preceq J \preceq N$ at all times.
$J$ and $h_j$ advance only on justification events, and $h_n, s_h$ only on height-advances, with $s_h$ set to the slot of the advancing block.
Consequently, when $\sigma[Y].h = h$, the ancestors of~$Y$ at state-height~$h$ are $\{B' \preceq Y : B'.\slot \geq \sigma[Y].s_h\}$.
\end{lemma}

\begin{proof}
Reading off Figure~\ref{alg:state-machine}: $s$ only increases (in \processSlot); $h$ only increases, by~$1$, in the height-advance branches of \processHeight, which also set $h_j, h_n \gets h$ on their respective updates and $s_h \gets L.\slot$.
Slots strictly increase along~$\preceq$ (Definition~\ref{def:block}), so $s_h$ is non-decreasing.
$F$ updates only via $F \gets J$, so $F \preceq J$ reduces to $J$-monotonicity.

\emph{Height-$h$ interval.}
Along any chain, $h$ increases monotonically; when $\sigma[Y].h = h$, the value $s_h$ was set at the block that advanced state-height to~$h$.
By strict slot increase along~$\preceq$ and the monotonicity of $(h, s_h)$, the ancestors $B' \preceq Y$ with $\sigma[B'].h = h$ are exactly those with $B'.\slot \geq s_h$.

\emph{Block fields $J$, $N$.}
At a height-advance on chain~$Y$ firing at target~$T$ with pre-state height~$h$, freshness forces $T.\slot \geq s_h$: directly for justification, since every contributing \just vote has target~$T$; for prefix notarization, since the firing~$T$ is some $\ntargets[i]$ value, hence has slot $\geq s_h$.
By the height-$h$ interval claim, $\sigma[T].h = h$, so the previous notarized block $N_{\mathrm{old}}$ with $\sigma[N_{\mathrm{old}}].h \leq h-1$ and $N_{\mathrm{old}} \preceq Y$ satisfies $T \succ N_{\mathrm{old}}$ by state-height monotonicity.
The justification branch sets $J = N = T$ jointly, the notarization branch only $N \gets T$: $N$ strictly advances at every height-advance, $J$ at every justification, and $J \preceq N$ is preserved.
\end{proof}

\begin{lemma}[Any chain past a finalized height contains the finalized block]\label{lem:mainsafety}
Unless $\geq n/3$ validators are slashable: if~$C$ is finalized at height~$h$, then $C \preceq B$ for every block~$B$ with $\sigma[B].h > h$.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:heightprog}, some $B^* \preceq B$ advanced height from~$h$ to~$h+1$ via a quorum~$Q$ at height~$h$ on~$B^*$'s chain; each $i \in Q$ contributed a fresh vote~$a_i$ at height~$h$ (freshness is automatic for \just votes; prefix-notarization entries in $\ntargets$ are also fresh by Figure~\ref{alg:state-machine}).
Finality of~$C$ at~$h$ gives a quorum~$Q_F$ each signing a vote~$b$ with $b.\finalizeHeight = h$ and $b.\finalizeTarget = C$.

By quorum intersection, $|Q \cap Q_F| \geq n/3$; pick $v \in Q \cap Q_F$ not slashable.
E1 (Definition~\ref{def:slashing}) applied to $v$'s votes $a \in Q$, $b \in Q_F$ with $a.\height = h = b.\finalizeHeight$ forces $a.\target = C$; freshness gives $C = a.\target \preceq B^* \preceq B$.
\end{proof}

\begin{lemma}[Finalized blocks form a chain]\label{lem:finchain}
Unless $\geq n/3$ validators are slashable: any two finalized checkpoints $(C, h), (C', h')$ are compatible, with $C \preceq C'$ whenever $h \leq h'$.
\end{lemma}

\begin{proof}
\emph{Case $h = h'$.}
Finality of~$C$ gives a quorum~$Q_F$ with $\finalizeHeight = h$, $\finalizeTarget = C$.
Finality of~$C'$ requires justification of~$C'$, giving a \just-quorum~$Q'$ at height~$h$ with target~$C'$.
By quorum intersection, $|Q_F \cap Q'| \geq n/3$; pick $v \in Q_F \cap Q'$ not slashable.
E1 (Definition~\ref{def:slashing}) applied to $v$'s votes in~$Q_F, Q'$ forces $C = C'$.

\emph{Case $h < h'$.}
Let~$Z$ be the chain on which~$C'$ was finalized, at the post-state where the finalize-quorum closed; then $\sigma[Z].h \geq h' > h$, $C' \preceq Z$, and $\sigma[C'].h = h'$ (Remark~\ref{rem:fresh-equiv}).
Lemma~\ref{lem:mainsafety} at~$Z$ gives $C \preceq Z$; Remark~\ref{rem:fresh-equiv} on~$C$'s own finalizing chain gives $\sigma[C].h = h$.
Both~$C, C'$ lie on~$Z$, so comparable; with $\sigma[C].h = h < h' = \sigma[C'].h$, state-height monotonicity (Lemma~\ref{lem:slotmono}) forces $C \prec C'$.
\end{proof}

\begin{theorem}[Accountable safety]\label{thm:safety}
Unless $\geq n/3$ validators are slashable, no two conflicting blocks can be finalized.
\end{theorem}

\begin{proof}
Immediate from Lemma~\ref{lem:finchain}: any two finalized blocks are compatible.
\end{proof}

\begin{remark}[Ancestor notarizations]\label{rem:notarization-safety}
Because \just votes bump $\ntargets$, a prefix notarization at height~$h$ on a chain containing a finalized~$C$ may fire at an ancestor $T^* \preceq C$.
No per-chain invariant requires $N \succeq C$; the store (Section~\ref{sec:store}) resolves such cases via lexicographic cert keys.
\end{remark}

\section{Fork-choice store}\label{sec:store}

A node may track multiple chains simultaneously; the \emph{store} is the node-local data structure that mediates which chain the node follows.
It maintains a single \emph{root block} $R$ together with a cached \emph{cert key} $\key_R$ for the event that last set $R$.
Each firing of \processHeight (a \emph{cert event}) is compared against $\key_R$ in lex order; whenever its key strictly dominates, $(R, \key_R)$ is updated to the new event.
The key packages height, a justified-bit, the notarization slot, the block slot, and the block hash, so that a justification strictly dominates a prior notarization-only event at the same block, a higher notarization slot outranks a lower one at the same (height, bit), and a higher-height cert always wins over a lower one.

\begin{definition}[Cert event and cert key]\label{def:cert-event}
When \processHeight fires a height advance on a chain (Figure~\ref{alg:state-machine}), we call the triggering event a \emph{cert event}, characterized by:
\begin{itemize}
  \item the block $X = \sigma.N$ at firing (equal to the justified target~$T$ in the justification branch, and to the prefix-notarization endpoint~$T^*$ in the notarization branch);
  \item the height~$h$ at which the advance fires, equal to the pre-advance $\sigma.h$ and to the post-advance $\sigma.h_n$;
  \item a \emph{bit} $b \in \{0, 1\}$: $b = 1$ for a justification, $b = 0$ for a prefix notarization without accompanying justification;
  \item the \emph{notarization slot} $s_n = \sigma.s_n$: for a bit-$0$ event, the common slot~$s$ shared by the $\geq 2n/3$ firing contributors, i.e., a slot such that $\geq 2n/3$ of the contributors with $\sigma.\ntargets$ covering~$X$ on~$\sigma.L$ satisfy $\sigma.\nslots = s$; for a bit-$1$ event (justification), $s_n = 0$.
\end{itemize}
The associated \emph{cert key} is the lexicographic tuple
\[
  \key(X, h, b, s_n) \;\coloneqq\; \bigl(h,\; b,\; s_n,\; -X.\slot,\; \hash(X)\bigr).
\]
\end{definition}

A justification also prefix-notarizes its block (the justification branch sets $(J, N) \gets (T, T)$ jointly), but we classify the combined event as a justification with key $\key(T, h, 1, s_n)$.
A given block may therefore fire a bit-$0$ cert event on one chain and a bit-$1$ cert event on another, with the bit-$1$ key strictly dominating at the same block.

\begin{definition}[Store]\label{def:store}
A node maintains a \emph{store}
\[
  \Sigma \;=\; (\sigma,\; \T,\; F,\; R,\; \key_R),
\]
where:
\begin{itemize}
  \item $\sigma$ is the \emph{state map}, assigning each accepted block its per-chain post-state (Section~1);
  \item $\T$ is the \emph{block tree}, the set of accepted blocks;
  \item $F$ is the \emph{store-finalized block};
  \item $R$ is the \emph{store root}: the block of the highest-key cert event ever observed;
  \item $\key_R$ is the cached cert key of the event that set $R$.
\end{itemize}
The genesis store has $\T = \{\genesis\}$, $F = R = \genesis$, and $\key_R = (0,\, 1,\, 0,\, \genesis.\slot,\, \hash(\genesis))$, treating genesis as a bit-$1$ cert event at height~$0$ with notarization slot~$0$.
\end{definition}

The capital~$\Sigma$ distinguishes the store from the per-chain state~$\sigma$.
We write $\Sigma.\sigma$, $\Sigma.\T$, $\Sigma.F$, $\Sigma.R$, $\Sigma.\key_R$ for field access, and $\Sigma.\key_R.h$, $\Sigma.\key_R.b$ for the height and justification-bit components of the cached key.

The acceptance routine \onBlock (Figure~\ref{alg:store}) admits $B$ only if $B.\parent \in \Sigma.\T$ and $\Sigma.F \preceq B$.
It runs \processSlot once per slot from $\Sigma.\sigma[B.\parent].s + 1$ up to $B.\slot$ (so any height advance enabled by the accumulated tally fires before~$B$'s votes are folded in), then $\processBlock(\cdot, B)$; the resulting post-state becomes $\Sigma.\sigma[B]$.
The post-state then offers two cert-event descriptors to \updateForkChoiceRoot: the justified checkpoint $(\sigma[B].h_j,\, 1,\, \sigma[B].J,\, \sigma[B].s_n)$ and the notarized checkpoint $(\sigma[B].h_n,\, 0,\, \sigma[B].N,\, \sigma[B].s_n)$; the bit-$1$ / bit-$0$ split reflects the distinct events (the justification may have fired at a strictly smaller height than the notarization).
Each call to \updateForkChoiceRoot first filters out any candidate~$C$ with $C \not\succeq \Sigma.F$ and then updates $(\Sigma.R,\, \Sigma.\key_R)$ iff the candidate's cert key strictly exceeds the current $\key_R$; this maintains $R$ incrementally as the lex-max cert event ever observed on a chain extending the current $F$, never recomputed.
Finally, \updateFinalized sets $\Sigma.F \gets \sigma[B].F$ iff $\sigma[B].F \succ \Sigma.F$ and $\sigma[B].F \preceq \Sigma.R$.

$\getConfirmed(\Sigma, \Omega)$ returns a block on the available chain that, under synchrony, is safe in the sense that it will not be reverted as long as more than half of the online validators are honest. Here $\Omega$ represents the state of the available chain --- whatever extra information the validator uses to disambiguate among descendants of $\Sigma.R$ in $\Sigma.\T$. The protocol fixes only that the returned block lies in $\Sigma.\T$ and descends from $\Sigma.R$; it does not constrain $\Omega$'s structure or how the choice depends on it. Different validators may pass different~$\Omega$, and a single validator may pass different~$\Omega$ at different call sites.

The $F$-filter never discards a key-dominant event: Theorem~\ref{thm:fleqr} below shows that $F$ advances only to blocks $\preceq R$, so every cert event whose key could beat $R$ remains on the $F$-chain.

\subsection{Store invariants and safety}

\subsubsection*{Unconditional invariants}

\begin{lemma}[Root-key monotonicity]\label{lem:Rs-key-monotone}
The cached cert key $\Sigma.\key_R$ is non-decreasing over time; in particular, its height component $\Sigma.\key_R.h$ is non-decreasing.
\end{lemma}

\begin{proof}
The cached key changes only inside \updateForkChoiceRoot, whose guard $\newKey > \Sigma.\key_R$ (lex) allows only strict increases; the height coordinate $h$ is the leading component of the key.
\end{proof}

\begin{theorem}[Local irreversibility of finality]\label{thm:finperm}
Once a node sets $\Sigma.F = F$, $\Sigma.F$ descends from~$F$ at all future times.
\end{theorem}

\begin{proof}
$\Sigma.F$ is updated only by \updateFinalized, whose guard $F_B \succ \Sigma.F$ forces strict extension along~$\preceq$.
\end{proof}

\begin{theorem}[$F \preceq R$]\label{thm:fleqr}
The store maintains $F \preceq R$ at all times.
\end{theorem}

\begin{proof}
By induction on store operations.
At genesis, $F = R = \genesis$, so $F \preceq R$.
Assume the invariant before a call to either mutator in Figure~\ref{alg:store}.

\emph{\updateForkChoiceRoot.}
$F$ is untouched, and $R$ either stays or is replaced by a candidate~$C$ that passed the $C \succeq \Sigma.F$ filter; either way $R \succeq F$.

\emph{\updateFinalized.}
$R$ is untouched, and the guard $F_B \preceq \Sigma.R$ ensures $F \gets F_B \preceq R$.
\end{proof}

\begin{remark}[Store-finalized invariant]\label{rem:fs-invariant}
At all times $\Sigma.F$ is either~$\genesis$ or a block that was finalized on some chain; moreover, whenever a justification $(J, h)$ has fired on some chain, the cert event $(h,\, 1,\, J,\, 0)$ was offered to \updateForkChoiceRoot, so $\Sigma.\key_R \geq (h,\, 1,\, 0,\, J.\slot,\, \hash(J))$ for every such~$(J, h)$ with $J \succeq \Sigma.F$.
$\Sigma.F$ only advances via \updateFinalized to $F_B = \sigma[B].F$, and $\sigma[B].F \neq \genesis$ requires the finality branch of \processHeight to have fired on~$B$'s chain, so $F_B$ is a block finalized on that chain.
\end{remark}

\begin{theorem}[Fork-choice consistency]\label{thm:fcconsistency}
Once a node sets $\Sigma.F = F$, $\getConfirmed(\Sigma, \Omega)$ returns a block descending from~$F$ at all future times, for every $\Omega$.
\end{theorem}

\begin{proof}
By Theorems~\ref{thm:finperm} and~\ref{thm:fleqr}, $F \preceq \Sigma.F \preceq \Sigma.R$, and $\getConfirmed(\Sigma, \Omega)$ returns a descendant of~$\Sigma.R$ regardless of~$\Omega$.
\end{proof}

\subsubsection*{Further properties under $f < n/3$}

\begin{lemma}[Checkpoint chain]\label{lem:certchain}
Unless $\geq n/3$ validators are slashable: if~$F$ is finalized at height~$h_F$ and a cert event $(C,\, h,\, b)$ has fired on any chain processed by the node with $h \geq h_F$, then $F$ and~$C$ are compatible, with $F \prec C$ when $h > h_F$.
\end{lemma}

\begin{proof}
If $h_F = 0$ then $F = \genesis$ and compatibility is trivial; assume $h_F \geq 1$.
The cert event fired at the post-state of some processed block~$B$ whose \processSlot call triggered \processHeight with $(\sigma[B].N,\, \sigma[B].h_n) = (C,\, h)$, so $C \preceq B$.
Every $h_n$-update is paired with $\sigma.h \gets \sigma.h + 1$ (Lemma~\ref{lem:slotmono}), hence $\sigma[B].h \geq h + 1 > h_F$; Lemma~\ref{lem:mainsafety} gives $F \preceq B$, and with $C \preceq B$ this yields $F \sim C$.
For $h > h_F$: $\sigma[F].h = h_F < h = \sigma[C].h$ by Remark~\ref{rem:fresh-equiv}, and state-height monotonicity along~$\preceq$ (Lemma~\ref{lem:slotmono}) forces $F \prec C$ on the comparable pair $F \sim C$.
\end{proof}

\begin{lemma}[Upgrade property]\label{lem:upgrade}
Unless $\geq n/3$ validators are slashable: if~$F$ is finalized at height~$h_F$ and some block~$B$ whose post-state has $(F, h_F)$ as its justified checkpoint has been processed by the node, then $R \succeq F$ at all future times.
\end{lemma}

\begin{proof}
Fix the moment when $\onBlock(B)$ offered the descriptor $(h_F,\, 1,\, F,\, 0)$ to \updateForkChoiceRoot, and let $F^0 = \Sigma.F$ at that moment.
Both $F^0$ and $F$ are finalized or genesis (Remark~\ref{rem:fs-invariant}), so Lemma~\ref{lem:finchain} gives $F^0 \preceq F$ or $F \preceq F^0$.

\emph{Case $F \preceq F^0$.}
$\Sigma.R \succeq F^0 \succeq F$ already holds by Theorem~\ref{thm:fleqr}; Theorem~\ref{thm:finperm} preserves $\Sigma.F \succeq F^0$ and Theorem~\ref{thm:fleqr} preserves $\Sigma.R \succeq \Sigma.F$ at all future times, so $\Sigma.R \succeq F$ persists.

\emph{Case $F^0 \preceq F$.}
The filter passes, so after the call $\Sigma.\key_R \geq (h_F,\, 1,\, 0,\, F.\slot,\, \hash(F))$; Lemma~\ref{lem:Rs-key-monotone} preserves this bound thereafter.
Write $h_R \coloneqq \Sigma.\key_R.h$ and $\bit_R \coloneqq \Sigma.\key_R.b$; the bound gives $h_R \geq h_F$.
If $h_R > h_F$, Lemma~\ref{lem:certchain} applied to the cert event $(R,\, h_R,\, \bit_R)$ gives $F \prec R$.
If $h_R = h_F$, the bound forces $\bit_R = 1$ (strictly greater would require $\bit_R > 1$, impossible), so $R$ was justified at height~$h_F$ on some processed chain, giving a \just-quorum~$Q'$ of size $\geq 2n/3$ with target~$R$ at height~$h_F$.
Finality of~$F$ at~$h_F$ gives a quorum~$Q_F$ with $\finalizeHeight = h_F$ and $\finalizeTarget = F$.
By quorum intersection, $|Q' \cap Q_F| \geq n/3$; pick $v \in Q' \cap Q_F$ not slashable.
E1 (Definition~\ref{def:slashing}) applied to $v$'s votes in~$Q', Q_F$ forces $R = F$.
\end{proof}

\begin{theorem}[Local acceptance of finality updates]\label{thm:finlive}
Unless $\geq n/3$ validators are slashable: if a block~$B$ is processed by \onBlock and its post-state has finalized block~$F_B$, then after processing $F \succeq F_B$.
\end{theorem}

\begin{proof}
If $F \succeq F_B$ already, the claim holds.
Otherwise $F$ is genesis or finalized (Remark~\ref{rem:fs-invariant}), so Lemma~\ref{lem:finchain} gives $F \sim F_B$ and hence $F \prec F_B$.
$F_B$ being finalized in~$B$'s post-state requires some ancestor~$B' \preceq B$ to have post-state $(J_{B'},\, h_j^{B'}) = (F_B,\, h_{F_B})$; $B'$ was processed before~$B$ (parent-first), so Lemma~\ref{lem:upgrade} gives $R \succeq F_B$, and the \updateFinalized guard passes.
\end{proof}

\begin{theorem}[Lock-in]\label{thm:lockin}
Unless $\geq n/3$ validators are slashable: if block~$F$ is finalized at height~$h$ on any chain, and some block whose post-state has $(F, h)$ as its justified checkpoint has been processed by the node, then $R \succeq F$ at all future times, and $\getConfirmed(\Sigma, \Omega)$ always returns a descendant of~$F$, for every $\Omega$.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:upgrade} gives $R \succeq F$ directly.
$\getConfirmed(\Sigma, \Omega)$ (descending from~$R$ for every $\Omega$) therefore returns a descendant of~$F$.
\end{proof}

\begin{theorem}[Order independence]\label{thm:orderindep}
Unless $\geq n/3$ validators are slashable: the store state after processing a set of blocks depends only on which blocks have been processed, not on their order.
\end{theorem}

\begin{proof}
Each block's post-state is a deterministic function of itself and its parent's post-state, so $\sigma$ depends only on the processed set.

\emph{$F$.} By Lemma~\ref{lem:finchain}, $\{\sigma[B].F : B \in \T\}$ is a chain with maximum $F_{\max}$, attained at some processed~$B^\ast$.
Remark~\ref{rem:fs-invariant} forces $F$ into this chain, so $F \preceq F_{\max}$; once $B^\ast$ has been processed, Theorem~\ref{thm:finlive} gives $F \succeq F_{\max}$.

\emph{$R$.} The multiset~$D$ of descriptors offered to \updateForkChoiceRoot is a deterministic function of the processed set: each block contributes its post-state's two descriptors exactly once.
Write $K_{\max} \coloneqq (h_{F_{\max}},\, 1,\, 0,\, F_{\max}.\slot,\, \hash(F_{\max}))$.
For any $(h, b, C, s_n) \in D$ with $h \geq h_{F_{\max}}$, Lemma~\ref{lem:certchain} gives $C \succeq F_{\max}$ (strict when $h > h_{F_{\max}}$); at $h = h_{F_{\max}}$, Remark~\ref{rem:fs-invariant} with $F_{\max}$ justified gives $\Sigma.\key_R \geq K_{\max}$, forcing any competitive descriptor to have bit~$1$, hence a \just-quorum~$Q_C$ of size $\geq 2n/3$ targeting~$C$ at height~$h_{F_{\max}}$, while finality of~$F_{\max}$ gives a quorum~$Q_F$ with $\finalizeHeight = h_{F_{\max}}$ and $\finalizeTarget = F_{\max}$; quorum intersection and E1 (as in Lemma~\ref{lem:upgrade}) force $C = F_{\max}$.
Descriptors with $h < h_{F_{\max}}$ have key strictly below $K_{\max}$ on the leading coordinate, so once $\key_R \geq K_{\max}$ holds they are dominated.
Hence $R$'s final value is the lex-max over $\{(h, b, C, s_n) \in D : h \geq h_{F_{\max}}\}$, a deterministic function of the processed set.
\end{proof}

\begin{corollary}[Cross-node agreement]\label{cor:agreement}
Unless $\geq n/3$ validators are slashable: two nodes that have processed the same set of blocks have the same $(\sigma,\, \T,\, F,\, R,\, \key_R)$.
\end{corollary}

\begin{proof}
Immediate from Theorem~\ref{thm:orderindep}.
\end{proof}

\begin{lemma}[Cached key dominates every cert event]\label{rev:lem:Rs-key-dominates}
Unless $\geq n/3$ validators are slashable, every cert event $(X, h, b, s_n)$ that has fired on any chain processed by the node satisfies $\key(X, h, b, s_n) \leq \Sigma.\key_R$.
\end{lemma}

\begin{proof}
Fix the moment when the cert event fired and \onBlock offered its descriptor to \updateForkChoiceRoot. Let $F^0 = \Sigma.F$ at that moment, with finality height $h_{F^0}$ ($F^0$ is genesis or finalized by \Cref{rem:fs-invariant}).

\emph{Case $h > h_{F^0}$.} \Cref{lem:certchain} gives $F^0 \prec X$, so the filter $\Sigma.F \preceq X$ passes; \updateForkChoiceRoot raises $\Sigma.\key_R$ to $\geq \key(X, h, b, s_n)$.

\emph{Case $h = h_{F^0} \geq 1$.}
By \Cref{rem:fs-invariant}, $F^0$ being finalized at $h_{F^0}$ implies $\Sigma.\key_R \geq (h_{F^0},\, 1,\, 0,\, F^0.\slot,\, \hash(F^0))$ at the time of offering. We split on the bit~$b$ of the present cert event.

\emph{Sub-case $b = 0$.} The present cert key is $(h_{F^0},\, 0,\, s_n,\, X.\slot,\, \hash(X))$, whose bit coordinate is $0$, strictly less than the cached~$1$; hence $\Sigma.\key_R > \key(X, h, b, s_n)$ already, with no update needed.

\emph{Sub-case $b = 1$.} The cert event justifies $X$ at height~$h_{F^0}$, providing a \just-quorum~$Q'$ of size $\geq 2n/3$ with target~$X$ at height~$h_{F^0}$. Finality of $F^0$ at $h_{F^0}$ provides a quorum~$Q_F$ of size $\geq 2n/3$ with $\finalizeHeight = h_{F^0}$ and $\finalizeTarget = F^0$. Quorum intersection gives $|Q' \cap Q_F| \geq n/3$; pick $v \in Q' \cap Q_F$ not slashable. E1 (\Cref{def:slashing}) applied to $v$'s votes in $Q'$ and $Q_F$ forces $X = F^0$. The present cert key is then $(h_{F^0},\, 1,\, 0,\, F^0.\slot,\, \hash(F^0)) \leq \Sigma.\key_R$.

\emph{Case $h \leq h_{F^0} = 0$.} Then $F^0 = \genesis$ and $h = 0$. The genesis store sets $\Sigma.\key_R = (0,\, 1,\, 0,\, \genesis.\slot,\, \hash(\genesis))$ (\Cref{def:store}); any cert event at $h = 0$ has $b \leq 1$ and is dominated.

\emph{Case $h < h_{F^0}$ with $h_{F^0} \geq 1$.} By \Cref{rem:fs-invariant}, $F^0$ being finalized at $h_{F^0}$ implies $\Sigma.\key_R$ already satisfies $\Sigma.\key_R.h \geq h_{F^0} > h$ at the time of offering, so the present cert key is already strictly dominated.

In every case, $\Sigma.\key_R \geq \key(X, h, b, s_n)$ after the call; \Cref{lem:Rs-key-monotone} preserves the bound at all later times.
\end{proof}

\begin{corollary}[No cert event above the root]\label{rev:cor:no-high-cn}
Unless $\geq n/3$ validators are slashable, every cert event $(X, h, b, s_n)$ that has fired on any chain processed by the node satisfies $h \leq R.h$.
\end{corollary}

\begin{proof}
\Cref{rev:lem:Rs-key-dominates} gives $\key(X, h, b, s_n) \leq \Sigma.\key_R$. The leading coordinate of the cert key is the height, so $h \leq \Sigma.\key_R.h = R.h$.
\end{proof}

\begin{corollary}[Processed-block height bound]\label{rev:cor:processed-height-bound}
Unless $\geq n/3$ validators are slashable, every processed block~$B$ satisfies $\sigma[B].h \leq R.h + 1$.
\end{corollary}

\begin{proof}
If $\sigma[B].h \geq 1$, \Cref{lem:heightprog} produces a cert event at height $\sigma[B].h - 1$ on $B$'s chain (the most recent height advance leading to $\sigma[B].h$); by \Cref{rev:cor:no-high-cn}, $\sigma[B].h - 1 \leq R.h$.
\end{proof}

\begin{lemma}[State-height range for blocks descending from $R$]\label{rev:lem:confirmed-h-range}
Unless $\geq n/3$ validators are slashable, every block $B \in \Sigma.\T$ with $\Sigma.R \preceq B$ satisfies $\sigma[B].h \in \{\Sigma.\key_R.h,\, \Sigma.\key_R.h + 1\}$. In particular, $\sigma[\getConfirmed(\Sigma, \Omega)].h \in \{\Sigma.\key_R.h,\, \Sigma.\key_R.h + 1\}$ for every auxiliary state~$\Omega$.
\end{lemma}

\begin{proof}
Write $h \coloneqq \Sigma.\key_R.h$ and $R \coloneqq \Sigma.R$. We show $\sigma[R].h = h$. If $R = \genesis$, $\sigma[\genesis].h = 0 = h$ directly from the genesis state. Otherwise the cert event that established $(R, \key_R)$ fired on some chain in $\Sigma.\T$ with pre-advance $\sigma[L].h = h$ at firing block~$L \succeq R$, and freshness places $R \preceq L$ with $R.\slot \geq \sigma[L].s_h$; by \Cref{rem:fresh-equiv}, $\sigma[R].h = \sigma[L].h = h$.

\Cref{lem:slotmono} (state-height monotonicity along $\preceq$) applied to $R \preceq B$ then gives $\sigma[B].h \geq \sigma[R].h = h$. \Cref{rev:cor:processed-height-bound} gives $\sigma[B].h \leq h+1$. Hence $\sigma[B].h \in \{h, h+1\}$.

The store invariant ($\getConfirmed(\Sigma, \Omega) \succeq \Sigma.R$, \Cref{alg:store}) gives the in-particular claim.
\end{proof}

\section{Inactivity leak}

This section extends the per-chain state machine of Figure~\ref{alg:state-machine} with a penalty counter and a new transition function \processInactivityPenalties, invoked each slot by an extended \processSlot (Figure~\ref{alg:leak-processslot}); \processBlock, \processHeight, and \processVote are unchanged.
Write $T = \getConfirmed(\Sigma, \Omega)$ for the \emph{canonical chain}, with $\Omega$ the validator's auxiliary state.

Two results follow.
\emph{Tightness} (Theorem~\ref{thm:tightness}) is purely state-logic: on any fixed chain, the total penalty increment over a non-finality period is at least $(T-1) \cdot n/3$, with no hypothesis on synchrony, honesty, or the slashable set.
\emph{Fairness} (Theorem~\ref{thm:fairness}) is local to a single honest~$i$: the vote~$i$ produces per the honest rule (Definition~\ref{def:voting-rule}), once included on any extension of~$i$'s canonical chain still at the same state-height, exempts~$i$ from Layer~1 penalties at that state-height.
Layer-2 exemption is out of scope: it depends on quorum formation on a common target, not on any local property of~$i$.

\subsection{Penalty units}

Augment the per-chain state tuple with one non-negative counter $\sigma.\penalties[1..n]$, initialized to~$0$ at genesis; no other field is added or modified.
Figure~\ref{alg:leak-processslot} replaces \processSlot from Figure~\ref{alg:state-machine} with a version that snapshots the pre-advance state and invokes \processInactivityPenalties after \processHeight.
The store of Section~\ref{sec:store} invokes this extended \processSlot in place of the base one.

\begin{figure}[H]
\caption{Penalty-tracking state extension (leak section)}\label{alg:leak-processslot}
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]

\Function{\processSlot}{$\sigma$} \Comment{extended: tracks penalties}
  \State $\sigma.s \gets \sigma.s + 1$
  \State $\sigma_{\mathrm{pre}} \gets \sigma$ \Comment{snapshot of pre-advance state}
  \State $\sigma \gets \processHeight(\sigma)$
  \State $\sigma \gets \processInactivityPenalties(\sigma_{\mathrm{pre}},\; \sigma)$
  \State \Return $\sigma$

\vspace{3pt}

\Function{\processInactivityPenalties}{$\sigma_{\mathrm{pre}},\; \sigma_{\mathrm{post}}$}
  \If{$\sigma_{\mathrm{pre}}.h = \sigma_{\mathrm{post}}.h$} \Comment{Layer 1: stall}
    \For{each $i \in \{1, \ldots, n\}$}
      \If{$\sigma_{\mathrm{post}}.\ntargets[i] = \bot$}
        \State $\sigma_{\mathrm{post}}.\penalties[i] \gets \sigma_{\mathrm{post}}.\penalties[i] + 1$
      \EndIf
    \EndFor
  \EndIf
  \If{$\sigma_{\mathrm{pre}}.J = \sigma_{\mathrm{post}}.J$} \Comment{Layer 2 target: justification did \emph{not} advance}
    \State $T^* \gets \arg\max_{T' \in \{\sigma_{\mathrm{pre}}.\jtargets[i] \,:\, \sigma_{\mathrm{pre}}.\jtargets[i] \neq \bot\}} |\{i : \sigma_{\mathrm{pre}}.\jtargets[i] = T'\}|$ \Comment{$T^* \gets \bot$ if the set is empty; ties broken deterministically}
    \For{each $i \in \{1, \ldots, n\}$}
      \If{$\sigma_{\mathrm{pre}}.\jtargets[i] \neq T^*$ \textbf{or} $\sigma_{\mathrm{pre}}.\jtargets[i] = \bot$}
        \State $\sigma_{\mathrm{post}}.\penalties[i] \gets \sigma_{\mathrm{post}}.\penalties[i] + 1$
      \EndIf
    \EndFor
  \EndIf
  \If{$\sigma_{\mathrm{pre}}.F = \sigma_{\mathrm{post}}.F$ \textbf{and} $\sigma_{\mathrm{pre}}.F \prec \sigma_{\mathrm{pre}}.J$} \Comment{Layer 2 finalize: $F$ did not advance, pending finality}
    \For{each $i \in \{1, \ldots, n\}$}
      \If{$i \notin \sigma_{\mathrm{pre}}.P$}
        \State $\sigma_{\mathrm{post}}.\penalties[i] \gets \sigma_{\mathrm{post}}.\penalties[i] + 1$
      \EndIf
    \EndFor
  \EndIf
  \State \Return $\sigma_{\mathrm{post}}$
\EndFunction

\end{algorithmic}
\end{boxedminipage}
\end{center}
\end{figure}

\paragraph{Reading Figure~\ref{alg:leak-processslot}.}
\processInactivityPenalties checks three independent guards; any subset may fire at a given slot.

\emph{Layer~1 (stall).} Fires when the slot did not advance height.
Every~$i$ with $\sigma_{\mathrm{post}}.\ntargets[i] = \bot$ is bumped, i.e.\ every validator that failed to cast a fresh vote at the current height ($\ntargets$ is not reset at a stall).

\emph{Layer~2 target.} Fires when the justification branch of \processHeight did not fire (else $J$ would strictly advance by Lemma~\ref{lem:slotmono}).
$T^*$ is the plurality target across $\sigma_{\mathrm{pre}}.\jtargets$ (deterministic tiebreak), or $\bot$ if no \just votes have been registered; the second disjunct of the guard bumps non-voters when $T^* = \bot$ (in which case all~$n$ validators are bumped).
Since no $2n/3$-quorum on a single target formed, the plurality count is $< 2n/3$, so $|\{i : \sigma_{\mathrm{pre}}.\jtargets[i] \neq T^* \vee \sigma_{\mathrm{pre}}.\jtargets[i] = \bot\}| > n/3$.

\emph{Layer~2 finalize.} Fires when a pending-finality gap is open at the start of the slot ($\sigma_{\mathrm{pre}}.F \prec \sigma_{\mathrm{pre}}.J$) and $F$ did not advance; every $i \notin \sigma_{\mathrm{pre}}.P$ is bumped.
The finality branch of \processHeight, when it fires, sets $\sigma_{\mathrm{post}}.F = \sigma_{\mathrm{pre}}.J \succ \sigma_{\mathrm{pre}}.F$, so this guard is skipped automatically.

\paragraph{Canonical-freshness.}
A vote~$v$ processed at the current slot on canonical chain~$T$ at state-height $h_c = \sigma_{\mathrm{post}}.h$ bumps $\sigma.\ntargets[v.\validator]$ iff $v$ is \emph{canonical-fresh}:
\[
  v.\height = h_c \;\wedge\; v.\target \preceq T \;\wedge\; v.\target.\slot \geq \sigma_{\mathrm{post}}.s_h,
\]
equivalently $v.\target \preceq T$ with $\sigma[v.\target].h = h_c$ (Remark~\ref{rem:fresh-equiv}).

\subsection{Leak tightness}

\begin{definition}[Non-finality period]\label{def:non-finality-period}
Fix a chain and write $\sigma_{\mathrm{pre}}^{(t)}, \sigma_{\mathrm{post}}^{(t)}$ for its pre/post-\processHeight snapshots at slot~$k$.
A \emph{non-finality period} is a maximal interval $[t_1, t_T]$ of consecutive slots on which $F$ is stuck, i.e., $\sigma_{\mathrm{post}}^{(t)}.F = \sigma_{\mathrm{pre}}^{(t)}.F$ throughout.
Each such slot falls into exactly one category:
\begin{itemize}
  \item \emph{pending-stuck}: $\sigma_{\mathrm{pre}}^{(t)}.F \prec \sigma_{\mathrm{pre}}^{(t)}.J$ (pending-finality gap open at the start);
  \item \emph{stable-stuck}: $\sigma_{\mathrm{pre}}^{(t)}.F = \sigma_{\mathrm{pre}}^{(t)}.J$ and $J$ does not advance at~$t$;
  \item \emph{transition}: $\sigma_{\mathrm{pre}}^{(t)}.F = \sigma_{\mathrm{pre}}^{(t)}.J$ and the justification branch fires at~$t$ (opening a gap by slot's end).
\end{itemize}
\end{definition}

\begin{theorem}[Non-finality period penalty bound]\label{thm:tightness}
Over any non-finality period of length $T \geq 1$ on any chain, the total penalty increment across all validators assessed by \processInactivityPenalties on that chain is at least $(T - 1) \cdot n/3$.
\end{theorem}

\begin{proof}
Fix a non-finality period $[t_1, t_T]$.
We show (i) at most one slot is transition, and (ii) every pending-stuck or stable-stuck slot contributes $> n/3$ to the total penalty increment.

\emph{(i) At most one transition slot.}
After a transition at~$t$, $\sigma_{\mathrm{post}}^{(t)}.J \succ \sigma_{\mathrm{pre}}^{(t)}.F$; since $F$ does not advance in the period and $J$ is monotone across slots, $\sigma_{\mathrm{pre}}^{(t')}.F \prec \sigma_{\mathrm{pre}}^{(t')}.J$ at every later $t' > t$, making every such $t'$ pending-stuck.

\emph{(ii) Per-slot charge $> n/3$.}

\emph{Pending-stuck.} The Layer~2 finalize guard fires at~$t$.
The finality branch of \processHeight did not fire (else $F$ would advance), so $|\sigma_{\mathrm{pre}}^{(t)}.P| < 2n/3$ and $|\{i : i \notin \sigma_{\mathrm{pre}}^{(t)}.P\}| > n/3$.

\emph{Stable-stuck.} The Layer~2 target guard fires at~$t$.
The justification branch did not fire, so the plurality count on $\sigma_{\mathrm{pre}}^{(t)}.\jtargets$ is $< 2n/3$ (counting $T^* = \bot$ as~$0$); hence the bumped set has size $> n/3$.

At most one slot is transition, so at least $T - 1$ slots are pending-stuck or stable-stuck; each contributes $> n/3$ by~(ii), and Layer~1 adds only non-negative charges.
Summing gives total increment $\geq (T-1) \cdot n/3$.
\end{proof}

\subsection{Honest voting rule}

We state the honest rule explicitly, as it underpins the leak fairness proof.

\begin{definition}[Honest voting rule]\label{def:voting-rule}
Let $V$ be the set of votes $i$ has previously cast, and write $\currentSlot$ for the node's protocol time (a slot counter shared across chains).
At each voting opportunity, $i$ constructs its vote by $\constructVote(V,\, \Sigma,\, \Omega)$, defined in Figure~\ref{alg:voting-rule}.
The first-vote check in the (R1) branch ensures at most one \just vote per height~$h_c$, and the target-selection clause preserves E1-safety under view shifts.
\end{definition}

\begin{figure}[H]
\caption{Honest voting rule}\label{alg:voting-rule}
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]

\Function{\constructVote}{$V,\; \Sigma,\; \Omega$}
  \State $T \gets \getConfirmed(\Sigma,\; \Omega)$
  \State $h_c \gets \sigma[T].h$; \;\; $h_j \gets \sigma[T].h_j$; \;\; $J \gets \sigma[T].J$
  \vspace{3pt}
  \State \textbf{-- Target selection --}
  \If{$\exists\, v \in V$ with $v.\finalizeHeight = h_c$ and $v.\finalizeTarget \neq \bot$}
    \State $T_c \gets v.\finalizeTarget$ \Comment{$i$ is locked at $h_c$}
  \Else
    \State $T_c \gets T$ \label{line:unlocked-target}
  \EndIf
  \vspace{3pt}
  \State \textbf{-- Vote kind and finalize fields --}
  \If{$\not \exists v \in V$ s.t. $v.\height = h_c$} \Comment{first vote at $h_c$: cast \just (R1)}
    \If{every $v \in V$ with $v.\height = h_j$ has $v.\target = J$}
      \State $(f_h,\, f_t) \gets (h_j,\, J)$
    \Else
      \State $(f_h,\, f_t) \gets (\bot,\, \bot)$
    \EndIf
    \State \Return $(i,\; h_c,\; \currentSlot,\; T_c,\; \just,\; f_h,\; f_t)$
  \Else \Comment{already voted at $h_c$: cast \notar (R2)}
    \State \Return $(i,\; h_c,\; \currentSlot,\; T_c,\; \notar,\; \bot,\; \bot)$
  \EndIf
\EndFunction

\end{algorithmic}
\end{boxedminipage}
\end{center}
\end{figure}

Only the (R1) branch produces a \just vote and, gated by the first-vote check, fires at most once per~$h_c$, ensuring per-height \just-uniqueness.
The (R2) branch is the \emph{resubmission} fallback: fired at every subsequent voting opportunity while canonical stays at~$h_c$, so when the (R1) \just quorum did not form on canonical (heterogeneous targets across views), the \notar vote's prefix counting may still notarize a common ancestor.

Checking \emph{every} prior vote of~$i$ at~$h_j$ (not just the (R1) one) is required because a (R2) at~$h_j$ may have chosen a target differing from the (R1) target if~$i$'s view shifted; committing $f_t = J$ at height~$h_j$ requires (by E1) that every $h_j$-vote of~$i$ have target~$J$, so in case of a split the conditional fires $(\bot, \bot)$: $i$ misses the finalization at~$h_j$ but remains E1-safe.

\begin{lemma}[Honest E1 safety]\label{lem:honest-e1-safety}
If validator~$i$ follows Definition~\ref{def:voting-rule}, then $i$ never produces a vote-pair $(v_1, v_2)$ with $v_2.\finalizeTarget \neq \bot$, $v_1.\height = v_2.\finalizeHeight$, and $v_1.\target \neq v_2.\finalizeTarget$.
\end{lemma}

\begin{proof}
Write $h_\ast \coloneqq v_2.\finalizeHeight$ and $T_\ast \coloneqq v_2.\finalizeTarget \neq \bot$.
Only (R1) produces finalize-bearing votes, so $v_2$ is an (R1) vote with $(f_h, f_t) = (h_j, J) = (h_\ast, T_\ast)$ read from the canonical state at~$v_2$'s firing.

\emph{Case~A ($v_1$ cast before~$v_2$).}
The (R1) finalize-field conditional at~$v_2$ attached $(h_\ast, T_\ast)$ only because every prior vote of~$i$ at height~$h_\ast$ had target~$T_\ast$; in particular $v_1.\target = T_\ast$.

\emph{Case~B ($v_1$ cast after~$v_2$).}
At $v_1$'s firing, the voting height is $h_c = v_1.\height = h_\ast$, and~$i$ has the prior vote~$v_2$ with $\finalizeHeight = h_\ast = h_c$ and $\finalizeTarget = T_\ast \neq \bot$: $i$ is \emph{locked at $h_c$}, and the target-selection clause of Definition~\ref{def:voting-rule} forces $T_c = T_\ast$ regardless of~$i$'s canonical view at that moment.
Hence $v_1.\target = T_c = T_\ast$.

\emph{Case~C ($v_1 = v_2$).}
Vacuous: $v_2.\height = h_c \neq h_\ast = v_2.\finalizeHeight$ (every height advance sets $h_j \gets h$ before $h \gets h + 1$, so $h_j < h$ in any post-advance state).
\end{proof}

\subsection{Leak fairness}

Fairness is local to a single honest validator~$i$: we fix~$i$, its local store~$\Sigma$, its auxiliary state~$\Omega$, and the canonical chain $T = \getConfirmed(\Sigma, \Omega)$.
The claim is that the vote~$v$ produced by Definition~\ref{def:voting-rule}, once included on any extension of~$T$ still at the same state-height, exempts~$i$ from Layer~1 penalties at that state-height.
Height advances reset $\jtargets$ and $\ntargets$ (Figure~\ref{alg:state-machine}), so Layer-2 state at the next height is independent of~$i$'s prior votes and no ``lock'' issue propagates.

The proof proceeds via four supporting lemmas: per-height justification uniqueness (Lemma~\ref{lem:just-unique-height}), root-height exclusivity of the notarize-only bit (Lemma~\ref{lem:root-h-notarize-only}), the absence of finalize votes without a matching justification cert event (Lemma~\ref{lem:no-finalize-without-just}), and a canonical-freshness statement for~$i$'s vote (Lemma~\ref{lem:canonical-fresh-honest-vote}).

\begin{lemma}[Per-height justification uniqueness]\label{lem:just-unique-height}
If $f < n/3$ and all honest validators follow Definition~\ref{def:voting-rule}: for any two blocks $C_1, C_2$ justified at the same height~$h$ on any chains processed by the node, $C_1 = C_2$.
\end{lemma}

\begin{proof}
Each justification at height~$h$ on a chain yields a \just-quorum $Q_k$ ($k \in \{1, 2\}$) of size $\geq 2n/3$ signing a \just vote at height~$h$ for $C_k$; quorum intersection gives $|Q_1 \cap Q_2| \geq n/3$.
Since $f < n/3$, some $v \in Q_1 \cap Q_2$ is honest; by Definition~\ref{def:voting-rule}, only (R1) emits a \just vote and it fires at most once per height, so~$v$'s single \just vote at~$h$ has both $C_1$ and $C_2$ as target, forcing $C_1 = C_2$.
\end{proof}

\begin{lemma}[Root-height exclusivity of the notarize-only bit]\label{lem:root-h-notarize-only}
Unless $\geq n/3$ validators are slashable: if $\Sigma.\key_R.b = 0$, then no justification at height $\Sigma.\key_R.h$ has fired on any processed chain.
\end{lemma}

\begin{proof}
Write $h_R \coloneqq \Sigma.\key_R.h$ and suppose some justification at~$h_R$ fired on a processed chain, producing $J_\ast$ justified at~$h_R$.
\onBlock offered the descriptor $(h_R,\, 1,\, J_\ast,\, 0)$ to \updateForkChoiceRoot at that time, with key $(h_R,\, 1,\, 0,\, J_\ast.\slot,\, \hash(J_\ast))$.
Let $F^0 \coloneqq \Sigma.F$ at that moment; $F^0$ is finalized or genesis (Remark~\ref{rem:fs-invariant}) at some height $h_{F^0}$.
When $F^0$ was set, the cert event $(F^0,\, h_{F^0},\, 1,\, 0)$ had already raised the cache height to $\geq h_{F^0}$ (the \updateFinalized guard $F_B \preceq R$ ensures the cache covered it), and Lemma~\ref{lem:Rs-key-monotone} preserves $h_{F^0} \leq h_R$.
We show $F^0 \preceq J_\ast$, so the filter in \updateForkChoiceRoot passes.
If $h_R > h_{F^0}$, Lemma~\ref{lem:certchain} applied to $F^0$ and the cert event at $J_\ast$ gives $F^0 \prec J_\ast$.
If $h_R = h_{F^0}$, finality of $F^0$ at~$h_R$ gives a finalize-quorum $Q_F$ with $\finalizeHeight = h_R$ and $\finalizeTarget = F^0$, and $J_\ast$ being justified at~$h_R$ gives a \just-quorum~$Q'$ targeting~$J_\ast$ at~$h_R$; quorum intersection and E1 (as in Lemma~\ref{lem:upgrade}) force $F^0 = J_\ast$.
Either way $F^0 \preceq J_\ast$, so \updateForkChoiceRoot raises the cache to $\Sigma.\key_R \geq (h_R,\, 1,\, 0,\, J_\ast.\slot,\, \hash(J_\ast))$ at that moment; Lemma~\ref{lem:Rs-key-monotone} preserves this bound.
At the present moment $\Sigma.\key_R.h = h_R$, so lex-comparison with the first coordinate fixed forces $\Sigma.\key_R.b \geq 1$, contradicting $\key_R.b = 0$.
\end{proof}

\begin{lemma}[Finalize vote implies justification]\label{lem:no-finalize-without-just}
If~$i$ has cast a vote~$v$ with $v.\finalizeHeight = h'$ and $v.\finalizeTarget = T_f \neq \bot$, then~$T_f$ was justified at height~$h'$ on some processed chain by the time~$v$ was cast.
\end{lemma}

\begin{proof}
Only (R1) produces $\finalizeHeight \neq \bot$, so~$v$ came from (R1) at an earlier moment with~$i$'s local store~$\Sigma_0$ and auxiliary state~$\Omega_0$; by the (R1) finalize-field conditional, $(f_h, f_t) = (h_j, J) = (h', T_f)$ read from~$i$'s then-canonical $T_0 = \getConfirmed(\Sigma_0, \Omega_0)$, so $\sigma[T_0].h_j = h'$ and $\sigma[T_0].J = T_f$.
A state has $(h_j, J) = (h', T_f)$ only after the justification branch of \processHeight fired on that chain with target~$T_f$ at pre-state height~$h'$, i.e.\ some ancestor $B' \preceq T_0$ triggered justification of~$T_f$ at height~$h'$.
\end{proof}

\begin{lemma}[Canonical-fresh honest vote]\label{lem:canonical-fresh-honest-vote}
Unless $\geq n/3$ validators are slashable, and assuming $f < n/3$ and all honest validators follow Definition~\ref{def:voting-rule}: let~$i$ be honest with local store~$\Sigma$ and auxiliary state~$\Omega$, $T = \getConfirmed(\Sigma, \Omega)$, and $h_c = \sigma[T].h$.
Let~$v$ be the vote~$i$ produces at the current slot.
Then $v.\target \preceq T$ with $\sigma[v.\target].h = h_c$.
In particular, $v$ is canonical-fresh on every extension $T' \succeq T$ with $\sigma[T'].h = h_c$.
\end{lemma}

\begin{proof}
Write $h_R \coloneqq \Sigma.\key_R.h$ and $\bit_R \coloneqq \Sigma.\key_R.b$.
$T$ descends from~$R$ (Figure~\ref{alg:store}), so $R \preceq T$; the cached cert event fired at height~$h_R$ on a processed chain with $\sigma[R].h = h_R$ (Remark~\ref{rem:fresh-equiv}, genesis trivial).
State-height monotonicity (Lemma~\ref{lem:slotmono}) gives $h_c \geq h_R$.
Every state-height advance on~$T$ fires a cert event whose descriptor is offered to \updateForkChoiceRoot, raising $\key_R.h$ if it exceeds the cached value; hence no advance from $h_R + 1$ to $h_R + 2$ has fired on~$T$, so $h_c \in \{h_R,\, h_R + 1\}$.
By Definition~\ref{def:voting-rule}, $v.\height = h_c$ and $v.\target = T_c$, where $T_c = T$ in the unlocked case (line~\ref{line:unlocked-target} of \constructVote{}), and $T_c = L$ in the locked case, where~$L$ is a lock target from a prior (R1) vote of~$i$ with $\finalizeHeight = h_c$.

\emph{Case $h_c = h_R + 1$.}
A lock at~$h_c$ would, by Lemma~\ref{lem:no-finalize-without-just}, require some~$J_\ast$ justified at~$h_c$ on a processed chain.
The filter-passing argument of Lemma~\ref{lem:root-h-notarize-only} applies (with $h_R$ there replaced by~$h_c$, using $h_{F^0} \leq h_R < h_c$), so \updateForkChoiceRoot raised the cache to $\Sigma.\key_R.h \geq h_c = h_R + 1$ and Lemma~\ref{lem:Rs-key-monotone} preserves this, contradicting $\key_R.h = h_R$.
So~$i$ is unlocked at~$h_c$ and $T_c = T$.

\emph{Case $h_c = h_R$ with $\bit_R = 0$.}
By Lemma~\ref{lem:root-h-notarize-only}, no justification at~$h_c$ has fired on any processed chain; by Lemma~\ref{lem:no-finalize-without-just},~$i$ has not cast a finalize-bearing vote at~$h_c$, so~$i$ is unlocked and $T_c = T$.

\emph{Case $h_c = h_R$ with $\bit_R = 1$.}
If~$i$ is unlocked, $T_c = T$ directly.
Otherwise~$i$ is locked at~$h_c$ with target~$L$; Lemma~\ref{lem:no-finalize-without-just} gives $L$ justified at~$h_c$ on some processed chain, and $\bit_R = 1$ gives $R$ justified at~$h_R = h_c$ on some processed chain.
Lemma~\ref{lem:just-unique-height} then gives $L = R \preceq T$, with $\sigma[T_c].h = \sigma[R].h = h_c$.
\end{proof}

\begin{theorem}[Layer~1 leak fairness]\label{thm:fairness}
Unless $\geq n/3$ validators are slashable, and assuming $f < n/3$ and all honest validators follow Definition~\ref{def:voting-rule}: let~$i$ be an honest validator, $\Sigma$ its local store, $\Omega$ its auxiliary state, $T = \getConfirmed(\Sigma, \Omega)$ its canonical chain, and $h_c = \Sigma.\sigma[T].h$ the current canonical state-height.
Let $v$ be the vote~$i$ produces at the current slot.
Let $\Sigma'$ be any store obtained by processing additional blocks and votes on top of~$\Sigma$, with auxiliary state $\Omega'$, such that $T' = \getConfirmed(\Sigma', \Omega')$ is an extension of~$T$ with $\Sigma'.\sigma[T'].h = h_c$, and suppose~$v$ is included on~$T'$.
Then for every slot processed on~$T'$ from that point onward while canonical state-height stays at~$h_c$, the Layer~1 increment to $\sigma.\penalties[i]$ on canonical is zero.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:canonical-fresh-honest-vote} gives $T_c \coloneqq v.\target \preceq T \preceq T'$ with $\sigma[T_c].h = h_c$.
The hypothesis $\Sigma'.\sigma[T'].h = h_c$ with Lemma~\ref{lem:slotmono} gives $\Sigma'.\sigma[T'].s_h = \Sigma.\sigma[T].s_h$; Remark~\ref{rem:fresh-equiv} then gives $T_c.\slot \geq \Sigma'.\sigma[T'].s_h$.
So~$v$ is canonical-fresh on~$T'$, and processing~$v$ sets $\ntargets[i] \neq \bot$.
$\ntargets$ resets only at a height advance, which does not fire while state-height stays at~$h_c$, so $\ntargets[i] \neq \bot$ persists and Layer~1 does not bump~$i$.
\end{proof}

\begin{remark}[Weakening $f < n/3$ under a double-justify slashing rule]\label{rem:canonical-fresh-slashable}
The $f < n/3$ hypothesis in \Cref{lem:canonical-fresh-honest-vote} is needed only by the invocation of \Cref{lem:just-unique-height} to collapse the lock target~$L$ onto $R$ in the $h_c = h_R$, $\bit_R = 1$ case; \Cref{lem:just-unique-height}'s proof uses \Cref{def:voting-rule}'s behavioural constraint (at most one \just vote per height) to close quorum intersection.
If \Cref{def:slashing} is extended with an additional slashing rule declaring any pair of \just votes at the same height with different targets slashable, then the intersection of two \just-quorums at height~$h$ with different targets forces every shared signer to be slashable; assuming fewer than $n/3$ validators are slashable yields one non-slashable signer, whose single \just vote fixes $C_1 = C_2$.
Under this strengthened slashing definition, $f < n/3$ in the hypotheses of \Cref{lem:just-unique-height,lem:canonical-fresh-honest-vote} (and downstream lemmas that inherit it) may be replaced by the weaker ``unless $\geq n/3$ validators are slashable.''
\end{remark}

\section{Healing from asynchrony}\label{sec:healing}

\subsection{Model and view-merge}\label{rev:sec:healing-model}

\begin{definition}[Slots and voting time]\label{rev:def:slots}
Physical time is partitioned into an infinite sequence of discrete \emph{slots} indexed by $k \in \mathbb{N}$.
Slot~$k$ has a \emph{start time} $\start(k) \in \mathbb{R}_{\geq 0}$ with $\start(k+1) \geq \start(k) + 2\Delta$, and a \emph{voting time} $\vote(k) \coloneqq \start(k) + \Delta$ at which every honest validator~$i$ invokes \Cref{def:voting-rule} on its local store~$\Sigma_i$ to produce at most one slot-$k$ vote.
\end{definition}

\begin{definition}[Syncer]\label{rev:def:syncer}
A publicly known function $\syncer \colon \mathbb{N} \to \{1, \ldots, n\}$ assigns to each slot~$k$ a validator $\syncer(k)$.
During the portion of slot~$k$ preceding $\vote(k)$, $\syncer(k)$ runs the \emph{view-merge protocol} (\Cref{app:view-merge}).
\end{definition}

\begin{assumption}[View-merge guarantee]\label{rev:ass:view-merge}
If $\syncer(k)$ is honest, then there exists a store $\comm{\Sigma}^{\vote(k)}$ such that every honest validator~$i$ has $\Sigma_i = \comm{\Sigma}^{\vote(k)}$ at time $\vote(k)$.
\end{assumption}

\begin{definition}[Time-dependent dishonest count]\label{rev:def:f-t}
For each time~$t$, $f^t$ denotes the number of validators that are dishonest at time~$t$. We assume $f^t$ is monotonically non-decreasing in~$t$: once a validator is dishonest, it remains so.
\end{definition}

\begin{definition}[Online-honest majority]\label{rev:def:online-honest-majority}
The \emph{\OHM} holds at a given time iff more than half of the validators that are online at that time are honest.
\end{definition}

\begin{definition}[All-honest-awake at slot $k$]\label{rev:def:all-honest-awake}
The \emph{\AHA} condition holds at slot~$k$ iff every honest validator is online at $\vote(k)$.
\end{definition}

\begin{definition}[Confirmation delay]\label{rev:def:confirmation-delay}
Let $\delta_c$ denote the time taken to confirm a block proposed by an honest proposer on the available chain.
\end{definition}

\begin{definition}[Honest available-chain proposer at slot $k$]\label{rev:def:honest-available-chain-proposer}
The \emph{\HAP} condition holds at slot~$k$ iff some honest validator proposes a block on the available chain in $[\vote(k),\, \vote(k+1) - \delta_c)$.
\end{definition}

\begin{definition}[Honest available-chain relay at slot $k$]\label{rev:def:honest-available-chain-relay}
The \emph{\HAR} condition holds at slot~$k$ iff every vote cast by an honest validator at $\vote(k)$ is received by some honest validator that proposes a block on the available chain at some time in $[\vote(k),\, \vote(k+1) - \delta_c)$, before that proposal.
\end{definition}

For an honest validator~$v$ and a time~$t$, write $\Sigma_v^t$ for $v$'s local store at time~$t$, $\Omega_v^t$ for its auxiliary state at time~$t$, $R_v^t$ as shorthand for $\Sigma_v^t.R$, and $\getConfirmed_v^t$ as shorthand for $\getConfirmed(\Sigma_v^t,\, \Omega_v^t)$.

\subsection{Properties of \getConfirmed (interaction with the available chain)}\label{rev:sec:getConfirmed-properties}

\begin{property}[Target compatibility]\label{rev:prop:getConfirmed-compat}
Assume $\start(k) - \Delta \geq \GST$, the \OHM holds, and $R_v^{\vote(k)} \sim R_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$.
If there exists an honest validator $v_\ell$ such that $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R_{v}^t$ for every honest validator~$v$ and every time $t \in [\vote(k),\, \vote(k+1)]$, then for every honest validators~$v,v'$ and times $t',t \in [\vote(k),\, \vote(k+1)], \getConfirmed_v^{t} \sim \getConfirmed_{v'}^{t'}$.
\end{property}

\begin{property}[Target monotonicity under stable root]\label{rev:prop:target-monotonicity}
Assume $\start(k) - \Delta \geq \GST$, the \OHM holds, and $R_v^{\vote(k)} \sim R_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$.
If there exists an honest validator $v_\ell$ such that $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R_{v}^t$ for every honest validator~$v$ and every time $t \in [\vote(k),\, \vote(k+1)]$, then for every honest validator~$v$ and every pair of times $t, t' \in [\vote(k),\, \vote(k+1)]$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
\end{property}

\begin{lemma}[Two-slot target compatibility and monotonicity]\label{rev:lem:two-slot-target-compat-monotone}
  If, for $k' \in [k,k+1]$,
  \begin{enumerate}
    \item for every honest validators~$v,v'$ and times $t',t \in [\vote(k'),\, \vote(k'+1)], \getConfirmed_v^{t} \sim \getConfirmed_{v'}^{t'}$
    \item for every honest validator~$v$ and every pair of times $t, t' \in [\vote(k'),\, \vote(k'+1)]$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$
  \end{enumerate}
  then,
  \begin{enumerate}
    \item for every honest validators~$v,v'$ and times $t',t \in [\vote(k),\, \vote(k+2)], \getConfirmed_v^{t} \sim \getConfirmed_{v'}^{t'}$
    \item for every honest validator~$v$ and every pair of times $t, t' \in [\vote(k),\, \vote(k+2)]$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$
  \end{enumerate}  
\end{lemma}
\begin{proof}
Without loss of generality, assume $t \leq t'$ (since~$\sim$ is symmetric, it suffices to consider $t \leq t'$ for both conclusions). If $t, t' \in [\vote(k'),\, \vote(k'+1)]$ for some $k' \in \{k, k+1\}$, both conclusions follow directly from hypotheses~(1) and~(2) at that~$k'$. The remaining case is $t \in [\vote(k),\, \vote(k+1)]$ and $t' \in [\vote(k+1),\, \vote(k+2)]$, which we treat below.

\emph{Conclusion~2.}
Hypothesis~(2) at $k' = k$ applied to $t \leq \vote(k+1)$ gives $\getConfirmed_v^{\vote(k+1)} \succeq \getConfirmed_v^t$. Hypothesis~(2) at $k' = k+1$ applied to $\vote(k+1) \leq t'$ gives $\getConfirmed_v^{t'} \succeq \getConfirmed_v^{\vote(k+1)}$. Composing, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.

\emph{Conclusion~1.}
Hypothesis~(1) at $k' = k+1$ applied to the pair of times $t', t' \in [\vote(k+1),\, \vote(k+2)]$ and the validators $v, v'$ gives $\getConfirmed_v^{t'} \sim \getConfirmed_{v'}^{t'}$. Conclusion~2 above gives $\getConfirmed_v^t \preceq \getConfirmed_v^{t'}$. Both $\getConfirmed_v^t$ and $\getConfirmed_{v'}^{t'}$ are $\sim$-comparable to $\getConfirmed_v^{t'}$, hence $\sim$-comparable to each other (any two blocks both $\sim$-comparable to a common block are $\sim$-comparable, since one is an ancestor of the common block and the other is either an ancestor or a descendant of it).
\end{proof}

\begin{property}[Honest-vote inclusion under stable root]\label{rev:prop:honest-vote-inclusion}
Assume $\start(k) - \Delta \geq \GST$, the \OHM holds, the \HAR condition at slot~$k$ holds (\Cref{rev:def:honest-available-chain-relay}), and $R_v^{\vote(k)} \sim R_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$.
If there exists an honest validator $v_\ell$ such that
$\getConfirmed_{v_\ell}^{\vote(k)}\succeq R_{v}^t$ for every honest validator~$v$ and every time $t \in [\vote(k),\, \vote(k+1)]$, then,
for every honest validator~$v$, the chain ending at $\getConfirmed_v^{\vote(k+1)}$ contains every vote cast by an honest validator at $\vote(k)$
\end{property}


\begin{property}[Target dominance under stable root]\label{rev:prop:target-dominance}
Assume $\start(k) - \Delta \geq \GST$, the \OHM holds, the \HAP condition at slot~$k$ holds (\Cref{rev:def:honest-available-chain-proposer}), and $R_v^{\vote(k)} \sim R_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$.
If there exists an honest validator $v_\ell$ such that $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R_{v}^t$ for every honest validator~$v$ and every time $t \in [\vote(k),\, \vote(k+1)]$, then for every honest validator~$v$, $\getConfirmed_v^{\vote(k+1)} \succeq \getConfirmed_{v_\ell}^{\vote(k)}$.
\end{property}

\subsection{Protocol modifications}\label{rev:sec:healing-protocol}

\paragraph{Virtual-block cert events.}
For this section, the store considers candidate cert events not only from on-chain block processing but also from \emph{virtual extensions} of every sufficient head in $\Sigma.\T$: using the votes in $\Sigma.V$, the following routine hypothesizes, for each candidate head, a block that carries the applicable subset on top of it and offers the resulting cert-event descriptors to \updateForkChoiceRoot. The candidate heads are the canonical block $\getConfirmed(\Sigma, \Omega)$ together with every leaf of $\Sigma.\T$ at state-height~$\geq \Sigma.\key_R.h$.
The virtual blocks are never inserted into $\Sigma.\T$ and the hypothetical states are discarded; only $(\Sigma.R,\, \Sigma.\key_R)$ may change, via the standard cert-key guard.
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]
\Function{\updateRootFromVotes}{$\Sigma,\; \Omega$}
  \State $\mathcal{L} \gets \{\getConfirmed(\Sigma,\; \Omega)\} \cup \{B \in \Sigma.\T : B \text{ is a leaf of } \Sigma.\T \land \sigma[B].h \geq \Sigma.\key_R.h\}$
  \For{each $L \in \mathcal{L}$}
    \State $\sigma' \gets \Sigma.\sigma[L]$ \Comment{working copy, not persisted}
    \State $V' \gets \{v \in \Sigma.V : v.\height = \sigma'.h \land v.\target \preceq L \land v.\target.\slot \geq \sigma'.s_h\}$ \Comment{fresh on $\sigma'$}
    \State let $B_V$ be a virtual block with $\parent = L$, $\slot = \sigma'.s + 1$, and votes~$V'$
    \State $\sigma' \gets \processSlot(\sigma')$ \Comment{advances $\sigma'.s$; runs \processHeight}
    \State $\sigma' \gets \processBlock(\sigma',\; B_V)$
    \State $\sigma' \gets \processHeight(\sigma')$ \Comment{fires any cert event enabled by $V'$}
    \State $\Sigma \gets \updateForkChoiceRoot(\Sigma,\; (\sigma'.h_j,\; 1,\; \sigma'.J,\; \sigma'.s_n))$
    \State $\Sigma \gets \updateForkChoiceRoot(\Sigma,\; (\sigma'.h_n,\; 0,\; \sigma'.N,\; \sigma'.s_n))$
  \EndFor
  \State \Return $\Sigma$
\EndFunction
\end{algorithmic}
\end{boxedminipage}
\end{center}
\noindent In particular, whenever received-but-not-yet-on-chain votes would justify or notarize a block on any sufficient head with a cert key strictly higher than $\key_R$, $R$ advances accordingly.

\paragraph{View-merge processing.}
The view-merge protocol (\Cref{app:view-merge}) delivers to each honest validator the syncer's block tree $\comm{\T}$ and message set $\comm{V}$; no other store components are transferred. Upon receipt at $\vote(k)$, the validator folds the newly-delivered blocks through \onBlock{} and then invokes \updateRootFromVotes{} on the enlarged $\Sigma.V$. Formally:
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]
\Function{\onViewMerge}{$\Sigma,\; \Omega,\; \comm{\T},\; \comm{V}$}
  \For{each $B \in \comm{\T} \setminus \Sigma.\T$ processed in any $\preceq$-respecting order}
    \State $\Sigma \gets \onBlock(\Sigma,\; B)$
  \EndFor
  \State $\Sigma.V \gets \Sigma.V \cup \comm{V}$
  \State $\Sigma \gets \updateRootFromVotes(\Sigma,\; \Omega)$
  \State \Return $\Sigma$
\EndFunction
\end{algorithmic}
\end{boxedminipage}
\end{center}
\noindent Because the other store components --- $\sigma,\, F,\, R,\, \key_R$ --- are pure functions of $(\Sigma.\T,\, \Sigma.V)$ under \Cref{alg:store} together with \updateRootFromVotes{}, honest validators that have merged the same $(\comm{\T},\, \comm{V})$ by time $\vote(k)$ obtain matching $\sigma,\, F,\, R,\, \key_R$ locally.

To state the conditions under which the protocol \emph{heals} after a prolonged period of asynchrony, we enrich the system model with slot-based timing and a per-slot view-merge mechanism.
The view-merge mechanism is detailed in \Cref{app:view-merge}; in this section we abstract it via an assumption sufficient for the healing arguments.


\paragraph{Modified \constructVote.}
We assume that each store $\Sigma$ carries an additional field $\Sigma.V$, the set of votes the owning validator has cast so far.
The voting rule is correspondingly rewritten to drop its $V$ parameter and to delegate unlocked target selection to \getConfirmed{}:
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]
\Function{\constructVote}{$\Sigma,\; \Omega$}
  \State $T \gets \getConfirmed(\Sigma,\; \Omega)$
  \State $h_c \gets \sigma[T].h$; \;\; $h_j \gets \sigma[T].h_j$; \;\; $J \gets \sigma[T].J$
  \vspace{3pt}
  \State \textbf{-- Target selection --}
  \If{$\exists\, v \in \Sigma.V$ with $v.\finalizeHeight = h_c$ and $v.\finalizeTarget \neq \bot$}
    \State $T_c \gets v.\finalizeTarget$ \Comment{$i$ is locked at $h_c$}
  \Else
    \State $T_c \gets T$
  \EndIf
  \vspace{3pt}
  \State \textbf{-- Vote kind and finalize fields --}
  \If{$\not \exists v \in \Sigma.V$ s.t. $v.\height = h_c$} \Comment{first vote at $h_c$: cast \just (R1)}
    \If{every $v \in \Sigma.V$ with $v.\height = h_j$ has $v.\target = J$}
      \State $(f_h,\, f_t) \gets (h_j,\, J)$
    \Else
      \State $(f_h,\, f_t) \gets (\bot,\, \bot)$
    \EndIf
    \State \Return $(i,\; h_c,\; \currentSlot,\; T_c,\; \just,\; f_h,\; f_t)$
  \Else \Comment{already voted at $h_c$: cast \notar (R2)}
    \State \Return $(i,\; h_c,\; \currentSlot,\; T_c,\; \notar,\; \bot,\; \bot)$
  \EndIf
\EndFunction
\end{algorithmic}
\end{boxedminipage}
\end{center}

\subsection{Auxiliary lemmas}\label{rev:sec:healing-auxiliary}

\begin{lemma}[Per-slot notarization uniqueness]\label{rev:lem:notar-unique-per-slot}
Assume $f^{\vote(s)} < n/3$ and all honest validators follow \Cref{def:voting-rule}. If blocks $X$ and $Y$ are each notarized at height~$h$ by a quorum whose contributing votes were all cast at a single slot~$s$, then $X \sim Y$.
\end{lemma}

\begin{proof}
Suppose for contradiction $X \not\sim Y$: $X$ and $Y$ lie on conflicting chains $L_X, L_Y$, and no block at state-height~$h$ lies on both chains (state-height is monotone along~$\preceq$, and $L_X, L_Y$ diverge strictly before any block at state-height~$h$).

Let $Q_X, Q_Y$ denote the slot-$s$ contributing quorums, with $|Q_X|, |Q_Y| \geq 2n/3$, and let $h_X, h_Y$ be their honest cardinalities.
By \Cref{def:voting-rule}, an honest validator casts at most one vote per slot; the target of that vote lies on at most one of $L_X, L_Y$ at state-height~$h$. Hence no honest validator is in both $Q_X$ and $Q_Y$, so
\[
  h_X + h_Y \;\leq\; n - f.
\]

At most $f$ adversary validators exist, each possibly equivocating to contribute once to each chain's slot-$s$ quorum; therefore
\[
  |Q_X| \leq h_X + f \quad\text{and}\quad |Q_Y| \leq h_Y + f.
\]
Combined with $|Q_X|, |Q_Y| \geq 2n/3$,
\[
  h_X \geq 2n/3 - f, \quad h_Y \geq 2n/3 - f, \quad\text{so}\quad h_X + h_Y \geq 4n/3 - 2f.
\]

Comparing the two bounds on $h_X + h_Y$ gives $4n/3 - 2f \leq n - f$, i.e.\ $f \geq n/3$, contradicting $f < n/3$.
\end{proof}

\begin{definition}[Healed state at slot $k$ with height $\comm{h}$]\label{rev:def:healed-state}
We say the system is in a \emph{healed state at slot $k$ with height $\comm{h}$} when:
\begin{enumerate}
  \item\label{rev:def:healed-state:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:def:healed-state:R} for any honest validator $v$, $R_v^{\vote(k)}.h \geq \comm{h}$;
  \item\label{rev:def:healed-state:gt-compat} for any two honest validators $v, v'$, $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$;
  % \item\label{rev:def:healed-state:no-just} no honest validator has cast any vote at height $> \comm{h}+1$ at any time strictly before $\vote(k)$;
  \item\label{rev:def:healed-state:target-cap} there exists an honest validator $v_\ell$ such that any honest vote sent for any height $\geq \comm{h}$ at a time strictly before $\vote(k)$ has target $\preceq \getConfirmed_{v_\ell}^{\vote(k)}$;
  % \item\label{rev:def:healed-state:just-bound} for every checkpoint $C$ for height $\comm{h}$ that can be justified by time $\vote(k)$, there exists an honest validator $v_\ell$ such that $C \preceq \getConfirmed_{v_\ell}^{\vote(k)}$;
  % \item\label{rev:def:healed-state:notar-bound} for any honest validator $v$, if $R_v^{\vote(k)}$ is notarized for slot $k_n$, then:
  % \begin{enumerate}
  %   \item\label{rev:def:healed-state:notar-bound:past} $k_n < k$;
  %   \item\label{rev:def:healed-state:notar-bound:coverage} for every $k' \in [k_n, k)$ and every checkpoint $C$ for height $\comm{h}$ that is notarized for slot $k'$ at any time no later than $\vote(k)$, there exists an honest validator $v_\ell$ such that $C \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
  % \end{enumerate}
\end{enumerate}
\end{definition}

\begin{lemma}[Intra-slot target compatibility and monotonicity]\label{rev:lem:intra-slot-target-compat-monotone}
If
\begin{enumerate}
  \item\label{rev:istcm:f} honest validators follow \Cref{def:voting-rule};
  \item\label{rev:istcm:ohm} the \OHM holds at slot~$k$;
  \item\label{rev:istcm:hs} the system is in a healed state at slot $k$ with height $\comm{h}$ (\Cref{rev:def:healed-state}),
\end{enumerate}
then
  \begin{enumerate}
    \item for every honest validators~$v,v'$ and times $t, t' \in [\vote(k),\, \vote(k+1)]$, $\getConfirmed_v^{t} \sim \getConfirmed_{v'}^{t'}$;
    \item for every honest validator~$v$ and every pair of times $t, t' \in [\vote(k),\, \vote(k+1)]$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
  \end{enumerate}
\end{lemma}

\begin{proof}
We discharge the preamble (R-compatibility at $\vote(k)$) and premise ($\exists$ honest~$v_\ell$ with $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R_v^t$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$) of \Cref{rev:prop:getConfirmed-compat,rev:prop:target-monotonicity}; the lemma's conclusions follow from those properties' conclusions.

\emph{Preamble and premise at $t = \vote(k)$.}
By condition~\ref{rev:def:healed-state:gt-compat}, the honest $\getConfirmed_v^{\vote(k)}$ are pairwise compatible; let~$D$ be the chain extending them, and pick $v_\ell$ honest with $\getConfirmed_{v_\ell}^{\vote(k)} = \max_\preceq$ on~$D$. By the store invariant ($R \preceq \getConfirmed$, \Cref{alg:store}), each $R_v^{\vote(k)} \preceq \getConfirmed_v^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k)}$ on~$D$, establishing pairwise compatibility of the $R_v^{\vote(k)}$ and the premise at $t = \vote(k)$.

\emph{Premise at $t \in (\vote(k),\, \vote(k+1)]$.}
$R_v$ updates only via cert events with key strictly dominating $\Sigma_v.\key_R$; condition~\ref{rev:def:healed-state:R} and \Cref{lem:Rs-key-monotone} give $\Sigma_v.\key_R.h \geq \comm{h}$, so any updating cert has height $h^* \geq \comm{h}$. Such a cert is witnessed by a $\geq 2n/3$ vote quorum, of which $\geq 2n/3 - f \geq 1$ are honest (under \OHM, $f < 2n/3$). The cert target is $\preceq$ some honest contribution: bit-$1$ (justification) shares its target across the quorum directly; bit-$0$ (prefix-notarization) has cert target $\preceq \ntargets[i]$ for every contributor~$i$, and for an honest~$i$ this is the $\preceq$-max of $i$'s fresh-vote targets at height~$h^*$, each cast at some $\tau \leq \vote(k)$. So it suffices to show every honest vote~$u$ at height~$\geq \comm{h}$ cast at some $\tau \leq \vote(k)$ has $u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
\begin{itemize}
  \item \emph{$\tau < \vote(k)$.} Condition~\ref{rev:def:healed-state:target-cap} gives the bound, with the condition's witness lifted to~$v_\ell$ by maximality on~$D$.
  \item \emph{$\tau = \vote(k)$.} $u$ is the slot-$k$ vote of some honest~$i$ with target~$T_c$ from \Cref{def:voting-rule}: unlocked, $T_c = \getConfirmed_i^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k)}$ by maximality; locked, $T_c = L$ was justified at height~$u.\height$ on some chain at some $\tau'' < \vote(k)$ via a $\geq 2n/3$ \just-quorum targeting~$L$ (\Cref{lem:no-finalize-without-just}), of which $\geq 2n/3 - f \geq 1$ honest contributors each cast a before-$\vote(k)$ \just vote with target~$L$ at height~$u.\height \geq \comm{h}$, so the previous bullet gives $L \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
\end{itemize}
\end{proof}

\subsection{Healing persistence}\label{rev:sec:healing-persistence}


\begin{lemma}[Bound on virtual-block advance]\label{rev:lem:virtual-block-advance}
Take an honest validator with local store $\Sigma$. Let $h \coloneqq \Sigma.\key_R.h$ before \updateRootFromVotes is invoked, and let $h' \coloneqq \Sigma.\key_R.h$ immediately after. If $h' > h$, then $h' = h+1$, and every block $B \in \Sigma.\T$ satisfies $\sigma[B].h \leq h+1$.
\end{lemma}

\begin{proof}
\Cref{rev:cor:processed-height-bound} at the pre-call moment gives $\sigma[B].h \leq h+1$ for every $B \in \Sigma.\T$, in particular $\sigma[L].h \leq h+1$ where $L \coloneqq \getConfirmed(\Sigma, \Omega)$.

\updateRootFromVotes performs two \processHeight{} invocations: one inside \processSlot{} (operating on $\sigma' = \sigma[L]$ with $\sigma'.V = \sigma[L].V$, before $B_V$'s votes are folded in), and one at the end (operating on $\sigma'$ enriched with $B_V$'s votes). Each increments $\sigma'.h$ by at most~$1$ and fires at most one cert event with key-height equal to the pre-advance value of $\sigma'.h$.

\emph{First invocation.}
If $\sigma[L].V$ contained a $\geq 2n/3$ quorum at any height $>h$, that quorum would already have fired a cert event during $L$'s on-chain processing, advancing $\Sigma.\key_R$ above $h$ — contradicting $\Sigma.\key_R.h = h$. Hence the first invocation fires (if at all) at $\sigma'.h \leq h$, with cert key-height $\leq h$, leaving $\sigma'.h \leq h+1$.

\emph{Final invocation.}
$\sigma'.h \leq h+1$ entering this invocation; its cert key-height (if any) is $\leq h+1$.

The cert keys passed to \updateForkChoiceRoot{} are derived from $\sigma'.h_j$ and $\sigma'.h_n$, themselves bounded by either pre-call values ($\sigma[L].h_j, \sigma[L].h_n \leq \sigma[L].h \leq h+1$) or new firings ($\leq h+1$ above). So $\Sigma.\key_R.h \leq h+1$ after the call; with $h' > h$, $h' = h+1$.

Finally, \updateRootFromVotes does not insert blocks into $\Sigma.\T$ (the virtual block $B_V$ is discarded), so $\sigma[B].h \leq h+1$ persists post-call.
\end{proof}

\begin{lemma}[Healing: persistence over one slot]\label{rev:lem:healing-persistence}
If
\begin{enumerate}
  \item\label{rev:hp:f} honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hp:ohm} the \OHM holds at slot~$k$;
  \item\label{rev:hp:hs} the system is in a healed state at slot $k$ with height $\comm{h}$ (\Cref{rev:def:healed-state}),
\end{enumerate}
then:
\begin{enumerate}
  \item the system is in a healed state at slot $k+1$ with height $\comm{h}$;
  \item for every honest validators~$v,v'$ and times $t, t' \in [\vote(k),\, \vote(k+1)]$, $\getConfirmed_v^{t} \sim \getConfirmed_{v'}^{t'}$;
  \item for every honest validator~$v$ and every pair of times $t, t' \in [\vote(k),\, \vote(k+1)]$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
\end{enumerate}
\end{lemma}



\begin{proof}
Write $h \coloneqq \comm{h}$. By condition~\ref{rev:def:healed-state:gt-compat} the honest $\getConfirmed_v^{\vote(k)}$ are pairwise compatible; let~$D$ be the chain extending them. Pick $v^*_\ell$ honest with $\getConfirmed_{v^*_\ell}^{\vote(k)} = \max_\preceq \{\getConfirmed_i^{\vote(k)} : i \text{ honest}\}$ on~$D$.

\emph{Conclusions~(2) and~(3).}
\Cref{rev:lem:intra-slot-target-compat-monotone} at slot~$k$ (hypotheses~\ref{rev:hp:f}, \ref{rev:hp:ohm}, \ref{rev:hp:hs} discharge~\ref{rev:istcm:f}, \ref{rev:istcm:ohm}, \ref{rev:istcm:hs} verbatim) yields conclusions~(2) and~(3) over $[\vote(k),\, \vote(k+1)]$.

\emph{Conclusion~(1): healed-state conditions at slot~$k+1$ with height~$h$.}
Pick $v_\ell$ honest with $\getConfirmed_{v_\ell}^{\vote(k+1)} = \max_\preceq \{\getConfirmed_i^{\vote(k+1)} : i \text{ honest}\}$ on the chain extending the honest $\getConfirmed^{\vote(k+1)}$ (well-defined by conclusion~(2)).
\begin{itemize}
  \item \emph{\ref{rev:def:healed-state:sync}.}
  $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$ by slot spacing and condition~\ref{rev:def:healed-state:sync}.

  \item \emph{\ref{rev:def:healed-state:R}.}
  By \Cref{lem:Rs-key-monotone}, $\Sigma_v.\key_R.h$ is non-decreasing in time, so $R_v^{\vote(k+1)}.h \geq R_v^{\vote(k)}.h \geq h$ by condition~\ref{rev:def:healed-state:R}.

  \item \emph{\ref{rev:def:healed-state:gt-compat}.}
  Conclusion~(2) at $t = t' = \vote(k+1)$.

  \item \emph{\ref{rev:def:healed-state:target-cap}.}
  Take any honest vote~$u$ sent at any time $\tau < \vote(k+1)$ with $u.\height \geq h$; we show $u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k+1)}$. Either $\tau < \vote(k)$ or $\tau = \vote(k)$ (the only honest voting times in $[0,\, \vote(k+1))$).

  \emph{Case $\tau < \vote(k)$.}
  Condition~\ref{rev:def:healed-state:target-cap} gives $u.\target \preceq \getConfirmed_{v^*_\ell}^{\vote(k)}$. Conclusion~(3) at $v^*_\ell$ lifts to $\getConfirmed_{v^*_\ell}^{\vote(k+1)} \succeq \getConfirmed_{v^*_\ell}^{\vote(k)}$, and maximality of~$v_\ell$ at $\vote(k+1)$ gives $\getConfirmed_{v_\ell}^{\vote(k+1)} \succeq \getConfirmed_{v^*_\ell}^{\vote(k+1)}$. Composing, $u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k+1)}$.

  \emph{Case $\tau = \vote(k)$.}
  $u$ is the slot-$k$ vote of some honest~$i$ with target~$T_c$ produced by \Cref{def:voting-rule}: either $T_c = \getConfirmed_i^{\vote(k)}$ (unlocked) or $T_c = L$ (locked), where $L$ is the lock target from a prior $i$-vote with $\finalizeTarget = L$ at $\finalizeHeight = u.\height$.

  In the unlocked sub-case, maximality of~$v^*_\ell$ on~$D$ gives $\getConfirmed_i^{\vote(k)} \preceq \getConfirmed_{v^*_\ell}^{\vote(k)}$, and the monotonicity~+ maximality argument from Case~$\tau < \vote(k)$ closes $T_c \preceq \getConfirmed_{v_\ell}^{\vote(k+1)}$.

  In the locked sub-case, by \Cref{lem:no-finalize-without-just} the lock target~$L$ was justified at height~$u.\height$ on some chain at some time $\tau' < \vote(k)$, via a $\geq 2n/3$ \just-quorum targeting~$L$; of these contributors, $\geq 2n/3 - f \geq 1$ are honest (under \OHM, $f < 2n/3$). Pick any honest contributor~$j$, whose \just vote~$u'$ has $u'.\target = L$, $u'.\height = u.\height \geq h$, and was cast at $\tau' < \vote(k)$. Condition~\ref{rev:def:healed-state:target-cap} applied to~$u'$ gives $L \preceq \getConfirmed_{v^*_\ell}^{\vote(k)}$, and the same monotonicity~+ maximality argument closes $T_c = L \preceq \getConfirmed_{v_\ell}^{\vote(k+1)}$.
\end{itemize}
\end{proof}

\begin{lemma}[Healing: persistence over unbounded times]\label{rev:lem:healing-unbounded}
If
\begin{enumerate}
  \item\label{rev:hu:f} honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hu:ohm} the \OHM holds at every slot $k' \geq k$;
  \item\label{rev:hu:hs} the system is in a healed state at slot $k$ with height $\comm{h}$ (\Cref{rev:def:healed-state}),
\end{enumerate}
then,
\begin{enumerate}
  \item for every honest validators~$v,v'$ and times $t, t' \geq \vote(k)$, $\getConfirmed_v^{t} \sim \getConfirmed_{v'}^{t'}$;
  \item for every honest validator~$v$ and every pair of times $t, t' \geq \vote(k)$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
\end{enumerate}
\end{lemma}

\begin{proof}
We iterate \Cref{rev:lem:healing-persistence}: starting from $K_0 = k$, set $K_{n+1} = K_n + 1$. By induction, the hypotheses of \Cref{rev:lem:healing-persistence} hold at every slot~$K_n$.

\emph{Hypothesis propagation.}
For $n = 0$, the hypotheses hold by assumption with $K_0 = k$. Suppose they hold at slot~$K_n$ with height~$\comm{h}_n$. \Cref{rev:lem:healing-persistence} at slot~$K_n$ yields some $\comm{h}_{n+1} \in \{\comm{h}_n,\, \comm{h}_n + 1\}$ such that conclusion~(1) gives that the system is in a healed state at slot~$K_{n+1} = K_n + 1$ with height~$\comm{h}_{n+1}$, discharging hypothesis~\ref{rev:hp:hs} of \Cref{rev:lem:healing-persistence} at slot~$K_{n+1}$. The remaining hypotheses inherit from the present lemma's hypotheses~\ref{rev:hu:f} and~\ref{rev:hu:ohm}.

\emph{Conclusions over each interval.}
For each $n \geq 0$, conclusions~(2) and~(3) of \Cref{rev:lem:healing-persistence} applied at slot~$K_n$ yield, for $t, t' \in [\vote(K_n),\, \vote(K_{n+1})]$, $\getConfirmed_v^t \sim \getConfirmed_{v'}^{t'}$ and (when $t \leq t'$) $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.

\emph{Stitching across intervals.}
We show by induction on $n$ that the lemma's two conclusions hold for all $t, t' \in [\vote(k),\, \vote(K_n)]$.

\textbf{Base ($n = 1$).}
Direct from the previous paragraph applied at slot $K_0 = k$.

\textbf{Inductive step ($n \to n+1$).}
Assume the conclusions hold over $[\vote(k),\, \vote(K_n)]$. The previous paragraph gives them over $[\vote(K_n),\, \vote(K_{n+1})]$. To extend to $[\vote(k),\, \vote(K_{n+1})]$, consider $t \in [\vote(k),\, \vote(K_n)]$ and $t' \in [\vote(K_n),\, \vote(K_{n+1})]$ (the only non-trivial mixed case, with $t \leq t'$):
\begin{itemize}
  \item \emph{Monotonicity}: by the inductive hypothesis, $\getConfirmed_v^{\vote(K_n)} \succeq \getConfirmed_v^t$; by the previous paragraph, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^{\vote(K_n)}$. Composing, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
  \item \emph{Compatibility}: by the previous paragraph, $\getConfirmed_v^{t'} \sim \getConfirmed_{v'}^{t'}$; by mono just established, $\getConfirmed_v^t \preceq \getConfirmed_v^{t'}$. Hence $\getConfirmed_v^t$ and $\getConfirmed_{v'}^{t'}$ are both $\sim$-comparable to $\getConfirmed_v^{t'}$ — one as an ancestor, the other as ancestor or descendant — so they are $\sim$-comparable to each other.
\end{itemize}

\emph{Conclusion.}
Since the $K_n$'s are unbounded, every $t, t' \geq \vote(k)$ lie in $[\vote(k),\, \vote(K_n)]$ for some sufficiently large $n$. The lemma's conclusions follow.
\end{proof}

\subsection{Healing advances by $N$ slots}\label{rev:sec:healing-advances}


\begin{lemma}[Two-slot healing from a justified common root]\label{rev:lem:healing-from-justified}
Assume:
\begin{enumerate}
  \item\label{rev:hfj:f} $f^{\vote(k+2)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hfj:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:hfj:ohm} the \OHM holds at slots~$k$ and $k+1$;
  \item\label{rev:hfj:aha} the \AHA condition holds at slots~$k$ and $k+1$;
  \item\label{rev:hfj:har} the \HAR condition holds at slot~$k$;
  \item\label{rev:hfj:r} for every honest validator~$v$, $R_v^{\vote(k)}$ is the same justified block; denote it by $\comm{R}^{\vote(k)}$;
  % \item\label{rev:hfj:gt} $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getConfirmed_v^{\vote(k)}].h = \comm{R}^{\vote(k)}.h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item\label{rev:hfj:nh} no honest validator has cast any vote at height $> \comm{R}^{\vote(k)}.h$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then the system is in a healed state at slot $k+2$ with height $h+1$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Let $h \coloneqq \comm{R}^{\vote(k)}.h$. By \Cref{rev:lem:confirmed-h-range} applied per-honest, $\sigma[\getConfirmed_v^{\vote(k)}].h \in \{R_v^{\vote(k)}.h,\, R_v^{\vote(k)}.h + 1\} = \{h, h+1\}$ for every honest~$v$, splitting into two cases.

\begin{description}
\item[{Case 1: $\sigma[\getConfirmed_v^{\vote(k)}].h = h$ for every honest~$v$.}]\hfill

\emph{Step 1: $R_v^t = \comm{R}^{\vote(k)}$ throughout $[\vote(k),\, \vote(k+1)]$.}
By hypothesis~\ref{rev:hfj:r}, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is justified at height~$h$ in every honest's view, so $\Sigma_v.\key_R \geq (h, 1, 0, -\comm{R}^{\vote(k)}.\slot, \hash(\comm{R}^{\vote(k)}))$ at $\vote(k)$. Hypothesis~\ref{rev:hfj:nh} excludes before-$\vote(k)$ honest votes at height~$>h$; the case assumption forces every slot-$k$ honest vote at canonical state-height~$h$ exactly; no voting event lies in $(\vote(k),\, \vote(k+1))$. Hence honest contributions at height~$>h$ in $[\vote(k),\, \vote(k+1)]$ are zero, dishonest alone are $\leq f < 2n/3$ (hypothesis~\ref{rev:hfj:ohm}), so no cert at height~$>h$ fires. \Cref{lem:Rs-key-monotone} keeps $\Sigma_v.\key_R.h \leq h$, and the bit-$1$ cached key dominates any bit-$0$ candidate at the same height. \Cref{lem:just-unique-height} (under $f < n/3$ from hypothesis~\ref{rev:hfj:f}) makes $\comm{R}^{\vote(k)}$ the unique justified block at height~$h$. Hence $R_v^t = \comm{R}^{\vote(k)}$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$.

\emph{Step 2: pairwise compatibility, monotonicity, and slot-$k$ vote inclusion at $\vote(k+1)$.}
The preamble (R-compatibility at $\vote(k)$) of \Cref{rev:prop:getConfirmed-compat,rev:prop:target-monotonicity,rev:prop:honest-vote-inclusion} is trivial since $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ uniformly. Pick any honest~$v_\ell$; by Step~1, $\getConfirmed_{v_\ell}^{\vote(k)} \succeq \comm{R}^{\vote(k)} = R_v^t$ for every honest~$v$ and $t \in [\vote(k),\, \vote(k+1)]$, discharging the premise. Then:
\begin{itemize}
  \item \Cref{rev:prop:getConfirmed-compat} gives pairwise compatibility of the honest $\getConfirmed_v^{\vote(k+1)}$; let~$D'$ be the chain extending them.
  \item \Cref{rev:prop:target-monotonicity} gives $\getConfirmed_v^{\vote(k+1)} \succeq \getConfirmed_v^{\vote(k)}$ for every honest~$v$.
  \item \Cref{rev:prop:honest-vote-inclusion} (under hypothesis~\ref{rev:hfj:har}) gives that the chain at $\getConfirmed_v^{\vote(k+1)}$ contains every slot-$k$ honest vote; in particular, every slot-$k$ honest vote target $T_i = \getConfirmed_i^{\vote(k)}$ lies on~$D'$.
\end{itemize}

\emph{Step 3: canonical state-height reaches~$h+1$ at $\vote(k+1)$.}
By Step~2, the chain at $\getConfirmed_v^{\vote(k+1)}$ contains the $\geq 2n/3$ slot-$k$ honest \just votes, each with target $T_i \in D'$ at state-height~$h$. After processing on~$D'$, each honest contributor~$i$ has $\sigma.\ntargets[i] = T_i$ and $\sigma.\nslots[i] = k$ at state-height~$h$. The prefix-notar $\arg\max_\preceq$ in \processHeight{} fires a bit-$0$ cert at $(h, 0, k, -B^\star.\slot, \hash(B^\star))$ for some $B^\star \in D'$ at state-height~$h$, advancing canonical state-height to~$h+1$ (the cert key is dominated by the cached bit-$1$, so $R_v$ doesn't update). Combining with $\sigma[\getConfirmed_v^{\vote(k+1)}].h \leq R_v^{\vote(k+1)}.h + 1 = h+1$ (\Cref{rev:cor:processed-height-bound}) gives $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h+1$ for every honest~$v$.

\emph{Step 4: slot-$(k+1)$ honest votes form a same-slot cert at~$h+1$ on~$D'$.}
By the voting rule, each honest~$v$ casts a slot-$(k+1)$ vote at canonical state-height~$h_c = h+1$. The (R1) branch fires (no prior honest vote at height~$h+1$: hypothesis~\ref{rev:hfj:nh} excludes before-$\vote(k)$, Step~3 places slot-$k$ at height~$h$ exactly), and the lock check fails (hypothesis~\ref{rev:hfj:nh} also excludes before-$\vote(k)$ honest finalize-bearing at height~$h+1$; slot-$k$ has $\finalizeHeight \leq h - 1 < h+1$). So the slot-$(k+1)$ vote target is $T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)} \in D'$ at state-height~$h+1$.

By synchrony (hypothesis~\ref{rev:hfj:sync} + slot spacing $\geq 2\Delta$), every slot-$(k+1)$ honest vote is delivered to every honest by $\vote(k+1) + \Delta \leq \vote(k+2) - \Delta$. By hypothesis~\ref{rev:hfj:aha} (\AHA at slot~$k+1$), all $\geq n - f > 2n/3$ honest validators cast such a vote. In each honest's $\Sigma.V$ at $\vote(k+2)$, the $\geq 2n/3$ slot-$(k+1)$ honest votes target blocks on~$D'$ at state-height~$h+1$.

\emph{Step 5: $R_v^{\vote(k+2)}$ is a uniform bit-$0$ notarization at height~$h+1$ with notarization slot~$k+1$.}
\updateRootFromVotes{} on each honest aggregates the slot-$(k+1)$ honest votes (uniform across honest by synchrony) into a virtual block extending its canonical (on~$D'$). The prefix-notar $\arg\max_\preceq$ over the slot-$(k+1)$ same-slot quorum at height~$h+1$ on~$D'$ selects some $B^{\star\star} \in D'$ at state-height~$h+1$, with $\geq 2n/3$ honest contributors having $B^{\star\star} \preceq T_v^{(k+1)}$. The selection depends only on the honest contributions (which dominate the $\geq 2n/3$ count alone) and $D'$'s linear structure, so $B^{\star\star}$ is uniform across honest validators. The cert key $(h+1, 0, k+1, -B^{\star\star}.\slot, \hash(B^{\star\star}))$ strictly dominates the cached $(h, 1, 0, \ldots)$ on the leading coordinate, so $R_v^{\vote(k+2)} = B^{\star\star}$ for every honest~$v$, with cert-key height~$h+1$ and notarization slot~$k+1 = (k+2) - 1$.

\emph{Step 6: healed-state conditions at slot~$k+2$ with height~$h+1$.}
We verify each condition of \Cref{rev:def:healed-state}.
\begin{itemize}
  \item \emph{\ref{rev:def:healed-state:sync}.}
  $\start(k+2) - \Delta \geq \start(k) + 4\Delta - \Delta = \start(k) + 3\Delta \geq \GST$ by hypothesis~\ref{rev:hfj:sync} and slot spacing.

  \item \emph{\ref{rev:def:healed-state:R}.}
  Step~5 gives $R_v^{\vote(k+2)}.h = h+1$.

  \item \emph{\ref{rev:def:healed-state:gt-compat}.}
  Apply \Cref{rev:prop:getConfirmed-compat} at slot~$k+1$. Preamble: $R_v^{\vote(k+1)} = \comm{R}^{\vote(k)}$ uniformly (Step~1), trivially pairwise compatible. Premise: pick $v_\ell$ honest with $\getConfirmed_{v_\ell}^{\vote(k+1)}$ the $\preceq$-max on~$D'$. For $t \in [\vote(k+1),\, \vote(k+2)]$, $R_v^t \in \{\comm{R}^{\vote(k)},\, B^{\star\star}\}$; $\getConfirmed_{v_\ell}^{\vote(k+1)} \succeq \comm{R}^{\vote(k)}$ trivially, and $B^{\star\star} \preceq T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)} \preceq \getConfirmed_{v_\ell}^{\vote(k+1)}$ by Step~5 and maximality. The conclusion gives pairwise compatibility of the honest $\getConfirmed_v^{\vote(k+2)}$.

  \item \emph{\ref{rev:def:healed-state:target-cap}.}
  Before-$\vote(k)$ honest votes at height~$\geq h+1$: ruled out by hypothesis~\ref{rev:hfj:nh}. Slot-$k$ honest votes: at height~$h$, not~$\geq h+1$. Slot-$(k+1)$ honest votes: at height~$h+1$ with target $T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)}$. \Cref{rev:prop:target-monotonicity} at slot~$k+1$ (premise discharged as in the previous bullet) gives $\getConfirmed_v^{\vote(k+2)} \succeq \getConfirmed_v^{\vote(k+1)} = T_v^{(k+1)}$; by the previous bullet's pairwise compatibility, picking $v_\ell$ honest with $\getConfirmed_{v_\ell}^{\vote(k+2)}$ the $\preceq$-max gives $T_v^{(k+1)} \preceq \getConfirmed_v^{\vote(k+2)} \preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$.
\end{itemize}
All four conditions verified, the system is in a healed state at slot~$k+2$ with height~$h+1$.


\item[{Case 2: $\sigma[\getConfirmed_v^{\vote(k)}].h = h+1$ for some honest~$v$.}]\hfill

\emph{Step 1: cert-target compatibility throughout $[\vote(k),\, \vote(k+1)]$.}
By hypothesis~\ref{rev:hfj:r}, $\Sigma_v.\key_R \geq (h, 1, 0, -\comm{R}^{\vote(k)}.\slot, \hash(\comm{R}^{\vote(k)}))$ at $\vote(k)$. Hypothesis~\ref{rev:hfj:nh} excludes before-$\vote(k)$ honest votes at height~$>h$; \Cref{rev:cor:processed-height-bound} bounds slot-$k$ honest votes at heights~$\leq h+1$; no voting event lies in $(\vote(k),\, \vote(k+1))$. Honest contributions at height~$>h+1$ are zero, dishonest alone are $\leq f < 2n/3$ (hypothesis~\ref{rev:hfj:ohm}), so no cert at height~$>h+1$ fires; \Cref{lem:Rs-key-monotone} keeps $\Sigma_v.\key_R.h \leq h+1$.

We classify cert events whose key may dominate the cached $(h, 1, 0, \ldots)$ in $v$'s view in the interval. At height~$h$, bit-$1$ uniquely targets $\comm{R}^{\vote(k)}$ (\Cref{lem:just-unique-height} under $f<n/3$, hypothesis~\ref{rev:hfj:f}); bit-$0$ at height~$h$ is dominated. At height~$h+1$, bit-$1$ uniquely targets some $J^*$ (\Cref{lem:just-unique-height}); bit-$0$ slot-$k$ same-slot certs at $h+1$ are pairwise compatible (\Cref{rev:lem:notar-unique-per-slot}). For any cert at height~$h+1$, the witnessing quorum has $\geq 2n/3 - f \geq 1$ honest contributors; each honest contribution at slot~$k$ at height~$h+1$ is a slot-$k$ vote by some honest~$i$ with $\sigma[\getConfirmed_i^{\vote(k)}].h = h+1$, and the (R1) clause of \Cref{def:voting-rule} (unlocked, since hypothesis~\ref{rev:hfj:nh} excludes any before-$\vote(k)$ honest vote at height~$>h$, in particular finalize-bearing ones at $h+1$) gives the contributor's vote target equal to $\getConfirmed_i^{\vote(k)}$. So every cert target $T^{**}$ at height~$h+1$ satisfies $T^{**} \preceq \getConfirmed_i^{\vote(k)}$ for some honest~$i$ at state-height~$h+1$, hence lies on the chain extending $\comm{R}^{\vote(k)}$ through $\getConfirmed_i^{\vote(k)}$ (since $\getConfirmed_i^{\vote(k)} \succeq \comm{R}^{\vote(k)}$ by the store invariant). Combined with the pairwise-compatibility statements above, $\comm{R}^{\vote(k)}$, $J^*$ (when applicable), every slot-$k$ same-slot cert target at $h+1$, and the $\getConfirmed_i^{\vote(k)}$ at state-height~$h+1$ that bound them all lie on a common chain~$D$ extending $\comm{R}^{\vote(k)}$.

\emph{Step 2: pairwise compatibility, monotonicity, and slot-$k$ vote inclusion at $\vote(k+1)$.}
The preamble of \Cref{rev:prop:getConfirmed-compat,rev:prop:target-monotonicity,rev:prop:honest-vote-inclusion} (R-compatibility at $\vote(k)$) is trivial since $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ uniformly. Pick $v_\ell$ honest with $\sigma[\getConfirmed_{v_\ell}^{\vote(k)}].h = h+1$ (case assumption guarantees one) and $\getConfirmed_{v_\ell}^{\vote(k)}$ the $\preceq$-max on~$D$ among the honest \getConfirmed{}s at $\vote(k)$ at state-height~$h+1$ on~$D$. By Step~1, every cert target~$T^{**}$ at height~$\leq h+1$ that may update $R_v$ in the interval satisfies $T^{**} \preceq \getConfirmed_i^{\vote(k)}$ for some honest~$i$ on~$D$, and $\getConfirmed_i^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k)}$ by maximality; together with $\comm{R}^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k)}$ (store invariant), every $R_v^t$ for $t \in [\vote(k),\, \vote(k+1)]$ satisfies $R_v^t \preceq \getConfirmed_{v_\ell}^{\vote(k)}$, discharging the premise. Then:
\begin{itemize}
  \item \Cref{rev:prop:getConfirmed-compat} gives pairwise compatibility of the honest $\getConfirmed_v^{\vote(k+1)}$; let~$D'$ be the chain extending them ($D \subseteq D'$ since $\getConfirmed_{v_\ell}^{\vote(k)} \in D$ and $\getConfirmed_{v_\ell}^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k+1)} \in D'$ by the next bullet).
  \item \Cref{rev:prop:target-monotonicity} gives $\getConfirmed_v^{\vote(k+1)} \succeq \getConfirmed_v^{\vote(k)}$ for every honest~$v$.
  \item \Cref{rev:prop:honest-vote-inclusion} (under hypothesis~\ref{rev:hfj:har}) gives that the chain at $\getConfirmed_v^{\vote(k+1)}$ contains every slot-$k$ honest vote; in particular, every slot-$k$ honest vote target lies on~$D'$.
\end{itemize}

\emph{Step 3: state-height at $\vote(k+1)$ lies in $\{h+1, h+2\}$.}
Pick any honest~$v_h$ with $\sigma[\getConfirmed_{v_h}^{\vote(k)}].h = h+1$ (case assumption guarantees one). Its canonical at $\vote(k)$ lies on a chain~$L_{v_h}$ where the cert-at-$h$ targeting $\comm{R}^{\vote(k)}$ has fired before-$\vote(k)$ (state-height~$h+1$ requires it); the before-$\vote(k)$ honest \just votes that established the $\geq 2n/3$ \just-quorum at $\comm{R}^{\vote(k)}$ on~$L_{v_h}$ are in blocks on~$L_{v_h}$. By Step~2's monotonicity, $\getConfirmed_{v_h}^{\vote(k+1)} \succeq \getConfirmed_{v_h}^{\vote(k)}$, so the chain at $\getConfirmed_{v_h}^{\vote(k+1)}$ extends $L_{v_h}$ at least up to $\getConfirmed_{v_h}^{\vote(k)}$; this initial segment of~$L_{v_h}$ thus lies on~$D'$. The bit-$1$ cert at height~$h$ targeting $\comm{R}^{\vote(k)}$ thus fires on~$D'$, advancing canonical state-height to~$h+1$. Combined with $\sigma[\getConfirmed_v^{\vote(k+1)}].h \leq R_v^{\vote(k+1)}.h + 1 \leq h+2$ (\Cref{rev:cor:processed-height-bound}), $\sigma[\getConfirmed_v^{\vote(k+1)}].h \in \{h+1, h+2\}$ for every honest~$v$. We sub-split on the value.

\begin{description}
\item[{Case 2.A: $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h+1$ for every honest~$v$.}]\hfill

\emph{Step 4: slot-$(k+1)$ honest votes target blocks on~$D'$ at state-height~$h+1$.}
By the voting rule, each honest~$v$ casts a slot-$(k+1)$ vote at canonical state-height~$h_c = h+1$. The lock check at $h+1$ fails: hypothesis~\ref{rev:hfj:nh} excludes any before-$\vote(k)$ honest vote at height~$>h$, in particular finalize-bearing at $h+1$; slot-$k$ (R1) honest votes have $\finalizeHeight = h_j \leq h < h+1$ (canonical $h_j$ equals~$h$ on chains where the cert-at-$h$ has fired, and is $\leq h-1$ otherwise). Hence the slot-$(k+1)$ vote target is $T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)} \in D'$ at state-height~$h+1$.

By synchrony (hypothesis~\ref{rev:hfj:sync} + slot spacing $\geq 2\Delta$), every slot-$(k+1)$ honest vote is delivered to every honest by $\vote(k+1) + \Delta \leq \vote(k+2) - \Delta$. By hypothesis~\ref{rev:hfj:aha} (\AHA at slot~$k+1$), all $\geq n - f > 2n/3$ honest cast such a vote. In each honest's $\Sigma.V$ at $\vote(k+2)$, the slot-$(k+1)$ honest votes target blocks on~$D'$ at state-height~$h+1$.

\emph{Step 5: $R_v^{\vote(k+2)}.h = h+1$ for every honest~$v$, with target on~$D'$.}
\updateRootFromVotes{} on each honest aggregates the slot-$(k+1)$ honest votes into a virtual block extending its canonical chain (on~$D'$); the prefix-notar $\arg\max_\preceq$ over the slot-$(k+1)$ same-slot quorum at height~$h+1$ on~$D'$ selects some $B^{\star\star}_v \in D'$ at state-height~$h+1$, firing a bit-$0$ cert event with key $(h+1, 0, k+1, -B^{\star\star}_v.\slot, \hash(B^{\star\star}_v))$. The $\arg\max$ depends on $\Sigma_v.V$ (which has uniform honest content by synchrony) and~$D'$'s linear structure; honest contributions alone form a $\geq n - f > 2n/3$ quorum, so $B^{\star\star}_v$ is uniform across honest validators.

We compare the new cert key to $\Sigma_v.\key_R$:
\begin{itemize}
  \item If $\Sigma_v.\key_R.h < h+1$: the slot-$(k+1)$ key dominates on the leading coordinate; $R_v$ updates to~$B^{\star\star}_v$.
  \item If $\Sigma_v.\key_R = (h+1, 0, k', \ldots)$ with $k' \leq k$ (slot-$\geq k+1$ contributions do not exist before $\vote(k+1)$): the slot-$(k+1)$ key dominates on the $s_n$ coordinate; $R_v$ updates to~$B^{\star\star}_v$.
  \item If $\Sigma_v.\key_R = (h+1, 1, 0, J^*)$ for some $J^*$ justified at height~$h+1$ in $v$'s view: the cached bit-$1$ dominates and $R_v$ stays at~$J^*$. The bit-$1$ quorum has $\geq 2n/3 - f$ honest contributors at common target $J^* = \getConfirmed_i^{\vote(k)}$ for some honest~$i$ at state-height~$h+1$, and Step~2's vote inclusion places that target on~$D'$.
\end{itemize}
In every case $R_v^{\vote(k+2)}.h = h+1$ and $R_v^{\vote(k+2)} \in D'$ at state-height~$h+1$.

\emph{Step 6: healed-state conditions at slot~$k+2$ with height~$h+1$.}
We verify each condition of \Cref{rev:def:healed-state}.
\begin{itemize}
  \item \emph{\ref{rev:def:healed-state:sync}.}
  $\start(k+2) - \Delta \geq \start(k) + 4\Delta - \Delta = \start(k) + 3\Delta \geq \GST$ by hypothesis~\ref{rev:hfj:sync} and slot spacing.

  \item \emph{\ref{rev:def:healed-state:R}.}
  Step~5 gives $R_v^{\vote(k+2)}.h = h+1$.

  \item \emph{\ref{rev:def:healed-state:gt-compat}.}
  Apply \Cref{rev:prop:getConfirmed-compat} at slot~$k+1$. Preamble: $R_v^{\vote(k+1)} \in D'$ (Step~1 if no cert at $h+1$ has updated~$R_v$ by $\vote(k+1)$, Step~5's classification otherwise), pairwise compatible. Premise: pick $v_\ell'$ honest with $\getConfirmed_{v_\ell'}^{\vote(k+1)}$ the $\preceq$-max on~$D'$ among honest \getConfirmed{}s at $\vote(k+1)$. For $t \in [\vote(k+1),\, \vote(k+2)]$, $R_v^t \in D'$ at state-height~$h+1$ (Step~5 covers $t = \vote(k+2)$ and the bit-$1$/bit-$0$ classification covers earlier $t$). Each such $R_v^t$ is~$\preceq \getConfirmed_j^{\vote(k+1)}$ for some honest~$j$ (slot-$k$ certs by Step~2's vote inclusion + maximality, slot-$(k+1)$ certs by Step~5's $\arg\max$ giving $B^{\star\star}_v \preceq \getConfirmed_v^{\vote(k+1)}$ for $\geq 2n/3$ honest); maximality of~$v_\ell'$ then gives $R_v^t \preceq \getConfirmed_{v_\ell'}^{\vote(k+1)}$. The conclusion gives pairwise compatibility of the honest $\getConfirmed_v^{\vote(k+2)}$.

  \item \emph{\ref{rev:def:healed-state:target-cap}.}
  Before-$\vote(k)$ honest votes at height~$\geq h+1$: ruled out by hypothesis~\ref{rev:hfj:nh}. Slot-$k$ honest votes at height~$\geq h+1$: those at height~$h+1$ exactly, with target $T_i = \getConfirmed_i^{\vote(k)} \in D'$ (Step~2's vote inclusion). \Cref{rev:prop:target-monotonicity} (premise as in the previous bullet) gives $\getConfirmed_i^{\vote(k+2)} \succeq \getConfirmed_i^{\vote(k+1)} \succeq T_i$; pairwise compatibility (previous bullet) and picking $v_\ell$ honest with $\getConfirmed_{v_\ell}^{\vote(k+2)}$ the $\preceq$-max gives $T_i \preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$. Slot-$(k+1)$ honest votes at height~$h+1$ with target $T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)}$: same monotonicity~+ maximality argument gives $T_v^{(k+1)} \preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$.
\end{itemize}
All four conditions verified, the system is in a healed state at slot~$k+2$ with height~$h+1$.


\item[{Case 2.B: $\sigma[\getConfirmed_{v_*}^{\vote(k+1)}].h = h+2$ for some honest~$v_*$.}]\hfill

\emph{Step 4: $v_*$'s view contains a cert event at height~$h+1$ with target on~$D'$, fired during onBlock processing of some block~$B^* \in \Sigma_{v_*}.\T$.}
\Cref{rev:cor:processed-height-bound} applied to $\Sigma_{v_*}^{\vote(k+1)}$ gives $R_{v_*}^{\vote(k+1)}.h \geq h+1$. Step~1's bound $\Sigma_v.\key_R.h \leq h+1$ then forces $R_{v_*}^{\vote(k+1)}.h = h+1$. By Step~1's classification of cert events that may dominate the cached key, the corresponding cert event has target $T^* \in D'$. Since $\sigma[\getConfirmed_{v_*}^{\vote(k+1)}].h = h+2$, the cert event also fired on $v_*$'s canonical chain (the height-advance from $h+1$ to $h+2$ requires processHeight to fire the cert at~$h+1$ on canonical), at processSlot for some block~$B^* \in \Sigma_{v_*}^{\vote(k+1)}.\T$ on canonical.

\emph{Step 5: every honest validator has $R_v^{\vote(k+2)}.h \geq h+1$ with target on~$D'$.}
By synchrony (hypothesis~\ref{rev:hfj:sync} + slot spacing $\geq 2\Delta$, post-GST), every block in $\Sigma_{v_*}^{\vote(k+1)}.\T$ is delivered to every honest by $\vote(k+2) - \Delta$, and honest validators process received blocks via onBlock in $\preceq$-respecting order. In particular, every honest~$v$ processes~$B^*$ together with all of its ancestors by $\vote(k+2)$. The state map is a deterministic function of the chain (\Cref{thm:orderindep}), so $\sigma_v[B^*.\parent] = \sigma_{v_*}[B^*.\parent]$, and the same processSlot/processHeight that fired the cert at~$h+1$ in $v_*$'s view fires it in $v$'s view; the descriptor offered to \updateForkChoiceRoot raises $\Sigma_v.\key_R$ to at least $(h+1, b, s_n, -T^*.\slot, \hash(T^*))$. \Cref{lem:Rs-key-monotone} preserves the bound, giving $R_v^{\vote(k+2)}.h \geq h+1$.

A subsequent cert in $v$'s view at height~$>h+1$ during $[\vote(k+1),\, \vote(k+2)]$ has height~$\leq h+2$ (no honest votes at height~$>h+2$ exist before $\vote(k+2)$: hypothesis~\ref{rev:hfj:nh} excludes before-$\vote(k)$, \Cref{rev:cor:processed-height-bound} bounds slot-$k$ honest at $\leq h+1$ and slot-$(k+1)$ honest at $\leq h+2$; dishonest alone $\leq f < 2n/3$). Such a cert at~$h+2$ has target $\preceq$ some slot-$(k+1)$ honest contributor's target at $h+2$, which equals $\getConfirmed_w^{\vote(k+1)}$ for some honest~$w$ at state-height~$h+2$ on~$D'$ (R1 unlocked, since hypothesis~\ref{rev:hfj:nh} excludes before-$\vote(k)$ honest at height $>h$ and slot-$k$ R1 honest votes have $\finalizeHeight \leq h$, both strictly below $h+2$). Hence $R_v^{\vote(k+2)} \in D'$.

\emph{Step 6: healed-state conditions at slot~$k+2$ with height~$h+1$.}
We verify each condition of \Cref{rev:def:healed-state}.
\begin{itemize}
  \item \emph{\ref{rev:def:healed-state:sync}.} As in Case~2.A: $\start(k+2) - \Delta \geq \start(k) + 3\Delta \geq \GST$.

  \item \emph{\ref{rev:def:healed-state:R}.} Step~5 gives $R_v^{\vote(k+2)}.h \geq h+1$ for every honest~$v$.

  \item \emph{\ref{rev:def:healed-state:gt-compat}.}
  Apply \Cref{rev:prop:getConfirmed-compat} at slot~$k+1$. Preamble: $R_v^{\vote(k+1)} \in D'$ for every honest~$v$ (Step~1 if no cert at~$h+1$ has updated~$R_v$ by $\vote(k+1)$, Step~5's classification otherwise), pairwise compatible. Premise: pick $v_\ell'$ honest with $\getConfirmed_{v_\ell'}^{\vote(k+1)}$ the $\preceq$-max on~$D'$ among honest \getConfirmed{}s at $\vote(k+1)$ (well-defined since $v_*$'s $\getConfirmed^{\vote(k+1)}$ at state-height~$h+2$ lies on~$D'$). For $t \in [\vote(k+1),\, \vote(k+2)]$, every cert target reachable as $R_v^t$ is on~$D'$ at chain-position bounded by some honest's $\getConfirmed^{\vote(k)}$ or $\getConfirmed^{\vote(k+1)}$ (slot-$k$ certs by Step~2's vote inclusion + maximality, slot-$(k+1)$ certs at~$h+2$ as in Step~5, slot-$(k+1)$ certs at~$h+1$ symmetrically), and maximality of~$v_\ell'$ gives $R_v^t \preceq \getConfirmed_{v_\ell'}^{\vote(k+1)}$. The conclusion gives pairwise compatibility of the honest $\getConfirmed_v^{\vote(k+2)}$.

  \item \emph{\ref{rev:def:healed-state:target-cap}.}
  Before-$\vote(k)$ honest votes at height~$\geq h+1$: ruled out by hypothesis~\ref{rev:hfj:nh}. Slot-$k$ honest votes at height~$\geq h+1$: at~$h+1$ exactly, with target $\getConfirmed_i^{\vote(k)} \in D'$ (Step~2's vote inclusion). Slot-$(k+1)$ honest votes at height~$\geq h+1$: at~$h+1$ or~$h+2$, with target $T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)} \in D'$ (R1 unlocked: hypothesis~\ref{rev:hfj:nh} excludes before-$\vote(k)$ honest at height $>h$ in particular finalize-bearing at~$h+1$ or~$h+2$; slot-$k$ R1 honest votes have $\finalizeHeight \leq h < h+1$). \Cref{rev:prop:target-monotonicity} (premise as in the previous bullet) gives $\getConfirmed_v^{\vote(k+2)} \succeq \getConfirmed_v^{\vote(k+1)}$ for every honest~$v$ and similarly $\getConfirmed_i^{\vote(k+2)} \succeq \getConfirmed_i^{\vote(k)}$; pairwise compatibility (previous bullet) and picking $v_\ell$ honest with $\getConfirmed_{v_\ell}^{\vote(k+2)}$ the $\preceq$-max give every such target~$\preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$.
\end{itemize}
All four conditions verified, the system is in a healed state at slot~$k+2$ with height~$h+1$.

\begin{lemma}
Assume:
\begin{enumerate}
  \item $f^{\vote(k+3)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item syncers of slots in $[k+1]$ is honest.
  \item the \OHM holds at slots in $[k, k+1, k+2]$;
  \item the \AHA condition holds at slots in $[k, k+1, k+2]$;
  \item the \HAR condition holds at slot in $[k, k+1]$;
  \item for every honest validator~$v$, $R_v^{\vote(k)}$  a notarized block for the same height $h$ and slot $k-1$.
  % \item $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getConfirmed_v^{\vote(k)}].h = \comm{R}^{\vote(k)}.h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item no honest validator has cast any vote at height $> \comm{R}^{\vote(k)}.h$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then the system is in a healed state at some slot in  $[k+2,k+3]$ with height $h+1$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Let $h \coloneqq \comm{R}^{\vote(k)}.h$. By \Cref{rev:lem:confirmed-h-range} applied per-honest, $\sigma[\getConfirmed_v^{\vote(k)}].h \in \{h, h+1\}$ for every honest~$v$, splitting into two cases.

\begin{description}
\item[{Case 1: $\sigma[\getConfirmed_v^{\vote(k)}].h = h$ for every honest~$v$.}]\hfill

\emph{Step 1: setup at $\vote(k)$.}
By hypothesis~7, each honest's cached key is $\Sigma_v.\key_R = (h, 0, k-1, -R_v^{\vote(k)}.\slot, \hash(R_v^{\vote(k)}))$. By \Cref{rev:lem:notar-unique-per-slot} (under $f<n/3$ from hypothesis~1), the honest $R_v^{\vote(k)}$ are pairwise compatible; let~$C_R$ be the chain extending them, with each $R_v^{\vote(k)} \in C_R$ at chain-height~$h$. Hypothesis~8 excludes any before-$\vote(k)$ honest vote at height~$>h$. \Cref{lem:no-finalize-without-just} combined with \Cref{lem:Rs-key-monotone} (cached~$.h \leq h$ keeps cached at bit-$0$ at~$h$ never bit-$1$ at~$h$) excludes any honest lock at height~$h$ before $\vote(k)$. Hence every slot-$k$ honest vote at canonical state-height~$h$ is unlocked with target $T_i = \getConfirmed_i^{\vote(k)}$ on a chain extending $R_i^{\vote(k)} \in C_R$.

\emph{Step 2: slot-$k$ same-slot bit-$0$ cert at $h$ slot~$k$.}
By hypothesis~5 (\AHA at slot~$k$) and hypothesis~6 (\HAR at slot~$k$), all $\geq n-f > 2n/3$ honest cast at $\vote(k)$, and an honest proposer at some slot in $[\vote(k), \vote(k+1) - \delta_c)$ proposes a block~$B^h$ on the available chain that contains every slot-$k$ honest vote; by synchrony (hypothesis~2 + slot spacing), $B^h$ is delivered to every honest by $\vote(k+1)$. Inside $B^h$'s on-chain processing (or via virtual-block aggregation on any leaf at state-height~$\geq h$, by the extended \updateRootFromVotes{}), the prefix-notar $\arg\max_\preceq$ in \processHeight{} fires a bit-$0$ cert at $(h, 0, k, -B^\star.\slot, \hash(B^\star))$ for some $B^\star$ on a chain extending~$C_R$. \Cref{rev:lem:notar-unique-per-slot} pins all such targets across honest views to a common chain. The cert key strictly dominates the cached~$(h, 0, k-1, \ldots)$ on the $s_n$ coordinate, so $R_v$ updates within height~$h$ to a target in this common chain.

\item[{Sub-case 1.A: no bit-$1$ cert at height~$h$ has fired in any honest's view by $\vote(k+1)$.}]\hfill

\emph{Step 3: canonical state-height reaches~$h+1$ at $\vote(k+1)$.}
\Cref{rev:prop:honest-vote-inclusion} (under hypothesis~6 + the discharged premise: $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R_v^t$ for honest~$v_\ell$ chosen as the $\preceq$-max honest \getConfirmed{} on~$C_R$ at $\vote(k)$, and $t \in [\vote(k),\, \vote(k+1)]$ — every cert target reachable as $R_v^t$ in this interval lies on~$C_R$ at chain-height~$\leq h$ by Step~1's notar-uniqueness + Step~2) gives that the chain at $\getConfirmed_v^{\vote(k+1)}$ contains every slot-$k$ honest vote. The slot-$k$ same-slot cert at~$h$ thus fires on canonical, advancing canonical state-height to~$h+1$. \Cref{rev:prop:getConfirmed-compat} gives pairwise compatibility of the honest $\getConfirmed_v^{\vote(k+1)}$; let~$D'$ be the chain extending them. Combined with \Cref{rev:cor:processed-height-bound} ($\sigma[\getConfirmed_v^{\vote(k+1)}].h \leq h+1$), $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h+1$ for every honest~$v$.

\emph{Step 4: slot-$(k+1)$ honest votes target blocks on~$D'$ at state-height~$h+1$.}
By the voting rule, each honest~$v$ casts a slot-$(k+1)$ vote at canonical state-height~$h_c = h+1$. Lock-check at~$h+1$ fails: hypothesis~8 excludes before-$\vote(k)$ honest votes at height~$>h$; slot-$k$ R1/R2 honest votes have $\finalizeHeight \leq h_j \leq h-1 < h+1$ (in this sub-case canonical $h_j < h$, since no bit-$1$ at~$h$ has fired). Hence $T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)} \in D'$ at state-height~$h+1$. By synchrony + hypothesis~5 (\AHA at $k+1$), all $\geq n-f > 2n/3$ slot-$(k+1)$ honest votes are delivered to every honest by $\vote(k+2) - \Delta$.

\emph{Step 5: $R_v^{\vote(k+2)}.h = h+1$ for every honest, target on~$D'$.}
The extended \updateRootFromVotes{} on each honest aggregates the slot-$(k+1)$ honest votes into virtual blocks extending each leaf at state-height~$\geq R.h = h$, including the leaf on~$D'$ that hosts the slot-$(k+1)$ targets. The prefix-notar $\arg\max_\preceq$ over the slot-$(k+1)$ same-slot quorum at~$h+1$ on~$D'$ selects $B^{\star\star} \in D'$ at state-height~$h+1$, firing a bit-$0$ cert with key $(h+1, 0, k+1, -B^{\star\star}.\slot, \hash(B^{\star\star}))$. Even if a late bit-$1$ cert at~$h$ would otherwise compete (cached~$.h$ stays~$\leq h$ in this sub-case otherwise), the new key dominates on the leading coordinate, and the leaf-coverage of the extended \updateRootFromVotes{} ensures the slot-$(k+1)$ cert is offered for every leaf of state-height~$\geq h$ (so $R_v$ updates regardless of which leaf the validator's $\getConfirmed$ lands on). Hence $R_v^{\vote(k+2)} = B^{\star\star}$ uniformly with $R_v^{\vote(k+2)}.h = h+1$.

\emph{Step 6: healed-state conditions at slot~$k+2$ with height~$h+1$.}
\begin{itemize}
  \item \emph{\ref{rev:def:healed-state:sync}}: $\start(k+2) - \Delta \geq \GST$ from hypothesis~2 + slot spacing.
  \item \emph{\ref{rev:def:healed-state:R}}: Step~5.
  \item \emph{\ref{rev:def:healed-state:gt-compat}}: \Cref{rev:prop:getConfirmed-compat} at slot~$k+1$, premise discharged via Step~3's pairwise compatibility on~$D'$ + Step~5's $B^{\star\star} \in D'$ at chain-height~$h+1$ bounded by the $\preceq$-max honest $\getConfirmed^{\vote(k+1)}$ on~$D'$.
  \item \emph{\ref{rev:def:healed-state:target-cap}}: before-$\vote(k)$ honest at $\geq h+1$ ruled out by hypothesis~8; slot-$k$ honest at~$h$, not~$\geq h+1$; slot-$(k+1)$ honest at~$h+1$ with target on~$D'$, bounded by $\getConfirmed_{v_\ell}^{\vote(k+2)}$ via \Cref{rev:prop:target-monotonicity} + maximality.
\end{itemize}
The system is in a healed state at slot~$k+2$ with height~$h+1$.

\item[{Sub-case 1.B: a bit-$1$ cert at height~$h$ has fired in some honest's view by $\vote(k+1)$.}]\hfill

By view-merge under honest $\syncer(k+1)$ (hypothesis~3, \Cref{rev:ass:view-merge}), every honest's store at $\vote(k+1)$ equals the common $\comm{\Sigma}^{\vote(k+1)}$; the bit-$1$ cert at~$h$ targeting some $J^*$ is in every honest's view, with $R_v^{\vote(k+1)} = J^*$ uniformly (the bit-$1$ key $(h, 1, 0, \ldots)$ dominates any cached bit-$0$ at the same height). Hence $\comm{R}^{\vote(k+1)} = J^*$ is justified at height~$h$ in every honest's view at $\vote(k+1)$.

The hypotheses of \Cref{rev:lem:healing-from-justified} apply at slot~$k+1$ with $\comm{R}^{\vote(k+1)} = J^*$:
\begin{itemize}
  \item \ref{rev:hfj:f}: $f^{\vote((k+1)+2)} = f^{\vote(k+3)} < n/3$ from hypothesis~1.
  \item \ref{rev:hfj:sync}: $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$ from hypothesis~2 + slot spacing.
  \item \ref{rev:hfj:ohm}: \OHM at slots~$k+1$ and~$k+2$ from hypothesis~4.
  \item \ref{rev:hfj:aha}: \AHA at slots~$k+1$ and~$k+2$ from hypothesis~5.
  \item \ref{rev:hfj:har}: \HAR at slot~$k+1$ from hypothesis~6.
  \item \ref{rev:hfj:r}: $R_v^{\vote(k+1)} = J^*$ uniformly justified at height~$h$, just established.
  \item \ref{rev:hfj:nh}: hypothesis~8 excludes before-$\vote(k)$ honest at height~$>h$; slot-$k$ honest at~$h$ exactly (Step~1); no voting in $(\vote(k),\, \vote(k+1))$.
\end{itemize}
Invoking \Cref{rev:lem:healing-from-justified} at slot~$k+1$ places the system in a healed state at slot~$(k+1)+2 = k+3$ with height~$h+1$.

In either sub-case, the system is in a healed state at some slot in~$\{k+2, k+3\}$ with height~$h+1$.

\item[{Case 2: $\sigma[\getConfirmed_v^{\vote(k)}].h = h+1$ for some honest~$v$.}]\hfill

\emph{Step 1: setup at $\vote(k)$.}
By hypothesis~7, $\Sigma_v.\key_R = (h, 0, k-1, -R_v^{\vote(k)}.\slot, \hash(R_v^{\vote(k)}))$ at $\vote(k)$ for every honest~$v$. By \Cref{rev:lem:notar-unique-per-slot} (under $f<n/3$, hypothesis~1), the honest $R_v^{\vote(k)}$ are pairwise compatible; let~$C_R$ be the chain extending them. Since cached is bit-$0$ at height~$h$, by \Cref{lem:Rs-key-monotone} no bit-$1$ cert at height~$h$ has fired in any honest's view by $\vote(k)$ (the bit-$1$ key $(h, 1, 0, \ldots)$ would have raised the cache). Hypothesis~8 excludes any before-$\vote(k)$ honest vote at height~$>h$.

\emph{Step 2: slot-$k$ honest votes are unlocked with target $\getConfirmed_v^{\vote(k)}$.}
By \Cref{rev:lem:confirmed-h-range}, $\sigma[\getConfirmed_v^{\vote(k)}].h \in \{h, h+1\}$ for every honest~$v$. Each honest~$v$ casts a slot-$k$ vote at canonical state-height $h_v \in \{h, h+1\}$. The lock-checks fail: hypothesis~8 excludes before-$\vote(k)$ honest finalize-bearing at height~$h+1$, ruling out a lock at~$h+1$; \Cref{lem:no-finalize-without-just} together with Step~1 (no bit-$1$ at~$h$ in any honest's view) excludes any prior $\finalizeHeight = h$ vote (which would require a justification at~$h$ to have fired), ruling out a lock at~$h$. Hence the slot-$k$ vote target is $T_v = \getConfirmed_v^{\vote(k)}$, on a chain extending $R_v^{\vote(k)} \in C_R$.

\begin{description}
\item[{Case 2.A: no bit-$1$ cert at height~$h$ has fired in any honest's view by $\vote(k+1)$.}]\hfill

\emph{Step 3: cert-target compatibility throughout $[\vote(k),\, \vote(k+1)]$.}
The case-$2.A$ assumption (no bit-$1$ at~$h$ by $\vote(k+1)$) plus Step~1 (no bit-$1$ at~$h$ before~$\vote(k)$, forced by hypothesis~7) rules out bit-$1$ at~$h$ throughout $[\vote(k),\, \vote(k+1)]$. We classify cert events whose key may dominate the cached at any honest's view in the interval:
\begin{itemize}
  \item \emph{Height~$h$:} only bit-$0$ same-slot certs at slot $s \in \{k-1, k\}$ (slot-$\geq k+1$ contributions don't exist before $\vote(k+1)$, slot-$\leq k-2$ would be dominated by cached's $s_n = k-1$ on the inverted-slot key when ties on $s_n$ via hash). All such cert targets are pairwise compatible (\Cref{rev:lem:notar-unique-per-slot}) and lie on the chain extending $C_R$ — call this~$D$.
  \item \emph{Height~$h+1$:} bit-$0$ slot-$k$ same-slot at $h+1$ (from slot-$k$ honest contributions at~$h+1$, for those honest with $\sigma[\getConfirmed_v^{\vote(k)}].h = h+1$), or bit-$1$ at $h+1$ targeting some $J^{**}$. By Step~2, slot-$k$ honest contributions at~$h+1$ have target $\getConfirmed_v^{\vote(k)} \in D$, so the prefix-notar $\arg\max_\preceq$ for bit-$0$ at~$h+1$ lies on~$D$; the bit-$1$ target is also $\getConfirmed_i^{\vote(k)} \in D$ for some honest contributor~$i$ at state-height~$h+1$.
\end{itemize}
Hence every cert target reachable as $R_v^t$ for $t \in [\vote(k),\, \vote(k+1)]$ lies on the common chain~$D$ extending~$C_R$ at chain-height~$\leq h+1$.

\emph{Step 4: pairwise compatibility, monotonicity, vote-inclusion at $\vote(k+1)$.}
\Cref{rev:prop:getConfirmed-compat,rev:prop:target-monotonicity,rev:prop:honest-vote-inclusion} apply at slot~$k$. Preamble: $R_v^{\vote(k)} \in C_R \subseteq D$ pairwise compatible (Step~1). Premise: pick $v_\ell$ honest with $\getConfirmed_{v_\ell}^{\vote(k)}$ the $\preceq$-max honest \getConfirmed{} on~$D$ at $\vote(k)$; by Step~3, $R_v^t \preceq \getConfirmed_{v_\ell}^{\vote(k)}$ for every honest~$v$ and $t \in [\vote(k),\, \vote(k+1)]$. \HAR{} at slot~$k$ from hypothesis~6 powers \Cref{rev:prop:honest-vote-inclusion}. Conclusions:
\begin{itemize}
  \item Pairwise compatibility of honest $\getConfirmed_v^{\vote(k+1)}$; let~$D'$ be the chain extending them ($D \subseteq D'$ by maximality + monotonicity).
  \item $\getConfirmed_v^{\vote(k+1)} \succeq \getConfirmed_v^{\vote(k)}$ for every honest~$v$.
  \item The chain at $\getConfirmed_v^{\vote(k+1)}$ contains every slot-$k$ honest vote; in particular, every slot-$k$ honest vote target lies on~$D'$.
\end{itemize}

\emph{Step 5: state-height at $\vote(k+1)$ lies in $\{h+1, h+2\}$.}
By hypothesis~6, \HAR{} at slot~$k$ holds, and \HAR{}'s definition guarantees an honest proposer of a block on the available chain in $[\vote(k),\, \vote(k+1) - \delta_c)$, discharging \HAP{} at slot~$k$. The case-$2$ assumption gives some honest validator at canonical state-height~$h+1$ at $\vote(k)$, so the $\preceq$-max honest $\getConfirmed_{v_\ell}^{\vote(k)}$ on~$D$ has state-height~$h+1$. \Cref{rev:prop:target-dominance} applies at slot~$k$ (preamble: $R_v^{\vote(k)} \in C_R$ pairwise compatible (Step~1); premise: discharged via Step~3 and maximality of~$v_\ell$, exactly as for the discharge of \Cref{rev:prop:getConfirmed-compat,rev:prop:target-monotonicity,rev:prop:honest-vote-inclusion} in Step~4). Its conclusion gives $\getConfirmed_v^{\vote(k+1)} \succeq \getConfirmed_{v_\ell}^{\vote(k)}$ for every honest~$v$; hence $\sigma[\getConfirmed_v^{\vote(k+1)}].h \geq \sigma[\getConfirmed_{v_\ell}^{\vote(k)}].h = h+1$ (state-height monotonicity along~$\preceq$, \Cref{lem:slotmono}). Combined with \Cref{rev:cor:processed-height-bound} ($\sigma[\getConfirmed_v^{\vote(k+1)}].h \leq R_v^{\vote(k+1)}.h + 1 \leq h+2$), $\sigma[\getConfirmed_v^{\vote(k+1)}].h \in \{h+1, h+2\}$ for every honest~$v$. (How the slot-$k$ honest votes happen to split between heights~$h$ and~$h+1$ at $\vote(k)$ doesn't enter the argument; whether or not a slot-$k$ same-slot cert at a single height happens to form, the target-dominance step alone forces canonical state-height to~$\geq h+1$ at $\vote(k+1)$.) We sub-split on the value.

\begin{description}
\item[{Case 2.A.A: $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h+1$ for every honest~$v$.}]\hfill

\emph{Step 6: slot-$(k+1)$ honest votes target blocks on~$D'$ at state-height~$h+1$.}
By the voting rule, each honest~$v$ casts a slot-$(k+1)$ vote at canonical state-height $h_c = h+1$ (case-$2.A.A$). Lock-check at~$h+1$ fails: hypothesis~8 excludes before-$\vote(k)$ honest votes at height~$>h$; slot-$k$ R2 honest votes at~$h$ have $\finalizeHeight \leq h_j \leq h-1$ (in Case~$2.A$ no bit-$1$ at~$h$ has fired by $\vote(k+1)$, so canonical $h_j \leq h-1$ at $\vote(k)$); slot-$k$ R1 honest votes at~$h+1$ have $\finalizeHeight \leq h$; neither equals~$h+1$. Hence $T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)} \in D'$ at state-height~$h+1$. By synchrony + hypothesis~5 (\AHA at $k+1$), all $\geq n-f > 2n/3$ slot-$(k+1)$ honest votes are delivered to every honest by $\vote(k+2) - \Delta$.

\emph{Step 7: $R_v^{\vote(k+2)}.h = h+1$ for every honest, target on~$D'$.}
The extended \updateRootFromVotes{} on each honest aggregates the slot-$(k+1)$ honest votes into virtual blocks extending each leaf at state-height~$\geq R.h$ (which is $h$ or $h+1$ in Case~$2.A$, since no bit-$1$ at~$h$ has fired so cached~$.h \in \{h, h+1\}$ from at most a slot-$\leq k$ same-slot bit-$0$ at~$h$ or~$h+1$). The prefix-notar $\arg\max_\preceq$ over the slot-$(k+1)$ same-slot quorum at~$h+1$ on~$D'$ selects $B^{\star\star} \in D'$ at state-height~$h+1$, firing a bit-$0$ cert with key $(h+1, 0, k+1, -B^{\star\star}.\slot, \hash(B^{\star\star}))$. The new key dominates any cached at height~$\leq h$ on the leading coordinate, and dominates any cached at height~$h+1$ (necessarily slot-$\leq k$ same-slot bit-$0$, since no bit-$1$ at~$h+1$ has fired with quorum from contributors at~$h+1$ before $\vote(k+1)$ in this sub-case) on the $s_n$ coordinate ($k+1 > k$). The leaf-coverage of the extended \updateRootFromVotes{} ensures the slot-$(k+1)$ cert is offered for every leaf at state-height~$\geq R.h$, so $R_v$ updates regardless of which leaf hosts $\getConfirmed_v$. Hence $R_v^{\vote(k+2)} = B^{\star\star}$ uniformly with $R_v^{\vote(k+2)}.h = h+1$, target on~$D'$ at chain-height~$h+1$.

\emph{Step 8: healed-state conditions at slot~$k+2$ with height~$h+1$.}
Pick~$v_\ell$ honest with $\getConfirmed_{v_\ell}^{\vote(k+1)} = \max_\preceq$ on~$D'$ among honest \getConfirmed{} at $\vote(k+1)$.
\begin{itemize}
  \item \emph{\ref{rev:def:healed-state:sync}}: $\start(k+2) - \Delta \geq \GST$ from hypothesis~2 + slot spacing.
  \item \emph{\ref{rev:def:healed-state:R}}: $R_v^{\vote(k+2)}.h = h+1$ uniformly by Step~7.
  \item \emph{\ref{rev:def:healed-state:gt-compat}}: \Cref{rev:prop:getConfirmed-compat} at slot~$k+1$. Standing assumptions: hypothesis~4 (\OHM at $k+1$); Step~4 ($R$-pairwise compat on~$D'$ at $\vote(k+1)$ via the store invariant). Premise discharged via Step~7's $B^{\star\star} \in D'$ at chain-height~$h+1$ + slot-$(k+1)$ honest contributions targeting $T_v^{(k+1)} \in D'$ from Step~6 + the bound on cert events in $[\vote(k+1),\, \vote(k+2)]$ as in Step~3.
  \item \emph{\ref{rev:def:healed-state:target-cap}}: any honest vote at height~$\geq h+1$ cast strictly before $\vote(k+2)$ has target $\preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$:
  \begin{itemize}
    \item before-$\vote(k)$ honest at $\geq h+1$: ruled out by hypothesis~8.
    \item slot-$k$ honest at $\geq h+1$: target $\getConfirmed_v^{\vote(k)} \in D \subseteq D'$ (Step~2 + Step~4 chain extension), $\preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$ by maximality + \Cref{rev:prop:target-monotonicity}.
    \item slot-$(k+1)$ honest at $\geq h+1$: target $\getConfirmed_v^{\vote(k+1)} \in D'$ by Step~6, $\preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$ by maximality + \Cref{rev:prop:target-monotonicity}.
  \end{itemize}
\end{itemize}
The system is in a healed state at slot~$k+2$ with height~$h+1$.

\item[{Case 2.A.B: $\sigma[\getConfirmed_{v_*}^{\vote(k+1)}].h = h+2$ for some honest~$v_*$.}]%
\footnote{The proof here relies on \onBlock{} (\Cref{alg:store}) processing every admissible block, regardless of whether it descends from the validator's current fork-choice: an honest~$v$ folds the chain through $v_*$'s height-$(h+2)$ block via \onBlock{} on receipt, firing the cert at~$h+1$ on-chain in $v$'s view independently of $v$'s fork-choice. If \onBlock{} were restricted to blocks descending from the current fork-choice, the synchrony argument in Step~6 below would not apply, and the proof would split on whether a bit-$1$ cert at height~$h$ fires in $[\vote(k+1),\, \vote(k+2)]$. (\textbf{Case~$2.A.B.A$, no bit-$1$ at~$h$ by $\vote(k+2)$.}) Under that case-restriction, every cert event reachable as $R_v^t$ for $t \in [\vote(k+1),\, \vote(k+2)]$ has target on~$D'$ at chain-height~$\leq h+2$: bit-$1$ at~$h$ is excluded by the case assumption, and the remaining cert events (slot-$\leq k$ same-slot bit-$0$ at~$h$ on~$D'$ by \Cref{rev:lem:notar-unique-per-slot}; bit-$0$ or bit-$1$ at~$h+1$ bounded as in Step~3; slot-$(k+1)$ same-slot at~$h+1$/$h+2$ targeting $T_v^{(k+1)} \in D'$ via Step~7) are all on~$D'$. Apply \Cref{rev:prop:target-dominance,rev:prop:getConfirmed-compat,rev:prop:target-monotonicity,rev:prop:honest-vote-inclusion} at slot~$k+1$ (\HAR{} from hypothesis~6, $R$-pairwise compat from Step~4, premise from the cert-target compatibility just shown with $v_\ell'$ chosen as the $\preceq$-max honest \getConfirmed{} on~$D'$ at $\vote(k+1)$ at $\sigma$-height~$h+2$); the conclusions give $\sigma[\getConfirmed_v^{\vote(k+2)}].h \geq h+2$ uniformly, $R_v^{\vote(k+2)}.h \geq h+1$ uniformly with target on~$D'$, and pairwise compatibility of $\getConfirmed_v^{\vote(k+2)}$ on a chain~$D''$ extending~$D'$. The healed-state conditions at slot~$k+2$ with height~$h+1$ then follow as in Step~7 below. (\textbf{Case~$2.A.B.B$, bit-$1$ at~$h$ fires by $\vote(k+2)$.}) Target-dominance over $[\vote(k+1),\, \vote(k+2)]$ is unavailable: the bit-$1$ cert at~$h$ has target $J^*$ at height~$h$ possibly off~$D'$, breaking the premise discharge. An honest~$v$ at $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h+1$ with $R_v^{\vote(k+1)}.h = h$ has cached dominated on the bit coordinate by the new bit-$1$ at~$h$, updating $R_v^{\vote(k+2)}$ to $(h, 1, 0, J^*)$ at height~$h$, breaking \ref{rev:def:healed-state:R} at slot~$k+2$ with height~$h+1$. Possible routes include invoking \Cref{rev:lem:healing-from-justified} at slot~$k+2$ to reach healed at slot~$k+4$ with height~$h+1$ (under a strengthened hypothesis~$f^{\vote(k+4)} < n/3$), or sharpening the case to ensure $R_v^{\vote(k+2)}.h \geq h+1$ uniformly via the cert-at-$h+1$ from Step~6.}\hfill

\emph{Step 6: synchrony lifts $R$ to height~$\geq h+1$ uniformly at $\vote(k+2)$.}
By case-$2.A.B$, $v_*$'s canonical at $\vote(k+1)$ contains a block~$B^{(h+2)}$ with $\sigma[B^{(h+2)}].h = h+2$. The state-height advance from~$h+1$ to~$h+2$ during $B^{(h+2)}$'s on-chain processing fires a cert event at height~$h+1$ via \processHeight{}, witnessed by votes embedded in $B^{(h+2)}$'s ancestor blocks. The cert is unique on the chain through~$B^{(h+2)}$ (\Cref{rev:lem:notar-unique-per-slot} under $f<n/3$, hypothesis~1) with target on~$D'$ at chain-height~$h+1$ (since $B^{(h+2)} \preceq \getConfirmed_{v_*}^{\vote(k+1)} \in D'$, by Step~3's classification).

By GST + slot spacing (hypothesis~2), $B^{(h+2)}$ and every ancestor in $v_*$'s view at $\vote(k+1)$ are delivered to every honest by $\vote(k+1) + \Delta \leq \vote(k+2) - \Delta$. Each honest~$v$ folds every received block through \onBlock{} (\Cref{alg:store}, regardless of fork-choice descent), so the on-chain cert at~$h+1$ fires identically in $v$'s view (deterministic from chain history) and is offered to \updateForkChoiceRoot{}. Its key $(h+1, b, s_n, \ldots)$ strictly dominates the cached key $(h, 0, k-1, \ldots)$ from Step~1 on the leading coordinate, so $R_v$ updates to the cert's target on the chain through~$B^{(h+2)}$. Hence $R_v^{\vote(k+2)}.h \geq h+1$ uniformly across honest, with $R_v^{\vote(k+2)} \in D'$ at chain-height~$h+1$. Any bit-$1$ cert at height~$h$ that may fire in $[\vote(k+1),\, \vote(k+2)]$ has key $(h, 1, 0, \ldots)$ dominated on the leading coordinate by $R_v^{\vote(k+2)}$'s key and does not affect the final state.

\emph{Step 7: setup at $\vote(k+1)$.}
By Step~4, honest $\getConfirmed_v^{\vote(k+1)}$ are pairwise compatible on~$D'$. Some honest are at state-height~$h+1$, others at~$h+2$ (case-$2.A.B$). Slot-$(k+1)$ honest votes:
\begin{itemize}
  \item For $v$ at state-height~$h+1$: cast at height~$h+1$, R1 unlocked (hypothesis~8 excludes before-$\vote(k)$ honest at $>h$; slot-$k$ R1 honest votes at~$h$ have $\finalizeHeight \leq h-1$, slot-$k$ R1 honest votes at~$h+1$ have $\finalizeHeight \leq h$, neither equals~$h+1$). Target $T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)} \in D'$.
  \item For $v$ at state-height~$h+2$: cast at height~$h+2$, R1 unlocked (analogous: no prior vote with $\finalizeHeight = h+2$, since :nh excludes before-$\vote(k)$ at $>h$ and slot-$k$ honest~$\finalizeHeight \leq h$). Target $T_v^{(k+1)} = \getConfirmed_v^{\vote(k+1)} \in D'$ at state-height~$h+2$.
\end{itemize}

\emph{Step 8: pairwise compatibility at $\vote(k+2)$.}
\Cref{rev:prop:getConfirmed-compat} applies at slot~$k+1$. Standing assumptions: hypothesis~4 (\OHM at $k+1$); Step~4 ($R$-pairwise compat on~$D'$ at $\vote(k+1)$ via the store invariant). Premise: pick~$v_\ell'$ honest with $\getConfirmed_{v_\ell'}^{\vote(k+1)}$ the $\preceq$-max honest \getConfirmed{} on~$D'$ at $\vote(k+1)$ ($\sigma$-height $h+2$ by case-$2.A.B$). For every honest~$v$ and $t \in [\vote(k+1),\, \vote(k+2)]$, $R_v^t$ has cert key with leading coordinate $\leq h+1$ (\Cref{lem:Rs-key-monotone} + Step~6) and target on~$D'$ at chain-height~$\leq h+1$: the cached at $\vote(k+1)$ lies on~$D'$ at $\leq h+1$ (Step~5), and once $B^{(h+2)}$ is processed in $v$'s view (within $\Delta$ of $\vote(k+1)$, well before $\vote(k+2)$) $R_v$ takes the cert at~$h+1$ from $B^{(h+2)}$'s history on~$D'$ at~$h+1$ (Step~6) and stays there for the remainder of the interval. Hence $R_v^t \preceq \getConfirmed_{v_\ell'}^{\vote(k+1)}$, discharging the premise. The conclusion gives pairwise compatibility of $\getConfirmed_v^{\vote(k+2)}$ on a chain~$D''$ extending~$D'$.

\emph{Step 9: healed-state conditions at slot~$k+2$ with height~$h+1$.}
Pick~$v_\ell$ honest with $\getConfirmed_{v_\ell}^{\vote(k+2)} = \max_\preceq$ on~$D''$ among honest \getConfirmed{} at $\vote(k+2)$.
\begin{itemize}
  \item \emph{\ref{rev:def:healed-state:sync}}: $\start(k+2) - \Delta \geq \GST$ from hypothesis~2 and slot spacing.
  \item \emph{\ref{rev:def:healed-state:R}}: $R_v^{\vote(k+2)}.h \geq h+1$ uniformly by Step~6.
  \item \emph{\ref{rev:def:healed-state:gt-compat}}: pairwise compatibility of honest $\getConfirmed_v^{\vote(k+2)}$ on~$D''$ by Step~8.
  \item \emph{\ref{rev:def:healed-state:target-cap}}: any honest vote at height~$\geq h+1$ cast strictly before $\vote(k+2)$ has target $\preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$:
  \begin{itemize}
    \item before-$\vote(k)$ honest at $\geq h+1$: ruled out by hypothesis~8.
    \item slot-$k$ honest at $\geq h+1$: target $\getConfirmed_v^{\vote(k)} \in D \subseteq D''$ (Step~5 chain extension), $\preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$ by maximality.
    \item slot-$(k+1)$ honest at $\geq h+1$: target $\getConfirmed_v^{\vote(k+1)} \in D' \subseteq D''$ by Step~7 and Step~8's chain extension, $\preceq \getConfirmed_{v_\ell}^{\vote(k+2)}$ by maximality.
  \end{itemize}
\end{itemize}
The system is in a healed state at slot~$k+2$ with height~$h+1$.
\end{description}

\item[{Case 2.B: a bit-$1$ cert at height~$h$ has fired in some honest's view by $\vote(k+1)$.}]\hfill

By view-merge under honest $\syncer(k+1)$ (hypothesis~3, \Cref{rev:ass:view-merge}), every honest's store at $\vote(k+1)$ equals the common $\comm{\Sigma}^{\vote(k+1)}$; the bit-$1$ cert at~$h$ targeting some $J^*$ is in every honest's view, with $R_v^{\vote(k+1)} = J^*$ uniformly (the bit-$1$ key $(h, 1, 0, \ldots)$ dominates the cached bit-$0$ at the same height from Step~1). Hence $\comm{R}^{\vote(k+1)} = J^*$ is justified at height~$h$ in every honest's view at $\vote(k+1)$.

The hypotheses of \Cref{rev:lem:healing-from-justified} apply at slot~$k+1$ with $\comm{R}^{\vote(k+1)} = J^*$:
\begin{itemize}
  \item \ref{rev:hfj:f}: $f^{\vote((k+1)+2)} = f^{\vote(k+3)} < n/3$ from outer hypothesis~1.
  \item \ref{rev:hfj:sync}: $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$ from outer hypothesis~2 + slot spacing.
  \item \ref{rev:hfj:ohm}: \OHM at slots~$k+1$ and~$k+2$ from outer hypothesis~4.
  \item \ref{rev:hfj:aha}: \AHA at slots~$k+1$ and~$k+2$ from outer hypothesis~5.
  \item \ref{rev:hfj:har}: \HAR at slot~$k+1$ from outer hypothesis~6.
  \item \ref{rev:hfj:r}: $R_v^{\vote(k+1)} = J^*$ uniformly justified at height~$h$, just established.
  \item \ref{rev:hfj:nh}: no honest validator has cast a vote at height~$> h$ strictly before $\vote(k+1)$. Outer hypothesis~8 excludes before-$\vote(k)$ honest at $>h$, and no voting event lies in $(\vote(k),\, \vote(k+1))$. \textbf{However}, Step~2 of Case~$2$ admits slot-$k$ honest votes at height~$h+1$ (those with $\sigma[\getConfirmed_v^{\vote(k)}].h = h+1$), cast at $\vote(k) < \vote(k+1)$. \emph{Flag: \ref{rev:hfj:nh} as currently stated is violated by these slot-$k$ honest votes at $h+1$, so Case~$2.B$ is not a verbatim replay of Case~$1.B$.} A clean discharge would either (a)~refine \Cref{rev:lem:healing-from-justified} so that \ref{rev:hfj:nh}'s exclusion is at heights $>\comm{R}^{\vote(k)}.h+1$ (admitting slot-$k$ honest at the next height), or (b)~separately argue that the slot-$k$ honest at $h+1$ on $\comm{\Sigma}^{\vote(k+1)}$ form a slot-$k$ same-slot cert at~$h+1$ during view-merge processing, advancing $\comm{R}^{\vote(k+1)}$ to height~$h+1$ uniformly and re-applying \Cref{rev:lem:healing-from-justified} with $\comm{h} = h+1$ (under which \ref{rev:hfj:nh} holds vacuously).
\end{itemize}

Subject to resolving the flagged hypothesis-mismatches, invoking \Cref{rev:lem:healing-from-justified} at slot~$k+1$ places the system in a healed state at slot~$(k+1)+2 = k+3$ with height~$h+1$.
\end{description}
\end{description}

\subsection{Old Healing advances by $N$ slots}%\label{rev:sec:healing-advances}

\begin{lemma}[Healing: one-slot advance]\label{rev:lem:healing-one-slot}
Assume:
\begin{enumerate}
  \item\label{rev:hone:f} $f^{\vote(k+1)} < n/3$ and all honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hone:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:hone:ohm} the \OHM holds at slot~$k$;
  \item\label{rev:hone:aha} the \AHA condition holds at slot~$k$;
  \item\label{rev:hone:gt} for every pair of honest validators $v, v'$, $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$;
  \item\label{rev:hone:sh} $\sigma[\getConfirmed_v^{\vote(k)}].h$ is the same for every honest validator~$v$; denote this common value by $\comm{h}$;
  \item\label{rev:hone:nh} no honest validator has cast any vote at height $\geq \comm{h}$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then the system is in a healed state at slot $k+1$ with height $\comm{h}$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
\emph{Step 1: every honest casts a \just vote at height $\comm{h}$ at $\vote(k)$.}
By hypothesis~\ref{rev:hone:sh}, $v$'s canonical state-height is $h_c = \sigma[\getConfirmed_v^{\vote(k)}].h = \comm{h}$.
Hypothesis~\ref{rev:hone:nh} rules out any earlier honest vote at height $\geq \comm{h}$, so the (R1) branch of \Cref{def:voting-rule} fires at every honest, producing
\[
  v^{\vote(k)}_i \;=\; \bigl(i,\; \comm{h},\; k,\; T_i,\; \just,\; f_h,\; f_t\bigr),
\]
with $T_i = \getConfirmed_i^{\vote(k)}$ satisfying $\sigma[T_i].h = \comm{h}$.

\emph{Step 2: all honest targets $T_i$ lie on a single chain~$C$.}
Hypothesis~\ref{rev:hone:gt} makes the $T_i$'s pairwise $\preceq$-compatible; combined with hypothesis~\ref{rev:hone:sh} (all at state-height $\comm{h}$), they lie on a single chain~$C$, linearly ordered under~$\preceq$.

\emph{Step 3: slot-$k$ honest votes and their target blocks reach every honest by $\vote(k+1)$.}
Hypothesis~\ref{rev:hone:sync} together with the slot spacing $\vote(k+1) - \vote(k) \geq 2\Delta$ imply that every vote and every block broadcast by an honest validator at or before $\vote(k)$ is delivered to every honest by $\vote(k+1)$.
In particular each $T_i$ is in $\Sigma_{v_i}^{\vote(k)}.\mathcal{T}$ (since $v_i$ voted for $T_i$), hence in every honest's store by $\vote(k+1)$ via store monotonicity.
The virtual-block cert-events rule of \Cref{sec:healing} lets every honest aggregate those received votes into a virtual block extending a canonical chain on~$C$, offering the resulting cert-event descriptors to \updateForkChoiceRoot.

\emph{Step 4: a same-slot cert event on $C$ at state-height $\comm{h}$ fires in every honest's store.}
Fix an honest node~$v$. Its virtual block on chain~$C$ carries all $\geq 2n/3$ honest slot-$k$ \just votes (received by Step~3), together with any dishonest slot-$k$ votes $v$ has received that are fresh on~$C$.
After \processBlock{}, each honest contributor~$i$ satisfies $\sigma.\ntargets[i] = T_i$ and $\sigma.\nslots[i] = k$ (the slot-$k$ vote is the first at height $\comm{h}$, so the slot-gated update fires) and $\sigma.\jtargets[i] = T_i$ (Step~1 votes are \just).
In the revised \processHeight{} of \Cref{alg:state-machine}, the candidate set $\{(B, s) : B \preceq \sigma.L \wedge |\{i : B \preceq \ntargets[i] \preceq \sigma.L \wedge \nslots[i] = s\}| \geq 2n/3\}$ is non-empty at $s = k$: for any $B \in C$ with $B \preceq T_i$ for every honest~$i$, the $\geq 2n/3$ honest slot-$k$ contributors satisfy the count. The $\arg\max$ by~$\preceq$ selects some $(T^\star_v, k)$ with $T^\star_v \in C$ at state-height $\comm{h}$, firing a bit-$0$ cert event with key $(\comm{h},\, 0,\, k,\, T^\star_v.\slot,\, \hash(T^\star_v))$.

If, additionally, $\geq 2n/3$ contributors share a common $\jtargets$ value~$T$, the justification branch of \processHeight{} also fires and produces a bit-$1$ cert event with key $(\comm{h},\, 1,\, 0,\, T.\slot,\, \hash(T))$ at the same height; in that case $T \in C$ since the witnessing $\geq 2n/3 - f$ honest contributions have $\sigma.\jtargets[i] = T_i = T \in C$.

In every case, every cert event at height $\comm{h}$ that fires in $v$'s view has target on~$C$, since $\geq 2n/3$ contributions at height $\comm{h}$ can only come on a chain receiving honest contributions, which target~$C$ (dishonest alone have $\leq f < n/3$).

\emph{Step 5: $R_v$ advances to a cert event at $\comm{h}$; conditions~\ref{rev:def:healed-state:R} and~\ref{rev:def:healed-state:no-just} of the healed state.}
Before any cert event at height $\geq \comm{h}$ fires in $v$'s store, $\Sigma_v.\key_R.h < \comm{h}$: any cert event at height $h^* \geq \comm{h}$ requires a $\geq 2n/3$ quorum of votes at height $h^*$, and hypothesis~\ref{rev:hone:nh} rules out honest contributions at such heights before $\vote(k)$, leaving only dishonest ($\leq f < n/3$) --- insufficient. Hence no cert event at height $\geq \comm{h}$ exists globally before $\vote(k)$, so $\Sigma_v^{\vote(k)}.\key_R.h < \comm{h}$ for every honest~$v$.
In $[\vote(k),\, \vote(k+1))$ no honest vote at height $> \comm{h}$ is cast (Step~1 votes are at $\comm{h}$ exactly; no voting event lies in $(\vote(k),\, \vote(k+1))$), so by the same quorum count no cert event at height $> \comm{h}$ fires; hence by \Cref{lem:Rs-key-monotone} the bound $\Sigma_v.\key_R.h < \comm{h}$ persists until the first cert event at height $\comm{h}$ is processed in $v$'s view.
That cert event's key has leading coordinate $\comm{h} > \Sigma_v.\key_R.h$, so \updateForkChoiceRoot{}'s guard passes; $R_v$ advances to the target of the lex-maximum cert event at height $\comm{h}$ processed by~$v$ in $[\vote(k),\, \vote(k+1)]$.

By Step~4, every such cert event is either the slot-$k$ bit-$0$ notarization at some $T^\star_v \in C$ (key $(\comm{h}, 0, k, \ldots)$) or, when honest $\jtargets$ coincide, the bit-$1$ justification at the common $T \in C$ (key $(\comm{h}, 1, 0, \ldots)$). Bit-$1$ strictly dominates bit-$0$ at the same height, so:
\begin{itemize}
  \item If a justification cert at $\comm{h}$ has fired in $v$'s view by $\vote(k+1)$, $R_v^{\vote(k+1)}$ is that justified block (in particular justified at height $\comm{h}$).
  \item Otherwise $R_v^{\vote(k+1)} = T^\star_v$, which is bit-$0$ notarized at height $\comm{h}$ with notarization slot~$k$.
\end{itemize}
No cert event at height $> \comm{h}$ fires in the interval, so $R_v^{\vote(k+1)}.h = \comm{h}$ and $R_v^{\vote(k+1)}$ is either justified or notarized for slot~$k = (k+1)-1$, proving condition~\ref{rev:def:healed-state:R} of the healed state at slot~$k+1$ with height~$\comm{h}$.

\emph{Step 6: no honest \just vote at height $> \comm{h}$ before $\vote(k+1)$.}
Hypothesis~\ref{rev:hone:nh} rules out honest votes at height $\geq \comm{h}$ before $\vote(k)$; at $\vote(k)$ every honest casts at height $\comm{h}$ exactly; no voting event occurs in $(\vote(k),\, \vote(k+1))$.
Hence no honest has cast a \just vote at height $> \comm{h}$ strictly before $\vote(k+1)$, proving condition~\ref{rev:def:healed-state:no-just} of the healed state.

\emph{Step 7: any future justification at height $\comm{h}$ is bounded by an honest \getConfirmed{} at $\vote(k+1)$; condition~\ref{rev:def:healed-state:just-bound} of the healed state.}
By R1 (\Cref{def:voting-rule}), each honest casts a \just vote at height~$\comm{h}$ at most once across all time. Hypothesis~\ref{rev:hone:nh} excludes before-$\vote(k)$ honest \just at~$\comm{h}$, and Step~1 places exactly one such \just vote per honest at $\vote(k)$ with target $T_i = \getConfirmed_i^{\vote(k)}$. Hence for every honest~$i$, $i$'s sole \just contribution at height~$\comm{h}$ across all time is the slot-$k$ vote with target $T_i$.

Let $C$ be a checkpoint at height~$\comm{h}$ justified at any time, with witnessing $\geq 2n/3$ \just-quorum at target~$C$. The quorum has $\geq 2n/3 - f \geq 1$ honest contributors (under $f < n/3$); any such contributor~$i$ has $T_i = C$, i.e.\ $\getConfirmed_i^{\vote(k)} = C$. By target-monotonicity (\Cref{rev:prop:target-monotonicity}; preamble and premise discharged as in Step~10 below), $\getConfirmed_i^{\vote(k+1)} \succeq \getConfirmed_i^{\vote(k)} = C$. Condition~\ref{rev:def:healed-state:just-bound} of the healed state at slot~$k+1$ holds with $v_\ell = i$.

\emph{Step 8: condition~\ref{rev:def:healed-state:target-cap} of the healed state.}
Hypothesis~\ref{rev:hone:nh} rules out any honest vote at height $\geq \comm{h}$ before $\vote(k)$, hence in particular at height~$> \comm{h}$. Step~1 places slot-$k$ honest votes at height~$\comm{h}$ exactly, not~$> \comm{h}$. No voting event lies in $(\vote(k),\, \vote(k+1))$. Hence no honest vote at height~$> \comm{h}$ exists strictly before $\vote(k+1)$, and condition~\ref{rev:def:healed-state:target-cap} holds vacuously.

\emph{Step 9: condition~\ref{rev:def:healed-state:notar-bound} of the healed state.}
If $R_v^{\vote(k+1)}$ is justified at height~$\comm{h}$ (Step~5 first bullet), the antecedent of condition~\ref{rev:def:healed-state:notar-bound} is false and the condition holds vacuously. Otherwise $R_v^{\vote(k+1)} = T^\star_v$ is bit-$0$ notarized at~$\comm{h}$ with notarization slot $k_n^v = k$ (Step~5 second bullet), and $k_n^v = k < k+1$ holds. For coverage, the only relevant $k'$ is~$k$: any bit-$0$ notarization at~$\comm{h}$ for slot~$k$ targeting~$C$ is a slot-$k$ same-slot cert (Step~4) satisfying $C \preceq T_i$ for every honest contributor~$i$ (the prefix-notar $\arg\max$ in \processHeight{} lies below every contributing target), with $T_i = \getConfirmed_i^{\vote(k)}$. The witnessing quorum has $\geq 2n/3 - f \geq 1$ honest contributors (under $f < n/3$); pick any such~$i$. By target-monotonicity (\Cref{rev:prop:target-monotonicity}; discharged as in Step~10 below), $\getConfirmed_i^{\vote(k+1)} \succeq \getConfirmed_i^{\vote(k)} = T_i \succeq C$.

\emph{Step 10: pairwise compatibility of \getConfirmed{}s at $\vote(k+1)$; condition~\ref{rev:def:healed-state:gt-compat} of the healed state.}
\Cref{rev:prop:getConfirmed-compat}'s preamble requires $R_v^{\vote(k)} \sim R_{v'}^{\vote(k)}$ pairwise: hypothesis~\ref{rev:hone:gt} places all honest $\getConfirmed_v^{\vote(k)}$ on the common chain~$C$ at state-height~$\comm{h}$ (Step~2), and each $R_v^{\vote(k)}$ is an ancestor of $\getConfirmed_v^{\vote(k)}$ on~$C$ (by \Cref{alg:store}), so all honest $R_v^{\vote(k)}$ lie on~$C$ and are pairwise compatible.
The property's premise — existence of an honest $v_\ell$ with $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R_v^t$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$ — is discharged by picking $v_\ell$ with $\getConfirmed_{v_\ell}^{\vote(k)} = \max_\preceq \{\getConfirmed_i^{\vote(k)} : i \text{ honest}\}$ on~$C$ at state-height~$\comm{h}$: for every honest~$v$, $R_v^{\vote(k)} \in C$ at chain-height $< \comm{h}$ (Step~5) and any cert event updating $R_v$ in the interval has target on~$C$ at state-height~$\comm{h}$ (Step~4), so $R_v^t \in C$ at chain-height $\leq \comm{h}$ throughout, yielding $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R_v^t$.
The property's conclusion gives $\getConfirmed_v^{\vote(k+1)} \sim \getConfirmed_{v'}^{\vote(k+1)}$ for every honest pair, proving condition~\ref{rev:def:healed-state:gt-compat} of the healed state.

Condition~\ref{rev:def:healed-state:sync} (synchrony at slot~$k+1$) holds since $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$ by hypothesis~\ref{rev:hone:sync} and the slot spacing $\start(k+1) - \start(k) \geq 2\Delta$. All seven conditions of the healed state at slot~$k+1$ with height~$\comm{h}$ have been verified, so the lemma follows.
\end{proof}

\begin{lemma}[Two-slot healing from a notarized root, no justification completing]\label{rev:lem:healing-from-notarized-no-just}
Assume:
\begin{enumerate}
  \item\label{rev:hfnnj:f} $f^{\vote(k+2)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hfnnj:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:hfnnj:ohm} the \OHM holds at slots~$k$ and $k+1$;
  \item\label{rev:hfnnj:aha} the \AHA condition holds at slots~$k$ and $k+1$;
  \item\label{rev:hfnnj:har} the \HAR condition holds at slot~$k$;
  \item\label{rev:hfnnj:gt} $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getConfirmed_v^{\vote(k)}].h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item\label{rev:hfnnj:r} for every honest validator~$v$, $R_v^{\vote(k)}$ is a bit-$0$ notarized block at cert-key height $\comm{h}$ with notarization slot $k-1$;
  \item\label{rev:hfnnj:nh} no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$;
  \item\label{rev:hfnnj:nb1} no bit-$1$ cert at height $\comm{h}$ has fired in any honest's view at any time in $[\vote(k),\, \vote(k+1)]$.
\end{enumerate}
Then the system is in a healed state at slot $k+2$ with height $\comm{h}+1$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: structural facts at $\vote(k)$.}
By hypothesis~\ref{rev:hfnnj:gt}, the honest $\getConfirmed_v^{\vote(k)}$ are pairwise $\sim$-compatible and all at state-height~$h$; let~$D$ be the chain extending them. Each $\getConfirmed_v^{\vote(k)} \in D$ at state-height~$h$ (possibly at distinct chain-heights, since state-height is monotone non-decreasing along chain ancestry).
By \Cref{alg:store}, each $\getConfirmed_v^{\vote(k)}$ descends from $R_v^{\vote(k)}$, so $R_v^{\vote(k)} \in D$ as well; consistently, hypothesis~\ref{rev:hfnnj:r} places each $R_v^{\vote(k)}$ at a bit-$0$ slot-$(k-1)$ notarization at cert-key height~$h$, and by \Cref{rev:lem:notar-unique-per-slot} these notarizations are pairwise compatible (also putting the $R_v$ on~$D$). Each honest~$v$ has cached $\Sigma_v.\key_R = (h,\, 0,\, k-1,\, R_v.\slot,\, \hash(R_v))$; the first three coordinates are uniform, the fourth and fifth vary with~$R_v$.
Hypothesis~\ref{rev:hfnnj:nh} rules out any earlier honest vote at height $>h$, so at $\vote(k)$ every honest casts an (R1) \just vote at height~$h$ with target $T_v = \getConfirmed_v^{\vote(k)} \in D$. Let $T^\dagger \coloneqq \max_\preceq \{T_v : v \text{ honest}\}$ on~$D$.

\emph{Step 2: a slot-$k$ same-slot bit-$0$ cert at height~$h$ on~$D$ fires by $\vote(k+1)$ in every honest's view.}
By synchrony, every slot-$k$ honest \just vote (which also constitutes a slot-$k$ \notar contribution) is delivered to every honest by $\vote(k+1)$. In each honest~$v$'s aggregation, the prefix-notar $\arg\max$-by-$\preceq$ in \processHeight{} (\Cref{alg:state-machine}) produces a candidate $(B^\star_v, k)$ at state-height~$h$ with $B^\star_v \preceq T_i$ for $\geq 2n/3$ honest~$i$, hence $B^\star_v \in D$ (the common chain of honest targets) and $B^\star_v \preceq T^\dagger$. The candidate fires a bit-$0$ cert event with key $(h,\, 0,\, k,\, B^\star_v.\slot,\, \hash(B^\star_v))$. The choice $B^\star_v$ may differ across stores (depending on $\sigma_v.L$), but each is on~$D$ at state-height~$h$.

By the same dishonest-only-bound argument (hypothesis~\ref{rev:hfnnj:nh} excludes honest at $>h$ before $\vote(k)$; Step~1 places slot-$k$ honest votes at height~$h$ exactly; no voting event in $(\vote(k), \vote(k+1))$; dishonest alone has $\leq f < 2n/3$), no cert event at height $>h$ fires in any honest's view in $[\vote(k),\, \vote(k+1)]$.

\emph{Step 3: every honest's $R_v^t$ stays on~$D$ with $R_v^t \preceq T^\dagger$ throughout $[\vote(k),\, \vote(k+1)]$.}
By Step~1's per-honest cached key $(h,\, 0,\, k-1,\, R_v.\slot,\, \hash(R_v))$ and \Cref{lem:Rs-key-monotone}, every cert event that updates $R_v$ during $[\vote(k),\, \vote(k+1)]$ has key $(h',\, b',\, s',\, \ldots)$ strictly dominating it. The options are:
\begin{itemize}
  \item $h' > h$: ruled out by Step~2.
  \item $h' = h$, $b' = 1$: ruled out by hypothesis~\ref{rev:hfnnj:nb1}.
  \item $h' = h$, $b' = 0$, $s' > k-1$: then $s' \geq k$. Contributions at slot~$\geq k$ do not exist before $\vote(k)$, so the witnessing single-slot quorum at slot~$s'$ at height~$h$ is the slot-$k$ cert from Step~2 (target on~$D$ with target $\preceq T^\dagger$ by the prefix-notar $\arg\max$ argument).
\end{itemize}
Hence every cert updating $R_v$ during the interval is a slot-$k$ bit-$0$ notarization on~$D$ at target $\preceq T^\dagger$. Combined with $R_v^{\vote(k)} \in D$ at state-height~$h$ (Step~1), every $R_v^t$ in the interval lies on~$D$ with $R_v^t \preceq T^\dagger$.

\emph{Step 4: canonical state-height reaches $h+1$ by $\vote(k+1)$.}
Pick $v_\ell$ with $\getConfirmed_{v_\ell}^{\vote(k)} = T^\dagger$. By Step~3, $T^\dagger \succeq R_v^t$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$, discharging the premise of \Cref{rev:prop:honest-vote-inclusion} (whose preamble's compatibility hypothesis is satisfied since all honest $R_v^{\vote(k)}$ lie on~$D$ by Step~1, hence pairwise compatible). Its conclusion gives that the chain ending at $\getConfirmed_v^{\vote(k+1)}$ contains every slot-$k$ honest vote; the slot-$k$ cert-event firing block on~$D$ (Step~2) therefore lies on that chain, and its height-advance places $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h + 1$ for every honest~$v$.

\emph{Step 5: reduce to the one-slot advance at slot~$k+1$.}
The hypotheses of \Cref{rev:lem:healing-one-slot}, applied with~$k$ replaced by~$k+1$, are all met:
\begin{itemize}
  \item hypotheses~\ref{rev:hone:f} and~\ref{rev:hone:sync} are inherited (synchrony at slot~$k+1$: $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$);
  \item hypothesis~\ref{rev:hone:ohm} (\OHM at slot~$k+1$): from outer's hypothesis~\ref{rev:hfnnj:ohm};
  \item hypothesis~\ref{rev:hone:aha} (\AHA at slot~$k+1$): from outer's hypothesis~\ref{rev:hfnnj:aha};
  \item hypothesis~\ref{rev:hone:gt} (\getConfirmed{} pairwise compatibility at $\vote(k+1)$): \Cref{rev:prop:getConfirmed-compat}, with premise discharged as in Step~4, gives $\getConfirmed_v^{\vote(k+1)} \sim \getConfirmed_{v'}^{\vote(k+1)}$ for every honest pair;
  \item hypothesis~\ref{rev:hone:sh} (common state-height): $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h + 1$ is the common value across honest validators (Step~4);
  \item hypothesis~\ref{rev:hone:nh} (no honest vote at height $\geq h+1$ before $\vote(k+1)$): outer's hypothesis~\ref{rev:hfnnj:nh} rules out before-$\vote(k)$ honest votes at height $>h$ (a fortiori at $\geq h+1$); Step~1 places slot-$k$ honest votes at height~$h$ exactly; no voting event lies in $(\vote(k),\, \vote(k+1))$.
\end{itemize}
Invoking \Cref{rev:lem:healing-one-slot} at slot~$k+1$ places the system in a healed state at slot~$k+2$ with height~$h+1$, as required.
\end{proof}


\begin{lemma}[Two-slot healing from a notarized common root]\label{rev:lem:healing-from-notarized}
Assume:
\begin{enumerate}
  \item\label{rev:hfn:f} $f^{\vote(k+2)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hfn:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:hfn:syncers} $\syncer(k+1)$ is honest; let $\comm{\Sigma}^{\vote(k+1)}$ be the common store at $\vote(k+1)$;
  \item\label{rev:hfn:ohm} the \OHM holds at slots~$k+1$ and $k+2$;
  \item\label{rev:hfn:aha} the \AHA condition holds at slots~$k$, $k+1$, and $k+2$;
  \item\label{rev:hfn:har} the \HAR condition holds at slot~$k+1$;
  \item\label{rev:hfn:gt} $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getConfirmed_v^{\vote(k)}].h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item\label{rev:hfn:r} for every honest validator~$v$, $R_v^{\vote(k)}$ is a bit-$0$ notarized block at cert-key height $\comm{h}$ with notarization slot $k-1$;
  \item\label{rev:hfn:nh} no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then the system is in a healed state at slot $k+2$ with height $\comm{h}+1$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$. By view-merge + honest $\syncer(k+1)$ (hypothesis~\ref{rev:hfn:syncers}), all honest stores at $\vote(k+1)$ equal $\comm{\Sigma}^{\vote(k+1)}$.

\emph{Case 1: a bit-$1$ cert at height~$h$ has fired in $\comm{\Sigma}^{\vote(k+1)}$.}
Then $\comm{R}^{\vote(k+1)}$ is a justified block at height~$h$ in every honest's view at $\vote(k+1)$. The setup matches the hypotheses of \Cref{rev:lem:healing-from-justified} applied at slot~$k+1$: by view-merge, $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is justified in every honest's view, discharging hypothesis~\ref{rev:hfj:r} of from-justified; honest \getConfirmed{}s at $\vote(k+1)$ coincide (hence pairwise compatible) and remain at state-height~$h$ (the bit-$1$ cert's height coordinate is~$h$ in cert-key, and its height-advance fires only virtually under view-merge), discharging hypothesis~\ref{rev:hfj:gt}; outer's hypothesis~\ref{rev:hfn:nh} (no honest at $>h$ before $\vote(k)$) plus the slot-$k$ honest votes at height~$h$ exactly rules out honest votes at height $>h$ before $\vote(k+1)$, discharging hypothesis~\ref{rev:hfj:nh}. Invoking \Cref{rev:lem:healing-from-justified} places the system in a healed state at slot~$k+2$ with height~$h+1$.

\emph{Case 2: no bit-$1$ cert at height~$h$ in $\comm{\Sigma}^{\vote(k+1)}$.}
By view-merge, no honest's view contains a bit-$1$ cert at height~$h$ at any time in $[\vote(k),\, \vote(k+1)]$ (cert events are persistent within a store, and view-merge unions stores; if any honest had such a cert at any earlier time, it would appear in $\comm{\Sigma}^{\vote(k+1)}$). The hypotheses of \Cref{rev:lem:healing-from-notarized-no-just} are all met: \ref{rev:hfnnj:f} and~\ref{rev:hfnnj:sync} are inherited; \ref{rev:hfnnj:ohm}, \ref{rev:hfnnj:aha}, \ref{rev:hfnnj:har} are outer's \ref{rev:hfn:ohm}, \ref{rev:hfn:aha}, \ref{rev:hfn:har}; \ref{rev:hfnnj:gt}, \ref{rev:hfnnj:r}, \ref{rev:hfnnj:nh} are outer's \ref{rev:hfn:gt}, \ref{rev:hfn:r}, \ref{rev:hfn:nh}; \ref{rev:hfnnj:nb1} is just established. Invoking it places the system in a healed state at slot~$k+2$ with height~$h+1$.
\end{proof}

\begin{lemma}[Three-slot healing]\label{rev:lem:healing-three-slot}
Assume:
\begin{enumerate}
  \item\label{rev:h3s:f} $f^{\vote(k+3)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:h3s:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:h3s:syncers} $\syncer(k+1)$ and $\syncer(k+2)$ are honest;
  \item\label{rev:h3s:ohm} the \OHM holds at slots~$k+1$, $k+2$, and $k+3$;
  \item\label{rev:h3s:aha} the \AHA condition holds at slots~$k, k+1, k+2$, and $k+3$;
  \item\label{rev:h3s:har} the \HAR condition holds at slots~$k+1$ and $k+2$;
  \item\label{rev:h3s:gt} $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getConfirmed_v^{\vote(k)}].h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item\label{rev:h3s:nh} no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then the system is in a healed state at slot $k+3$ with height $\comm{h}+1$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: slot-$k$ honest votes form a cert at height~$h$ on a common chain.}
By hypothesis~\ref{rev:h3s:gt}, the honest $\getConfirmed_v^{\vote(k)}$ are pairwise compatible and at state-height~$h$; let~$C$ be the chain extending them. Each $\getConfirmed_v^{\vote(k)} \in C$ at state-height~$h$ (possibly at distinct chain-heights, since state-height is monotone non-decreasing along chain ancestry). Hypothesis~\ref{rev:h3s:nh} rules out any earlier honest vote at height $>h$, so at $\vote(k)$ every honest casts an (R1) \just vote at height~$h$ with target $T_i = \getConfirmed_i^{\vote(k)} \in C$. The $\geq 2n/3$ honest slot-$k$ \just votes (which also constitute slot-$k$ \notar contributions on~$C$) together with any slot-$k$ dishonest contributions fire a same-slot cert event at height~$h$ on~$C$ (bit-$1$ justification if the $T_i$ all coincide, bit-$0$ notarization at the prefix-notar $\arg\max$-by-$\preceq$ block on~$C$ otherwise).

\emph{Step 2: case split on $\comm{R}^{\vote(k+1)}$.}
Under view-merge + honest $\syncer(k+1)$, honest share $\comm{\Sigma}^{\vote(k+1)}$ at $\vote(k+1)$. By \Cref{rev:cor:no-high-cn} applied to $\comm{\Sigma}^{\vote(k+1)}$, no cert event at height $>h$ exists there: hypothesis~\ref{rev:h3s:nh} excludes honest contributions at height $>h$ before $\vote(k)$, Step~1 places slot-$k$ honest contributions at height~$h$ exactly, no voting event lies in $(\vote(k),\, \vote(k+1))$, and dishonest alone ($\leq f < n/3$) cannot form a $\geq 2n/3$ quorum.

\emph{Case 1: $\comm{R}^{\vote(k+1)}$ is a justified block at height~$h$.}
The setup at $\vote(k+1)$ satisfies the hypotheses of \Cref{rev:lem:healing-from-justified} applied at slot~$k+1$: synchrony $\start(k+1) - \Delta \geq \GST$ (from outer's hypothesis~\ref{rev:h3s:sync}); by view-merge + honest $\syncer(k+1)$ (outer's hypothesis~\ref{rev:h3s:syncers}), $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is justified in every honest's view (discharging hypothesis~\ref{rev:hfj:r}); honest \getConfirmed{}s at $\vote(k+1)$ coincide (hence pairwise compatible) and sit at state-height~$h$ (the justification's height-advance fires only virtually in $\comm{\Sigma}^{\vote(k+1)}$ and is not persisted to any actual block), discharging hypothesis~\ref{rev:hfj:gt}; and outer's hypothesis~\ref{rev:h3s:nh} together with Step~1 rules out any honest vote at height $>h$ before $\vote(k+1)$, discharging hypothesis~\ref{rev:hfj:nh}. Invoking \Cref{rev:lem:healing-from-justified} places the system in a healed state at slot~$k+3$ with height~$h+1$.

\emph{Case 2: $\comm{R}^{\vote(k+1)}$ is not a justified checkpoint.}
Then the cached cert at $\vote(k+1)$ is bit-$0$. By the same no-cert-above-$h$ argument its height is $\leq h$; among bit-$0$ certs at height~$h$, the slot-$k$ notarization of Step~1 strictly dominates any slot-$<k$ cert on the $s_n$ coordinate, and contributions at slot $\geq k$ do not exist before $\vote(k)$ for $s_n > k$ to occur. Hence $\comm{R}^{\vote(k+1)}$ is a bit-$0$ notarization at height~$h$ with notarization slot~$k$.

The setup at $\vote(k+1)$ satisfies the hypotheses of \Cref{rev:lem:healing-from-notarized} applied at slot~$k+1$: synchrony; $\syncer(k+2)$ honest (outer's hypothesis~\ref{rev:h3s:syncers}), discharging hypothesis~\ref{rev:hfn:syncers}; by view-merge + honest $\syncer(k+1)$ (outer's hypothesis~\ref{rev:h3s:syncers}), every honest's store at $\vote(k+1)$ coincides with $\comm{\Sigma}^{\vote(k+1)}$, so honest \getConfirmed{}s at $\vote(k+1)$ are equal (hence pairwise compatible) and at state-height~$h$ (same virtual-only argument as Case~1), discharging hypothesis~\ref{rev:hfn:gt}; by view-merge, $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is bit-$0$ notarized at height~$h$ with slot $k = (k+1)-1$ in every honest's view, discharging hypothesis~\ref{rev:hfn:r}; and outer's hypothesis~\ref{rev:h3s:nh} with Step~1 rules out honest votes at height $>h$ before $\vote(k+1)$, discharging hypothesis~\ref{rev:hfn:nh}. Invoking \Cref{rev:lem:healing-from-notarized} places the system in a healed state at slot~$k+3$ with height~$h+1$.
\end{proof}

\begin{lemma}[Three-slot healing, no justification completing]\label{rev:lem:healing-three-slot-no-just}
Assume:
\begin{enumerate}
  \item\label{rev:h3snj:f} $f^{\vote(k+3)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:h3snj:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:h3snj:ohm} the \OHM holds at slots~$k+1$ and $k+2$;
  \item\label{rev:h3snj:aha} the \AHA condition holds at slots~$k, k+1$, and $k+2$;
  \item\label{rev:h3snj:har} the \HAR condition holds at slot~$k+1$;
  \item\label{rev:h3snj:gt} $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getConfirmed_v^{\vote(k)}].h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item\label{rev:h3snj:nh} no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$;
  \item\label{rev:h3snj:nb1} no bit-$1$ cert at height $\comm{h}$ has fired in any honest's view at any time in $[\vote(k),\, \vote(k+2)]$.
\end{enumerate}
Then the system is in a healed state at slot $k+3$ with height $\comm{h}+1$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: slot-$k$ honest votes form a same-slot bit-$0$ cert at height~$h$ on a common chain.}
By hypothesis~\ref{rev:h3snj:gt}, the honest $\getConfirmed_v^{\vote(k)}$ are pairwise compatible and at state-height~$h$; let~$D$ be the chain extending them. Each $\getConfirmed_v^{\vote(k)} \in D$ at state-height~$h$. Hypothesis~\ref{rev:h3snj:nh} rules out earlier honest votes at height $>h$, so at $\vote(k)$ every honest casts an (R1) \just vote at height~$h$ with target $T_v = \getConfirmed_v^{\vote(k)} \in D$. Let $T^\dagger \coloneqq \max_\preceq \{T_v : v \text{ honest}\}$ on~$D$.

By synchrony, every slot-$k$ honest \just vote (which also constitutes a slot-$k$ \notar contribution) is delivered to every honest by $\vote(k+1)$. In each honest~$v$'s aggregation, the prefix-notar $\arg\max$-by-$\preceq$ in \processHeight{} produces a candidate $(B^\star_v, k)$ at state-height~$h$ on~$D$ with $B^\star_v \preceq T^\dagger$, firing a bit-$0$ cert event with key $(h,\, 0,\, k,\, B^\star_v.\slot,\, \hash(B^\star_v))$. By the dishonest-only-bound argument (hypothesis~\ref{rev:h3snj:nh} excludes honest at $>h$ before $\vote(k)$; slot-$k$ honest votes are at height~$h$ exactly; dishonest alone has $\leq f < 2n/3$), no cert event at height $>h$ fires in any honest's view in $[\vote(k),\, \vote(k+1)]$.

\emph{Step 2: per-honest $R_v^{\vote(k+1)}$ is bit-$0$ notarized on~$D$ at $h$ slot~$k$.}
By hypothesis~\ref{rev:h3snj:nb1}, no bit-$1$ cert at height~$h$ has fired in any honest's view in $[\vote(k),\, \vote(k+1)]$. By \Cref{rev:cor:no-high-cn} applied per-honest, no $C_N$ entry at height $>h$ exists in any honest's view at $\vote(k)$, and \Cref{lem:Rs-key-monotone} gives $\Sigma_v.\key_R.h \leq h$ at $\vote(k)$.
Hypothesis~\ref{rev:h3snj:gt} places $R_v^{\vote(k)}$ on~$D$ as an ancestor of $\getConfirmed_v^{\vote(k)} \in D$, so $R_v^{\vote(k)} \in D$ at chain-height~$\leq h$.
The slot-$k$ bit-$0$ cert at $(h, 0, k, \ldots)$ from Step~1 fires in $v$'s view by $\vote(k+1)$. Cached $\Sigma_v.\key_R$ is either at height $<h$ (dominated by the slot-$k$ key on the leading coordinate) or $(h, 0, s', \ldots)$ with $s' < k$ (slot-$\geq k$ contributions did not exist before $\vote(k)$, and bit-$1$ at~$h$ is excluded by~(5)) — in both cases the slot-$k$ key strictly dominates, so $R_v$ updates to $B^\star_v \in D$ at state-height~$h$. After update, $\Sigma_v.\key_R = (h, 0, k, B^\star_v.\slot, \hash(B^\star_v))$ and $R_v^{\vote(k+1)} = B^\star_v$.

\emph{Step 3: invoke \Cref{rev:lem:healing-from-notarized-no-just} at slot~$k+1$.}
The hypotheses, applied with $k$ replaced by $k+1$, are met:
\begin{itemize}
  \item \ref{rev:hfnnj:f} and~\ref{rev:hfnnj:sync} inherited; \ref{rev:hfnnj:ohm}, \ref{rev:hfnnj:aha}, \ref{rev:hfnnj:har} from outer's \ref{rev:h3snj:ohm}, \ref{rev:h3snj:aha}, \ref{rev:h3snj:har}.
  \item hypothesis~\ref{rev:hfnnj:gt} at $\vote(k+1)$: by Step~2, $R_v^{\vote(k+1)} = B^\star_v \in D$ at state-height~$h$, so the honest $R_v^{\vote(k+1)}$ are pairwise compatible (R-compatibility at $\vote(k+1)$); $\getConfirmed_v^{\vote(k+1)}$ descends from $R_v^{\vote(k+1)}$ at state-height~$h$ (no advancing cert past $B^\star_v$ has fired by $\vote(k+1)$ — hypothesis~\ref{rev:h3snj:nb1} excludes bit-$1$ at~$h$, slot-$\geq k+1$ contributions have not yet materialized at $\vote(k+1)$, and dishonest alone cannot form a $\geq 2n/3$ same-slot quorum). \Cref{rev:prop:getConfirmed-compat} (its preamble's R-compatibility at $\vote(k)$ premise discharged via outer's hypothesis~\ref{rev:h3snj:gt}: $R_v^{\vote(k)} \in D$ pairwise) applied with premise $T^\dagger \succeq R_v^t$ for honest~$v, t \in [\vote(k),\, \vote(k+1)]$ (Steps~1--2 give $R_v^t \in D$ at chain-height~$\leq h$, so $T^\dagger \succeq R_v^t$) gives $\getConfirmed_v^{\vote(k+1)} \sim \getConfirmed_{v'}^{\vote(k+1)}$ pairwise.
  \item hypothesis~\ref{rev:hfnnj:r} at $\vote(k+1)$: each honest~$v$ has $R_v^{\vote(k+1)} = B^\star_v$ bit-$0$ notarized at cert-key height~$h$ with notarization slot~$k = (k+1)-1$ (Step~2).
  \item hypothesis~\ref{rev:hfnnj:nh} at $\vote(k+1)$: hypothesis~\ref{rev:h3snj:nh} rules out honest at $>h$ before $\vote(k)$; slot-$k$ honest votes are at height~$h$ exactly (Step~1); no voting event lies in $(\vote(k),\, \vote(k+1))$; hence no honest cast at height $>h$ before $\vote(k+1)$.
  \item hypothesis~\ref{rev:hfnnj:nb1} at $\vote(k+1)$: hypothesis~\ref{rev:h3snj:nb1} covers $[\vote(k),\, \vote(k+2)] \supseteq [\vote(k+1),\, \vote(k+2)]$.
\end{itemize}
Invoking \Cref{rev:lem:healing-from-notarized-no-just} at slot~$k+1$ places the system in a healed state at slot~$k+3$ with height~$h+1$.
\end{proof}

\begin{lemma}[Four-slot healing]\label{rev:lem:healing-four-slot}
Assume:
\begin{enumerate}
  \item\label{rev:h4s:f} $f^{\vote(k+4)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:h4s:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:h4s:syncers} $\syncer(k+1)$, $\syncer(k+2)$ and $\syncer(k+3)$ are honest;
  \item\label{rev:h4s:ohm} the \OHM holds at slots~$k$, $k+2$, $k+3$, and $k+4$;
  \item\label{rev:h4s:aha} the \AHA condition holds at slots~$k, k+1, k+2, k+3$, and $k+4$;
  \item\label{rev:h4s:har} the \HAR condition holds at slots~$k$, $k+2$, and $k+3$;
  \item\label{rev:h4s:r} for every honest validator~$v$, $R_v^{\vote(k)}$ is the same justified block; denote it by $\comm{R}^{\vote(k)}$;
  \item\label{rev:h4s:gt} $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getConfirmed_v^{\vote(k)}].h = \comm{R}^{\vote(k)}.h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item\label{rev:h4s:nh} no honest validator has cast any vote at height $> \comm{h}+1$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then the system is in a healed state at slot $k+4$ with height $\comm{h}+2$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: canonical state-height reaches $h+1$ at $\vote(k+1)$, common across honest.}
By hypothesis~\ref{rev:h4s:r}, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is justified at height~$h$ in every honest's view; hence $\Sigma_v.\key_R \geq (h, 1, 0, \comm{R}^{\vote(k)}.\slot, \hash(\comm{R}^{\vote(k)}))$ at $\vote(k)$ for every honest~$v$. By hypothesis~\ref{rev:h4s:gt}, the honest $\getConfirmed_v^{\vote(k)}$ are pairwise compatible and at state-height~$h$; let~$D$ be the chain extending them. Each $\getConfirmed_v^{\vote(k)} \in D$ at state-height~$h$, descending from $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$, so $\comm{R}^{\vote(k)} \in D$ at chain-height~$h$ and $D$ extends~$\comm{R}^{\vote(k)}$ at chain-heights~$\geq h$. Hypothesis~\ref{rev:h4s:nh} allows honest votes at heights $\leq h+1$ before $\vote(k)$; in particular, honest may have cast \just votes at height~$h+1$ at earlier slots.

At $\vote(k)$ every honest casts an (R1) \just vote at height~$h$ with target $T_v = \getConfirmed_v^{\vote(k)} \in D$; the prefix-notar $\arg\max$-by-$\preceq$ in \processHeight{} fires a same-slot slot-$k$ cert at some $B^\star \in D$ at state-height~$h$, with key $(h, 0, k, \ldots)$ or $(h, 1, 0, \ldots)$, which is lex-dominated by (or tied to) the cached bit-$1$ key $(h, 1, 0, \ldots)$; this cert does not strictly advance~$R_v$. In $[\vote(k),\, \vote(k+1)]$, no cert event at height $>h+1$ can form (hypothesis~\ref{rev:h4s:nh} combined with Step-1 voting + $f<n/3$), so $\Sigma_v.\key_R.h \leq h+1$ throughout. Two sub-cases arise at $\vote(k+1)$; view-merge + honest $\syncer(k+1)$ makes the outcome common across honest.

\emph{Sub-case A: a cert at height $h+1$ has fired in $\comm{\Sigma}^{\vote(k+1)}$.}
This is possible when pre-existing honest \just votes at $h+1$, together with late dishonest contributions, complete a $\geq 2n/3$ quorum. The resulting cert strictly dominates the cached key, so $\comm{R}^{\vote(k+1)}$ is at cert-key height $h+1$. Canonical state-height at $\vote(k+1)$ is then $h+1$ (no cert event at height $>h+1$ can fire under the same bound).

\emph{Sub-case B: no cert at height $h+1$ has fired.}
Then $\comm{R}^{\vote(k+1)} = \comm{R}^{\vote(k)}$, so $R_v^t = \comm{R}^{\vote(k)}$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$. For any honest $v_\ell$, $\getConfirmed_{v_\ell}^{\vote(k)} \in D$ at state-height~$h$, so $\getConfirmed_{v_\ell}^{\vote(k)} \succeq \comm{R}^{\vote(k)} = R_v^t$ (by \Cref{alg:store}), discharging the premise of \Cref{rev:prop:honest-vote-inclusion}; its conclusion gives that the chain ending at $\getConfirmed_v^{\vote(k+1)}$ contains every slot-$k$ honest vote (all on~$D$ by Step~1 above), and the slot-$k$ cert event at $B^\star \in D$ lies on that chain, advancing state-height to $h+1$, i.e.\ $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h + 1$ for every honest~$v$.

Either way, canonical state-height at $\vote(k+1)$ is $h+1$ and common across honest.

\emph{Step 2: apply \Cref{rev:lem:healing-three-slot} at slot~$k+1$.}
The hypotheses of \Cref{rev:lem:healing-three-slot} applied with $k$ replaced by $k+1$ are all met:
\begin{itemize}
  \item \ref{rev:h3s:f} and~\ref{rev:h3s:sync} inherited (synchrony at slot $k+1$: $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$); \ref{rev:h3s:ohm}, \ref{rev:h3s:aha}, \ref{rev:h3s:har} from outer's \ref{rev:h4s:ohm}, \ref{rev:h4s:aha}, \ref{rev:h4s:har};
  \item hypothesis~\ref{rev:h3s:syncers}: outer's hypothesis~\ref{rev:h4s:syncers} provides $\syncer(k+2), \syncer(k+3)$ honest;
  \item hypothesis~\ref{rev:h3s:gt}: Step~1 gives $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h + 1$ as the common value across honest validators, and by view-merge + honest $\syncer(k+1)$ (outer's hypothesis~\ref{rev:h4s:syncers}) honest \getConfirmed{}s at $\vote(k+1)$ coincide, hence pairwise compatible;
  \item hypothesis~\ref{rev:h3s:nh}: outer's hypothesis~\ref{rev:h4s:nh} rules out honest votes at height $>h+1$ before $\vote(k)$; at $\vote(k)$ every honest casts at height~$h$ (Step~1); no voting event lies in $(\vote(k),\, \vote(k+1))$; hence no honest has cast at height $>h+1$ before $\vote(k+1)$.
\end{itemize}
Invoking \Cref{rev:lem:healing-three-slot} at slot~$k+1$ places the system in a healed state at slot~$(k+1)+3 = k+4$ with height~$(h+1)+1 = h+2$, as required.
\end{proof}

\begin{lemma}[Four-slot healing from a justified common root]\label{rev:lem:healing-four-slot-from-justified}
Assume:
\begin{enumerate}
  \item\label{rev:h4sfj:f} $f^{\vote(k+4)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:h4sfj:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:h4sfj:syncers} $\syncer(k+1)$, $\syncer(k+2)$ and $\syncer(k+3)$ are honest;
  \item\label{rev:h4sfj:ohm} the \OHM holds at every slot $k' \in \{k, k+1, k+2, k+3, k+4\}$;
  \item\label{rev:h4sfj:aha} the \AHA condition holds at every slot $k' \in \{k, k+1, k+2, k+3, k+4\}$;
  \item\label{rev:h4sfj:har} the \HAR condition holds at every slot $k' \in \{k, k+1, k+2, k+3\}$;
  \item\label{rev:h4sfj:r} for every honest validator~$v$, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is a justified block, and $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$; denote $\comm{h} \coloneqq \comm{R}^{\vote(k)}.h$.
\end{enumerate}
Then there exists $k' \in \{k+3, k+4\}$ such that the system is in a healed state at slot $k'$ with height $\comm{h}+2$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h} = \comm{R}^{\vote(k)}.h$.

\emph{Step 1: structural facts at $\vote(k)$.}
By hypothesis~\ref{rev:h4sfj:r}, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is justified at height~$h$ in every honest's view; hence $\Sigma_v.\key_R \geq (h, 1, 0, \comm{R}^{\vote(k)}.\slot, \hash(\comm{R}^{\vote(k)}))$ at $\vote(k)$ for every honest~$v$, and in particular $\Sigma_v.\key_R.h = h$. \Cref{rev:cor:processed-height-bound} applied per-honest to each $\Sigma_v^{\vote(k)}$ bounds every processed block by $\sigma.h \leq R_v^{\vote(k)}.h + 1 = h + 1$, so $\sigma[\getConfirmed_v^{\vote(k)}].h \in \{h, h+1\}$ for every honest~$v$. By \Cref{lem:Rs-key-monotone}, $\Sigma_v.\key_R.h \leq h$ at every earlier time, and \Cref{rev:cor:processed-height-bound} applied per-honest at every earlier time bounds processed blocks by state-height $\leq h+1$; hence no honest validator has cast any vote at height $> h+1$ at any time strictly before $\vote(k)$.

\emph{Step 2: case split on common state-height at $\vote(k)$.}

\emph{Case A: $\sigma[\getConfirmed_v^{\vote(k)}].h = h+1$ for every honest~$v$.}
The hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k$ with the property's $\comm{h}$ taken to be $h+1$: hypothesis~\ref{rev:h3s:gt} (\getConfirmed pairwise compatible from outer's hypothesis~\ref{rev:h4sfj:r} and common state-height from the case assumption); hypothesis~\ref{rev:h3s:nh} (no honest at $> h+1$ before $\vote(k)$) is Step~1; and \ref{rev:h3s:syncers} is inherited. \Cref{rev:lem:healing-three-slot} places the system in a healed state at slot~$k+3$ with height~$(h+1)+1 = h+2$.

\emph{Case B: $\sigma[\getConfirmed_v^{\vote(k)}].h = h$ for every honest~$v$.}
The hypotheses of \Cref{rev:lem:healing-four-slot} apply at slot~$k$: hypothesis~\ref{rev:h4s:syncers}: outer's hypothesis~\ref{rev:h4sfj:syncers} provides $\syncer(k+1), \syncer(k+2), \syncer(k+3)$ honest; hypothesis~\ref{rev:h4s:r}: by outer's hypothesis~\ref{rev:h4sfj:r}, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is the same justified block in every honest's view; hypothesis~\ref{rev:h4s:gt}: honest \getConfirmed{}s at $\vote(k)$ are pairwise compatible (outer's hypothesis~\ref{rev:h4sfj:r}) and at common state-height $= \comm{R}.h$ (case assumption); hypothesis~\ref{rev:h4s:nh} is Step~1. \Cref{rev:lem:healing-four-slot} places the system in a healed state at slot~$k+4$ with height~$h+2$.

\emph{Case C: state-heights at $\vote(k)$ are mixed (some honest at $h$, some at $h+1$).}
Let $T^\dagger$ denote the $\preceq$-maximum honest \getConfirmed{} at $\vote(k)$; by Step~1 and the case assumption, $\sigma[T^\dagger].h = h+1$. We sub-split on $\comm{R}^{\vote(k+1)}.h$ at $\vote(k+1)$ (well-defined and common across honest by view-merge + honest $\syncer(k+1)$).

\emph{Sub-case C.A: $\comm{R}^{\vote(k+1)}.h = h+1$.}
A cert at height $h+1$ has fired in $\comm{\Sigma}^{\vote(k+1)}$. The hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k+1$ with $\comm{h}' = h+1$: canonical state-height at $\vote(k+1)$ is exactly $h+1$ (\Cref{rev:cor:processed-height-bound} bounds it by $\Sigma.\key_R.h + 1 = h+2$, and the $\geq h+1$ direction follows from $\getConfirmed$ descending from~$R$ at height $h+1$); the no-honest-above bound is inherited from Step~1 plus the slot-$k$ honest votes lying at heights $\leq h+1$. \Cref{rev:lem:healing-three-slot} places the system in a healed state at slot~$(k+1)+3 = k+4$ with height~$(h+1)+1 = h+2$.

\emph{Sub-case C.B: $\comm{R}^{\vote(k+1)}.h = h$.}
By justification-uniqueness (\Cref{lem:just-unique-height}) the only justified block at height~$h$ is $\comm{R}^{\vote(k)}$, so $\comm{R}^{\vote(k+1)} = \comm{R}^{\vote(k)}$, giving $R_v^t = \comm{R}^{\vote(k)}$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$. Picking $v_\ell$ with $\getConfirmed_{v_\ell}^{\vote(k)} = T^\dagger$ discharges the premise of \Cref{rev:prop:target-dominance}; its conclusion gives $\getConfirmed_v^{\vote(k+1)} \succeq T^\dagger$ for every honest~$v$, so $\sigma[\getConfirmed_v^{\vote(k+1)}].h \geq h + 1$. \Cref{rev:cor:processed-height-bound} applied to $\comm{\Sigma}^{\vote(k+1)}$ bounds it by $\Sigma.\key_R.h + 1 = h + 1$, so canonical state-height at $\vote(k+1)$ is exactly $h+1$ and common across honest. Apply \Cref{rev:lem:healing-three-slot} at slot~$k+1$ with $\comm{h}' = h+1$ exactly as in Sub-case~C.A, placing the system in a healed state at slot~$k+4$ with height~$h+2$.

In every case, the system is in a healed state at slot~$k' \in \{k+3, k+4\}$ with height~$\comm{h}+2$.
\end{proof}

\begin{lemma}[Four-slot healing from a notarized common root]\label{rev:lem:healing-four-slot-from-notarized}
Assume:
\begin{enumerate}
  \item\label{rev:h4sfn:f} $f^{\vote(k+4)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:h4sfn:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:h4sfn:syncers} $\syncer(k+1)$, $\syncer(k+2)$ and $\syncer(k+3)$ are honest;
  \item\label{rev:h4sfn:ohm} the \OHM holds at every slot $k' \in \{k, k+1, k+2, k+3, k+4\}$;
  \item\label{rev:h4sfn:aha} the \AHA condition holds at every slot $k' \in \{k, k+1, k+2, k+3, k+4\}$;
  \item\label{rev:h4sfn:har} the \HAR condition holds at slots~$k+1$, $k+2$, and $k+3$;
  \item\label{rev:h4sfn:hap} the \HAP condition holds at slot~$k$;
  \item\label{rev:h4sfn:gt} $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$;
  \item\label{rev:h4sfn:r} for every honest validator~$v$, $R_v^{\vote(k)}$ is a bit-$0$ notarized block at cert-key height $\comm{h}$ with notarization slot $k-1$ and there exists an honest validator $v_\ell$ such that $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R_v^{\vote(k)}$.
\end{enumerate}
Then there exists $k' \in \{k+3, k+4\}$ such that the system is in a healed state at slot $k'$ with height $\comm{h}+2$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
\emph{Case A: $\sigma[\getConfirmed_v^{\vote(k)}].h = h$ for every honest~$v$.}
$\vote(k)$. Form a cert for height $h$ and slot $k$.

$\vote(k+1)$. by cases.
Case A.1: $\comm{R}$ is justified for height $h$. Invoke \Cref{rev:lem:healing-four-slot-from-justified}.

Case A.2: $\comm{R}$ is notarized for height $h+1$. Invoke \Cref{rev:lem:healing-three-slot}.

Case A.3: $\comm{R}$ is notarized for height $h$.
Then, it must be for either slot $k-1$ or slot $k$.
In either case, given \ref{rev:h4sfn:r}, we should be able to discharge the condition for \Cref{rev:prop:honest-vote-inclusion}, which implies the votes sent at $\vote(k)$ are included in the $\getConfirmed$ chain of every honest which has therefore advanced to height $\comm{h}+1$. THen apply \Cref{rev:lem:healing-from-notarized}.

  
\end{proof}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: structural facts at $\vote(k)$.}
By hypothesis~\ref{rev:h4sfn:r}, each honest~$v$ has cached $\Sigma_v.\key_R = (h, 0, k-1, R_v.\slot, \hash(R_v))$ at $\vote(k)$, so $R_v^{\vote(k)}.h = h$. By \Cref{rev:lem:notar-unique-per-slot}, the slot-$(k-1)$ notarizations at cert-key height~$h$ are pairwise compatible, so the honest $R_v^{\vote(k)}$ all lie on a single chain at state-height~$h$. \Cref{rev:cor:processed-height-bound} applied per-honest to each $\Sigma_v^{\vote(k)}$ bounds every processed block by state-height $\leq R_v^{\vote(k)}.h + 1 = h + 1$, so $\sigma[\getConfirmed_v^{\vote(k)}].h \in \{h, h+1\}$ for every honest~$v$. Combined with \Cref{lem:Rs-key-monotone} (so $\Sigma_v.\key_R.h \leq h$ at every earlier time) and \Cref{rev:cor:processed-height-bound} applied per-honest at every earlier time, no honest validator has cast any vote at height $> h+1$ at any time strictly before $\vote(k)$.

\emph{Step 2: case split on common state-height at $\vote(k)$.}

\emph{Case A: $\sigma[\getConfirmed_v^{\vote(k)}].h = h$ for every honest~$v$.}

\emph{Step A.1: no cert at height $> h$ fires in $[\vote(k), \vote(k+1)]$.}
By outer~\ref{rev:h4sfn:r} ($R_v^{\vote(k)}.h = h$) and \Cref{lem:Rs-key-monotone}, $\Sigma_v.\key_R.h \leq h$ at all $t \leq \vote(k)$. By \Cref{rev:cor:processed-height-bound}, every block processed in $\Sigma_v$ at any $t \leq \vote(k)$ has state-height $\leq h+1$, so honest votes before $\vote(k)$ are all at heights $\leq h+1$. The slot-$k$ honest cast at $h$ exactly (Case~A); no voting event lies in $(\vote(k), \vote(k+1))$. We rule out cert events at heights $> h$ in $[\vote(k), \vote(k+1)]$ as follows:
\begin{itemize}
  \item \emph{Heights $> h+1$:} honest contributions at heights $> h+1$ before $\vote(k+1)$ are zero (the height-bound rules out before-$\vote(k)$, and slot-$k$ honest are at $h$). Dishonest alone $\leq f < 2n/3$ (under outer~\ref{rev:h4sfn:f}) cannot form a $\geq 2n/3$ quorum.
  \item \emph{Same-slot cert at $h+1$ for some slot $s$:} for $s < k$, honest contributions at $h+1$ for slot $s$ are $< 2n/3 - f$ (else a slot-$s$ same-slot cert at $h+1$ would have fired, advancing $\Sigma_v.\key_R$ above~$h$, contradiction); slot-$k$ honest at $h+1$ is zero. Hence per-slot contributions at $h+1$ stay $< (2n/3 - f) + f = 2n/3$.
  \item \emph{Bit-$1$ justification cert at $h+1$ for any target $T$:} honest \just contributions at $h+1$ for $T$ (cumulated across all slots, with each honest contributing at most one by R1) are $< 2n/3 - f$ (else such a cert would have fired earlier, advancing $\Sigma_v.\key_R$). Plus dishonest \just for $T$ at $h+1$ $\leq f$. Total $< 2n/3$.
\end{itemize}
By \Cref{lem:Rs-key-monotone}, $\Sigma_v.\key_R.h \leq h$ throughout $[\vote(k), \vote(k+1)]$ for every honest~$v$.

\emph{Step A.2: state at $\vote(k+1)$ via view-merge.}
By outer~\ref{rev:h4sfn:syncers} ($\syncer(k+1)$ honest) and \Cref{rev:ass:view-merge}, every honest's store at $\vote(k+1)$ equals $\comm{\Sigma}^{\vote(k+1)}$. Hence $R_v^{\vote(k+1)}$ is uniform across honest~$v$ (it is determined by $\Sigma$ alone). The honest $\getConfirmed_v^{\vote(k+1)}$ values may differ across honest (they depend on each validator's auxiliary state $\Omega_v$ in addition to $\Sigma$), but they all lie on a common chain~$D$ extending the unified $R_v^{\vote(k+1)}$.

By Step~A.1, no cert at height $> h$ has fired in $[\vote(k), \vote(k+1)]$, so $\Sigma_v.\key_R.h \leq h$ at $\vote(k+1)$, hence $R_v^{\vote(k+1)}.h \leq h$. Combined with monotonicity of $\Sigma_v.\key_R$, the only cert that could update $R_v$ in $[\vote(k), \vote(k+1)]$ is the slot-$k$ same-slot cert at $h$ (target on the common chain $D$ by the prefix-notar $\arg\max$ argument), so $R_v^{\vote(k+1)}.h = h$ on $D$. Since $R$ is itself a cert target, $\sigma[R_v^{\vote(k+1)}].h = h$. As $\getConfirmed_v^{\vote(k+1)}$ descends from $R_v$ along $D$ and state-height is monotone non-decreasing along chain ancestry, $\sigma[\getConfirmed_v^{\vote(k+1)}].h \geq h$. By \Cref{rev:cor:processed-height-bound}, $\sigma[\getConfirmed_v^{\vote(k+1)}].h \leq R_v^{\vote(k+1)}.h + 1 = h+1$. Define $h_1 \coloneqq \sigma[\getConfirmed_v^{\vote(k+1)}].h \in \{h, h+1\}$ (uniform across honest).

\emph{Sub-case A.fast: $h_1 = h+1$.}
Every honest casts a slot-$(k+1)$ \just vote at canonical $h+1$ with target $\getConfirmed_v^{\vote(k+1)}$ (uniform across honest). By outer~\ref{rev:h4sfn:har} (\HAR at $k+1$) and \Cref{rev:prop:honest-vote-inclusion} at slot~$k+1$:
\begin{itemize}
  \item \emph{R-compatibility at $\vote(k+1)$}: $R_v^{\vote(k+1)}$ uniform across honest by view-merge, so trivially pairwise compatible.
  \item \emph{Premise} ($\exists v_\ell$ with $\getConfirmed_{v_\ell}^{\vote(k+1)} \succeq R_v^t$ for honest~$v$, $t \in [\vote(k+1), \vote(k+2)]$): $\getConfirmed_v^{\vote(k+1)}$ at chain-height $\geq h+1$ on~$D$ (since $\sigma.h = h+1$); $R_v^t$ stays on~$D$ at chain-height $\leq h+1$ throughout (any cert that updates $R_v$ in the interval has target on $D$ at chain-height $\leq h+1$ by the same dishonest-only-quorum bound and $\Cref{lem:Rs-key-monotone}$ argument as Step~A.1). Pick any honest~$v_\ell$.
\end{itemize}
The chain ending at $\getConfirmed_v^{\vote(k+2)}$ thus contains every slot-$(k+1)$ honest vote, in particular the $\geq 2n/3$ honest contributions at $h+1$ (under outer~\ref{rev:h4sfn:aha}, all honest are awake; with $f < n/3$, honest count $> 2n/3$). The slot-$(k+1)$ same-slot cert at $h+1$ thus fires via \processHeight{} (\Cref{alg:state-machine}, lines 313–332), advancing state-height to $h+2$.

By outer~\ref{rev:h4sfn:syncers} ($\syncer(k+2)$ honest) and \Cref{rev:ass:view-merge}, $\Sigma_v^{\vote(k+2)}$ is unified across honest, so $\sigma[\getConfirmed_v^{\vote(k+2)}].h = h+2$ uniformly. The $\getConfirmed_v^{\vote(k+2)}$ values lie on a common chain $D'$ extending~$D$ (via the slot-$(k+1)$ cert target).

Apply \Cref{rev:lem:healing-one-slot} at slot~$k+2$ with $\comm{h}_{\text{inner}} = h+2$:
\begin{itemize}
  \item \ref{rev:hone:f}: $f^{\vote(k+3)} \leq f^{\vote(k+4)} < n/3$ (monotonicity of $f^t$ + outer~\ref{rev:h4sfn:f}).
  \item \ref{rev:hone:sync}: $\start(k+2) - \Delta \geq \start(k) + \Delta \geq \GST$.
  \item \ref{rev:hone:ohm}: from outer~\ref{rev:h4sfn:ohm}.
  \item \ref{rev:hone:aha}: from outer~\ref{rev:h4sfn:aha}.
  \item \ref{rev:hone:gt}: honest $\getConfirmed_v^{\vote(k+2)}$ on common chain $D'$, hence pairwise compatible.
  \item \ref{rev:hone:sh}: state-height $h+2$ uniform (just established).
  \item \ref{rev:hone:nh}: before-$\vote(k)$ honest votes at $\leq h+1$ (Step~A.1); slot-$k$ at $h$; slot-$(k+1)$ at $h+1$; no voting in $(\vote(k+1), \vote(k+2))$. All heights $< h+2$.
\end{itemize}
Output: healed at slot~$k+3$ with height~$h+2$.

\emph{Sub-case A.slow: $h_1 = h$.}
Every honest casts a slot-$(k+1)$ \just vote at canonical $h$ with target $\getConfirmed_v^{\vote(k+1)} \in D$ at chain-height $h$ (the targets may differ across honest but all lie on~$D$). By outer~\ref{rev:h4sfn:har} and \Cref{rev:prop:honest-vote-inclusion} at slot~$k+1$ (preamble and premise discharged as in Sub-case A.fast, with chain-heights $\leq h$ throughout the interval), the chain ending at $\getConfirmed_v^{\vote(k+2)}$ contains every slot-$(k+1)$ honest vote. The slot-$(k+1)$ same-slot cert at $h$ thus fires via \processHeight{}, advancing state-height to $h+1$.

By outer~\ref{rev:h4sfn:syncers} ($\syncer(k+2)$ honest) and \Cref{rev:ass:view-merge}, $\Sigma_v^{\vote(k+2)}$ is unified, so $\sigma[\getConfirmed_v^{\vote(k+2)}].h = h+1$ uniformly; the $\getConfirmed_v^{\vote(k+2)}$ values lie on a common chain $D'$ extending~$D$.

Repeat at slot~$k+2$: slot-$(k+2)$ honest cast at canonical $h+1$. By outer~\ref{rev:h4sfn:har} (\HAR at $k+2$) and \Cref{rev:prop:honest-vote-inclusion} at slot~$k+2$ (preamble and premise discharged analogously, with $R_v^t$ at chain-height $\leq h+1$ on $D$), the chain at $\vote(k+3)$ contains every slot-$(k+2)$ honest vote. The slot-$(k+2)$ same-slot cert at $h+1$ fires via \processHeight{}, advancing state-height to $h+2$.

By outer~\ref{rev:h4sfn:syncers} ($\syncer(k+3)$ honest) and \Cref{rev:ass:view-merge}, $\Sigma_v^{\vote(k+3)}$ is unified, so $\sigma[\getConfirmed_v^{\vote(k+3)}].h = h+2$ uniformly; the $\getConfirmed_v^{\vote(k+3)}$ values lie on a common chain $D''$ extending~$D'$.

Apply \Cref{rev:lem:healing-one-slot} at slot~$k+3$ with $\comm{h}_{\text{inner}} = h+2$:
\begin{itemize}
  \item \ref{rev:hone:f}: $f^{\vote(k+4)} < n/3$ (outer~\ref{rev:h4sfn:f}).
  \item \ref{rev:hone:sync}: $\start(k+3) - \Delta \geq \GST$.
  \item \ref{rev:hone:ohm}, \ref{rev:hone:aha}: from outer~\ref{rev:h4sfn:ohm}, \ref{rev:h4sfn:aha}.
  \item \ref{rev:hone:gt}: honest $\getConfirmed_v^{\vote(k+3)}$ on common chain $D''$, hence pairwise compatible.
  \item \ref{rev:hone:sh}: state-height $h+2$ uniform.
  \item \ref{rev:hone:nh}: before-$\vote(k)$ at $\leq h+1$; slot-$k$ at $h$; slot-$(k+1)$ at $h$; slot-$(k+2)$ at $h+1$; no voting in gaps. All $< h+2$.
\end{itemize}
Output: healed at slot~$k+4$ with height~$h+2$.

\emph{Case B: $\sigma[\getConfirmed_v^{\vote(k)}].h = h+1$ for every honest~$v$.}
The hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k$ with $\comm{h}' = h+1$: hypothesis~\ref{rev:h3s:gt} is the case assumption; hypothesis~\ref{rev:h3s:nh} (no honest at $> h+1$ before $\vote(k)$) is Step~1; \ref{rev:h3s:syncers} is inherited. \Cref{rev:lem:healing-three-slot} places the system in a healed state at slot~$k+3$ with height~$(h+1)+1 = h+2$.

\emph{Case C: state-heights mixed at $\vote(k)$.}
Let $T^\dagger := \max_\preceq \{\getConfirmed_v^{\vote(k)} : v \text{ honest}\}$; by Step~1 and the case assumption, $\sigma[T^\dagger].h = h+1$. Sub-split on $\comm{R}^{\vote(k+1)}.h$ (well-defined and common across honest by view-merge + honest $\syncer(k+1)$).

\emph{Sub-case C.A: $\comm{R}^{\vote(k+1)}.h = h+1$.}
A cert at height $h+1$ has fired in $\comm{\Sigma}^{\vote(k+1)}$. Hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k+1$ with $\comm{h}' = h+1$: canonical state-height at $\vote(k+1)$ is exactly $h+1$ (\Cref{rev:cor:processed-height-bound} bounds it by $h+2$, and the $\geq h+1$ direction follows from $\getConfirmed$ descending from~$R$); the no-honest-above-$h+1$ bound is inherited from Step~1 plus slot-$k$ honest voting at heights $\leq h+1$. \Cref{rev:lem:healing-three-slot} places the system in a healed state at slot~$(k+1)+3 = k+4$ with height~$h+2$.

\emph{Sub-case C.B: $\comm{R}^{\vote(k+1)}.h = h$.}
By the same argument as Step~2 of \Cref{rev:lem:healing-from-notarized}, every cert event that updates $R_v$ during $[\vote(k),\, \vote(k+1)]$ is a slot-$k$ notarization with target on the chain of slot-$k$ honest contributions, hence $\preceq T^\dagger$. So $R_v^t \preceq T^\dagger$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$. Picking $v_\ell$ with $\getConfirmed_{v_\ell}^{\vote(k)} = T^\dagger$ discharges the premise of \Cref{rev:prop:target-dominance}; its conclusion gives $\getConfirmed_v^{\vote(k+1)} \succeq T^\dagger$ for every honest~$v$, so $\sigma[\getConfirmed_v^{\vote(k+1)}].h \geq h+1$. \Cref{rev:cor:processed-height-bound} bounds it by $\Sigma.\key_R.h + 1 = h+1$. Apply \Cref{rev:lem:healing-three-slot} at slot~$k+1$ with $\comm{h}' = h+1$ as in Sub-case~C.A, placing the system in a healed state at slot~$k+4$ with height~$h+2$.

In every case, the system is in a healed state at slot~$k' \in \{k+3, k+4\}$ with height~$\comm{h}+2$.
\end{proof}

\begin{lemma}[Five-slot healing]\label{rev:lem:healing-five-slot}
Assume:
\begin{enumerate}
  \item\label{rev:h5s:f} $f^{\vote(k+5)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:h5s:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:h5s:syncers} $\syncer(k)$, $\syncer(k+1)$, $\syncer(k+2)$, $\syncer(k+3)$ and $\syncer(k+4)$ are honest;
  \item\label{rev:h5s:ohm} the \OHM holds at every slot $k' \in \{k+1, k+2, k+3, k+4, k+5\}$;
  \item\label{rev:h5s:aha} the \AHA condition holds at every slot $k' \in \{k, k+1, k+2, k+3, k+4, k+5\}$;
  \item\label{rev:h5s:har} the \HAR condition holds at every slot $k' \in \{k+1, k+2, k+3, k+4\}$;
  \item\label{rev:h5s:gt} $\sigma[\getConfirmed_v^{\vote(k)}].h = \comm{R}^{\vote(k)}.h$ is the same for every honest validator~$v$; denote this common value by $\comm{h}$;
  \item\label{rev:h5s:nh} no honest validator has cast any vote at height $> \comm{h}+1$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then there exists $k' \in \{k+3, k+4, k+5\}$ such that the system is in a healed state at slot $k'$ with height $\comm{h}+2$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: $\Sigma.\key_R.h \leq h+1$ throughout $[\vote(k),\, \vote(k+1)]$.}
By \Cref{rev:ass:view-merge} and honest $\syncer(k)$, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ for every honest~$v$, with $\Sigma_v.\key_R.h = h$. By hypothesis~\ref{rev:h5s:nh} and \Cref{lem:Rs-key-monotone}, no honest casts at height $> h+1$ before $\vote(k)$ and the cached key remains $\Sigma_v.\key_R.h \leq h$ at $\vote(k)$. At $\vote(k)$ every honest casts at height~$h$ exactly (hypothesis~\ref{rev:h5s:gt}), forming slot-$k$ contributions at height~$h$. No cert event at height $> h+1$ can form in $[\vote(k),\, \vote(k+1)]$: any such cert needs a $\geq 2n/3$ quorum of votes at height $> h+1$, and hypothesis~\ref{rev:h5s:nh} plus Step-1 voting plus $f < n/3$ rule that out.

\emph{Step 2: case split on $\comm{R}^{\vote(k+1)}$.}
View-merge + honest $\syncer(k+1)$ gives common $\comm{\Sigma}^{\vote(k+1)}$. By Step~1 $\comm{R}^{\vote(k+1)}.h \in \{h, h+1\}$; by \Cref{lem:Rs-key-monotone} also $\comm{R}^{\vote(k+1)}.h \geq h$. The cases:
\begin{itemize}
  \item \emph{Case 1a:} $\comm{R}^{\vote(k+1)}$ is a bit-$1$ justification at height~$h$. (Either the cached $\comm{R}^{\vote(k)}$ was already justified at~$h$ and is unchanged, or the slot-$k$ honest \just votes' targets coincided to fire a justification at~$h$.) The hypotheses of \Cref{rev:lem:healing-four-slot-from-justified} apply at slot~$k+1$ with $\comm{h}_{\text{ffj}}' = h$: outer's hypothesis~\ref{rev:h5s:syncers} provides $\syncer(k+2), \syncer(k+3), \syncer(k+4)$ honest, discharging hypothesis~\ref{rev:h4sfj:syncers}; by view-merge + honest $\syncer(k+1)$ (outer's hypothesis~\ref{rev:h5s:syncers}), $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is the same justified block in every honest's view, and honest \getConfirmed{}s at $\vote(k+1)$ coincide (hence pairwise compatible), discharging hypothesis~\ref{rev:h4sfj:r}. Invoking it places the system in a healed state at slot~$k'_1 \in \{(k+1)+3,\, (k+1)+4\} = \{k+4,\, k+5\}$ with height~$h+2$.
  \item \emph{Case 1b:} $\comm{R}^{\vote(k+1)}$ is a bit-$1$ justification at height $h+1$. (Pre-existing honest \just votes at $h+1$, together with late dishonest contributions in $\comm{\Sigma}^{\vote(k+1)}$, completed a quorum.) The hypotheses of \Cref{rev:lem:healing-from-justified} apply at slot~$k+1$ with $\comm{h}_{\text{fj}}' = h+1$: by view-merge + honest $\syncer(k+1)$ (outer's hypothesis~\ref{rev:h5s:syncers}), $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is justified in every honest's view (discharging hypothesis~\ref{rev:hfj:r}); canonical state-height at $\vote(k+1)$ is $h+1$ by \Cref{rev:cor:processed-height-bound}, and view-merge gives uniform (hence pairwise compatible) honest \getConfirmed{}s at $\vote(k+1)$ (discharging hypothesis~\ref{rev:hfj:gt}); the no-honest-above-$h+1$ bound holds by Step~1 (discharging hypothesis~\ref{rev:hfj:nh}). Invoking it places the system in a healed state at slot~$(k+1)+2 = k+3$ with height~$h+2$.
  \item \emph{Case 2:} $\comm{R}^{\vote(k+1)}$ is bit-$0$ notarized at height $h+1$. The hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k+1$ with $\comm{h}_{\text{three}}' = h+1$. Invoking it places the system in a healed state at slot~$(k+1)+3 = k+4$ with height~$h+2$.
  \item \emph{Case 3:} $\comm{R}^{\vote(k+1)}$ is bit-$0$ notarized at height~$h$. The cached at $\vote(k)$ has $\Sigma.\key_R = (h, b, s, \ldots)$ with $b \in \{0, 1\}$ and $s < k$ (slot-$\geq k$ contributions don't exist before $\vote(k)$); $\comm{R}^{\vote(k+1)}$ being bit-$0$ at height~$h$ rules out the bit-$1$ branch and forces the slot-$k$ notarization (which has slot $k > s$) to dominate, so $\comm{R}^{\vote(k+1)}$ has notarization slot $k = (k+1) - 1$. The hypotheses of \Cref{rev:lem:healing-four-slot-from-notarized} apply at slot~$k+1$: outer's hypothesis~\ref{rev:h5s:syncers} provides $\syncer(k+2), \syncer(k+3), \syncer(k+4)$ honest, discharging hypothesis~\ref{rev:h4sfn:syncers}; by view-merge + honest $\syncer(k+1)$ (outer's hypothesis~\ref{rev:h5s:syncers}), every honest's store at $\vote(k+1)$ coincides with $\comm{\Sigma}^{\vote(k+1)}$, so honest \getConfirmed{}s at $\vote(k+1)$ are equal (hence pairwise compatible), discharging hypothesis~\ref{rev:h4sfn:gt}; by the same view-merge, $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is bit-$0$ notarized at~$h$ with slot $k = (k+1)-1$ in every honest's view, discharging hypothesis~\ref{rev:h4sfn:r}. Invoking it places the system in a healed state at slot~$k'_3 \in \{k+4,\, k+5\}$ with height~$h+2$.
\end{itemize}

In every case, the system is in a healed state at slot~$k' \in \{k+3, k+4, k+5\}$ with height~$\comm{h}+2$.
\end{proof}


\subsection{Main theorem}\label{rev:sec:healing-main}

\begin{theorem}
Assume:
\begin{enumerate}
  \item\label{rev:mt:f} $f^{\vote(k+5)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:mt:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:mt:syncers} $\syncer(k)$, $\syncer(k+1)$, $\syncer(k+2)$, $\syncer(k+3)$ and $\syncer(k+4)$ are honest;
  \item\label{rev:mt:ohm} the \OHM holds at every slot $k' \geq k+1$;
  \item\label{rev:mt:aha} the \AHA condition holds at slots $k$, $k+1$, $k+2$, $k+3$, $k+4$, and $k+5$;
  \item\label{rev:mt:har} the \HAR condition holds at slots $k+1$, $k+2$, $k+3$, and $k+4$;
  \item\label{rev:mt:gt} $\sigma[\getConfirmed_v^{\vote(k)}].h = \comm{R}^{\vote(k)}.h$ is the same for every honest validator~$v$; denote this common value by $\comm{h}$;
  \item\label{rev:mt:nh} no honest validator has cast any vote at height $> \comm{h}+1$ at any time strictly before $\vote(k)$.
\end{enumerate}
then,
\begin{enumerate}
  \item for every honest validators~$v,v'$ and times $t, t' \geq \vote(k+5)$, $\getConfirmed_v^{t} \sim \getConfirmed_{v'}^{t'}$;
  \item for every honest validator~$v$ and every pair of times $t, t' \geq \vote(k+5)$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
\end{enumerate}
\end{theorem}

\begin{proof}
Outer hypotheses~\ref{rev:mt:f}--\ref{rev:mt:nh} discharge the corresponding hypotheses \ref{rev:h5s:f}--\ref{rev:h5s:nh} of \Cref{rev:lem:healing-five-slot} (with outer's slot-$\geq k+1$ \OHM coverage from~\ref{rev:mt:ohm} subsuming the slot-$\{k+1, \ldots, k+5\}$ requirement of~\ref{rev:h5s:ohm}), which yields $k' \in \{k+3, k+4, k+5\}$ such that the system is in a healed state at slot~$k'$ with height~$\comm{h}+2$.

\Cref{rev:lem:healing-unbounded} applies at slot~$k'$: the voting rule clause of outer's hypothesis~\ref{rev:mt:f} discharges its hypothesis~\ref{rev:hu:f}, outer's hypothesis~\ref{rev:mt:ohm} discharges its hypothesis~\ref{rev:hu:ohm} (since $k' \geq k+1$), and the healed state just established is its hypothesis~\ref{rev:hu:hs}. Its conclusions give, for every pair of honest validators~$v, v'$ and every pair of times $t, t' \geq \vote(k')$, $\getConfirmed_v^t \sim \getConfirmed_{v'}^{t'}$ and (when $t \leq t'$) $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.

Since $k' \leq k+5$, every $t \geq \vote(k+5)$ also satisfies $t \geq \vote(k')$, so the two stated conclusions hold.
\end{proof}

Assume:
\begin{enumerate}
  \item\label{rev:h3sdv:f} $f^{\vote(k+3)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:h3sdv:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:h3sdv:syncers} $\syncer(k), \syncer(k+1), \syncer(k+2)$ are honest;
  \item\label{rev:h3sdv:ohm} the \OHM holds at every slot $k' \in \{k, k+1, k+2, k+3\}$;
  \item\label{rev:h3sdv:aha} the \AHA condition holds at every slot $k' \in \{k, k+1, k+2, k+3\}$;
  \item\label{rev:h3sdv:har} the \HAR condition holds at slots~$k+1$ and $k+2$;
  \item\label{rev:h3sdv:dual} (\emph{slot-$k$ dual-height voting}) at $\vote(k)$, every honest validator casts \emph{two} votes: one at height $R^{\vote(k)}.h$ and one at height $R^{\vote(k)}.h + 1$ (the latter conditional on a block at state-height $R^{\vote(k)}.h + 1$ existing in its store). Each is of whatever type \Cref{def:voting-rule} would normally produce at the height in question (\just on the first vote at the height, \notar otherwise; with the appropriate $(\finalizeHeight, \finalizeTarget)$ for the (R1) clause). At every voting time other than $\vote(k)$, validators follow \Cref{def:voting-rule} unchanged.
\end{enumerate}
Write $\comm{h} \coloneqq \comm{R}^{\vote(k)}.h$. Then the system is in a healed state at slot $k+3$ with height $\comm{h}+2$ (\Cref{rev:def:healed-state}).
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: structural facts.}
By \Cref{rev:ass:view-merge} and honest $\syncer(k)$, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ and $\Sigma_v^{\vote(k)} = \comm{\Sigma}^{\vote(k)}$ for every honest~$v$. By \Cref{rev:cor:processed-height-bound}, $\sigma[\getConfirmed_v^{\vote(k)}].h \in \{h, h+1\}$ for every honest~$v$. By \Cref{lem:Rs-key-monotone} and \Cref{rev:cor:processed-height-bound} applied at every earlier time, no honest validator has cast at height $>h+1$ before $\vote(k)$. By \Cref{rev:prop:getConfirmed-compat} (premise discharged by view-merge), honest \getConfirmed{}s at $\vote(k)$ are pairwise $\preceq$-compatible; hence both the height-$h$ and height-$(h+1)$ honest targets at $\vote(k)$ lie on a single chain~$C$ through $\comm{R}^{\vote(k)}$.

\emph{Step 2: a same-slot cert at height $h+1$ fires by $\vote(k+1)$ in every honest's view.}
Under the slot-$k$ dual-height extension every honest casts a slot-$k$ vote at height~$h$ \emph{and} a slot-$k$ vote at height $h+1$, both targeting blocks on~$C$. The slot-$k$ contributions at height $h+1$ thus include $\geq n - f > 2n/3$ honest~$\ntargets$ entries with $\nslots = k$ and target on~$C$. In each honest~$v$'s virtual-block aggregation by $\vote(k+1)$, the same-slot $\arg\max$ (\Cref{alg:state-machine}) admits a candidate $(B^\star, k)$ at height $h+1$ on~$C$, firing a cert event with key $(h+1,\, b,\, k,\, \ldots)$. This key strictly dominates the cached $\Sigma_v.\key_R = (h, *, *, \ldots)$ on the leading coordinate, so $R_v$ advances to $B^\star$. By view-merge + honest $\syncer(k+1)$, $\comm{R}^{\vote(k+1)}$ is the resulting cert event at height $h+1$, common across honest.

\emph{Step 3: reduce to a two-slot lemma at slot $k+1$.}
The setup at $\vote(k+1)$ matches the hypotheses of \Cref{rev:lem:healing-from-justified} (if $\comm{R}^{\vote(k+1)}$ is a bit-$1$ justification at $h+1$) or of \Cref{rev:lem:healing-from-notarized} (if $\comm{R}^{\vote(k+1)}$ is a bit-$0$ notarization at $h+1$ with notarization slot~$k$):
\begin{itemize}
  \item synchrony at slot $k+1$: $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$;
  \item by view-merge + honest $\syncer(k+1)$ (outer~(3)), $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is the appropriate justified or notarized block in every honest's view; for the notarized branch, $\syncer(k+2)$ honest is also inherited from outer~(3);
  \item $\sigma[\getConfirmed_v^{\vote(k+1)}].h = h+1$ common across honest, by view-merge + \Cref{rev:cor:processed-height-bound} bounding state-height by $\comm{R}^{\vote(k+1)}.h + 1 = h+2$ and the $\geq h+1$ direction following from $\getConfirmed$ descending from $\comm{R}^{\vote(k+1)}$;
  \item no honest at height $>h+1$ before $\vote(k+1)$, by Step~1 plus the slot-$k$ honest contributions lying at heights in $\{h, h+1\}$.
\end{itemize}
Invoking the appropriate sub-lemma at slot~$k+1$ places the system in a healed state at slot~$(k+1)+2 = k+3$ with height~$(h+1)+1 = h+2$, as required.
\end{proof}

\begin{remark}[Why the dual-height extension is needed]\label{rev:rem:dual-vote-needed}
Without the slot-$k$ dual-height extension, the \Cref{rev:cor:processed-height-bound} bound only constrains canonical state-heights to $\{h, h+1\}$, but does not force them to agree across honest. The adversary can split honest \getConfirmed{}s between the two heights via the per-validator $\Omega$ component of $\getConfirmed(\Sigma, \Omega)$, so the slot-$k$ honest contributions split between two heights:
\[
  h^*_h + h^*_{h+1} \;=\; n - f, \qquad h^*_h, h^*_{h+1} \;\geq\; 0.
\]
A same-slot cert at height $h^* \in \{h, h+1\}$ requires $\geq 2n/3$ contributors at $h^*$. Together with at most $f$ adversarial contributions, the threshold becomes $h^*_{h^*} \geq 2n/3 - f$. Both thresholds met simultaneously requires $h^*_h + h^*_{h+1} \geq 4n/3 - 2f$, i.e., $f \geq n/3$, contradicting $f < n/3$; so at most one height can carry a cert. The adversary can in fact split so that \emph{neither} reaches the threshold (e.g., $h^*_h = h^*_{h+1} = (n-f)/2$, giving honest mass $(n-f)/2 + f = (n+f)/2 < 2n/3$), blocking progress entirely. The split survives view-merge unchanged --- view-merge fixes $\Sigma$ but not the $\Omega$ argument of $\getConfirmed$ --- so neither extra honest syncers nor extra slots resolve it.

The dual-height extension forces every honest to contribute at \emph{both} heights, raising each height's honest count to $n - f > 2n/3$ and guaranteeing a cert at height $h+1$. The extension is restricted to slot~$k$ because exactly one ``alignment slot'' is enough: post-slot-$k$, the cert at $h+1$ has fired, $R$ has advanced, and standard view-merge dynamics realign honest \getConfirmed{}s for slots $k+1, k+2, \ldots$ without further dual-voting.
\end{remark}

\begin{remark}[A natural-looking fix that does not work]\label{rev:rem:conditional-dual-vote-fails}
A tempting refinement of \Cref{def:voting-rule} is the following ``conditional dual-vote'' rule: an honest validator with $\getConfirmed = T$ at state-height $h$ casts a vote at height~$h$ on $T$ as usual, and \emph{additionally} casts a vote at height~$h+1$ whenever its store contains some $T' \succeq T$ with $\sigma[T'].h = h+1$. Universally adopted, this would seem to overcome the split: validators whose $\getConfirmed$ already sits at $h+1$ vote at $h+1$ via \Cref{def:voting-rule}, and validators whose $\getConfirmed$ sits at $h$ but who can see an extension also vote at $h+1$, raising the height-$(h+1)$ honest count.

The rule fails at its premise: if $\getConfirmed_i = T$ has state-height $h$, no chain in $\Sigma_i.\T$ extends $T$ to a block at state-height $h+1$. By construction, $\getConfirmed$ is the deepest target on the canonical chain $\sigma_i.L \succeq T$, and $\sigma[\getConfirmed].h$ is the canonical state-height; if some $T' \succeq T$ on a chain in $\Sigma_i.\T$ had $\sigma[T'].h = h+1$, then $\getConfirmed$ would itself sit at state-height $\geq h+1$, contradicting $\sigma[\getConfirmed].h = h$. So the ``additionally vote at $h+1$'' branch never fires for the validators it was meant to recruit, and the rule reduces to \Cref{def:voting-rule} on those validators --- leaving the split untouched.
\end{remark}

\section{View-merge mechanism}\label{app:view-merge}

This appendix details the view-merge mechanism abstracted by \Cref{rev:ass:view-merge} and discharges that assumption under a synchronous-slot hypothesis.
Throughout, $\Delta$ denotes the synchronous network-delivery bound.

\begin{definition}[Slot timing]\label{rev:def:slot-timing}
Each slot~$k$ is anchored at a real-valued slot time $\start(k)$ with $\start(k+1) - \start(k) \geq 2\Delta$ and contains three ordered events:
\begin{itemize}
  \item \emph{Freeze time} $\start(k) - \Delta$: every validator~$i$ records its current local store as its slot-$k$ frozen snapshot $\Sigma^k_{\mathrm{frozen},i}$.
  \item \emph{Syncer time} $\start(k)$: $\syncer(k)$ broadcasts its current (unfrozen) local store to every validator; let $\Sigma^k_{\mathrm{sent}}$ be the broadcast content.
  \item \emph{Voting time} $\start(k) + \Delta$: every honest validator~$i$ invokes \Cref{def:voting-rule} on its slot-$k$ merged view $\Sigma^k_{\mathrm{merge},i} \coloneqq \Sigma^k_{\mathrm{frozen},i} \cup \Sigma^k_{\mathrm{sent}}$.
\end{itemize}
\end{definition}

\begin{definition}[Synchronous slot]\label{rev:def:synchrony-window}
Slot~$k$ is \emph{synchronous} if every message sent during $[\start(k) - 2\Delta, \start(k) + \Delta]$ is delivered to every honest validator within $\Delta$ of its sending time.
\end{definition}

\begin{lemma}[View-merge under an honest synchronous syncer]\label{rev:lem:view-merge}
If slot~$k$ is synchronous and $\syncer(k)$ is honest, then every honest~$i$ has $\Sigma^k_{\mathrm{frozen},i} \subseteq \Sigma^k_{\mathrm{sent}}$ and $\Sigma^k_{\mathrm{merge},i} = \Sigma^k_{\mathrm{sent}}$, discharging \Cref{rev:ass:view-merge} with $\comm{\Sigma}^{\vote(k)} = \Sigma^k_{\mathrm{sent}}$.
\end{lemma}

\begin{proof}
Any block or vote in $\Sigma^k_{\mathrm{frozen},i}$ was received by $\start(k) - \Delta$, hence sent by that time, hence delivered to $\syncer(k)$ by $\start(k)$ under \Cref{rev:def:synchrony-window}, so it belongs to $\Sigma^k_{\mathrm{sent}}$.
The broadcast at $\start(k)$ reaches every honest by $\start(k) + \Delta$, so $\Sigma^k_{\mathrm{merge},i}$ is well-defined and equals $\Sigma^k_{\mathrm{sent}}$.
\end{proof}
