\section{Model and definitions}\label{sec:model}

We consider a system of~$n$ validators of which at most~$f$ are adversarial, where $n \geq 3f + 1$.
A \emph{quorum} is any set of at least $2n/3$ validators.
Any two quorums overlap in at least $n/3 + 1 > f$ validators, so their intersection contains at least one honest validator.

\begin{definition}[Height]\label{def:height}
\emph{Height} is the finality-gadget counter, separate from slot.
The genesis state-height is~$1$ (with $\genesis$ treated as pre-justified and pre-finalized at height~$0$); height advances by exactly~$1$ via fixed-target justification (Definition~\ref{def:justification}) or a timeout (Definition~\ref{def:timeout}).
\end{definition}

\begin{definition}[Block]\label{def:block}
A \emph{block} has a \emph{parent} block, a \emph{slot} field, and a set of votes (Definition~\ref{def:vote}).
The \emph{genesis block}~$\genesis$ has no parent and slot~$0$.
Slots strictly increase along parent links: $B.\slot > B.\parent.\slot$ for every non-genesis~$B$.
\end{definition}

\begin{definition}[Chain]\label{def:chain}
A \emph{chain} is a path from~$\genesis$ to some block~$B$ via parent links.
Chains are in bijection with their tips, and we identify the two.
Blocks form a tree rooted at~$\genesis$; we write $B \preceq B'$ when $B$ is an ancestor of~$B'$ (or $B = B'$), and $B \sim B'$ (\emph{compatible}) when $B \preceq B'$ or $B' \preceq B$.
Two chains \emph{conflict} when their tips are not compatible.
\end{definition}

\begin{definition}[Vote]\label{def:vote}
A \emph{vote} is a signed tuple
\[
  (\validator,\; \height,\; \target,\; \finalizeHeight,\; \finalizeTarget),
\]
where $\validator$ is the signer, $\height$ is a height, and $\target \in \{\bot\} \cup \mathrm{Blocks}$.
A vote with $\target = \bot$ is a \emph{timeout vote}; a vote with $\target \neq \bot$ is a \emph{justification vote}.
The pair $(\finalizeHeight,\, \finalizeTarget)$ is either both~$\bot$ or a commitment to finalize a block at a height.
A vote included on a chain must reference a target (and finalize target, when present) that exists on that chain when non-$\bot$.
\end{definition}

\paragraph{State.}
The \emph{state after block~$B$}, written $\sigma[B]$, is the tuple
\[
  (L,\; s,\; h,\; s_h,\; \targets,\; \timeouts,\; J,\; h_j,\; F,\; P),
\]
with components:
\begin{itemize}
  \item $L$: the \emph{chain head} ($L = B$ after processing~$B$).
  \item $s$: the \emph{current slot}, incremented once per slot by \processSlot (Figure~\ref{alg:state-machine}).
  \item $h$: the \emph{current height}.
  \item $s_h$: the slot of the block whose processing last advanced~$h$ on this chain ($0$ if no advance has occurred).
  \item $\targets[1..n]$: for each validator~$i$, the target of~$i$'s justification vote at the current height ($\bot$ if none processed).
  \item $\timeouts[1..n]$: for each~$i$, a boolean that is \texttt{true} iff~$i$'s vote at the current height set the timeout marker --- either a timeout vote ($\target = \bot$) or a height-fresh justification vote on this chain.
  \item $J$, $h_j$: the most recently justified block and its height.
  \item $F$: the most recently finalized block.
  \item $P \subseteq \{1, \ldots, n\}$: validators who have cast a vote with $\finalizeHeight = h_j$ and $\finalizeTarget = J$.
\end{itemize}
The genesis state is
\[
  \sigma_{\mathrm{gen}} \coloneqq (\genesis,\; 0,\; 1,\; 0,\; \bot^n,\; \mathtt{false}^n,\; \genesis,\; 0,\; \genesis,\; \emptyset),
\]
with $\sigma[\genesis] \coloneqq \sigma_{\mathrm{gen}}$: $\genesis$ is treated as pre-justified and pre-finalized at height~$0$, so the genesis state-height is~$1$ and $h_j = 0$ records the genesis-justification.
When unambiguous we drop the chain prefix and write $L$, $s$, $h$, etc.\ for the components of $\sigma[L]$.

\begin{definition}[State-height]\label{def:sheight}
The \emph{state-height} of a block~$B$ is $\sigma[B].h$: the height after processing~$B$ on its own chain.
\end{definition}

\paragraph{Height freshness.}
A justification vote~$v$ (i.e., $v.\target = T \neq \bot$) is \emph{fresh} on a chain with head state $\sigma[L]$ iff
\[
  \sigma[L].h = v.\height \;\wedge\; T \preceq L \;\wedge\; T.\slot \geq \sigma[L].s_h:
\]
the chain is at~$v$'s height, $T$ lies on this chain, and $T$'s slot is no earlier than where the chain's current height began.
A timeout vote ($v.\target = \bot$) is fresh on the chain iff $\sigma[L].h = v.\height$.
Intuitively, a fresh justification vote attests that its signer saw~$T$ as a chain of height~$v.\height$.

\begin{remark}[Freshness via $T$'s own chain]\label{rem:fresh-equiv}
For a justification vote with target $T \preceq L$, the constraint $T.\slot \geq \sigma[L].s_h$ is equivalent to $\sigma[T].h = \sigma[L].h$.
State-height is $\preceq$-monotone (Lemma~\ref{lem:slotmono}), equals $\sigma[L].h$ on~$L$, and advanced to its current value at the block whose slot is $\sigma[L].s_h$; so the ancestors of~$L$ at that state-height are exactly those with slot $\geq \sigma[L].s_h$.
\end{remark}

Freshness ensures that every justification at height~$h$ is witnessed by targets whose own chains reached state-height~$h$, and pins both timeout and justification contributions to the chain's current height.

\begin{definition}[Fixed-target justification]\label{def:justification}
A block~$T$ is \emph{justified at height~$h$} on a chain with state $\sigma[L]$ with $\sigma[L].h = h$ iff
\[
  \bigl|\{i \,:\, \targets[i] = T\}\bigr| \;\geq\; 2n/3.
\]
Only votes with target exactly~$T$ contribute.
\end{definition}

\begin{definition}[Timeout]\label{def:timeout}
On a chain with head state $\sigma[L]$ at height~$h$, a \emph{timeout} fires iff
\[
  \bigl|\{i : \timeouts[i] = \mathtt{true}\}\bigr| \;\geq\; 2n/3.
\]
A timeout has no associated target block.
Both timeout votes and height-fresh justification votes (with target $\preceq L$ at the chain's current height) set $\timeouts[i] = \mathtt{true}$, so a justification quorum implies a timeout quorum.
The two height-advance branches are checked in order, justification first (Figure~\ref{alg:state-machine}).
\end{definition}

\begin{definition}[Finality]\label{def:finality}
The justified block~$J$ is \emph{finalized at height~$h_j$} once $|P| \geq 2n/3$.
\end{definition}

\begin{definition}[Slashing, rule E1]\label{def:slashing}
A validator is \emph{slashable} if it signed two votes~$a, b$ (possibly $a = b$) with
\[
  b.\finalizeTarget \neq \bot \;\wedge\; a.\height = b.\finalizeHeight \;\wedge\; a.\target \neq b.\finalizeTarget.
\]
That is, a validator committing $(\finalizeHeight,\, \finalizeTarget) = (h_f, T_f)$ must not sign any vote at height~$h_f$ with target $\neq T_f$.
Note that timeout votes ($a.\target = \bot$) are themselves in conflict with any commitment $b.\finalizeTarget \neq \bot$ at the same height, since $\bot \neq b.\finalizeTarget$.
\end{definition}

Two justification votes (or two timeout votes) at the same height are not by themselves slashable; per-height uniqueness of honest justification votes is imposed behaviorally (Section~\ref{sec:store}).
A validator who never sets $\finalizeTarget \neq \bot$ trivially avoids slashing.

\paragraph{State transitions.}
The state machine (Figure~\ref{alg:state-machine}) exposes two per-chain routines: \processSlot, invoked once per slot on every maintained chain, and \processBlock, invoked additionally whenever a block~$B$ is processed.

\processSlot increments~$s$ and calls \processHeight, which checks three branches in order: (i) finality, $F \gets J$ if $|P| \geq 2n/3$; (ii) justification, if some~$T$ has $\geq 2n/3$ entries in $\targets$; (iii) timeout cert, if $\geq 2n/3$ entries of $\timeouts$ are \texttt{true}.
Branches (ii) and (iii) both advance height and reset $\targets, \timeouts$, so at most one height-advance fires per invocation.

$\processBlock(\sigma, B)$ has precondition $\sigma.s = B.\slot$: the caller runs \processSlot up to $B.\slot$ first.
Then $L \gets B$, each vote~$v$ in~$B$ is fed through \processVote, and finally \finalizeProcessBlock is invoked.
A timeout vote ($v.\target = \bot$) at the current height sets $\timeouts[i] \gets \mathtt{true}$.
A height-fresh justification vote sets both $\targets[i] \gets v.\target$ and $\timeouts[i] \gets \mathtt{true}$.
The $P$-gate adds~$i$ to~$P$ iff $v.\finalizeHeight = \sigma.h_j$ and $v.\finalizeTarget = \sigma.J$; this is independent of freshness.
\finalizeProcessBlock then runs \processHeight on the post-vote state, so any cert event enabled by~$B$'s own votes fires in $B$'s post-state.
Height advances thus happen in two places: in \processSlot (which runs on every slot, including empty ones), and in \finalizeProcessBlock (at the end of \processBlock, capturing the effect of $B$'s votes).

\begin{algorithmic}[1]

\Function{\processSlot}{$\sigma$} \Comment{Runs once per slot on every chain, even without a block}
  \State $\sigma.s \gets \sigma.s + 1$
  \State $\sigma \gets \finalizeProcessBlock(\sigma)$
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\processBlock}{$\sigma,\; B$}
  \State \textbf{assert} $\sigma.s = B.\slot$ \Comment{Precondition: caller advanced $\sigma.s$ via \processSlot}
  \State $\sigma.L \gets B$
  \For{each vote $v$ in $B$} $\sigma \gets \processVote(\sigma,\, v)$ \EndFor
  \State $\sigma \gets \finalizeProcessBlock(\sigma)$
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\finalizeProcessBlock}{$\sigma$} \Comment{fires cert events enabled by $B$'s votes}
  \State $\sigma \gets \processHeight(\sigma)$
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\processVote}{$\sigma,\; v$}
  \State $i \gets v.\validator$
  \If{$v.\target = \bot$} \Comment{Timeout vote}
    \If{$v.\height = \sigma.h$}
      \State $\sigma.\timeouts[i] \gets \mathtt{true}$
    \EndIf
  \Else \Comment{Justification vote: apply height freshness}
    \State $\fresh \gets (v.\height = \sigma.h) \land (v.\target \preceq \sigma.L) \land (v.\target.\slot \geq \sigma.s_h)$
    \If{$\fresh$}
      \State $\sigma.\targets[i] \gets v.\target$
      \State $\sigma.\timeouts[i] \gets \mathtt{true}$
    \EndIf
  \EndIf
  \If{$v.\finalizeHeight = \sigma.h_j$ \textbf{and} $v.\finalizeTarget = \sigma.J$}
    \State $\sigma.P \gets \sigma.P \cup \{i\}$
  \EndIf
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\processHeight}{$\sigma$}
  \If{$|\sigma.P| \geq 2n/3$} \Comment{Finality}
    \State $\sigma.F \gets \sigma.J$
  \EndIf
  \State $T \gets \arg\max_{T' \in \{\sigma.\targets[i] \,:\, \sigma.\targets[i] \neq \bot\}} |\{i : \sigma.\targets[i] = T'\}|$ \Comment{$T \gets \bot$ if the set is empty}
  \If{$T \neq \bot$ \textbf{and} $|\{i : \sigma.\targets[i] = T\}| \geq 2n/3$} \Comment{Justification}
    \State $\sigma.J \gets T$, \; $\sigma.h_j \gets \sigma.h$, \; $\sigma.P \gets \emptyset$
    \State $\sigma.h \gets \sigma.h + 1$, \; $\sigma.s_h \gets \sigma.L.\slot$
    \State reset $\targets, \timeouts$
    \State \Return $\sigma$
  \EndIf
  \If{$|\{i : \sigma.\timeouts[i] = \mathtt{true}\}| \geq 2n/3$} \Comment{Timeout}
    \State $\sigma.h \gets \sigma.h + 1$, \; $\sigma.s_h \gets \sigma.L.\slot$
    \State reset $\targets, \timeouts$
  \EndIf
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\finalizeConflict}{$v_1, v_2$} \Comment{Does $v_1$ conflict with $v_2$'s finalize commitment?}
  \State \Return $v_2.\finalizeTarget \neq \bot \;\wedge\; v_1.\height = v_2.\finalizeHeight \;\wedge\; v_1.\target \neq v_2.\finalizeTarget$
\EndFunction
\vspace{3pt}

\Function{\isSlashable}{$v_1, v_2$}
  \State \Return $v_1.\validator = v_2.\validator$
  \Statex \hspace{3em}$\land \;(\finalizeConflict(v_1, v_2) \lor \finalizeConflict(v_2, v_1))$
\EndFunction

\end{algorithmic}

\section{Accountable safety}\label{sec:safety}

\begin{lemma}[Height progression]\label{lem:heightprog}
\processHeight increments~$h$ by at most~$1$ per invocation, and each increment is gated by a $\geq 2n/3$ justification or timeout quorum.
\end{lemma}

\begin{proof}
The only assignments to~$h$ are $h \gets h + 1$ in the justification and timeout-cert branches of \processHeight, each guarded by its $\geq 2n/3$ quorum and followed by a return or function exit, so at most one fires per call.
\end{proof}

\begin{lemma}[Checkpoint monotonicity]\label{lem:slotmono}
On any chain, the fields $s$, $h$, $h_j$, $s_h$, $F$, $J$ are non-decreasing along~$\preceq$, with $F \preceq J$ at all times.
$J$ and $h_j$ advance only on justification events, and $s_h$ only on height-advances, with $s_h$ set to the slot of the advancing block.
Consequently, when $\sigma[Y].h = h$, the ancestors of~$Y$ at state-height~$h$ are $\{B' \preceq Y : B'.\slot \geq \sigma[Y].s_h\}$.
\end{lemma}

\begin{proof}
Reading off Figure~\ref{alg:state-machine}: $s$ only increases (in \processSlot); $h$ only increases, by~$1$, in the height-advance branches of \processHeight, which also set $h_j \gets h$ on the justification branch and $s_h \gets L.\slot$ on either branch.
Slots strictly increase along~$\preceq$ (Definition~\ref{def:block}), so $s_h$ is non-decreasing.
$F$ updates only via $F \gets J$, so $F \preceq J$ reduces to $J$-monotonicity.

\emph{Height-$h$ interval.}
Along any chain, $h$ increases monotonically; when $\sigma[Y].h = h$, the value $s_h$ was set at the block that advanced state-height to~$h$.
By strict slot increase along~$\preceq$ and the monotonicity of $(h, s_h)$, the ancestors $B' \preceq Y$ with $\sigma[B'].h = h$ are exactly those with $B'.\slot \geq s_h$.

\emph{Block field $J$.}
At a justification advance on chain~$Y$ firing at target~$T$ with pre-state height~$h$, freshness forces $T.\slot \geq s_h$, since every contributing justification vote has target~$T$ and was processed only when fresh.
By the height-$h$ interval claim, $\sigma[T].h = h$, so the previous justified block $J_{\mathrm{old}}$ with $\sigma[J_{\mathrm{old}}].h \leq h-1$ and $J_{\mathrm{old}} \preceq Y$ satisfies $T \succ J_{\mathrm{old}}$ by state-height monotonicity.
Hence $J$ strictly advances along~$\preceq$ at every justification event.
\end{proof}

\begin{lemma}[Any chain past a finalized height contains the finalized block]\label{lem:mainsafety}
Unless $\geq n/3$ validators are slashable: if~$C$ is finalized at height~$h$, then $C \preceq B$ for every block~$B$ with $\sigma[B].h > h$.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:heightprog}, some $B^* \preceq B$ advanced height from~$h$ to~$h+1$ via a quorum~$Q$ at height~$h$ on~$B^*$'s chain (justification or timeout cert).
Finality of~$C$ at~$h$ gives a quorum~$Q_F$ each signing a finalize-bearing vote~$b$ with $b.\finalizeHeight = h$ and $b.\finalizeTarget = C$.
Quorum intersection gives $|Q \cap Q_F| \geq n/3$; pick $v \in Q \cap Q_F$ not slashable, with~$b \in Q_F$ its finalize-bearing vote and~$a$ its contribution to~$Q$ at height~$h$ on~$B^*$'s chain.
Non-slashability rules out $\finalizeConflict(a, b)$, so $a.\target \neq \bot$ and $a.\target = C$ (since $a.\height = h = b.\finalizeHeight$ and $b.\finalizeTarget = C \neq \bot$).
Hence~$a$ is a fresh justification vote (timeouts have $a.\target = \bot$) processed on~$B^*$'s chain at state-height~$h$, and freshness gives $C = a.\target \preceq B^* \preceq B$.
\end{proof}

\begin{lemma}[Finalized blocks form a chain]\label{lem:finchain}
Unless $\geq n/3$ validators are slashable: any two finalized checkpoints $(C, h), (C', h')$ are compatible, with $C \preceq C'$ whenever $h \leq h'$.
\end{lemma}

\begin{proof}
\emph{Case $h = h'$.}
Finality of~$C$ gives a quorum~$Q_F$ with $\finalizeHeight = h$, $\finalizeTarget = C$.
Finality of~$C'$ requires justification of~$C'$, giving a justification quorum~$Q'$ at height~$h$ with target~$C'$.
By quorum intersection, $|Q_F \cap Q'| \geq n/3$; pick $v \in Q_F \cap Q'$ not slashable.
E1 (Definition~\ref{def:slashing}) applied to $v$'s votes in~$Q_F, Q'$ forces $C = C'$.

\emph{Case $h < h'$.}
Let~$Z$ be the chain on which~$C'$ was finalized, at the post-state where the finalize-quorum closed; then $\sigma[Z].h \geq h' > h$, $C' \preceq Z$, and $\sigma[C'].h = h'$ (Remark~\ref{rem:fresh-equiv}).
Lemma~\ref{lem:mainsafety} at~$Z$ gives $C \preceq Z$; Remark~\ref{rem:fresh-equiv} on~$C$'s own finalizing chain gives $\sigma[C].h = h$.
Both~$C, C'$ lie on~$Z$, so comparable; with $\sigma[C].h = h < h' = \sigma[C'].h$, state-height monotonicity (Lemma~\ref{lem:slotmono}) forces $C \prec C'$.
\end{proof}

\begin{theorem}[Accountable safety]\label{thm:safety}
Unless $\geq n/3$ validators are slashable, no two conflicting blocks can be finalized.
\end{theorem}

\begin{proof}
Immediate from Lemma~\ref{lem:finchain}: any two finalized blocks are compatible.
\end{proof}
\section{Fork-choice store}\label{sec:store}

A node may track multiple chains simultaneously; the \emph{store} is the node-local data structure that mediates which chain the node follows.
It maintains a single \emph{root block}~$J$ --- the block of the highest-key justification ever observed --- together with that justification's height~$h_j$.
Each justification firing is compared against $(h_j, \hash(J))$ in lex order, and $(J, h_j)$ is updated whenever a new justification strictly dominates.
Because timeouts replace every prefix-notarization mechanism, the comparison uses the lexicographic pair $(h, \hash(C))$ associated with each justification firing $(C, h)$: height first, hash tiebreak; under $f < n/3$ the tiebreak is rarely needed and exists for robustness only.

\begin{definition}[Store]\label{def:store}
A node maintains a \emph{store}
\[
  \Sigma \;=\; (\sigma,\; \T,\; F,\; J,\; h_j,\; \hmax),
\]
where:
\begin{itemize}
  \item $\sigma$ is the \emph{state map}, assigning each accepted block its per-chain post-state (Section~\ref{sec:model});
  \item $\T$ is the \emph{block tree}, the set of accepted blocks;
  \item $F$ is the \emph{store-finalized block};
  \item $J$ is the \emph{store root}: the block of the highest-key justification ever observed;
  \item $h_j$ is the height component of that justification key;
  \item $\hmax \in \mathbb{N}$ is the maximum state-height $\sigma[B].h$ over $B \in \T$.
\end{itemize}
The genesis store has $\T = \{\genesis\}$, $F = J = \genesis$, $h_j = 0$, $\hmax = 1$ (matching $\sigma[\genesis].h = 1$ from Definition~\ref{def:height}).
Section~\ref{sec:healing} extends $\Sigma$ with a vote pool $\Sigma.\mathcal{V}$ for view-merge processing.
\end{definition}

The capital~$\Sigma$ distinguishes the store from the per-chain state~$\sigma$.
We write $\Sigma.\sigma$, $\Sigma.\T$, $\Sigma.F$, $\Sigma.J$, $\Sigma.h_j$, $\Sigma.\hmax$ for field access.
The store key associated with the root is $(\Sigma.h_j,\, \hash(\Sigma.J))$.

\paragraph{Viable subtree.}
The store also defines a \emph{viable subtree} $\viableTree(\Sigma) \subseteq \Sigma.\T$, gating which blocks $\getConfirmed$ may return.

\begin{definition}[Viable subtree]\label{def:viable}
A \emph{leaf} of $\Sigma.\T$ is a block with no proper descendant in $\Sigma.\T$.
Define \emph{viability} recursively:
\begin{itemize}
  \item a leaf~$B$ is \emph{viable} iff $\sigma[B].h \geq \Sigma.\hmax - 1$;
  \item an internal block~$B$ is viable iff some descendant of~$B$ in $\Sigma.\T$ is viable.
\end{itemize}
The \emph{viable subtree} is $\viableTree(\Sigma) \coloneqq \{B \in \Sigma.\T : B \text{ is viable}\}$.
Equivalently, $B \in \viableTree(\Sigma)$ iff some leaf $L \succeq B$ has $\sigma[L].h \geq \Sigma.\hmax - 1$.
\end{definition}

The viability filter discards branches whose tip lags the most recently advanced chain by more than one height.
Under $f < n/3$, Lemma~\ref{lem:mainsafety} forces every block at $\sigma$-height strictly greater than~$h_F$ to descend from~$F$, so when $\hmax > h_F$ any block at $\sigma$-height~$\hmax$ descends from~$F$ (and when $\hmax = h_F$, $F$ itself satisfies $\sigma[F].h = \hmax \geq \hmax - 1$); either way the filter is automatic for finalized blocks.
Blocks at $\sigma$-height $\hmax - 1$ may still conflict with~$F$, since mainsafety constrains only strictly higher heights --- this is why $\Sigma.J$ remains the primary walk-from block in the cascade whenever it has not slipped behind the frontier.
$\Sigma.F$ itself is always viable (Lemma~\ref{lem:F-viable} below): the \onBlock assertion $\Sigma.F \preceq B$ makes every newly admitted block a descendant of $\Sigma.F$, so the block driving any $\Sigma.\hmax$ bump is itself a viable witness for $\Sigma.F$.

\paragraph{Store mutators.}
The acceptance routine \onBlock (Figure~\ref{alg:store}) admits~$B$ only if $B.\parent \in \Sigma.\T$ and $\Sigma.F \preceq B$.
It runs \processSlot once per slot from $\Sigma.\sigma[B.\parent].s + 1$ up to $B.\slot$ (so any height advance enabled by the parent's accumulated tally fires before~$B$'s votes are folded in), then $\processBlock(\cdot, B)$ which folds in $B$'s votes and runs \finalizeProcessBlock to fire any further cert event enabled by them; the resulting post-state $\sigma'$ becomes $\Sigma.\sigma[B]$.
The post-state then offers the justified checkpoint $(\sigma'.J,\, \sigma'.h_j)$ to \updateJustified and the finalized block $\sigma'.F$ to \updateFinalized; $\hmax$ is bumped to $\max(\Sigma.\hmax,\, \sigma'.h)$ before either guard is evaluated, so the viability filter sees the new maximum.

$\updateJustified(\Sigma, J', h')$ first filters out any candidate~$J'$ with $J' \not\succeq \Sigma.F$ and then updates $(\Sigma.J,\, \Sigma.h_j)$ iff the candidate's justification key strictly exceeds $(\Sigma.h_j,\, \hash(\Sigma.J))$ in lex order; this maintains $J$ incrementally as the lex-max justification ever observed on a chain extending the current~$F$, never recomputed.
$\updateFinalized(\Sigma, F')$ sets $\Sigma.F \gets F'$ iff $F' \succ \Sigma.F$, $F' \preceq \Sigma.J$, \emph{and} $F' \in \viableTree(\Sigma)$; the final clause is the \emph{viability guard}, which prevents $\Sigma.F$ from settling at a block the height filter has rendered unreachable from $\getConfirmed$.

$\getConfirmed(\Sigma, \Omega)$ returns a block on the available chain that, under synchrony, is safe in the sense that it will not be reverted as long as more than half of the online validators are honest.
Here $\Omega$ represents the state of the available chain --- whatever extra information the validator uses to disambiguate among viable descendants of the walk-from block.
The protocol fixes the walk-from rule via a two-way cascade: walk from $\Sigma.J$ when $\Sigma.\hmax = \Sigma.h_j + 1$, and otherwise walk from $\Sigma.F$ (always viable, Lemma~\ref{lem:F-viable}).
A justification at height~$h$ on a processed chain forces that chain's state-height to~$h + 1$, so $\Sigma.\hmax \geq \Sigma.h_j + 1$ always; the gate therefore holds precisely when no timeout has advanced any chain's state-height further past $\Sigma.J$'s height.
When the gate holds, $\sigma[\Sigma.J].h = \Sigma.h_j = \Sigma.\hmax - 1$ meets the leaf-viability bound and $\Sigma.J \in \viableTree(\Sigma)$.
When it fails, $\Sigma.J$ is ``stale'' --- some chain has advanced past $\Sigma.J$'s height via timeout cert --- and the adversary cannot force canonical to track such a justification even if its key dominates by lex.
The choice among admissible descendants is left to~$\Omega$.
Different validators may pass different~$\Omega$, and a single validator may pass different~$\Omega$ at different call sites.

The $F$-filter in \updateJustified never discards a key-dominant event: Theorem~\ref{thm:fleqr} below shows that $F$ advances only to blocks $\preceq J$, so every justification whose key could beat the current root remains on the $F$-chain.

\subsection{Store invariants and safety}

\subsubsection*{Unconditional invariants}

\begin{lemma}[$h_j$ monotonicity]\label{lem:Rs-key-monotone}
$\Sigma.h_j$ is non-decreasing over time, and the justification key $(\Sigma.h_j,\, \hash(\Sigma.J))$ is non-decreasing in lex order.
\end{lemma}

\begin{proof}
Both $\Sigma.J$ and $\Sigma.h_j$ change only inside \updateJustified, whose guard enforces that the new key strictly exceeds the old in lex order; in particular the height coordinate $h_j$ is the leading component, so it is non-decreasing.
\end{proof}

\begin{theorem}[Local irreversibility of finality]\label{thm:finperm}
Once a node sets $\Sigma.F = F$, $\Sigma.F$ descends from~$F$ at all future times.
\end{theorem}

\begin{proof}
$\Sigma.F$ is updated only by \updateFinalized, whose guard $F' \succ \Sigma.F$ forces strict extension along~$\preceq$.
\end{proof}

\begin{theorem}[$F \preceq J$]\label{thm:fleqr}
The store maintains $\Sigma.F \preceq \Sigma.J$ at all times.
\end{theorem}

\begin{proof}
By induction on store operations.
At genesis, $F = J = \genesis$, so $F \preceq J$.
Assume the invariant before a call to either mutator in Figure~\ref{alg:store}.

\emph{\updateJustified.}
$F$ is untouched, and $J$ either stays or is replaced by a candidate~$J'$ that passed the $J' \succeq \Sigma.F$ filter; either way $J \succeq F$.

\emph{\updateFinalized.}
$J$ is untouched, and the guard $F' \preceq \Sigma.J$ ensures $F \gets F' \preceq J$.
\end{proof}

\begin{remark}[Store-finalized invariant]\label{rem:fs-invariant}
At all times $\Sigma.F$ is either $\genesis$ or a block that was finalized on some chain; moreover, whenever a justification with target~$J^\star$ has fired at height~$h^\star$ on some chain processed by the node, the justification $(J^\star,\, h^\star)$ was offered to \updateJustified once $J^\star \succeq \Sigma.F$ at the moment of offering.
$\Sigma.F$ only advances via \updateFinalized to $F' = \sigma[B].F$, and $\sigma[B].F \neq \genesis$ requires the finality branch of \processHeight to have fired on~$B$'s chain, so $F'$ is a block finalized on that chain.
\end{remark}

\begin{remark}[$\viableTree$ shrinks only when $\Sigma.\hmax$ grows]\label{rem:viable-monotone}
Adding a block to $\Sigma.\T$ never removes any block from $\viableTree(\Sigma)$ unless the addition also increases $\Sigma.\hmax$.
The viability threshold $\Sigma.\hmax - 1$ depends only on $\Sigma.\hmax$; while it is fixed, every previously viable leaf remains viable and every previously viable internal block retains its viable leaf witness, so $\viableTree(\Sigma)$ can only grow.
Only an $\Sigma.\hmax$ bump can disqualify leaves whose $\sigma$-height falls below the new threshold.
\end{remark}

\begin{lemma}[$\Sigma.F$ is always viable]\label{lem:F-viable}
$\Sigma.F \in \viableTree(\Sigma)$ at all times.
\end{lemma}

\begin{proof}
At genesis, $\Sigma.F = \genesis$ and $\Sigma.\hmax = 1 = \sigma[\genesis].h \geq \Sigma.\hmax - 1$, so $\genesis$ is a viable leaf.
Inductively, $\Sigma.F$ changes only via \updateFinalized, whose viability guard ensures the new value is viable.
Block additions that do not bump $\Sigma.\hmax$ preserve $\Sigma.F$'s viability by Remark~\ref{rem:viable-monotone}.
The remaining case is an $\Sigma.\hmax$ bump in \onBlock with new block~$B$ satisfying $\sigma[B].h > \Sigma.\hmax$; after the bump $\Sigma.\hmax = \sigma[B].h$, and the assertion $\Sigma.F \preceq B$ makes~$B$ --- a leaf at the moment of addition with $\sigma[B].h$ equal to the new $\Sigma.\hmax$ --- a viable leaf descendant of~$\Sigma.F$, witnessing $\Sigma.F \in \viableTree(\Sigma)$.
\end{proof}

\begin{corollary}[\getConfirmed is total]\label{cor:getConfirmed-total}
For every store~$\Sigma$ and auxiliary~$\Omega$, $\getConfirmed(\Sigma, \Omega)$ returns a block in $\viableTree(\Sigma)$ at $\sigma$-height $\geq \Sigma.\hmax - 1$.
\end{corollary}

\begin{proof}
If the cascade gate $\Sigma.\hmax = \Sigma.h_j + 1$ holds, $R = \Sigma.J$ with $\sigma[\Sigma.J].h = \Sigma.h_j = \Sigma.\hmax - 1$ (Remark~\ref{rem:fresh-equiv}); state-height monotonicity along $\preceq$ (Lemma~\ref{lem:slotmono}) gives every descendant of~$\Sigma.J$ in $\Sigma.\T$ a $\sigma$-height $\geq \Sigma.\hmax - 1$, so any leaf of $\Sigma.\T$ above $\Sigma.J$ is a viable leaf and witnesses $\Sigma.J \in \viableTree(\Sigma)$.
Otherwise $R = \Sigma.F$, and Lemma~\ref{lem:F-viable} gives a viable leaf descending from~$\Sigma.F$ in $\Sigma.\T$.
\end{proof}

\begin{theorem}[Fork-choice consistency]\label{thm:fcconsistency}
Once a node sets $\Sigma.F = F$, $\getConfirmed(\Sigma, \Omega)$ returns a block descending from~$F$ at all future times, for every $\Omega$.
\end{theorem}

\begin{proof}
By Theorems~\ref{thm:finperm} and~\ref{thm:fleqr}, $F \preceq \Sigma.F \preceq \Sigma.J$ at all future times.
Both branches of $\getConfirmed$ return a descendant of $\Sigma.J$ or $\Sigma.F$ (Corollary~\ref{cor:getConfirmed-total}); either descends from~$F$.
\end{proof}

\subsubsection*{Further properties under fewer than $n/3$ slashable validators}

\begin{lemma}[Justification chain]\label{lem:certchain}
Unless $\geq n/3$ validators are slashable: if~$F$ is finalized at height~$h_F$ and a justification $(C,\, h)$ has fired on any chain processed by the node with $h \geq h_F$, then $F$ and~$C$ are compatible, with $F \prec C$ when $h > h_F$.
\end{lemma}

\begin{proof}
If $h_F = 0$ then $F = \genesis$ and compatibility is trivial; assume $h_F \geq 1$.
The justification fired at the post-state of some processed block~$B$ whose \onBlock call (whether in the slot loop's \processSlot or in \processBlock's \finalizeProcessBlock) triggered the justification branch with $(\sigma[B].J,\, \sigma[B].h_j) = (C,\, h)$, so $C \preceq B$ and $\sigma[B].h \geq h + 1$ (justification advances height).
Lemma~\ref{lem:mainsafety} at $\sigma[B].h > h_F$ gives $F \preceq B$, and with $C \preceq B$ this yields $F \sim C$.
For $h > h_F$: $\sigma[F].h = h_F < h = \sigma[C].h$ by Remark~\ref{rem:fresh-equiv}, and state-height monotonicity along $\preceq$ (Lemma~\ref{lem:slotmono}) forces $F \prec C$ on the comparable pair.
\end{proof}

\begin{lemma}[Upgrade property]\label{lem:upgrade}
Unless $\geq n/3$ validators are slashable: if~$F$ is finalized at height~$h_F$ and some block~$B$ with $\sigma[B].J = F$ and $\sigma[B].h_j = h_F$ has been processed by the node, then $\Sigma.J \succeq F$ at all future times.
\end{lemma}

\begin{proof}
Let $F^0 \coloneqq \Sigma.F$ at the moment $\onBlock(B)$ offered $(F, h_F)$ to \updateJustified.
Both $F^0$ and $F$ are genesis or finalized (Remark~\ref{rem:fs-invariant}), so Lemma~\ref{lem:finchain} makes them comparable.
If $F \preceq F^0$, Theorem~\ref{thm:fleqr} gives $\Sigma.J \succeq \Sigma.F \succeq F^0 \succeq F$ at that moment, and Theorems~\ref{thm:finperm} and~\ref{thm:fleqr} preserve $\Sigma.J \succeq \Sigma.F \succeq F$ thereafter.
Otherwise $F^0 \prec F$ strictly: the $F$-filter $\Sigma.F \preceq F$ passes, so the call brings $(\Sigma.h_j, \hash(\Sigma.J)) \geq (h_F, \hash(F))$ in lex order, preserved by Lemma~\ref{lem:Rs-key-monotone}.
At any later moment, $\Sigma.h_j \geq h_F$, and the current $\Sigma.J$ is justified at some $h' \geq h_F$ on a processed chain: if $h' > h_F$, Lemma~\ref{lem:certchain} gives $F \prec \Sigma.J$; if $h' = h_F$, the justification quorum~$Q'$ for $\Sigma.J$ at $h_F$ and the finality quorum~$Q_F$ for~$F$ at $h_F$ intersect in $\geq n/3$, and a non-slashable $v \in Q' \cap Q_F$ forces $\Sigma.J = F$ by E1.
Either way $\Sigma.J \succeq F$.
\end{proof}

\begin{lemma}[Finalized blocks are viable]\label{lem:viable-finalized}
Unless $\geq n/3$ validators are slashable: if~$F$ is finalized at height~$h_F$ and some block~$B$ with $\sigma[B].J = F$ and $\sigma[B].h_j = h_F$ has been processed by the node, then $F \in \viableTree(\Sigma)$ at all future times.
\end{lemma}

\begin{proof}
Processing of~$B$ with $\sigma[B].h_j = h_F$ means the justification branch of \processHeight fired with target~$F$ at pre-state height~$h_F$ on~$B$'s chain, advancing~$\sigma$ to $\sigma[B].h \geq h_F + 1$; hence $\Sigma.\hmax \geq h_F + 1 > h_F$ at all future times.
Pick any leaf $L \in \Sigma.\T$ with $\sigma[L].h = \Sigma.\hmax$ (some block achieves~$\Sigma.\hmax$ by definition, and any leaf descendant preserves the maximum by state-height monotonicity).
Then $L$ is viable, and Lemma~\ref{lem:mainsafety} gives $F \preceq L$, so $L$ witnesses $F \in \viableTree(\Sigma)$.
\end{proof}

\begin{theorem}[Local acceptance of finality updates]\label{thm:finlive}
Unless $\geq n/3$ validators are slashable: if a block~$B$ is processed by \onBlock and $\sigma[B].F = F'$, then after processing $\Sigma.F \succeq F'$.
\end{theorem}

\begin{proof}
If $\Sigma.F \succeq F'$ already, processing~$B$ preserves the bound by monotonicity of~$\Sigma.F$ (Theorem~\ref{thm:finperm}).
Otherwise $\Sigma.F$ is genesis or finalized (Remark~\ref{rem:fs-invariant}), so Lemma~\ref{lem:finchain} gives $\Sigma.F \sim F'$ and hence $\Sigma.F \prec F'$.
$\sigma[B].F = F'$ requires some ancestor~$B' \preceq B$ with $\sigma[B'].J = F'$ and $\sigma[B'].h_j = h_{F'}$; $B'$ was processed before~$B$ (parent-first), so Lemma~\ref{lem:upgrade} gives $\Sigma.J \succeq F'$, hence the $F' \preceq \Sigma.J$ guard passes.
For the viability guard, Lemma~\ref{lem:viable-finalized} gives $F' \in \viableTree(\Sigma)$.
All three guards pass, so $\Sigma.F \gets F'$.
\end{proof}

\begin{theorem}[Lock-in]\label{thm:lockin}
Unless $\geq n/3$ validators are slashable: if block~$F$ is finalized at height~$h_F$ on any chain, and some block~$B$ with $\sigma[B].J = F$ and $\sigma[B].h_j = h_F$ has been processed by the node, then $\Sigma.J \succeq F$ at all future times, $F \in \viableTree(\Sigma)$ at all future times, and $\getConfirmed(\Sigma, \Omega)$ always returns a descendant of~$F$, for every $\Omega$.
\end{theorem}

\begin{proof}
$\Sigma.J \succeq F$ by Lemma~\ref{lem:upgrade}, so $\sigma[\Sigma.J].h = \Sigma.h_j \geq h_F$ by state-height monotonicity along $F \preceq \Sigma.J$ (using $\sigma[F].h = h_F$ from Remark~\ref{rem:fresh-equiv}); $F \in \viableTree(\Sigma)$ by Lemma~\ref{lem:viable-finalized}.
If $\Sigma.\hmax = \Sigma.h_j + 1$, $\getConfirmed$'s first branch fires and returns a descendant of $\Sigma.J \succeq F$.
Otherwise $\Sigma.\hmax \geq \Sigma.h_j + 2 \geq h_F + 2$, and \getConfirmed's second branch returns a block~$T$ with $\sigma[T].h \geq \Sigma.\hmax - 1 \geq h_F + 1 > h_F$, so Lemma~\ref{lem:mainsafety} gives $F \preceq T$.
\end{proof}

\begin{theorem}[Order independence]\label{thm:orderindep}
Unless $\geq n/3$ validators are slashable: the store state after processing a set of blocks depends only on which blocks have been processed, not on their order.
Consequently, two nodes that have processed the same set of blocks agree on $(\sigma,\, \T,\, F,\, J,\, h_j,\, \hmax)$, and hence on $\viableTree$.
\end{theorem}

\begin{proof}
Each block's post-state is a deterministic function of itself and its parent's post-state, so $\sigma$ depends only on the processed set; in particular $\Sigma.\hmax$ and $\viableTree(\Sigma)$ depend only on the set.

\emph{$F$.} By Lemma~\ref{lem:finchain}, $\{\sigma[B].F : B \in \T\}$ is a chain with maximum $F_{\max}$, attained at some processed~$B^\ast$.
Remark~\ref{rem:fs-invariant} forces $\Sigma.F$ into this chain, so $\Sigma.F \preceq F_{\max}$; once $B^\ast$ has been processed, Theorem~\ref{thm:finlive} gives $\Sigma.F \succeq F_{\max}$.

\emph{$J$.} The multiset~$D$ of cert descriptors offered to \updateJustified is a deterministic function of the processed set: each block contributes its post-state's justified checkpoint exactly once.
Write $K_{\max} \coloneqq (h_{F_{\max}},\, \hash(F_{\max}))$.
For any $(C, h) \in D$ with $h \geq h_{F_{\max}}$: if $h > h_{F_{\max}}$, Lemma~\ref{lem:certchain} gives $C \succ F_{\max}$; if $h = h_{F_{\max}}$, finality of $F_{\max}$ at $h_{F_{\max}}$ provides a finality quorum~$Q_F$ of size $\geq 2n/3$ with $\finalizeHeight = h_{F_{\max}}$ and $\finalizeTarget = F_{\max}$, while the descriptor's existence gives a justification quorum~$Q'$ at height~$h_{F_{\max}}$ targeting~$C$; quorum intersection picks a non-slashable $v \in Q' \cap Q_F$, and E1 (as in Lemma~\ref{lem:upgrade}) forces $C = F_{\max}$.
Descriptors with $h < h_{F_{\max}}$ have key strictly below $K_{\max}$ on the leading coordinate, so once $\Sigma.h_j \geq h_{F_{\max}}$ holds they are dominated.
Hence $\Sigma.J$'s final value is the lex-max over $\{(C, h) \in D : h \geq h_{F_{\max}}\}$, a deterministic function of the processed set.
\end{proof}

\begin{lemma}[Justifications stay at or below~$\Sigma.h_j$]\label{lem:no-high-just}
Unless $\geq n/3$ validators are slashable, every justification $(C, h)$ that has fired on any chain processed by the node satisfies $h \leq \Sigma.h_j$ at all subsequent times.
\end{lemma}

\begin{proof}
The justification fires inside an \onBlock call on some processed block~$B$, at which point Remark~\ref{rem:fs-invariant} gives the descriptor $(C, h)$ offered to \updateJustified.
$C$ lies on $B$'s chain (it is the justified target), and so does $\Sigma.F$ (by $\Sigma.F \preceq B$), so $\Sigma.F$ and~$C$ are comparable.
If $\Sigma.F \not\preceq C$, then $\Sigma.F \succ C$, so $h_F = \sigma[\Sigma.F].h \geq \sigma[C].h = h$ by state-height monotonicity (Lemma~\ref{lem:slotmono}, applied along $C \prec \Sigma.F$, using $\sigma[C].h = h$ from Remark~\ref{rem:fresh-equiv}).
Combining $\Sigma.F \preceq \Sigma.J$ (Theorem~\ref{thm:fleqr}) with $\sigma[\Sigma.J].h = \Sigma.h_j$ (Remark~\ref{rem:fresh-equiv}) and the same state-height monotonicity along $\Sigma.F \preceq \Sigma.J$ gives $\Sigma.h_j \geq h_F \geq h$.
Otherwise the $F$-filter passes and \updateJustified's lex test forces $\Sigma.h_j \geq h$ after the call.
Lemma~\ref{lem:Rs-key-monotone} preserves the bound at all later times.
\end{proof}

\section{Inactivity leak}\label{sec:leak}

This section extends the per-chain state machine of Section~\ref{sec:model} with a penalty counter and a new transition function \processInactivityPenalties, invoked at every \processHeight call by an extended \finalizeProcessBlock (Figure~\ref{alg:leak-processslot}); \processSlot picks up the extension transparently through its call to \finalizeProcessBlock, while \processBlock, \processHeight, and \processVote are unchanged.
Write $T = \getConfirmed(\Sigma, \Omega)$ for the \emph{canonical chain}, with $\Omega$ the validator's auxiliary state.

Two results follow.
Theorem~\ref{thm:tightness} is purely state-logic: on any fixed chain, the total penalty increment over a non-finality period is at least $(T-1) \cdot n/3$, with no hypothesis on synchrony, honesty, or the slashable set.
Theorem~\ref{thm:fairness} is local to a single honest~$i$: the vote~$i$ produces per Definition~\ref{def:voting-rule}, once included on any extension of~$i$'s canonical chain still at the same state-height, exempts~$i$ from Layer~1 penalties at that state-height.
Layer~2 exemption is out of scope: it depends on quorum formation on a common target, not on any local property of~$i$.

\subsection{Penalty units}

Augment the per-chain state tuple with one non-negative counter $\sigma.\penalties[1..n]$, initialized to~$0$ at genesis; no other field is added or modified.
Figure~\ref{alg:leak-processslot} replaces \finalizeProcessBlock from Section~\ref{sec:model} with a version that snapshots the pre-advance state and invokes \processInactivityPenalties after \processHeight; \processSlot inherits the change automatically since it calls \finalizeProcessBlock.
The store of Section~\ref{sec:store} invokes the extended routine in place of the base one.

\emph{Layer~1 (stall).} Fires when the slot did not advance height.
Every~$i$ with $\sigma_{\mathrm{post}}.\timeouts[i] = \mathtt{false}$ is bumped, i.e.\ every validator that failed to cast a vote setting the timeout marker at the current height ($\timeouts$ is not reset at a stall).
The marker is set by either a timeout vote (target~$\bot$) at the current height or a height-fresh justification vote with target on this chain (Section~\ref{sec:model}).

\emph{Layer~2 target.} Fires when the justification branch of \processHeight did not fire (else $J$ would strictly advance by Lemma~\ref{lem:slotmono}).
$T^*$ is the plurality target across $\sigma_{\mathrm{pre}}.\targets$ (deterministic tiebreak), or $\bot$ if no justification votes have been registered; the second disjunct of the guard bumps non-voters when $T^* = \bot$ (in which case all~$n$ validators are bumped).
Since no $2n/3$-quorum on a single target formed, the plurality count is $< 2n/3$, so $|\{i : \sigma_{\mathrm{pre}}.\targets[i] \neq T^* \vee \sigma_{\mathrm{pre}}.\targets[i] = \bot\}| > n/3$.

\emph{Layer~2 finalize.} Fires when a pending-finality gap is open at the start of the slot ($\sigma_{\mathrm{pre}}.F \prec \sigma_{\mathrm{pre}}.J$) and $F$ did not advance; every $i \notin \sigma_{\mathrm{pre}}.P$ is bumped.
The finality branch of \processHeight, when it fires, sets $\sigma_{\mathrm{post}}.F = \sigma_{\mathrm{pre}}.J \succ \sigma_{\mathrm{pre}}.F$, so this guard is skipped automatically.

\subsection{Leak tightness}

\begin{definition}[Non-finality period]\label{def:non-finality-period}
Fix a chain and write $\sigma_{\mathrm{pre}}^{(t)}, \sigma_{\mathrm{post}}^{(t)}$ for its pre/post-\processHeight snapshots at slot~$t$.
A \emph{non-finality period} is a maximal interval $[t_1, t_T]$ of consecutive slots on which $F$ is stuck, i.e., $\sigma_{\mathrm{post}}^{(t)}.F = \sigma_{\mathrm{pre}}^{(t)}.F$ throughout.
Each such slot falls into exactly one category:
\begin{itemize}
  \item \emph{pending-stuck}: $\sigma_{\mathrm{pre}}^{(t)}.F \prec \sigma_{\mathrm{pre}}^{(t)}.J$ (pending-finality gap open at the start);
  \item \emph{stable-stuck}: $\sigma_{\mathrm{pre}}^{(t)}.F = \sigma_{\mathrm{pre}}^{(t)}.J$ and $J$ does not advance at~$t$;
  \item \emph{transition}: $\sigma_{\mathrm{pre}}^{(t)}.F = \sigma_{\mathrm{pre}}^{(t)}.J$ and the justification branch fires at~$t$ (opening a gap by slot's end).
\end{itemize}
\end{definition}

\begin{theorem}[Non-finality period penalty bound]\label{thm:tightness}
Over any non-finality period of length $T \geq 1$ on any chain, the total penalty increment across all validators assessed by \processInactivityPenalties on that chain is at least $(T - 1) \cdot n/3$.
\end{theorem}

\begin{proof}
Fix a non-finality period $[t_1, t_T]$.
We show (i) at most one slot is transition, and (ii) every pending-stuck or stable-stuck slot contributes $> n/3$ to the total penalty increment.

\emph{(i) At most one transition slot.}
After a transition at~$t$, $\sigma_{\mathrm{post}}^{(t)}.J \succ \sigma_{\mathrm{pre}}^{(t)}.F$; since $F$ does not advance in the period and $J$ is monotone across slots, $\sigma_{\mathrm{pre}}^{(t')}.F \prec \sigma_{\mathrm{pre}}^{(t')}.J$ at every later $t' > t$, making every such $t'$ pending-stuck.

\emph{(ii) Per-slot charge $> n/3$.}

\emph{Pending-stuck.} The Layer~2 finalize guard fires at~$t$.
The finality branch of \processHeight did not fire (else $F$ would advance), so $|\sigma_{\mathrm{pre}}^{(t)}.P| < 2n/3$ and $|\{i : i \notin \sigma_{\mathrm{pre}}^{(t)}.P\}| > n/3$.

\emph{Stable-stuck.} The Layer~2 target guard fires at~$t$.
The justification branch did not fire, so the plurality count on $\sigma_{\mathrm{pre}}^{(t)}.\targets$ is $< 2n/3$ (counting $T^* = \bot$ as~$0$); hence the bumped set has size $> n/3$.

At most one slot is transition, so at least $T - 1$ slots are pending-stuck or stable-stuck; each contributes $> n/3$ by~(ii), and Layer~1 adds only non-negative charges.
Summing gives total increment $\geq (T-1) \cdot n/3$.
\end{proof}

\subsection{Honest voting rule}

We state the honest rule explicitly, as it underpins the leak fairness proof.
The rule consumes three pieces of validator-local state: the store~$\Sigma$ (drives fork-choice), the auxiliary~$\Omega$ (disambiguates among admissible \getConfirmed outputs), and a new \emph{vote record}~$\Delta$ (the validator's own past-vote summary, used to avoid slashable signatures).

\begin{definition}[Vote record]\label{def:voterecord}
The vote record~$\Delta$ associates to each height~$h$ a triple $(\Delta.T(h),\, \Delta.\tau(h),\, \Delta.\lambda(h)) \in (\mathrm{Block} \cup \{\bot\}) \times \mathtt{bool} \times \mathtt{bool}$, initialized to $(\bot,\, \mathtt{false},\, \mathtt{false})$:
$\Delta.T(h)$ is~$\bot$ until~$i$'s first non-$\bot$ vote at~$h$, after which it pins the target of that vote; $\Delta.\tau(h)$ records that~$i$ has voted timeout at~$h$; and $\Delta.\lambda(h)$ records that~$i$ has voted to finalize~$\Delta.T(h)$ at height~$h$ (i.e.\ has cast a finalize-bearing vote with $\finalizeHeight = h$ and $\finalizeTarget = \Delta.T(h)$).
$\Delta$ is updated only by \recordVote (Figure~\ref{alg:voting-rule}), invoked once per signed vote.
The honest reachable per-height configurations are $(\bot, \mathtt{false}, \mathtt{false})$, $(T, \mathtt{false}, \mathtt{false})$, $(T, \mathtt{false}, \mathtt{true})$, and $(T, \mathtt{true}, \mathtt{false})$; the configuration $(T, \mathtt{true}, \mathtt{true})$ is the slashable trace (a finalize-bearing vote at~$h$ alongside a timeout vote at~$h$).

\smallskip
\noindent\emph{Implementation.}
$\Delta$ may safely drop entries at heights below $\Sigma.\hmax - 1$; \getFinalVote then conservatively clears the finalize commitment at older heights (because the gate $\Delta.T(h_f) = T_f$ fails on a missing entry), but every emitted vote remains E1-safe (Lemma~\ref{lem:honest-e1-safety}).
$\Delta$ is the entire state needed for safe vote signing on the validator-client side, depending only on~$\Delta$, separated from fork-choice on the beacon-client side (depending on $\Sigma$ and~$\Omega$).
\end{definition}

\begin{definition}[Honest voting rule]\label{def:voting-rule}
At each voting opportunity, $i$ produces its vote by $\getFinalVote(\Delta,\, \getBaseVote(\Sigma, \Omega))$ (Figure~\ref{alg:voting-rule}) and signs the result.
$\getBaseVote$ reads fork-choice state to assemble an unconstrained candidate; $\getFinalVote$ adjusts it to respect~$\Delta$'s prior commitments and folds the new vote back into~$\Delta$ via \recordVote.
\end{definition}

\begin{figure}[H]
\caption{Honest voting rule}\label{alg:voting-rule}
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]

\Function{\getBaseVote}{$\Sigma,\; \Omega$} \Comment{the unconstrained vote}
  \State $T \gets \getConfirmed(\Sigma,\; \Omega)$
  \State $h_c \gets \sigma[T].h$; \;\; $h_f \gets \sigma[T].h_j$; \;\; $T_f \gets \sigma[T].J$
  \State \Return $(i,\; h_c,\; T,\; h_f,\; T_f)$
\EndFunction
\vspace{3pt}

\Function{\getFinalVote}{$\Delta,\; v$}
  \State $(i,\; h_c,\; T,\; h_f,\; T_f) \gets v$
  \State \textbf{assert} $h_c > h_f$ \Comment{vote-height strictly above the finalize-commitment height}
  \vspace{3pt}
  \If{$\Delta.\lambda(h_c) = \mathtt{true}$} \Comment{lock at $h_c$: re-submit lock target}
    \State $T \gets \Delta.T(h_c)$
  \ElsIf{$\Delta.T(h_c) \neq \bot$} \Comment{voted at $h_c$ before, not locked: timeout}
    \State $T \gets \bot$
  \EndIf
  \vspace{3pt}
  \If{$\Delta.T(h_f) \neq T_f$ \textbf{or} $\Delta.\tau(h_f) = \mathtt{true}$} \Comment{cannot E1-safely commit to finalize}
    \State $(h_f,\, T_f) \gets (\bot,\, \bot)$
  \EndIf
  \vspace{3pt}
  \State $v' \gets (i,\; h_c,\; T,\; h_f,\; T_f)$
  \State $\recordVote(\Delta,\; v')$
  \State \Return $v'$
\EndFunction
\vspace{3pt}

\Function{\recordVote}{$\Delta,\; v$}
  \State $h \gets v.\height$; \;\; $h_f \gets v.\finalizeHeight$
  \If{$v.\target \neq \bot$ \textbf{and} $\Delta.T(h) = \bot$}
    \State $\Delta.T(h) \gets v.\target$
  \ElsIf{$v.\target = \bot$}
    \State $\Delta.\tau(h) \gets \mathtt{true}$
  \EndIf
  \If{$v.\finalizeTarget \neq \bot$}
    \State $\Delta.\lambda(h_f) \gets \mathtt{true}$
  \EndIf
\EndFunction

\end{algorithmic}
\end{boxedminipage}
\end{center}
\end{figure}

The target adjustment in \getFinalVote produces the lock target when~$i$ is locked at~$h_c$; a timeout when~$i$ has voted at~$h_c$ without locking (so a second non-$\bot$ vote at~$h_c$ would violate per-height target consistency); or the unconstrained~$T$ otherwise (first vote at~$h_c$).
The finalize fields are kept iff the gate $\Delta.T(h_f) = T_f \wedge \Delta.\tau(h_f) = \mathtt{false}$ holds (so~$i$ previously voted~$T_f$ at~$h_f$ and has not voted timeout there); the assert $h_c > h_f$ rules out the degenerate case where the vote and the commitment would land at the same height (which would be E1-self-slashable when $T \neq T_f$).
The gate is the same in all three target branches, so locks attach finalize symmetrically with first-votes whenever the gate passes.

\begin{lemma}[Honest E1 safety]\label{lem:honest-e1-safety}
If validator~$i$ follows Definition~\ref{def:voting-rule}, then $i$ never produces a vote-pair $(v_1, v_2)$ with $v_2.\finalizeTarget \neq \bot$, $v_1.\height = v_2.\finalizeHeight$, and $v_1.\target \neq v_2.\finalizeTarget$.
\end{lemma}

\begin{proof}
Write $h_\ast \coloneqq v_2.\finalizeHeight$ and $T_\ast \coloneqq v_2.\finalizeTarget \neq \bot$.
$v_2$ retained finalize fields, so the gate inside \getFinalVote held at $v_2$'s firing: $\Delta.T(h_\ast) = T_\ast \neq \bot$, $\Delta.\tau(h_\ast) = \mathtt{false}$, and $v_2.\target \neq \bot$; the subsequent \recordVote then set $\Delta.\lambda(h_\ast) \gets \mathtt{true}$.
Since $v_2.\target \neq \bot$, the target-adjustment in \getFinalVote did not enter its timeout branch, so $v_2.\target = T$ (first-vote) or $v_2.\target = \Delta.T(h_c)$ (lock); either way $v_2.\target \neq \bot$ at $h_c = v_2.\height$, while $\Delta.T(h_\ast) = T_\ast \neq \bot$ at $h_\ast = v_2.\finalizeHeight$.

If $v_1.\height = h_\ast$ and $v_1$ was cast before~$v_2$: $\Delta.\tau$-monotonicity and $\Delta.\tau(h_\ast) = \mathtt{false}$ at $v_2$'s firing rule out $v_1$ being a timeout at~$h_\ast$, so $v_1.\target \neq \bot$.
$\Delta.T(h_\ast)$ is set once at~$i$'s first non-$\bot$ vote at~$h_\ast$ and pinned thereafter, and the target adjustment in \getFinalVote forces every subsequent non-$\bot$ vote of~$i$ at~$h_\ast$ to equal $\Delta.T(h_\ast)$ (lock branch returns $\Delta.T(h_\ast)$; first-vote branch is unreachable once $\Delta.T(h_\ast) \neq \bot$).
Hence $v_1.\target = \Delta.T(h_\ast) = T_\ast$.

If $v_1.\height = h_\ast$ and $v_1$ was cast after~$v_2$: at $v_1$'s firing $\Delta.\lambda(h_\ast) = \mathtt{true}$ (set by~$v_2$), so the lock branch fires and $v_1.\target = \Delta.T(h_\ast) = T_\ast$ ($\Delta.T(h_\ast)$ is fixed after~$v_2$'s firing).

If $v_1 = v_2$: $v_1.\height = v_2.\height = h_c$, so the hypothesis $v_1.\height = v_2.\finalizeHeight$ requires $h_c = h_\ast$.
At genesis ($h_c = h_j = 0$) this collapses with $J = \genesis$: $v_2.\target = \genesis$ in either the first-vote or lock branch ($\Delta.T(0)$ is either~$\bot$, in which case the unconstrained $T = \genesis$, or already $\genesis$, the only block at~$\sigma$-height~$0$); hence $v_1.\target = v_2.\target = \genesis = T_\ast$.
Post-genesis, every height advance sets $h_j \gets h$ before $h \gets h+1$, giving $h_c \geq h_j + 1 > h_\ast$ at any state where $v_2$ could fire, so $h_c = h_\ast$ does not occur.
\end{proof}

\begin{lemma}[Per-height justification uniqueness]\label{lem:just-unique-height}
If $f < n/3$ and all honest validators follow Definition~\ref{def:voting-rule}: for any two blocks $C_1, C_2$ justified at the same height~$h$ on any chains processed by the node, $C_1 = C_2$.
\end{lemma}

\begin{proof}
For $h = 0$, the only ``justification'' is the genesis pre-justification convention ($\sigma[\genesis].J = \genesis$, $h_j = 0$), which is unique by construction; the rest of the proof handles $h \geq 1$, where a $\geq 2n/3$ quorum is required.

Each justification at height~$h$ on a chain yields a quorum $Q_k$ of size $\geq 2n/3$; quorum intersection picks an honest~$v \in Q_1 \cap Q_2$ contributing a non-$\bot$-target vote at height~$h$ to each.
Per Definition~\ref{def:voterecord} and the target adjustment in \getFinalVote, every non-$\bot$-target vote of~$v$ at height~$h$ has target equal to $\Delta.T(h)$ (set once at $v$'s first such vote and pinned thereafter; subsequent non-$\bot$ votes at~$h$ come from the lock branch, which returns $\Delta.T(h)$).
Hence both contributions have the same target, $C_1 = C_2$.
\end{proof}

\subsection{Leak fairness}

Fairness is local to a single honest validator~$i$: we fix~$i$, its local store~$\Sigma$, its auxiliary state~$\Omega$, and the canonical chain $T = \getConfirmed(\Sigma, \Omega)$.
The claim is that the vote~$v$ produced by Definition~\ref{def:voting-rule}, once included on any extension of~$T$ still at the same state-height, exempts~$i$ from Layer~1 penalties at that state-height.
Height advances reset $\targets$ and $\timeouts$ (Section~\ref{sec:model}), so Layer~2 state at the next height is independent of~$i$'s prior votes and no ``lock'' issue propagates.

The proof rests on two facts about the canonical chain.
First, $T = \getConfirmed(\Sigma,\Omega)$ has $\sigma$-height $\geq \Sigma.\hmax - 1$ by the algorithm spec, so $h_c = \sigma[T].h \in \{\Sigma.\hmax - 1,\, \Sigma.\hmax\}$.
Second, every locally observed justification at height~$h$ is witnessed by some processed block whose post-state advanced to $\sigma$-height~$h+1$, so $h \leq \Sigma.\hmax - 1$.
Together: at $h_c = \Sigma.\hmax$ this node has not observed a justification at~$h_c$, so the lock branch is unreachable; at $h_c = \Sigma.\hmax - 1$ the lock branch forces the lock target to coincide with~$\Sigma.J$ via Lemma~\ref{lem:just-unique-height}, and $\Sigma.J$ is itself viable (its witnessing block sits at $\sigma$-height~$\Sigma.\hmax$ as a descendant of~$\Sigma.J$).

\begin{lemma}[Honest vote sets the timeout marker]\label{lem:honest-vote-sets-timeout}
Assume $f < n/3$ and all honest validators follow Definition~\ref{def:voting-rule}.
Let~$i$ be honest with local store~$\Sigma$ and auxiliary state~$\Omega$, $T = \getConfirmed(\Sigma, \Omega)$, and $h_c = \sigma[T].h$.
Let~$v$ be the vote~$i$ produces at the current slot.
Then either $v.\target = \bot$ with $v.\height = h_c$, or $v.\target \preceq T$ with $\sigma[v.\target].h = h_c$.
\end{lemma}

\begin{proof}
By Definition~\ref{def:voting-rule}, $v.\height = h_c$ in every branch.
The elif-timeout branch ($\Delta.T(h_c) \neq \bot$ with $\Delta.\lambda(h_c) = \mathtt{false}$) emits $v.\target = \bot$, satisfying the first disjunct.
The first-vote-at-$h_c$ branch sets $v.\target = T$ with $\sigma[T].h = h_c$ and $T \preceq T$, satisfying the second disjunct.

The remaining branch is the lock branch.
Let $k \coloneqq v_0.\finalizeHeight$ for the earlier finalize-bearing vote~$v_0$ of~$i$ whose \recordVote call set $\Delta.\lambda(k) \gets \mathtt{true}$; the lock-branch firing condition $\Delta.\lambda(h_c) = \mathtt{true}$ at the current slot requires $k = h_c$, so the current vote's $h_c$ equals $v_0$'s finalize-height.
Write $T_{\mathrm{lock}} \coloneqq v_0.\finalizeTarget$.
For $v_0$ to retain its finalize fields, the gate inside \getFinalVote had to hold at $v_0$'s firing: $\Delta.T(k) = v_0.\finalizeTarget = T_{\mathrm{lock}}$ (with $\Delta.\tau(k) = \mathtt{false}$).
Since $\Delta.T(k)$ is pinned once set (cf.\ the target adjustment in \getFinalVote), it remains $T_{\mathrm{lock}}$ at $v$'s firing, and the lock branch returns $v.\target = \Delta.T(h_c) = T_{\mathrm{lock}}$.
Since $v_0$ is finalize-bearing, \getBaseVote populated its finalize fields from $i$'s then-canonical state, giving $(v_0.\finalizeHeight, v_0.\finalizeTarget) = (k, T_{\mathrm{lock}}) = (\sigma[T^0].h_j, \sigma[T^0].J)$ at $v_0$'s firing, where~$T^0$ is~$i$'s canonical chain at that earlier time.
The justification branch of \processHeight thus fired with target $T_{\mathrm{lock}}$ at pre-state height~$k$ inside some \onBlock call on a processed block~$B^\star$, whose post-state has $\sigma[B^\star].h \geq k + 1$ (height-advance).
Hence $\Sigma.\hmax \geq k + 1 = h_c + 1$, i.e., $h_c \leq \Sigma.\hmax - 1$.

Since $T = \getConfirmed(\Sigma, \Omega)$ has $\sigma[T].h \geq \Sigma.\hmax - 1$ (algorithm spec), combining with the previous inequality yields $h_c = \sigma[T].h = \Sigma.\hmax - 1$ and $\sigma[B^\star].h = \Sigma.\hmax$.
The same height-advance argument applied to $\Sigma.J$'s own justification (its witnessing block has $\sigma$-height $\geq \Sigma.h_j + 1$, and that $\sigma$-height is $\leq \Sigma.\hmax$ by definition of~$\hmax$) gives $\Sigma.h_j \leq \Sigma.\hmax - 1 = h_c$, while Lemma~\ref{lem:no-high-just} applied to $T_{\mathrm{lock}}$'s justification at~$h_c$ gives $h_c \leq \Sigma.h_j$.
Hence $\Sigma.h_j = h_c$, and Lemma~\ref{lem:just-unique-height} forces $\Sigma.J = T_{\mathrm{lock}}$.

$\Sigma.h_j = h_c = \Sigma.\hmax - 1$ gives $\Sigma.\hmax = \Sigma.h_j + 1$, satisfying the cascade gate, so \getConfirmed{} walks from~$\Sigma.J$.
The return descends from~$\Sigma.J = T_{\mathrm{lock}}$ (existence: $B^\star \succeq \Sigma.J$ with $\sigma[B^\star].h = \Sigma.\hmax$ provides the required witness in $\viableTree(\Sigma)$ at $\sigma$-height~$\Sigma.\hmax$, since any leaf $L \succeq B^\star$ inherits $\sigma[L].h = \Sigma.\hmax$ by state-height monotonicity).
Hence $T \succeq \Sigma.J = T_{\mathrm{lock}}$, with $\sigma[T_{\mathrm{lock}}].h = h_c$ by Remark~\ref{rem:fresh-equiv} on the chain witnessing $T_{\mathrm{lock}}$'s justification.
\end{proof}

\begin{theorem}[Layer~1 leak fairness]\label{thm:fairness}
Assume $f < n/3$ and all honest validators follow Definition~\ref{def:voting-rule}.
Let~$i$ be an honest validator, $\Sigma$ its local store, $\Omega$ its auxiliary state, $T = \getConfirmed(\Sigma, \Omega)$ its canonical chain, and $h_c = \sigma[T].h$ the current canonical state-height.
Let $v$ be the vote~$i$ produces at the current slot.
Let $\Sigma'$ be any store obtained by processing additional blocks and votes on top of~$\Sigma$, with auxiliary state $\Omega'$, such that $T' = \getConfirmed(\Sigma', \Omega')$ is an extension of~$T$ with $\Sigma'.\sigma[T'].h = h_c$, and suppose~$v$ is included on~$T'$.
Then for every slot processed on~$T'$ from that point onward, while canonical continues to extend~$T'$ and canonical state-height stays at~$h_c$, the Layer~1 increment to $\sigma.\penalties[i]$ on canonical is zero.
\end{theorem}

\begin{proof}
We show that processing~$v$ on~$T'$ sets $\timeouts[i] = \mathtt{true}$ in $\sigma[T']$.
By Section~\ref{sec:model}, $\timeouts[i]$ is set at processing time iff either (a)~$v$ is a timeout vote at the current state-height ($v.\target = \bot$ and $v.\height = \sigma[T'].h = h_c$), or (b)~$v$'s target is height-fresh on~$T'$ ($v.\target \preceq T'$ with $\sigma[v.\target].h = h_c$, equivalently $v.\target.\slot \geq \sigma[T'].s_h$ via Remark~\ref{rem:fresh-equiv}).

Lemma~\ref{lem:honest-vote-sets-timeout} gives the dichotomy.
In the timeout sub-case, $v = (i, h_c, \bot, \bot, \bot)$ with $\sigma[T'].h = h_c$ satisfies (a) directly.
In the non-timeout sub-case, $v.\target \preceq T \preceq T'$ with $\sigma[v.\target].h = h_c$, so~$v$ is height-fresh on~$T'$ and (b) holds.

Either way, $\timeouts[i] = \mathtt{true}$ in $\sigma[T']$.
$\timeouts$ is per-state and resets only at a height advance; on any canonical chain extending~$T'$, $\sigma$ at $T'$'s descendants inherits $\timeouts[i] = \mathtt{true}$ from $\sigma[T']$, and no height advance fires while canonical state-height stays at~$h_c$, so $\timeouts[i] = \mathtt{true}$ persists on canonical and Layer~1 does not bump~$i$.
\end{proof}

\section{Healing from asynchrony}\label{sec:healing}

This section shows that, after a prolonged period of asynchrony during which the gate $\Sigma.\hmax = \Sigma.h_j + 1$ may have failed and validators may have been driven into divergent local stores, three consecutive honest syncer slots --- under standard synchrony, online-honest-majority, available-chain-relay, and all-honest-awake assumptions --- restore a healed state in which all honest validators agree on the canonical chain and on a fresh justification at the new frontier.

The core simplification compared with the prefix-notarization design is that the cascade gate ($\Sigma.\hmax = \Sigma.h_j + 1$) automatically discards stale justifications that fall below the frontier, so the adversary cannot stage an unbounded ``staircase'' of reorgs from old held justifications.
The bound is tight against an adversary holding two justification cards: (i)~the unique justification at $\Sigma.\hmax^0 - 1$ on a sibling chain, and (ii)~the unique justification at $\Sigma.\hmax^0$ on another chain whose containing block bumps $\Sigma.\hmax$ by 1 on release.
By justification uniqueness, the adversary holds at most one $J$ per height, and quorum-overlap under $f < n/3$ rules out a third disruptive card (e.g.\ a timeout-only chain at $\Sigma.\hmax^0 + 1$ would require an honest contribution overlapping with~(ii)'s, and the disjoint-quorum bound forbids both).
Each card forces at most one canonical reorg in the slot it is played; staggered across slots~$k+1$ and~$k+2$, both cards are spent by slot~$k+2$, and slot~$k+3$ produces the final fresh frontier-justification.

\subsection{Model and view-merge}

\begin{definition}[Slots and voting time]\label{def:slots}
Physical time is partitioned into discrete \emph{slots} indexed by $k \in \mathbb{N}$, with start time $\start(k)$ satisfying $\start(k+1) \geq \start(k) + 2\Delta$ for the network delay bound~$\Delta$, and \emph{voting time} $\vote(k) \coloneqq \start(k) + \Delta$ at which every honest validator~$i$ invokes \getFinalVote{} (Definition~\ref{def:voting-rule}) to produce at most one slot-$k$ vote.
\end{definition}

\begin{definition}[Syncer]\label{def:syncer}
A publicly known function $\syncer \colon \mathbb{N} \to \{1, \ldots, n\}$ assigns to each slot~$k$ a validator $\syncer(k)$, who runs the view-merge protocol during $[\start(k),\, \vote(k))$.
\end{definition}

\begin{assumption}[View-merge guarantee]\label{ass:view-merge}
If $\syncer(k)$ is honest, every honest validator~$i$ has $\Sigma_i = \comm{\Sigma}^{\vote(k)}$ at $\vote(k)$ for some store $\comm{\Sigma}^{\vote(k)}$.
\end{assumption}

\begin{definition}[Time-dependent dishonest count]\label{rev:def:f-t}
For each time~$t$, $f^t$ denotes the number of validators that are dishonest at time~$t$. We assume $f^t$ is monotonically non-decreasing in~$t$: once a validator is dishonest, it remains so.
\end{definition}

\begin{definition}[Standing assumptions for slot $k$]\label{def:standing}
We say slot~$k$ satisfies the \emph{standing assumptions} when (a)~$\start(k) - \Delta \geq \GST$ (synchrony), (b)~more than half of the validators online at~$\vote(k)$ are honest (\emph{online-honest majority}), (c)~every honest validator is online at~$\vote(k)$ (\emph{all-honest-awake}), and (d)~every vote cast by an honest validator at~$\vote(k)$ is received by some honest proposer that builds an available-chain block before $\vote(k+1) - \delta_c$ (\emph{honest-relay}, with $\delta_c$ the available-chain confirmation delay).
\end{definition}

For an honest validator~$v$ and time~$t$, write $\Sigma_v^t,\, \Omega_v^t$ for $v$'s store and auxiliary state at~$t$, and $\getConfirmed_v^t \coloneqq \getConfirmed(\Sigma_v^t,\, \Omega_v^t)$.

\subsection{Protocol modifications}

\paragraph{Virtual-block cert events.}
For this section, the store admits cert-event candidates not only from on-chain block processing but also from \emph{virtual extensions} of canonical: using the votes in the local pool, a routine~\updateJFromVotes hypothesizes a virtual block with parent~$\getConfirmed(\Sigma,\Omega)$ that carries the height-fresh subset of pool votes and offers the resulting cert-key descriptor to \updateJustified.
Virtual blocks are never inserted into $\Sigma.\T$ and the hypothetical states are discarded; only $(\Sigma.J,\, \Sigma.h_j)$ may change.
We further extend $\Sigma$ with a \emph{vote pool} $\Sigma.\mathcal{V}$ populated by view-merge --- a fresh field, distinct from the per-validator vote record $\Delta$ of Definition~\ref{def:voterecord}.

\paragraph{View-merge processing.}
When the syncer's view-merge delivers $(\comm{\T},\, \comm{V})$ at~$\vote(k)$, each honest validator folds new blocks through \onBlock{} in $\preceq$-respecting order and unions $\comm{V}$ into $\Sigma.\mathcal{V}$, then invokes \updateJFromVotes{}.

\paragraph{Voting-time snapshot for \getConfirmed{}.}
Between consecutive voting times, $\getConfirmed_v$ reads from the store snapshot taken at the most recent voting time: for any time $t \in [\vote(k),\, \vote(k{+}1))$, $\getConfirmed_v^t = \getConfirmed(\Sigma_v^{\vote(k)},\, \Omega_v^t)$.
The live store $\Sigma_v$ continues to evolve via \onBlock{} and view-merge processing during the interval, but those updates become visible to $\getConfirmed_v$ only from the next voting time $\vote(k{+}1)$ onward.

\paragraph{Voting-rule modification: timeout-shift on a lagging $\getConfirmed$.}
The honest voting rule of Definition~\ref{def:voting-rule} is extended as follows.
Let $T \coloneqq \getConfirmed(\Sigma_v^{\vote(k)},\, \Omega_v)$ and $h_c \coloneqq \sigma[T].h$ at voting time $\vote(k)$.
If $h_c < \Sigma_v^{\vote(k)}.\hmax - 1$ (the canonical state-height lags strictly more than one behind the maximum state-height in $v$'s tree), the validator emits a timeout vote at height $\Sigma_v^{\vote(k)}.\hmax - 1$:
\[
  v \;\coloneqq\; (i,\; \Sigma_v^{\vote(k)}.\hmax - 1,\; \bot,\; \bot,\; \bot),
\]
and \recordVote{} is invoked on $v$ as usual (so $\Delta.\tau(\Sigma_v^{\vote(k)}.\hmax - 1) \gets \mathtt{true}$).
Otherwise (if $h_c \geq \Sigma_v^{\vote(k)}.\hmax - 1$), the vote is produced by $\getFinalVote(\Delta,\, \getBaseVote(\Sigma_v^{\vote(k)},\, \Omega_v))$ as in Definition~\ref{def:voting-rule}.

\subsection{Properties of \getConfirmed{} under stable cascade root}\label{sec:getConfirmed-properties}

For an honest validator~$v$ and time~$t$, write
\[
  R_v^t \;\coloneqq\; \begin{cases}\Sigma_v^t.J & \text{if } \Sigma_v^t.\hmax = \Sigma_v^t.h_j + 1, \\ \Sigma_v^t.F & \text{otherwise,}\end{cases}
\]
for the cascade root selected inside $\getConfirmed_v^t$ (Figure~\ref{alg:store}).

For an honest validator~$v$ and time~$t$, write
\[
  \V_v^t \;\coloneqq\; \{B \in \T'(\Sigma_v^t) \,:\, B \succeq R_v^t \;\wedge\; \sigma[B].h \geq \Sigma_v^t.\hmax-1\}
\]
for the set of \uviable\RSfn{Need to find a different terms as viable has already been used} blocks --- those that $\getConfirmed(\Sigma_v^t,\, \cdot)$ may return as the auxiliary state varies.

The healing arguments rest on four utility lemmas: pairwise compatibility and intra-slot monotonicity of $\getConfirmed$, inclusion of slot-$k$ honest votes in the slot-$(k{+}1)$ canonical chain, and dominance of $v_\ell$'s slot-$k$ canonical at every honest's slot-$(k{+}1)$ canonical.
All four share the same preamble: pairwise $\sim$-compatible cascade roots at $\vote(k)$ and an honest~$v_\ell$ whose $\getConfirmed_{v_\ell}^{\vote(k)}$ dominates every honest~$R$ throughout $[\vote(k),\, \vote(k{+}1)]$.

\begin{property}[Cross-time, cross-validator target compatibility]\label{prop:getConfirmed-compat}
Assume all honest follow Definition~\ref{def:voting-rule}, slot~$k$ satisfies clauses (a), (b), and (d) of the standing assumptions (Definition~\ref{def:standing}), and, for any honest validator $v$, there exists blocks $R' \in \V_v^{\vote(k)}$ and $R'' \in \V_v^{\vote(k+1)}$ such that $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R'$ and $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R''$.
Then $\getConfirmed_v^t \sim \getConfirmed_{v'}^{t'}$ for every honest pair $v, v'$ and times $t, t' \in [\vote(k),\, \vote(k{+}1)]$.
\end{property}

\begin{property}[Intra-slot target monotonicity]\label{prop:getConfirmed-monotone}
Under the hypotheses of Property~\ref{prop:getConfirmed-compat}, for every honest~$v$ and times $t \leq t' \in [\vote(k),\, \vote(k{+}1)]$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
\end{property}

\begin{property}[Honest-vote inclusion in slot-$(k{+}1)$ canonical]\label{prop:vote-inclusion}
Under the hypotheses of Property~\ref{prop:getConfirmed-compat}, for every honest~$v$, the chain ending at $\getConfirmed_v^{\vote(k+1)}$ contains every vote cast by an honest validator at~$\vote(k)$.
\end{property}

\begin{property}[Slot-$k$ honest-vote carrier in slot-$(k{+}1)$ trees]\label{prop:vote-carrier}
Assume all honest follow Definition~\ref{def:voting-rule}, slot~$k$ satisfies clauses~(a), (b), and~(d) of the standing assumptions (Definition~\ref{def:standing}), and, for every honest validator $v$, there exists a block $R' \in \V_v^{\vote(k)}$ with $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R'$.
Then for every honest validator $v$, the block tree $\Sigma_v^{\vote(k+1)}.\T$ contains a block $B \succeq \getConfirmed_{v_\ell}^{\vote(k)}$ that carries every vote cast by an honest validator at $\vote(k)$.
\end{property}

\begin{property}[Slot-$(k{+}1)$ canonical extends $v_\ell$'s slot-$k$ canonical]\label{prop:target-dominance}
Under the hypotheses of Property~\ref{prop:getConfirmed-compat}, for every honest~$v$, $\getConfirmed_v^{\vote(k+1)} \succeq \getConfirmed_{v_\ell}^{\vote(k)}$.
\end{property}

\begin{property}[Cross-validator visibility of \getConfirmed]\label{prop:getConfirmed-visible}
Assume $\start(k) - \Delta \geq \GST$ and the online-honest majority holds at $\vote(k)$ (Definition~\ref{def:standing}(a,b)).
For every honest~$v,v'$, $\forall B \preceq \getConfirmed_v^{\vote(k)}, B\in \Sigma_{v'}^{\vote(k)}.\T$.
That is, every honest validator's slot-$k$ canonical block has already been admitted into every other honest validator's block tree by~$\vote(k)$.
\end{property}

\subsection{Healed state}\label{sec:healed}

\begin{definition}[Healed state at slot~$k$ with frontier height~$\comm{h}$]\label{def:healed}
The system is in a \emph{healed state at slot~$k$ with frontier height~$\comm{h}$} when there exists an honest validator $v_\ell$ such that:
\begin{enumerate}
  \item\label{rev:def:healed-state:h} for every honest~$v$, $\Sigma_v^{\vote(k)}.\hmax \geq \comm{h}+1$;
  \item\label{rev:def:healed-state:compat} for every honest pair $v, v'$, $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$;
  \item\label{rev:def:healed-state:cap} every honest vote~$u$ cast at any time strictly before~$\vote(k)$ with $u.\height \geq \comm{h}$ has $u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$;
  \item\label{rev:def:healed-state:vote-h} every honest vote~$u$ cast at any time strictly before $\vote(k)$ has $u.\height \leq \sigma[\getConfirmed_{v_\ell}^{\vote(k)}].h$.
  \item\label{rev:def:healed-state:F-cap} if $\sigma[\Sigma_v^t.F].h < \comm{h}$ for any honest validator $v$ and time $t$, then $\Sigma_v^t.F \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
\end{enumerate}
\end{definition}

The standing assumptions of slot~$k$ are not part of the healed-state definition; persistence and advance below pair the healed-state property at slot~$k$ with the standing assumptions of subsequent slots.

\subsection{Persistence}\label{sec:healing-persistence}

\begin{lemma}[Healing persistence over one slot]\label{lem:healing-persistence}
If
\begin{enumerate}
  \item\label{rev:hp:f} honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hp:ohm} the online-honest-majority holds at slot~$k$;
  \item\label{rev:hp:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:hp:relay} the honest-relay condition (Definition~\ref{def:standing}(d)) holds at slot~$k$;
  \item\label{rev:hp:hs} the system is in a healed state at slot $k$ with frontier $\comm{h}$,
\end{enumerate}
then:
\begin{enumerate}
  \item the system is in a healed state at slot~$k+1$ with frontier height $\comm{h}' \geq \comm{h}$.
  \item for every honest validators~$v,v'$ and times $t, t' \in [\vote(k),\, \vote(k+1)]$, $\getConfirmed_v^{t} \sim \getConfirmed_{v'}^{t'}$;
  \item for every honest validator~$v$ and every pair of times $t, t' \in [\vote(k),\, \vote(k+1)]$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
\end{enumerate}
\end{lemma}

\begin{proof}

\emph{Setup.}
Definition~\ref{def:healed} names some honest $v_\ell^0$ with conditions~\ref{rev:def:healed-state:cap}, \ref{rev:def:healed-state:vote-h}, and \ref{rev:def:healed-state:F-cap}.
By condition~\ref{rev:def:healed-state:compat}, the set $\{\getConfirmed_v^{\vote(k)} : v \text{ honest}\}$ is pairwise $\sim$-comparable, hence totally $\preceq$-ordered;
let $v_\ell$ be an honest $\preceq$-max in this set, and write
\[
  h^\ast \;\coloneqq\; \sigma[\getConfirmed_{v_\ell}^{\vote(k)}].h.
\]
Since $v_\ell$ is $\preceq$-max, $\getConfirmed_{v_\ell^0}^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k)}$, so conditions~\ref{rev:def:healed-state:cap}, \ref{rev:def:healed-state:vote-h}, and~\ref{rev:def:healed-state:F-cap} also hold with $v_\ell$ in place of $v_\ell^0$ (each is a $\preceq$-upper-bound statement against $\getConfirmed_{v_\ell^0}^{\vote(k)}$);
in particular every honest vote $u$ cast strictly before $\vote(k)$ has $u.\height \leq h^\ast$.

For honest votes cast at $\vote(k)$:
$\preceq$-maximality plus state-height monotonicity give $\sigma[\getConfirmed_v^{\vote(k)}].h \leq h^\ast$ for every honest $v$, and Definition~\ref{def:voting-rule} sets $u.\height = \sigma[\getConfirmed_v^{\vote(k)}].h$ in every branch, so $u.\height \leq h^\ast$ at $\vote(k)$ as well.
Hence every honest vote $u$ cast at any time $\leq \vote(k)$ has $u.\height \leq h^\ast$.
Moreover, the target of a non-$\bot$ slot-$k$ honest vote at height $h_c$ is either $\getConfirmed_v^{\vote(k)}$ (first non-$\bot$ vote at $h_c$, which yields $u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$ by $\preceq$-maximality) or $\Delta.T(h_c)$ (lock branch, which unfolds to $u_0.\target$ for $v$'s first non-$\bot$ vote $u_0$ at height $h_c$ cast strictly before $\vote(k)$; condition~\ref{rev:def:healed-state:cap} (when $h_c \geq \comm{h}$) gives $u_0.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$).
Hence every non-$\bot$ honest vote $u$ cast at any time $\leq \vote(k)$ with $u.\height \geq \comm{h}$ has $u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.

\emph{Outline.}
Steps~1--3 establish $\getConfirmed_{v_\ell}^{\vote(k)} \in \V_v^t$ for every honest $v$ and $t \in [\vote(k), \vote(k{+}1)]$.
Step~4 applies Properties~\ref{prop:getConfirmed-compat} and~\ref{prop:getConfirmed-monotone}, yielding conclusions~2 and~3 of the lemma directly, and condition~\ref{rev:def:healed-state:compat} at $\vote(k{+}1)$ as a slot-$k{+}1$ specialisation.
Steps~5--7 verify conditions~\ref{rev:def:healed-state:h}, \ref{rev:def:healed-state:cap}, and~\ref{rev:def:healed-state:vote-h} at $\vote(k{+}1)$ with $\comm{h}' \coloneqq \comm{h}$, completing conclusion~1.

\medskip

\emph{Step 1: justifications at heights $h \in [\comm{h}, h^\ast]$ have target $\preceq \getConfirmed_{v_\ell}^{\vote(k)}$, and no justification at heights $> h^\ast$ exists.}
Any justification at height~$h$ targeting some $T^\ast$ has $|\targets[T^\ast]| \geq 2n/3$ on a chain processed by some honest validator.
Under online-honest majority ($f < n/2$), $\geq 2n/3 - f > n/6 > 0$ honest validators contribute, each with $\targets[i] = T^\ast$ set by \processVote{} on a non-$\bot$ vote at height $h$ with target $T^\ast$.
For $h > h^\ast$: such a honest vote contradicts the Setup's height bound.
For $h \in [\comm{h}, h^\ast]$: the Setup's target bound on non-$\bot$ honest votes at heights $\geq \comm{h}$ gives $T^\ast = u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.

\emph{Step 2: $\Sigma_v^t.\hmax \leq h^\ast + 1$ throughout $[\vote(k), \vote(k{+}1)]$.}
Growth of $\hmax$ to $\sigma$-height $> h^\ast + 1$ requires a height advance at some height $> h^\ast$, hence a justification or timeout cert at $> h^\ast$.
Step~1 rules out justifications.
A timeout cert at height $h > h^\ast$ needs marks at $h$, contributed by $\bot$-votes at $h$ or height-fresh non-$\bot$ votes at $h$;
both forms of honest contribution have $u.\height = h > h^\ast$ and are ruled out by condition~\ref{rev:def:healed-state:vote-h}.
The remaining contributors are dishonest, $\leq f < n/2$ under the online-honest majority --- below the $2n/3$ threshold.

\emph{Step 3: $\getConfirmed_{v_\ell}^{\vote(k)} \in \V_v^t$.}
Property~\ref{prop:getConfirmed-visible} (under hypotheses~\ref{rev:hp:sync} and~\ref{rev:hp:ohm}) places $\getConfirmed_{v_\ell}^{\vote(k)} \in \Sigma_v^{\vote(k)}.\T \subseteq \Sigma_v^t.\T$, with $\sigma$-height $h^\ast$ in $v$'s store (Theorem~\ref{thm:orderindep}).
By Step~2, $h^\ast \geq \Sigma_v^t.\hmax - 1$, so $\getConfirmed_{v_\ell}^{\vote(k)}$ is its own viable witness in $\T'(\Sigma_v^t)$.

It remains to show $R_v^t \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
Note $\sigma[R_v^t].h \leq \Sigma_v^t.h_j \leq \Sigma_v^t.\hmax - 1 \leq h^\ast$ by Step~2.
\begin{itemize}
  \item \emph{Gate fires.} $R_v^t = \Sigma_v^t.J$ at height $\Sigma_v^t.h_j = \Sigma_v^t.\hmax - 1$, which lies in $[\comm{h}, h^\ast]$ by condition~\ref{rev:def:healed-state:h}.
  Step~1 gives $\Sigma_v^t.J \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
  \item \emph{Gate fails.} $R_v^t = \Sigma_v^t.F$ is finalized (Remark~\ref{rem:fs-invariant}) at some height $h_F \leq h^\ast$.
  If $h_F < \comm{h}$, condition~\ref{rev:def:healed-state:F-cap} gives $\Sigma_v^t.F \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
  Otherwise $h_F \in [\comm{h}, h^\ast]$, and finalization of $F$ at $h_F$ requires $F$ to also be justified at $h_F$ on a chain processed by $v$;
  Step~1 then gives $F \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
\end{itemize}

\medskip

\emph{Step 4: conclusions 2 and 3, and condition~\ref{rev:def:healed-state:compat} at $\vote(k{+}1)$.}
Step~3 instantiates the premise of Property~\ref{prop:getConfirmed-compat} with $R' = R'' = \getConfirmed_{v_\ell}^{\vote(k)}$, which lies in both $\V_v^{\vote(k)}$ and $\V_v^{\vote(k+1)}$ (Step~3 covers all $t \in [\vote(k), \vote(k{+}1)]$) and is reflexively $\succeq$ itself.
The property yields conclusion~2: $\getConfirmed_v^t \sim \getConfirmed_{v'}^{t'}$ for every honest $v, v'$ and $t, t' \in [\vote(k), \vote(k{+}1)]$.
Property~\ref{prop:getConfirmed-monotone}, which inherits the same premise, yields conclusion~3: $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$ for every honest $v$ and $t \leq t' \in [\vote(k), \vote(k{+}1)]$.
The $t = t' = \vote(k{+}1)$ specialisation of conclusion~2 is condition~\ref{rev:def:healed-state:compat} at $\vote(k{+}1)$.

\emph{Step 5: condition~\ref{rev:def:healed-state:h} at $\vote(k{+}1)$ with $\comm{h}' \coloneqq \comm{h}$.}
$\hmax$ is non-decreasing in $t$, so $\Sigma_v^{\vote(k+1)}.\hmax \geq \Sigma_v^{\vote(k)}.\hmax \geq \comm{h} + 1 = \comm{h}' + 1$.

\emph{Step 6: condition~\ref{rev:def:healed-state:cap} at $\vote(k{+}1)$.}
By Step~4, the honest canonical blocks at $\vote(k{+}1)$ form a chain; pick $v_\ell^{(k+1)}$ honest achieving the $\preceq$-max.
Property~\ref{prop:getConfirmed-monotone} gives $\getConfirmed_{v_\ell}^{\vote(k+1)} \succeq \getConfirmed_{v_\ell}^{\vote(k)}$, so by maximality
\[
  \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)} \;\succeq\; \getConfirmed_{v_\ell}^{\vote(k+1)} \;\succeq\; \getConfirmed_{v_\ell}^{\vote(k)}.
\]
For any honest vote $u$ cast strictly before $\vote(k{+}1)$ with $u.\height \geq \comm{h}' = \comm{h}$:
\begin{itemize}
  \item if cast strictly before $\vote(k)$, condition~\ref{rev:def:healed-state:cap} at slot~$k$ combined with the chain extension above gives $u.\target \preceq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$;
  \item if cast at $\vote(k)$, Property~\ref{prop:vote-inclusion} places $u.\target$ on the chain at $\getConfirmed_v^{\vote(k+1)}$ for every honest $v$, hence $\preceq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$.
\end{itemize}

\emph{Step 7: condition~\ref{rev:def:healed-state:vote-h} at $\vote(k{+}1)$.}
Write $h^{\ast\prime} \coloneqq \sigma[\getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}].h$.
For any honest vote $u$ cast at any time $\leq \vote(k{+}1)$:
\begin{itemize}
  \item if cast at $\leq \vote(k)$, condition~\ref{rev:def:healed-state:vote-h} at slot~$k$ gives $u.\height \leq h^\ast \leq h^{\ast\prime}$ (using Step~6's chain extension and state-height monotonicity);
  \item if cast at $\vote(k{+}1)$, Definition~\ref{def:voting-rule} gives $u.\height = \sigma[\getConfirmed_{v}^{\vote(k+1)}].h$ for the casting validator $v$, which is $\leq h^{\ast\prime}$ by maximality of $v_\ell^{(k+1)}$.
\end{itemize}

\emph{Step 8: condition~\ref{rev:def:healed-state:F-cap} at $\vote(k{+}1)$.}
For any honest validator $v$ and time $t$ with $\sigma[\Sigma_v^t.F].h < \comm{h}' = \comm{h}$:
condition~\ref{rev:def:healed-state:F-cap} at slot~$k$ (propagated to $v_\ell$ in the Setup) gives $\Sigma_v^t.F \preceq \getConfirmed_{v_\ell}^{\vote(k)}$, and Step~6's chain extension $\getConfirmed_{v_\ell}^{\vote(k)} \preceq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$ yields $\Sigma_v^t.F \preceq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$.
\end{proof}

\begin{lemma}[Healing: persistence over unbounded times]\label{rev:lem:healing-unbounded}
If
\begin{enumerate}
  \item\label{rev:hu:f} honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hu:ohm} the online-honest-majority holds at every slot $k' \geq k$;
  \item\label{rev:hu:sync} $\start(k') - \Delta \geq \GST$ for every slot $k' \geq k$;
  \item\label{rev:hu:relay} the honest-relay condition (Definition~\ref{def:standing}(d)) holds at every slot $k' \geq k$;
  \item\label{rev:hu:hs} the system is in a healed state at slot $k$ with height $\comm{h}$ (\Cref{def:healed}),
\end{enumerate}
then,
\begin{enumerate}
  \item for every honest validators~$v,v'$ and times $t, t' \geq \vote(k)$, $\getConfirmed_v^{t} \sim \getConfirmed_{v'}^{t'}$;
  \item for every honest validator~$v$ and every pair of times $t, t' \geq \vote(k)$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
\end{enumerate}
\end{lemma}

\begin{proof}
We induct on $K \geq k$, with inductive claim $H(K)$:
\begin{enumerate}
  \item the system is in a healed state at every slot $k' \in [k, K{+}1]$;
  \item for every honest $v, v'$ and $t, t' \in [\vote(k), \vote(K{+}1)]$, $\getConfirmed_v^t \sim \getConfirmed_{v'}^{t'}$;
  \item for every honest $v$ and $t \leq t' \in [\vote(k), \vote(K{+}1)]$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
\end{enumerate}

\emph{Base ($K = k$).}
Hypothesis~\ref{rev:hu:hs} gives the system healed at slot~$k$.
Lemma~\ref{lem:healing-persistence} applies at slot~$k$ (its hypotheses follow from~\ref{rev:hu:f}, \ref{rev:hu:ohm}, \ref{rev:hu:sync}, \ref{rev:hu:relay} at $k$, and~\ref{rev:hu:hs}), yielding the system healed at $k+1$ (so~(i)) and conclusions~2 and~3 over $[\vote(k), \vote(k{+}1)]$ (so~(ii) and~(iii)).

\emph{Inductive step.}
Suppose $H(K)$ holds; we show $H(K{+}1)$.
By~(i) at slot $K{+}1$ together with hypotheses~\ref{rev:hu:f}, \ref{rev:hu:ohm}, \ref{rev:hu:sync}, \ref{rev:hu:relay} at $K{+}1$, Lemma~\ref{lem:healing-persistence} applies at slot $K{+}1$, yielding the system healed at $K{+}2$ (extending~(i)) and conclusions~2 and~3 over $[\vote(K{+}1), \vote(K{+}2)]$.

To extend~(ii) and~(iii) to $[\vote(k), \vote(K{+}2)]$, the only new case is $t \in [\vote(k), \vote(K{+}1)]$ and $t' \in [\vote(K{+}1), \vote(K{+}2)]$ (or vice versa, handled by symmetry of~$\sim$ and by interchanging the time-ordering).

\emph{Extending~(iii):}
the inductive~(iii) gives $\getConfirmed_v^t \preceq \getConfirmed_v^{\vote(K{+}1)}$, and the new lemma instance's conclusion~3 gives $\getConfirmed_v^{\vote(K{+}1)} \preceq \getConfirmed_v^{t'}$;
chaining yields $\getConfirmed_v^t \preceq \getConfirmed_v^{t'}$.

\emph{Extending~(ii):}
the inductive~(iii) gives $\getConfirmed_v^t \preceq \getConfirmed_v^{\vote(K{+}1)}$, and the new lemma instance's conclusion~2 gives $\getConfirmed_v^{\vote(K{+}1)} \sim \getConfirmed_{v'}^{t'}$.
In the block tree, $A \preceq B$ and $C \sim B$ imply $A \sim C$:
either $C \preceq B$, in which case $A$ and $C$ are both ancestors of $B$, hence on the chain to $B$ and pairwise comparable;
or $C \succeq B$, in which case $A \preceq B \preceq C$.
Applied with $A = \getConfirmed_v^t$, $B = \getConfirmed_v^{\vote(K{+}1)}$, $C = \getConfirmed_{v'}^{t'}$, this gives $\getConfirmed_v^t \sim \getConfirmed_{v'}^{t'}$.

\emph{Conclusion.}
By induction, $H(K)$ holds for every $K \geq k$. Conclusions~1 and~2 of the lemma are the union $\bigcup_{K \geq k} H(K)$.
\end{proof}


\subsection{Healing after $N$ slots}

\begin{lemma}[Intra-slot target compatibility and monotonicity]\label{rev:lem:intra-slot-target-compat-monotone}
If
\begin{enumerate}
  \item\label{rev:istcm:f} honest validators follow \Cref{def:voting-rule};
  \item\label{rev:istcm:ohm} the honest-majority holds at slot~$k$;
  \item\label{rev:istcm:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:istcm:relay} the honest-relay condition (Definition~\ref{def:standing}(d)) holds at slot~$k$;
  \item\label{rev:istcm:hs} the system is in a healed state at slot $k$ with frontier $\comm{h}$
\end{enumerate}
then
\begin{enumerate}
  \item for every honest validators $v, v'$ and times $t, t' \in [\vote(k), \vote(k+1)]$, $\getConfirmed_v^t \sim \getConfirmed_{v'}^{t'}$;
  \item for every honest validator $v$ and every pair of times $t, t' \in [\vote(k), \vote(k+1)]$ with $t \leq t'$, $\getConfirmed_v^{t'} \succeq \getConfirmed_v^t$.
\end{enumerate}
\end{lemma}

\begin{proof}
\emph{Setup.}
By Definition~\ref{def:healed} (hypothesis~\ref{rev:istcm:hs}), there exists an honest $v_\ell^0$ for which conditions \ref{rev:def:healed-state:cap} and \ref{rev:def:healed-state:vote-h} hold.
By condition \ref{rev:def:healed-state:compat}, $\{\getConfirmed_v^{\vote(k)} : v \text{ honest}\}$ is pairwise $\sim$-comparable, hence totally $\preceq$-ordered;
let $v_\ell$ be an honest $\preceq$-max in this set, and write $h^\ast \coloneqq \sigma[\getConfirmed_{v_\ell}^{\vote(k)}].h$.
$\preceq$-maximality plus $\getConfirmed_{v_\ell^0}^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k)}$ propagates conditions \ref{rev:def:healed-state:cap}, \ref{rev:def:healed-state:vote-h}, and \ref{rev:def:healed-state:F-cap} to $v_\ell$ (each is a $\preceq$-upper-bound statement against $\getConfirmed_{v_\ell^0}^{\vote(k)}$, hence remains true with the larger bound $\getConfirmed_{v_\ell}^{\vote(k)}$).

We discharge the premise of Property~\ref{prop:getConfirmed-compat} with $R' = R'' \coloneqq \getConfirmed_{v_\ell}^{\vote(k)}$ by showing this block lies in $\V_v^t$ for every honest $v$ and every $t \in [\vote(k), \vote(k+1)]$.

\emph{Step 1: every honest vote $u$ processed by $\Sigma_v^t$ for $t \in [\vote(k), \vote(k+1)]$ has $u.\height \leq h^\ast$, and any non-$\bot$ such vote at height $\geq \comm{h}$ has $u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.}
Honest validators cast no votes in the open interval $(\vote(k), \vote(k+1))$, and $\vote(k+1)$-votes are not yet folded into $\Sigma_v^t$ for $t \in [\vote(k), \vote(k+1)]$, so every honest vote present in $\Sigma_v^t$ was cast at some time $\leq \vote(k)$.
For $u$ cast strictly before $\vote(k)$: condition~\ref{rev:def:healed-state:vote-h} gives $u.\height \leq h^\ast$, and condition~\ref{rev:def:healed-state:cap} gives $u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$ when $u.\height \geq \comm{h}$.
For $u$ cast at $\vote(k)$ by some honest validator $v$: Definition~\ref{def:voting-rule} sets $u.\height = h_c = \sigma[\getConfirmed_v^{\vote(k)}].h$ in every branch (the assignment in \getBaseVote{} and the unchanged height field in \getFinalVote{}), and $\preceq$-maximality of $v_\ell$ together with state-height monotonicity along $\preceq$ (Lemma~\ref{lem:slotmono}) gives $u.\height \leq h^\ast$.
The target of a non-$\bot$ slot-$k$ honest vote at height $h_c$ is either $\getConfirmed_v^{\vote(k)}$ (first non-$\bot$ vote at $h_c$, target adjustment in \getFinalVote takes the unconstrained branch) or $\Delta.T(h_c)$ (lock branch); the first case yields $u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$ by $\preceq$-maximality, the second case unfolds to $\Delta.T(h_c) = u_0.\target$ for $v$'s first non-$\bot$ vote $u_0$ at height $h_c$ cast strictly before $\vote(k)$, and condition~\ref{rev:def:healed-state:cap} (when $h_c \geq \comm{h}$) gives $u_0.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.

\emph{Step 2: $\Sigma_v^t.\hmax \leq h^\ast + 1$ throughout $[\vote(k), \vote(k+1)]$.}
A bump of $\hmax$ above $h^\ast + 1$ would require a height advance at some height $> h^\ast$ on a chain processed by $v$ (Lemma~\ref{lem:heightprog}), gated by a $\geq 2n/3$ quorum at that height.
Under hypothesis~\ref{rev:istcm:ohm} ($f < n/2$), $2n/3 - f > 2n/3 - n/2 = n/6 > 0$, so every such quorum contains at least one honest contribution at height $> h^\ast$:
either a non-$\bot$ vote (justification branch) or a $\bot$-vote/justification recorded into $\timeouts$ (timeout branch).
Both kinds of honest contribution are honest votes at height $> h^\ast$, ruled out by Step~1.

\emph{Step 3: $\getConfirmed_{v_\ell}^{\vote(k)} \in \V_v^t$.}
\emph{Tree membership.}
Property~\ref{prop:getConfirmed-visible} (under hypotheses~\ref{rev:istcm:sync} and~\ref{rev:istcm:ohm}) places $\getConfirmed_{v_\ell}^{\vote(k)} \in \Sigma_v^{\vote(k)}.\T \subseteq \Sigma_v^t.\T$, with $\sigma$-height $h^\ast$ in $v$'s store (Theorem~\ref{thm:orderindep}).
\emph{Viability.}
By Step~2, $h^\ast \geq \Sigma_v^t.\hmax - 1$, so $\getConfirmed_{v_\ell}^{\vote(k)}$ is its own viable witness when it is a leaf, and otherwise any leaf descendant in $\Sigma_v^t.\T$ has $\sigma$-height $\geq h^\ast$ (Lemma~\ref{lem:slotmono}) and is viable;
so $\getConfirmed_{v_\ell}^{\vote(k)} \in \viableTree(\Sigma_v^t)$.
\emph{Walk-from dominance.}
$\sigma[R_v^t].h \leq \Sigma_v^t.h_j \leq \Sigma_v^t.\hmax - 1 \leq h^\ast$.
\begin{itemize}
  \item \emph{Gate fails.}
  $R_v^t = \Sigma_v^t.F$ is finalized at some height $h_F$ (Remark~\ref{rem:fs-invariant}).
  Gate failure forces $\Sigma_v^t.\hmax > \Sigma_v^t.h_j + 1$, so $h_F \leq \Sigma_v^t.h_j < \Sigma_v^t.\hmax - 1 \leq h^\ast$, hence $h_F < h^\ast$.
  If $h_F < \comm{h}$, condition~\ref{rev:def:healed-state:F-cap} gives $F \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
  Otherwise $h_F \in [\comm{h}, h^\ast - 1]$, and finalization of $F$ at $h_F$ requires $F$ to also be justified at $h_F$ on a chain processed by $v$;
  the corresponding height-$h_F$ justification quorum has $\geq 2n/3 - f > n/6$ honest contributors with $\targets[i] = F$, set by \processVote{} on a non-$\bot$ vote at height $h_F \geq \comm{h}$ with target $F$, processed by $\Sigma_v^t$.
  By Step~1, this contributor's target satisfies $F \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
  \item \emph{Gate fires.}
  $R_v^t = \Sigma_v^t.J$ at height $h_j = \Sigma_v^t.\hmax - 1$.
  By condition~\ref{rev:def:healed-state:h} and monotonicity of $\hmax$, $\Sigma_v^t.\hmax \geq \comm{h}+1$, hence $h_j \geq \comm{h}$.
  Combined with the upper bound $h_j \leq h^\ast$, $h_j \in [\comm{h}, h^\ast]$.
  The corresponding justification quorum has $|\targets[J]| \geq 2n/3$ at height $h_j$, with $\geq 2n/3 - f > n/6$ honest contributions (hypothesis~\ref{rev:istcm:ohm}); each such honest validator $i$ has $\targets[i] = J$, set by \processVote on $i$'s non-$\bot$ vote at height $h_j$ with $u.\target = J$.
  This vote is processed by $\Sigma_v^t$ (otherwise the cert would not appear in $v$'s view at time $t$), so Step~1 gives $J = u.\target \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
\end{itemize}
Hence $R_v^t \preceq \getConfirmed_{v_\ell}^{\vote(k)}$, completing $\getConfirmed_{v_\ell}^{\vote(k)} \in \V_v^t$.

\emph{Step 4: applying the property.}
Reflexivity yields $\getConfirmed_{v_\ell}^{\vote(k)} \succeq R'$ and $\succeq R''$, discharging both existential clauses of the premise.
Property~\ref{prop:getConfirmed-compat} delivers conclusion~1.
Property~\ref{prop:getConfirmed-monotone}, inheriting the same premise, delivers conclusion~2.
\end{proof}

\begin{lemma}[Two-slot healing from a justification cert]\label{lem:healing-from-cert}
Assume:
\begin{enumerate}
  \item\label{rev:hfcert:f} $f^{\vote(k+2)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hfcert:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:hfcert:standing} standing assumptions are satisfied by slots in $[k,k+1,k+2]$;
  \item\label{rev:hfcert:compat} for every honest pair $v, v'$, $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$;
  \item\label{rev:hfcert:hj} $\Sigma_v^{\vote(k)}.J$ for every honest $v$ is at $\sigma$-height $\comm{h}$;
  \item\label{rev:hfcert:hmax} $\Sigma_v^{\vote(k)}.\hmax \leq \comm{h}+1$ for every honest $v$;
  \item\label{rev:hfcert:nh} no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then the system is in a healed state at slot $k+2$ with frontier $\comm{h}+1$.
\end{lemma}

\begin{proof}[Filled proof of \Cref{lem:healing-from-cert}]
\emph{Setup.}
By hypothesis~\ref{rev:hfcert:compat}, $\{\getConfirmed_v^{\vote(k)} : v \text{ honest}\}$ is pairwise $\sim$-comparable, hence totally $\preceq$-ordered; let $v_\ell$ be a $\preceq$-max element.
By hypothesis~\ref{rev:hfcert:hj} and per-height uniqueness (\Cref{lem:just-unique-height}), $\Sigma_v^{\vote(k)}.J = J$ at $\sigma$-height $\comm{h}$ uniformly across honest $v$.
A height-$\comm{h}$ cert advances state-height to $\comm{h}+1$ at any block carrying it, so $\hmax^{\vote(k)} \geq \comm{h}+1$;
combined with hypothesis~\ref{rev:hfcert:hmax}, $\hmax^{\vote(k)} = \comm{h}+1$ uniformly.
The cascade gate $\hmax = h_j+1$ fires, $R_v^{\vote(k)} = J$ at $\sigma$-height $\comm{h}$, and \Cref{cor:getConfirmed-total} together with $\preceq$-maximality of $v_\ell$ chains $R_v^{\vote(k)} \preceq \getConfirmed_v^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k)}$;
all honest's $\getConfirmed^{\vote(k)}$ extend $J$ on a common chain, with $\sigma$-height in $\{\comm{h}, \comm{h}+1\}$.

\begin{description}
  \item[Case 1: the height of all honest $\getConfirmed^{\vote(k)}$ is $\comm{h}$.]
  Each honest validator votes at height $\comm{h}$ at $\vote(k)$ (its canonical state-height).
  By \Cref{def:voting-rule}, the lock branch is unreachable (no honest is locked at $\comm{h}$, since lock at $\comm{h}$ would require a prior finalize-bearing vote with $\finalizeHeight = \comm{h}$, and the assert $h_c > h_f$ in \getFinalVote{} forces that earlier vote's height $> \comm{h}$, contradicting hypothesis~\ref{rev:hfcert:nh});
  each honest is therefore in the elif-branch (timeout, target $\bot$) if its first vote at $\comm{h}$ was non-$\bot$, or in the default branch (non-$\bot$, target equal to the slot-$k$ canonical) if its first vote at $\comm{h}$ was a timeout.
  Both branches set $\timeouts$ at height $\comm{h}$ via \processVote{} (timeouts directly; non-$\bot$ votes via the side effect $\sigma.\timeouts[i] \gets \mathtt{true}$ in the justification branch), so the $\geq 2n/3$ honest contributions form a height-$\comm{h}$ timeout quorum.
  A height-$\comm{h}$ \emph{justification} quorum may or may not also form, depending on whether the default-branch contributions concentrate on a single target; either way a height-$\comm{h}$ cert fires.
  \Cref{prop:vote-carrier} at slot~$k$ (premise discharged via $R' = \getConfirmed_{v_\ell}^{\vote(k)} \in \V_v^{\vote(k)}$, viable since $\sigma$-height $\comm{h} \geq \hmax-1 = \comm{h}$, with walk-from dominance via $R = J \preceq \getConfirmed_{v_\ell}^{\vote(k)}$) yields a slot-$k$ carrier $B^\sharp_v \in \Sigma_v^{\vote(k+1)}.\T$ extending $\getConfirmed_{v_\ell}^{\vote(k)}$;
  \finalizeProcessBlock{} fires the cert and $\sigma[B^\sharp_v].h = \comm{h}+1$.
  At $\vote(k+1)$, by \Cref{prop:vote-inclusion} (same premise), $B^\sharp_v \preceq \getConfirmed_v^{\vote(k+1)}$ and \Cref{lem:slotmono} gives $\sigma[\getConfirmed_v^{\vote(k+1)}].h \geq \comm{h}+1$;
  all honest have their chain at height $\comm{h}+1$ by $\vote(k+1)$.

  \item[Case 2: the height of some honest $\getConfirmed^{\vote(k)}$ is $\comm{h}+1$.]
  By \Cref{prop:target-dominance} at slot~$k$ (premise discharged as in Case~1, with $v_\ell$ chosen as the honest with $\getConfirmed^{\vote(k)}.h = \comm{h}+1$), every honest $v$ has $\getConfirmed_v^{\vote(k+1)} \succeq \getConfirmed_{v_\ell}^{\vote(k)}$, so $\sigma[\getConfirmed_v^{\vote(k+1)}].h \geq \comm{h}+1$;
  all honest have their chain at height $\comm{h}+1$ by $\vote(k+1)$.
\end{description}

Then, all honest have their chain at height $\comm{h}+1$ by $\vote(k+1)$.

We know that all honest votes for height $\comm{h}+1$ are $\preceq$ the $\getConfirmed$ of $v_\ell$ at time $\vote(k)$:
in Case~1 there are no such votes (vacuously true);
in Case~2 honest first-vote at $\comm{h}+1$ (no honest voted at $\comm{h}+1$ before $\vote(k)$, by hypothesis~\ref{rev:hfcert:nh}), so by \Cref{def:voting-rule}'s default branch $u.\target = \getConfirmed_v^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
Then, any justified for height $\comm{h}+1$ has its target ($= u.\target$ for an honest contributor by \Cref{lem:just-unique-height}'s per-height uniqueness and the first-vote argument above) $\preceq \getConfirmed_{v_\ell}^{\vote(k)}$.

Then, by \Cref{prop:getConfirmed-compat} at slot~$k$ (premise discharged as before), all honest's slot-$(k+1)$ canonicals are pairwise $\sim$-compat;
combined with $\sigma[\getConfirmed_v^{\vote(k+1)}].h \geq \comm{h}+1$ above and the upper bound $\hmax^{\vote(k+1)} \leq \comm{h}+2$ (no honest vote at $> \comm{h}+1$ before $\vote(k+1)$), all honest vote for height $\comm{h}+1$ at $\vote(k+1)$, forming a $\geq 2n/3$ height-$(\comm{h}+1)$ quorum that fires a cert (timeout or justification) at $\vote(k+1)$.

By \Cref{prop:vote-carrier} at slot~$k+1$ (premise discharged via $R' = \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)} \in \V_v^{\vote(k+1)}$ for a $\preceq$-max honest $v_\ell^{(k+1)}$), the slot-$(k+1)$ carrier $B^{\sharp\sharp}_v$ is in every honest's tree at $\vote(k+2)$ with $B^{\sharp\sharp}_v \succeq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$ and $\sigma[B^{\sharp\sharp}_v].h = \comm{h}+2$, fixing $\Sigma_v^{\vote(k+2)}.\hmax = \comm{h}+2$ uniformly.
Pick a $\preceq$-max honest $v_\ell^{(k+2)}$ at $\vote(k+2)$ (well-defined by \Cref{prop:getConfirmed-compat} at slot~$k+1$).
We verify each clause of \Cref{def:healed} at slot~$k+2$ with $\comm{h}' \coloneqq \comm{h}+1$:
\begin{itemize}
  \item \emph{\Cref{rev:def:healed-state:h}.} $\Sigma_v^{\vote(k+2)}.\hmax = \comm{h}+2 = \comm{h}'+1$.
  \item \emph{\Cref{rev:def:healed-state:compat}.} \Cref{prop:getConfirmed-compat} at slot~$k+1$ gives pairwise $\sim$-compat at $\vote(k+2)$.
  \item \emph{\Cref{rev:def:healed-state:cap}.} An honest vote $u$ cast strictly before $\vote(k+2)$ with $u.\height \geq \comm{h}+1$: by the height bounds, $u$ is cast at $\vote(k)$ or $\vote(k+1)$. By \Cref{def:voting-rule}, $u.\target \preceq \getConfirmed_v^{t}$ for the casting validator $v$ at the casting time $t$. \Cref{prop:target-dominance} at slot~$k+1$ gives $\getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)} \succeq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$, applied at slot~$k$ similarly chains $\getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)} \succeq \getConfirmed_{v_\ell}^{\vote(k)}$, and $\preceq$-maximality at each slot pushes $u.\target$ all the way up to $\getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)}$.
  \item \emph{\Cref{rev:def:healed-state:vote-h}.} Max honest vote height before $\vote(k+2)$ is $\comm{h}+1 \leq \sigma[\getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)}].h$ (which is $\geq \comm{h}+1$ by \Cref{prop:vote-inclusion} at slot~$k+1$ placing $B^{\sharp\sharp}_v$ on the chain to $\getConfirmed_v^{\vote(k+2)}$ plus \Cref{lem:slotmono}).
  \item \emph{\Cref{rev:def:healed-state:F-cap}.} For honest $v$ and time $t$ with $\sigma[\Sigma_v^t.F].h < \comm{h}+1$: \Cref{lem:mainsafety} (under hypothesis~\ref{rev:hfcert:f}'s $f<n/3$) on $\getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)}$ at $\sigma$-height $\geq \comm{h}+1 > \sigma[\Sigma_v^t.F].h$ gives $\Sigma_v^t.F \preceq \getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)}$.
\end{itemize}
Hence the system is in a healed state at slot~$k+2$ with frontier height $\comm{h}+1$.
\end{proof}

\begin{lemma}[Two-slot healing from a flat frontier]\label{lem:healing-2slot-flat-frontier}
Assume:
\begin{enumerate}
  \item\label{rev:hfj:f} $f^{\vote(k+3)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hfj:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:hfj:syncers} the syncer for slot~$k+1$ is honest;
  \item\label{rev:hfj:standing} standing assumptions are satisfied by slots in $[k,k+1,k+2,k+3]$;
  \item\label{rev:hfj:compat} for every honest pair $v, v'$, $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$;
  \item\label{rev:hfj:gt-hmax} for every honest validators $v, v'$, $\sigma[\getConfirmed_v^{\vote(k)}].h = \Sigma_{v'}^{\vote(k)}.\hmax$; denote this common value by $\comm{h}$;
  \item\label{rev:hfj:nh} no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$.
  % \item\label{rev:hfj:ohm} the \OHM holds at slots~$k$ and $k+1$;
  % \item\label{rev:hfj:aha} the \AHA condition holds at slots~$k$ and $k+1$;
  % \item\label{rev:hfj:har} the \HAR condition holds at slot~$k$;
  % \item\label{rev:hfj:r} for every honest validator~$v$, $R_v^{\vote(k)}$ is the same justified block; denote it by $\comm{R}^{\vote(k)}$;
  % \item\label{rev:hfj:gt} $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getConfirmed_v^{\vote(k)}].h = \comm{R}^{\vote(k)}.h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  % \item\label{rev:hfj:nh} no honest validator has cast any vote at height $> \comm{R}^{\vote(k)}.h$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then the system is in a healed state at some slot $k_h \in \{k+2, k+3\}$ with frontier $\comm{h}+1$.
\end{lemma}

\begin{proof}
\emph{Setup.}
By hypothesis~\ref{rev:hfj:gt-hmax}, $\sigma[\getConfirmed_v^{\vote(k)}].h = \Sigma_{v'}^{\vote(k)}.\hmax = \comm{h}$ for every honest $v, v'$.
By hypothesis~\ref{rev:hfj:compat}, $\{\getConfirmed_v^{\vote(k)} : v \text{ honest}\}$ is pairwise $\sim$-comparable, hence totally $\preceq$-ordered; let $v_\ell$ be a $\preceq$-max element.
By Definition~\ref{def:voting-rule}, every honest $v$'s slot-$k$ vote $u_v$ has $u_v.\height = \sigma[\getConfirmed_v^{\vote(k)}].h = \comm{h}$, and under hypothesis~\ref{rev:hfj:f}'s $f^{\vote(k)} \leq f^{\vote(k+2)} < n/3$ the honest contribution count exceeds $2n/3$;
so all honest cast a height-$\comm{h}$ vote at $\vote(k)$, on the common chain ending at $\getConfirmed_{v_\ell}^{\vote(k)}$ (by pairwise compat plus first-vote/lock target argument from \Cref{def:voting-rule}, every non-$\bot$ such vote has target $\preceq \getConfirmed_{v_\ell}^{\vote(k)}$).

\emph{Discharge of \Cref{prop:vote-carrier} at slot $k$.}
\Cref{prop:getConfirmed-visible} (under hypothesis~\ref{rev:hfj:sync} and the online-honest-majority clause of hypothesis~\ref{rev:hfj:standing}) places $\getConfirmed_{v_\ell}^{\vote(k)} \in \Sigma_v^{\vote(k)}.\T$ for every honest $v$, with $\sigma$-height $\comm{h} \geq \hmax - 1$ (Theorem~\ref{thm:orderindep}, hypothesis~\ref{rev:hfj:gt-hmax}), making it viable;
\Cref{cor:getConfirmed-total} plus $\preceq$-maximality of $v_\ell$ chains $R_v^{\vote(k)} \preceq \getConfirmed_{v_\ell}^{\vote(k)}$.
So $R' \coloneqq \getConfirmed_{v_\ell}^{\vote(k)} \in \V_v^{\vote(k)}$ for every honest $v$, and \Cref{prop:vote-carrier} delivers, for every honest $v$, a block $B^\sharp_v \in \Sigma_v^{\vote(k+1)}.\T$ with $B^\sharp_v \succeq \getConfirmed_{v_\ell}^{\vote(k)}$ that carries every slot-$k$ honest vote.
Since the $\geq 2n/3$ height-$\comm{h}$ slot-$k$ honest votes are recorded by \processVote{} on $B^\sharp_v$'s chain (pre-state $\sigma.h = \comm{h}$ since $\hmax^{\vote(k)} = \comm{h}$ rules out an earlier height-$\comm{h}$ cert on this chain), \finalizeProcessBlock{} fires the height-advance branch and $\sigma[B^\sharp_v].h = \comm{h}+1$;
hence $\Sigma_v^{\vote(k+1)}.\hmax \geq \comm{h}+1$.
Conversely, hypothesis~\ref{rev:hfj:nh} bounds every honest vote cast strictly before $\vote(k)$ to height $\leq \comm{h}$, slot-$k$ honest votes are at $\comm{h}$, and honest cast no votes in $(\vote(k), \vote(k+1))$;
under $f<n/3$ no $\geq 2n/3$ quorum at height $> \comm{h}$ assembles before $\vote(k+1)$, so $\hmax^{\vote(k+1)} = \comm{h}+1$ uniformly.

\begin{description}
  \item[Case 1: no justified block for height $\comm{h}$ is received by $\vote(k+1)$.]
  Let $v_m$ be any honest with $\getConfirmed_{v_m}^{\vote(k)}$ a $\preceq$-min element of $\{\getConfirmed_v^{\vote(k)} : v \text{ honest}\}$.
  Max height that can be achieved up to $\vote(k+1)$ is $\comm{h}+1$ (just shown).
  By \Cref{prop:vote-carrier}, $\Sigma_v^{\vote(k+1)}.\hmax = \comm{h}+1$ for every honest $v$.
  Case~1's hypothesis says no height-$\comm{h}$ justification has fired by $\vote(k+1)$, so $\Sigma_v^{\vote(k+1)}.h_j \leq \comm{h}-1$ for every honest $v$, the cascade gate $\hmax = h_j+1$ fails ($\comm{h}+1 \neq h_j+1$), and $R_v^{\vote(k+1)} = \Sigma_v^{\vote(k+1)}.F$ at some height $h_F \leq h_j \leq \comm{h}-1 < \comm{h}$.
  Hence $\getConfirmed_{v_m}^{\vote(k)}$ at $\sigma$-height $\comm{h}$ is \uviable until at least $\vote(k+1)$ for every honest $v$:
  tree-membership and viability hold as in the discharge above, and walk-from dominance follows from \Cref{lem:mainsafety} (under hypothesis~\ref{rev:hfj:f}'s $f<n/3$) on $\getConfirmed_{v_m}^{\vote(k)}$ at $\sigma$-height $\comm{h} > h_F$.

  Then, from \Cref{prop:vote-inclusion,prop:getConfirmed-compat,prop:target-dominance} (premise discharged with $R' = R'' \coloneqq \getConfirmed_{v_m}^{\vote(k)}$), the chain to every honest's $\getConfirmed^{\vote(k+1)}$ contains $B^\sharp_v$, so $\sigma[\getConfirmed_v^{\vote(k+1)}].h \geq \comm{h}+1$ (\Cref{lem:slotmono});
  combined with $\hmax = \comm{h}+1$ (upper bound), $\sigma[\getConfirmed_v^{\vote(k+1)}].h = \comm{h}+1$ uniformly, with pairwise $\sim$-compat.
  So all honest vote for height $\comm{h}+1$ at $\vote(k+1)$, forming a $\geq 2n/3$ height-$(\comm{h}+1)$ quorum on the common chain.

  $\vote(k+2)$: \Cref{prop:vote-carrier} at slot $k+1$ (premise discharged identically with $R' = \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$ for a $\preceq$-max honest $v_\ell^{(k+1)}$, viable since $\sigma$-height $\comm{h}+1 \geq \hmax^{\vote(k+1)}-1 = \comm{h}$, with walk-from dominance via $R^{\vote(k+1)} = F \preceq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$ by mainsafety) yields a slot-$(k+1)$ carrier $B^{\sharp\sharp}_v \in \Sigma_v^{\vote(k+2)}.\T$ with $B^{\sharp\sharp}_v \succeq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$ and $\sigma[B^{\sharp\sharp}_v].h = \comm{h}+2$, fixing $\Sigma_v^{\vote(k+2)}.\hmax = \comm{h}+2$ uniformly.
  Pick a $\preceq$-max honest $v_\ell^{(k+2)}$ at $\vote(k+2)$ (well-defined by \Cref{prop:getConfirmed-compat} at slot~$k+1$, applied with the same premise discharge).
  We verify each clause of \Cref{def:healed} at slot~$k+2$ with $\comm{h}' \coloneqq \comm{h}+1$:
  \begin{itemize}
    \item \emph{\Cref{rev:def:healed-state:h}.} $\Sigma_v^{\vote(k+2)}.\hmax = \comm{h}+2 = \comm{h}'+1$.
    \item \emph{\Cref{rev:def:healed-state:compat}.} \Cref{prop:getConfirmed-compat} at slot~$k+1$ gives pairwise $\sim$-compat at $\vote(k+2)$.
    \item \emph{\Cref{rev:def:healed-state:cap}.} An honest vote $u$ cast strictly before $\vote(k+2)$ with $u.\height \geq \comm{h}+1$ must be at $\vote(k+1)$ (by hypothesis~\ref{rev:hfj:nh} and the $\comm{h}$-bound on slot-$k$ votes, no honest votes at $\geq \comm{h}+1$ before $\vote(k+1)$). \Cref{def:voting-rule}'s default branch gives $u.\target = \getConfirmed_v^{\vote(k+1)}$ for the casting validator $v$; \Cref{prop:target-dominance} at slot~$k+1$ gives $\getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)} \succeq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$, and $\preceq$-maximality of $v_\ell^{(k+1)}$ gives $\getConfirmed_v^{\vote(k+1)} \preceq \getConfirmed_{v_\ell^{(k+1)}}^{\vote(k+1)}$; chaining yields $u.\target \preceq \getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)}$.
    \item \emph{\Cref{rev:def:healed-state:vote-h}.} Max honest vote height before $\vote(k+2)$ is $\comm{h}+1 \leq \sigma[\getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)}].h$ (the latter is $\geq \comm{h}+1$ by \Cref{prop:vote-inclusion} at slot~$k+1$ placing $B^{\sharp\sharp}_v$ on the chain to $\getConfirmed_v^{\vote(k+2)}$ plus \Cref{lem:slotmono}).
    \item \emph{\Cref{rev:def:healed-state:F-cap}.} For honest $v$ and time $t$ with $\sigma[\Sigma_v^t.F].h < \comm{h}+1$: \Cref{lem:mainsafety} (under hypothesis~\ref{rev:hfj:f}'s $f<n/3$) applied to $\getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)}$ at $\sigma$-height $\geq \comm{h}+1 > \sigma[\Sigma_v^t.F].h$ gives $\Sigma_v^t.F \preceq \getConfirmed_{v_\ell^{(k+2)}}^{\vote(k+2)}$.
  \end{itemize}
  Hence the system is in a healed state at slot~$k+2$ with frontier height $\comm{h}+1$.

  \item[Case 2: a justified block for height $\comm{h}$ is received by $\vote(k+1)$.]
  View-merge at slot~$k+1$ (under hypothesis~\ref{rev:hfj:syncers}) propagates the height-$\comm{h}$ justification cert to every honest by $\vote(k+1)$, and \updateJustified{} accepts the lex-higher key, so $\Sigma_v^{\vote(k+1)}.h_j = \comm{h}$ uniformly.
  Apply \Cref{lem:healing-from-cert} at slot~$k+1$ with cert height $\comm{h}$, discharging its hypotheses:
  \ref{rev:hfcert:f} via $f^{\vote(k+3)} < n/3$ from \ref{rev:hfj:f};
  \ref{rev:hfcert:sync} via $\start(k+1) > \start(k) \geq \GST + \Delta$ from \ref{rev:hfj:sync};
  \ref{rev:hfcert:standing} via \ref{rev:hfj:standing} at slots $\{k+1, k+2, k+3\} \subseteq [k,k+1,k+2,k+3]$;
  \ref{rev:hfcert:compat} via \Cref{prop:getConfirmed-compat} at slot~$k$ with $R' = R'' \coloneqq \getConfirmed_{v_\ell}^{\vote(k)}$ (viable at both times by the cascade analysis: at $\vote(k+1)$, $R = J$ at $\sigma$-height $\comm{h}$ with $\sigma[\getConfirmed_{v_\ell}^{\vote(k)}].h = \comm{h}$ on the same chain by \Cref{cor:getConfirmed-total} plus per-height uniqueness);
  \ref{rev:hfcert:hj} from view-merge above;
  \ref{rev:hfcert:hmax} from $\Sigma_v^{\vote(k+1)}.\hmax = \comm{h}+1$ (Setup);
  \ref{rev:hfcert:nh} from \ref{rev:hfj:nh} together with the $\comm{h}$-bound on slot-$k$ honest votes.
  Conclusion: healed state at slot $(k+1)+2 = k+3$ with frontier $\comm{h}+1$.
\end{description}
\end{proof}

\begin{lemma}\label{lem:healing-from-compat}
Assume:
\begin{enumerate}
  \item\label{rev:hfc:f} $f^{\vote(k+5)} < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item\label{rev:hfc:sync} $\start(k) - \Delta \geq \GST$;
  \item\label{rev:hfc:syncers} the syncers for slots $[k,k+1,k+2,k+3]$ are honest;
  \item\label{rev:hfc:standing} standing assumptions are satisfied by slots in $[k,k+1,k+2,k+3,k+4,k+5]$;
  \item\label{rev:hfc:compat} for every honest pair $v, v'$, $\getConfirmed_v^{\vote(k)} \sim \getConfirmed_{v'}^{\vote(k)}$.
  % \item for every honest validators $v, v'$, $\Sigma_{v'}^{\vote(k)}.\hmax = \Sigma_{v'}^{\vote(k)}.\hmax$; denote this common value by $\comm{h}$;
  % \item no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then the system is in a healed state at some slot $k_h \leq k+5$ with some frontier height $\comm{h} \geq \comm{\hmax}+1$, where $\comm{\hmax}$ denotes the common value of $\Sigma_v^{\vote(k)}.\hmax$ across honest $v$ (well-defined by the slot-$k$ honest syncer).
\end{lemma}

\emph{Setup.}

            View-merge at slot~$k+2$ (hypothesis~\ref{rev:hfc:syncers}) propagates the height-$\comm{\hmax}$ cert and \updateJustified{} pins $\Sigma_v^{\vote(k+2)}.h_j = \comm{\hmax}$;
            the slot-$(k+2)$ honest proposer extends the cert's target, lifting state-height to $\comm{\hmax}+1$ before $\vote(k+2)$, so $\Sigma_v^{\vote(k+2)}.\hmax = \comm{\hmax}+1$ uniformly.
            Apply \Cref{lem:healing-from-cert} at slot~$k+2$ with cert height $\comm{\hmax}$, discharging:
            \ref{rev:hfcert:f} via $f^{\vote(k+4)} \leq f^{\vote(k+5)} < n/3$ from \ref{rev:hfc:f};
            \ref{rev:hfcert:sync} via $\start(k+2) > \start(k) \geq \GST + \Delta$ from \ref{rev:hfc:sync};
            \ref{rev:hfcert:standing} at $\{k+2, k+3, k+4\} \subseteq [k,\dots,k+5]$ from \ref{rev:hfc:standing};
            \ref{rev:hfcert:compat} at $\vote(k+2)$ from cascade fire ($R = J$ at $\sigma$-height $\comm{\hmax}$, every honest's $\getConfirmed^{\vote(k+2)} \succeq J$ on $J$'s chain by \Cref{cor:getConfirmed-total});
            \ref{rev:hfcert:hj}, \ref{rev:hfcert:hmax} from view-merge / state-advance above;
            \ref{rev:hfcert:nh} from the Setup's vote-history bound and the $\comm{\hmax}$-bound on slot-$k$ and slot-$(k+1)$ honest votes (Case~2.B.B / Case~2.B.B.A).
            Conclusion: healed state at slot $(k+2)+2 = k+4$ with frontier $\comm{\hmax}+1 \geq \comm{\hmax}+1$.
        \end{description}
    \end{description}
  \end{description}
\end{description}

In every case (other than Case~2.A, flagged), the system reaches a healed state at some slot $k_h \leq k+5$ with frontier $\geq \comm{\hmax}+1$.
\end{proof}



\subsection{Card inventory and three-slot advance}\label{sec:card-and-advance}

The adversary may withhold and selectively release justification certificates to disrupt canonical inside \getConfirmed{}.
A potential-function bookkeeping argument shows that under $f < n/3$ at most two such cards can ever influence canonical, and three honest-syncer slots suffice to absorb them and reach a healed state.
The bound and the advance fold into a single lemma; no per-slot enumeration of adversary release patterns is needed.

\begin{lemma}[Card inventory and three-slot advance]\label{lem:healing-3slot}
Assume $f < n/3$ and that all honest validators follow Definition~\ref{def:voting-rule}.
Let~$k$ be a slot, and let $h^0 \coloneqq \comm{\Sigma}^{\vote(k+1)}.\hmax$ (uniform across honest stores at $\vote(k+1)$ by Assumption~\ref{ass:view-merge} under the honest slot-$(k+1)$ syncer).
If slots~$k+1, k+2, k+3$ each satisfy the standing assumptions and have honest syncers, then there exists $k_h \in \{k+1, k+2, k+3\}$ such that the system is in a healed state at slot~$k_h$ with some frontier height $\comm{h} \geq h^0 - 1$.
\end{lemma}

\begin{proof}
\emph{Step 1: Card inventory.}
By Lemma~\ref{lem:just-unique-height}, every justification at any height on any chain processed by the node is unique; carrying-block heights follow from the height-advance rule (Lemma~\ref{lem:slotmono}: a justification at height~$h$ produces a containing block of $\sigma$-height $h+1$).
Three observations bound the adversary's disruptive inventory at slot $k+1$:

\begin{itemize}
  \item\emph{Heights $> h^0$ do not exist.}
  A $\geq 2n/3$ justification quorum at height~$h$ requires $\geq 2n/3 - f > n/3$ honest contributions; each honest's voting height $h_c = \sigma[\getConfirmed_v].h$ at any prior voting time satisfies $h_c \leq \Sigma_v.\hmax \leq h^0$ by the definition of~$h^0$.
  Hence no honest validator has cast a non-$\bot$ vote at height~$> h^0$, and no justification at height~$> h^0$ exists.

  \item\emph{Heights $< h^0 - 1$ fail the cascade gate post-release.}
  Releasing a justification at height $h < h^0 - 1$ on any chain sets $\Sigma.h_j \gets h$ and bumps $\Sigma.\hmax$ to at most $\max(\Sigma.\hmax, h + 1) \leq h^0$.
  Since $\Sigma.\hmax \geq h^0 > h + 1 = \Sigma.h_j + 1$ post-release, the gate fails, $R = \Sigma.F$ inside \getConfirmed{}, and the released card does not shift canonical.

  \item\emph{The two surviving cards.}
  Let~$J^*$ denote the unique justification at $h^0 - 1$ when present, with carrier $B^*$ at $\sigma[B^*].h = h^0$, and $J^\dagger$ the unique justification at $h^0$ when present, with carrier $B^\dagger$ at $\sigma[B^\dagger].h = h^0 + 1$.
  On release of card~$J^*$: $\Sigma.h_j \gets h^0 - 1$, $\Sigma.\hmax = h^0$, gate fires.
  On release of card~$J^\dagger$: $\Sigma.h_j \gets h^0$, $\Sigma.\hmax \gets h^0 + 1$, gate fires.
  When both exist, the disjoint-quorum bound under $f < n/3$ forces $J^* \preceq J^\dagger$ on a common chain: the height advance from~$h^0 - 1$ to~$h^0$ on $J^\dagger$'s chain via a timeout cert would require a $\geq 2n/3$ timeout quorum at height~$h^0 - 1$ disjoint (in honest validators) from $J^*$'s justification quorum at the same height (Definition~\ref{def:voting-rule}: at most one vote per height), giving $4n/3 - 2f \leq n - f$, i.e., $f \geq n/3$, ruled out by the standing $f < n/3$.
\end{itemize}

\emph{Step 2: Potential function.}
For $\ell \in \{k+1, k+2, k+3\}$, define on the post-merge common store $\comm{\Sigma}^{(\ell)} \coloneqq \comm{\Sigma}^{\vote(\ell)}$:
\begin{itemize}
  \item a card $c \in \{J^*, J^\dagger\}$ is \emph{live at}~$\ell$ iff the adversary still holds $c$ \emph{and} releasing $c$ would update $\comm{\Sigma}^{(\ell)}.J$ via the lex test in \updateJustified{} --- equivalently (using Lemma~\ref{lem:just-unique-height} to discard hash tiebreaks), iff $c$ is held and $\mathrm{height}(c) > \comm{\Sigma}^{(\ell)}.h_j$;
  \item $N^{(\ell)}$ counts live cards at~$\ell$;
  \item $\Phi^{(\ell)} \coloneqq \min(\comm{\Sigma}^{(\ell)}.h_j,\; h^0) + N^{(\ell)}$.
\end{itemize}

The bound $\Phi^{(\ell)} \leq h^0$ holds in all cases:
$\comm{\Sigma}^{(\ell)}.h_j \geq h^0$ kills both cards ($N = 0$, $\Phi = h^0$); $\comm{\Sigma}^{(\ell)}.h_j = h^0 - 1$ kills $J^*$ ($N \leq 1$, $\Phi \leq h^0$); $\comm{\Sigma}^{(\ell)}.h_j \leq h^0 - 2$ admits $N \leq 2$ ($\Phi \leq h^0$).

\emph{Step 3: $\Phi$ is monotone non-decreasing.}
Three transition classes can change $\Phi$ in $[\vote(\ell),\, \vote(\ell+1)]$.

\emph{(T1) Live card $c$ released and absorbed.}
Pre-release $h_j^{\mathrm{old}} < \mathrm{height}(c)$ (liveness). Post-release $h_j \gets \mathrm{height}(c)$, so $\Delta \min(h_j, h^0) = \mathrm{height}(c) - h_j^{\mathrm{old}} \geq 1$; $c$ leaves the live set ($\Delta N \ni \{-1\}$); a second card $c'$ leaves the live set iff $\mathrm{height}(c') \leq \mathrm{height}(c)$.
Direct enumeration on $\{(c, c')\} \in \{(J^*, J^\dagger), (J^\dagger, J^*)\}$, using $\mathrm{height}(J^*) = h^0 - 1 < h^0 = \mathrm{height}(J^\dagger)$ and the corresponding liveness thresholds:
\begin{itemize}
  \item $c = J^*$ released, $J^\dagger$ live: $\Delta \min = h^0 - 1 - h_j^{\mathrm{old}} \geq 1$, $\Delta N = -1$ (only $J^*$ removed; $J^\dagger$ remains live since $h^0 > h^0 - 1$); $\Delta \Phi \geq 0$;
  \item $c = J^\dagger$ released, $J^*$ live ($h_j^{\mathrm{old}} \leq h^0 - 2$): $\Delta \min = h^0 - h_j^{\mathrm{old}} \geq 2$, $\Delta N = -2$ (both cards removed); $\Delta \Phi \geq 0$;
  \item $c = J^\dagger$ released, $J^*$ not live: $\Delta \min = h^0 - h_j^{\mathrm{old}} \geq 1$, $\Delta N = -1$; $\Delta \Phi \geq 0$.
\end{itemize}

\emph{(T2) Honest virtual cert via \updateJFromVotes{} or on-chain cert event.}
The cert offers a key at height $h^* \in \{h_c^{(\ell)},\, h^0 - 1,\, h^0\}$, with $h_c^{(\ell)} \coloneqq \sigma[\getConfirmed^{\vote(\ell)}].h \geq \comm{\Sigma}^{(\ell)}.\hmax - 1 \geq h^0 - 1$.
\updateJustified{} updates $h_j$ iff the lex test passes, in which case $\Delta \min(h_j, h^0) \geq 1$ and $\Delta N$ is at most the number of cards with height in $(h_j^{\mathrm{old}}, h^*]$, which is at most $\Delta \min(h_j, h^0)$ since payloads are distinct.
$\Delta \Phi \geq 0$.
If lex fails (e.g., gate firing with $h_c^{(\ell)} = h_j^{\mathrm{old}}$ giving a hash-tied cert at the unique target by Lemma~\ref{lem:just-unique-height}): no-op, $\Delta \Phi = 0$.

\emph{(T3) View-merge.}
$\comm{\Sigma}^{(\ell+1)}.h_j$ takes the cross-honest maximum, weakly $\geq$ pre-merge values.
$N$ shrinks weakly. $\Delta \Phi \geq 0$.

\emph{Step 4: Frontier advance.}
We claim $\comm{\Sigma}^{(\ell+1)}.h_j \geq h^0 - 1$ for $\ell \in \{k+1, k+2\}$.
Slot-$\ell$ honest votes have $.\height = h_c^{(\ell)} \geq h^0 - 1$ (uniformity from view-merge plus Lemma~\ref{lem:honest-vote-sets-timeout}); their targets lie on the chain~$D^{(\ell)}$ ending at $\getConfirmed_{v_\ell}^{\vote(\ell)}$ at $\sigma$-height $h_c^{(\ell)}$.
By Definition~\ref{def:standing}(d), an honest available-chain proposer aggregates these slot-$\ell$ votes into a slot-$\ell$ block delivered to every honest validator before $\vote(\ell+1) - \Delta$, and \updateJFromVotes{} fires either the on-chain or the virtual cert at $h_c^{(\ell)}$.
Either canonical at $\vote(\ell+1)$ extends $D^{(\ell)}$, so the cert's lex test passes and $\comm{\Sigma}^{(\ell+1)}.h_j \geq h_c^{(\ell)} \geq h^0 - 1$; or canonical reorganizes onto the chain of a card released in $[\vote(\ell), \vote(\ell+1))$, in which case the released card's height ($h^0 - 1$ or $h^0$) bounds $\comm{\Sigma}^{(\ell+1)}.h_j$ from below by $h^0 - 1$.

\emph{Step 5: Cascade gate at $\vote(\ell)$ for $\ell \in \{k+2, k+3\}$.}
Step~4 gives $\comm{\Sigma}^{(\ell)}.h_j \geq h^0 - 1$.
$\comm{\Sigma}^{(\ell)}.\hmax \in \{h^0,\, h^0 + 1\}$: $\hmax = h^0 + 1$ requires $B^\dagger$ to have entered $\Sigma.\T$, which forces $\comm{\Sigma}^{(\ell)}.h_j \geq h^0$; $\hmax = h^0$ admits $\comm{\Sigma}^{(\ell)}.h_j \in \{h^0 - 1\}$.
Combined with the protocol invariant $h_j \leq \hmax - 1$:
\begin{itemize}
  \item $\hmax = h^0,\, h_j = h^0 - 1$: gate fires;
  \item $\hmax = h^0 + 1,\, h_j = h^0$: gate fires.
\end{itemize}

\emph{Step 6: Identifying $k_h$.}
Let $\ell_d$ be the largest $\ell \in \{k+1, k+2\}$ such that a live card was released-and-absorbed in $[\vote(\ell), \vote(\ell+1))$ (set $\ell_d = k$ if no such slot exists, treating slot~$k+1$ as the first opportunity).
Set $k_h \coloneqq \ell_d + 1 \in \{k+1, k+2, k+3\}$ when $\ell_d \geq k+1$; and $k_h \coloneqq k+1$ when $\ell_d = k$.

By Step~4, $\comm{\Sigma}^{(k_h)}.h_j \geq h^0 - 1$, and Step~5 gives gate firing.
View-merge under the honest slot-$k_h$ syncer makes $\Sigma_v^{\vote(k_h)} = \comm{\Sigma}^{(k_h)}$ uniform across honest, hence $\getConfirmed_v^{\vote(k_h)} \sim \getConfirmed_{v'}^{\vote(k_h)}$ (condition~\ref{rev:def:healed-state:compat} of Definition~\ref{def:healed}).
Set $\comm{h} \coloneqq \comm{\Sigma}^{(k_h)}.h_j \geq h^0 - 1$ and $R_v^{\vote(k_h)} = \comm{\Sigma}^{(k_h)}.J$ at $\sigma$-height $\comm{h}$ (condition~\ref{rev:def:healed-state:h}).

\emph{Condition~(3) at $\vote(k_h)$.}
Pick $v_\ell$ honest with $\preceq$-maximal $\getConfirmed^{\vote(k_h)}$ on the common chain (well-defined by condition~\ref{rev:def:healed-state:compat}).
For an honest vote~$u$ cast strictly before~$\vote(k_h)$ with $u.\height \geq \comm{h}$:
\begin{itemize}
  \item $u$ contributes to the justification at $\mathrm{height}(c) \in \{h^0 - 1, h^0\}$ for some absorbed card~$c$: by Lemma~\ref{lem:just-unique-height}, the unique justified target at $\mathrm{height}(c)$ is~$c$ on its chain $\preceq \comm{\Sigma}^{(k_h)}.J$ post-absorption, and freshness of $u$ on its contributing chain forces $u.\target = c$ on canonical at $\vote(k_h)$;
  \item $u$ is a slot-$\ell$ honest vote for $\ell \in \{k+1, \ldots, k_h - 1\}$: by view-merge uniformity at $\vote(\ell)$ together with Lemma~\ref{lem:honest-vote-sets-timeout}, all honest at $\vote(\ell)$ target the same block $T_\ell$ on the common canonical chain at $\sigma$-height $h_c^{(\ell)} \in \{h^0 - 1, h^0\}$.
  Definition~\ref{def:standing}(b,c) gives $\geq 2n/3$ honest contributions, forming a justification quorum at $h_c^{(\ell)}$ targeting $T_\ell$ (delivered via on-chain processing or \updateJFromVotes{} by $\vote(\ell+1) - \Delta$).
  Lemma~\ref{lem:just-unique-height} then forces $T_\ell$ to coincide with the unique justified block at $h_c^{(\ell)}$, namely $J^*$ when $h_c^{(\ell)} = h^0 - 1$ and $J^\dagger$ when $h_c^{(\ell)} = h^0$ (creating $J^*$ or $J^\dagger$ as a newly justified block when no card-borne justification pre-existed at that height).
  Both $J^*$ and $J^\dagger$ lie on canonical at $\vote(k_h)$: by Step~1, $J^* \preceq J^\dagger$ when both exist, and Lemma~\ref{lem:Rs-key-monotone} together with the per-height uniqueness places the unique justified block of $\comm{\Sigma}^{(k_h)}.h_j$ on canonical via $\Sigma.J$-monotonicity along absorbed-card transitions.
  Hence $u.\target \in \{J^*, J^\dagger\} \preceq \getConfirmed_{v_\ell}^{\vote(k_h)}$.
\end{itemize}
This is condition~\ref{rev:def:healed-state:cap} at slot~$k_h$ with frontier height $\comm{h} \geq h^0 - 1$.
\end{proof}

\subsection{Main theorem}\label{sec:healing-main}

\begin{theorem}[Healing]\label{thm:healing}
Assume $f < n/3$ and that all honest validators follow Definition~\ref{def:voting-rule}.
Let~$k$ be a slot such that slots~$k+1, k+2, k+3$ each satisfy the standing assumptions and have honest syncers.
Then the system is in a healed state at some slot $k_h \in \{k+1, k+2, k+3\}$, and remains in a healed state at every subsequent slot satisfying the standing assumptions and with an honest syncer.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:healing-3slot} delivers $k_h \in \{k+1, k+2, k+3\}$ with the system healed at slot~$k_h$.
Lemma~\ref{lem:healing-persistence} extends inductively: given the healed state at slot~$\ell \geq k_h$, if slot~$\ell+1$ satisfies the standing assumptions and has an honest syncer, then the system is healed at slot~$\ell+1$ with frontier height non-decreasing.
\end{proof}
