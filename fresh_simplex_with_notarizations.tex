\section{Model and definitions}

We consider a system of~$n$ validators of which at most~$f$ are adversarial, where $n \geq 3f + 1$.
A \emph{quorum} is any set of at least $2n/3$ validators.
Any two quorums overlap in at least $n/3 + 1 > f$ validators, so their intersection contains at least one honest validator.

\begin{definition}[Height]\label{def:height}
\emph{Height} is the finality-gadget counter, separate from slot.
The genesis height is~$0$; height advances by exactly~$1$ via fixed-target justification (Definition~\ref{def:justification}) or prefix notarization (Definition~\ref{def:notarization}).
\end{definition}

\begin{definition}[Block]\label{def:block}
A \emph{block} has a \emph{parent} block, a \emph{slot} field, and a set of votes (Definition~\ref{def:vote}).
The \emph{genesis block}~$\genesis$ has no parent and slot~$0$.
Slots strictly increase along parent links: $B.\slot > B.\parent.\slot$ for every non-genesis~$B$.
\end{definition}

\begin{definition}[Chain]\label{def:chain}
A \emph{chain} is a path from~$\genesis$ to some block~$B$ via parent links.
Chains are in bijection with their tips, and we identify the two.
Blocks form a tree rooted at~$\genesis$; we write $B \preceq B'$ when $B$ is an ancestor of~$B'$ (or $B = B'$), and $B \sim B'$ (\emph{compatible}) when $B \preceq B'$ or $B' \preceq B$.
Two chains \emph{conflict} when their tips are not compatible.
\end{definition}

\begin{definition}[Vote]\label{def:vote}
A \emph{vote} is a signed tuple
\[
  (\validator,\; \height,\; \slot,\; \target,\; \kind,\; \finalizeHeight,\; \finalizeTarget),
\]
where $\validator$ is the signer, $\height$ is a height, $\slot$ a slot, $\target$ a block (never~$\bot$), and $\kind \in \{\just,\, \notar\}$.
The pair $(\finalizeHeight,\, \finalizeTarget)$ is either both~$\bot$ or a commitment to finalize a block at a height.
A vote included on a chain must reference a target (and finalize target, when present) that exists on that chain.
\end{definition}

\paragraph{State.}
The \emph{state after block~$B$}, written $\sigma[B]$, is the tuple
\[
  (L,\; s,\; h,\; s_h,\; \jtargets,\; \ntargets,\; \nslots,\; J,\; h_j,\; N,\; h_n,\; s_n,\; F,\; P),
\]
with components:
\begin{itemize}
  \item $L$: the \emph{chain head} ($L = B$ after processing~$B$).
  \item $s$: the \emph{current slot}, incremented once per slot by \processSlot (Figure~\ref{alg:state-machine}).
  \item $h$: the \emph{current height}.
  \item $s_h$: the slot of the block whose processing last advanced~$h$ on this chain ($0$ if no advance has occurred).
  \item $\jtargets[1..n]$: for each validator~$i$, the target of~$i$'s \just vote at the current height ($\bot$ if none processed).
  \item $\ntargets[1..n]$: for each~$i$, the target of the latest-slot fresh vote of~$i$ at the current height, across both kinds ($\bot$ if none processed). Both \just and \notar votes can set $\ntargets[i]$; a later-slot vote overwrites any earlier entry.
  \item $\nslots[1..n]$: for each~$i$, the $\slot$ of the vote that set $\ntargets[i]$ ($0$ if none).
  \item $J$, $h_j$: the most recently justified block and its height.
  \item $N$, $h_n$: the most recently notarized (or justified) block and its height; $J$ is itself a notarization (Definition~\ref{def:notarization}).
  \item $s_n$: the \emph{notarization slot} of the most recent prefix-notarization-only advance --- the common slot~$s$ shared by the $\geq 2n/3$ contributors of the firing quorum, i.e., a slot such that $\geq 2n/3$ of the contributors whose $\ntargets$ covers~$N$ satisfy $\nslots = s$. Justifications do not set $s_n$; the field is reset to $0$ on a justification advance, and initializes to $0$.
  \item $F$: the most recently finalized block.
  \item $P \subseteq \{1, \ldots, n\}$: validators who have cast a vote with $\finalizeHeight = h_j$ and $\finalizeTarget = J$.
\end{itemize}
The genesis state is
\[
  \sigma_{\mathrm{gen}} \coloneqq (\genesis,\; 0,\; 0,\; 0,\; \bot^n,\; \bot^n,\; 0^n,\; \genesis,\; 0,\; \genesis,\; 0,\; 0,\; \genesis,\; \emptyset),
\]
with $\sigma[\genesis] \coloneqq \sigma_{\mathrm{gen}}$.
When unambiguous we drop the chain prefix and write $L$, $s$, $h$, etc.\ for the components of $\sigma[L]$.

\begin{definition}[State-height]\label{def:sheight}
The \emph{state-height} of a block~$B$ is $\sigma[B].h$: the height after processing~$B$ on its own chain.
\end{definition}

\paragraph{Height freshness.}
A vote~$v$ with target~$T$ is processed on a chain with head state $\sigma[L]$ only if
\[
  \sigma[L].h = v.\height \;\wedge\; T \preceq L \;\wedge\; T.\slot \geq \sigma[L].s_h:
\]
the chain is at~$v$'s height, $T$ lies on this chain, and $T$'s slot is no earlier than where the chain's current height began.
Intuitively, $v$ attests that its signer saw~$T$ as a chain of height~$v.\height$.

\begin{remark}[Freshness via $T$'s own chain]\label{rem:fresh-equiv}
Given $T \preceq L$, the constraint $T.\slot \geq \sigma[L].s_h$ is equivalent to $\sigma[T].h = \sigma[L].h$.
State-height is $\preceq$-monotone (Lemma~\ref{lem:slotmono}), equals $\sigma[L].h$ on~$L$, and advanced to its current value at the block whose slot is $\sigma[L].s_h$; so the ancestors of~$L$ at that state-height are exactly those with slot $\geq \sigma[L].s_h$.
\end{remark}

Freshness guarantees that every justification or notarization at height~$h$ is witnessed by targets whose own chains reached state-height~$h$.

\begin{definition}[Fixed-target justification]\label{def:justification}
A block~$T$ is \emph{justified at height~$h$} on a chain with state $\sigma[L]$ with $\sigma[L].h = h$ iff
\[
  \bigl|\{i \,:\, \jtargets[i] = T\}\bigr| \;\geq\; 2n/3.
\]
Only votes with target exactly~$T$ contribute.
\end{definition}

\begin{definition}[Prefix notarization]\label{def:notarization}
On a chain with head state $\sigma[L]$ at height~$h$, a block~$T$ is \emph{notarized at height~$h$ with slot~$s$} iff~$T$ is the $\preceq$-maximum block for which some slot~$s$ satisfies
\[
  \bigl|\{i : T \preceq \ntargets[i] \preceq L \;\wedge\; \nslots[i] = s\}\bigr| \;\geq\; 2n/3.
\]
If no such pair exists, nothing is notarized; when~$T$ exists the witnessing~$s$ is unique, since the sets indexed by distinct~$s$ are disjoint and cannot both exceed~$n/3$.
Notarization uses \emph{prefix counting}: \just or \notar votes targeting descendants of~$T$ contribute via $\ntargets$, but only the latest-slot entry per validator counts, and the firing quorum must share a single $\nslots$ value~$s$.
A justification of~$T$ is paired with a notarization cert event at~$T$: the justification branch of \processHeight sets $(J, N) \gets (T, T)$ jointly (Figure~\ref{alg:state-machine}).
\end{definition}

\begin{definition}[Finality]\label{def:finality}
The justified block~$J$ is \emph{finalized at height~$h_j$} once $|P| \geq 2n/3$.
\end{definition}

\begin{definition}[Slashing, rule E1]\label{def:slashing}
A validator is \emph{slashable} if it signed two votes~$a, b$ (possibly $a = b$) with
\[
  b.\finalizeTarget \neq \bot \;\wedge\; a.\height = b.\finalizeHeight \;\wedge\; a.\target \neq b.\finalizeTarget.
\]
That is, a validator committing $(\finalizeHeight,\, \finalizeTarget) = (h_f, T_f)$ must not sign any vote at height~$h_f$ with target $\neq T_f$.
\end{definition}

Two \just (or two \notar) votes at the same height are not by themselves slashable; per-height uniqueness of honest \just votes is imposed behaviorally (Definition~\ref{def:voting-rule}).
A validator who never sets $\finalizeTarget \neq \bot$ trivially avoids slashing.

\paragraph{State transitions.}
The state machine (Figure~\ref{alg:state-machine}) exposes two per-chain routines: \processSlot, invoked once per slot on every maintained chain, and \processBlock, invoked additionally whenever a block~$B$ is processed.

\processSlot increments~$s$ and calls \processHeight, which in turn checks three branches in order: (i) finality, $F \gets J$ if $|P| \geq 2n/3$; (ii) justification, if some~$T$ has $\geq 2n/3$ entries in $\jtargets$; (iii) prefix notarization, at the firing block~$T^*$ located by the walk of Definition~\ref{def:notarization}.
Branches (ii) and (iii) both advance height and reset $\jtargets, \ntargets$, so at most one height-advance fires per invocation.

$\processBlock(\sigma, B)$ has precondition $\sigma.s = B.\slot$: the caller runs \processSlot up to $B.\slot$ first.
Then $L \gets B$, and each vote~$v$ in~$B$ is fed through \processVote.
A fresh vote updates $\ntargets[i] \gets \max_{\preceq}(\ntargets[i],\, v.\target)$ (well-defined as both arguments lie on~$L$ at height $\sigma.h$, treating~$\bot$ as~$\genesis$); if also $v.\kind = \just$, then $\jtargets[i] \gets v.\target$.
The $P$-gate adds~$i$ to~$P$ iff $v.\finalizeHeight = \sigma.h_j$ and $v.\finalizeTarget = \sigma.J$; this is independent of freshness.
\processBlock never advances~$h$: height advances happen only in \processSlot, which runs on every slot (including empty ones).

\begin{algorithmic}[1]

\Function{\processSlot}{$\sigma$} \Comment{Runs once per slot on every chain, even without a block}
  \State $\sigma.s \gets \sigma.s + 1$
  \State $\sigma \gets \processHeight(\sigma)$
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\processBlock}{$\sigma,\; B$}
  \State \textbf{assert} $\sigma.s = B.\slot$ \Comment{Precondition: caller advanced $\sigma.s$ via \processSlot}
  \State $\sigma.L \gets B$
  \For{each vote $v$ in $B$} $\sigma \gets \processVote(\sigma,\, v)$ \EndFor
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\processVote}{$\sigma,\; v$}
  \State $i \gets v.\validator$
  \State $\fresh \gets (v.\height = \sigma.h) \land (v.\target \preceq \sigma.L) \land (v.\target.\slot \geq \sigma.s_h)$
  \If{$\fresh$ \textbf{and} $v.\slot > \sigma.\nslots[i]$}
    \State $\sigma.\ntargets[i] \gets v.\target$ \Comment{both kinds bump $\ntargets$, slot-gated}
    \State $\sigma.\nslots[i]  \gets v.\slot$
  \EndIf
  \If{$\fresh$ \textbf{and} $v.\kind = \just$}
    \State $\sigma.\jtargets[i] \gets v.\target$
  \EndIf
  \If{$v.\finalizeHeight = \sigma.h_j$ \textbf{and} $v.\finalizeTarget = \sigma.J$}
    \State $\sigma.P \gets \sigma.P \cup \{i\}$
  \EndIf
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\processHeight}{$\sigma$}
  \If{$|\sigma.P| \geq 2n/3$} \Comment{Finality}
    \State $\sigma.F \gets \sigma.J$
  \EndIf
  \State $T \gets \arg\max_{T' \in \{\sigma.\jtargets[i] \,:\, \sigma.\jtargets[i] \neq \bot\}} |\{i : \sigma.\jtargets[i] = T'\}|$ \Comment{$T \gets \bot$ if the set is empty;}
  \If{$T \neq \bot$ \textbf{and} $|\{i : \sigma.\jtargets[i] = T\}| \geq 2n/3$} \Comment{Justification (also a notarization)}
    \State $\sigma.J \gets T$, \; $\sigma.h_j \gets \sigma.h$, \; $\sigma.P \gets \emptyset$
    \State $\sigma.N \gets T$, \; $\sigma.h_n \gets \sigma.h$, \; $\sigma.s_n \gets 0$
    \State $\sigma.h \gets \sigma.h + 1$, \; $\sigma.s_h \gets \sigma.L.\slot$
    \State reset $\jtargets, \ntargets, \nslots$
    \State \Return $\sigma$
  \EndIf
  \State $(B^*, s^*) \gets \arg\max_{\preceq}\,\bigl\{(B, s) : B \preceq \sigma.L \;\wedge\; |\{i : B \preceq \sigma.\ntargets[i] \preceq \sigma.L \wedge \sigma.\nslots[i] = s\}| \geq 2n/3\bigr\}$
  \Statex \hspace{6.1em} \Comment{$\arg\max$ by $\preceq$ on~$B$; $s$ given $B$ is unique; $(B^*, s^*) \gets \bot$ if the set is empty}
  \If{$(B^*, s^*) \neq \bot$} \Comment{Prefix notarization fires}
    \State $\sigma.N \gets B^*$, \; $\sigma.h_n \gets \sigma.h$, \; $\sigma.s_n \gets s^*$
    \State $\sigma.h \gets \sigma.h + 1$, \; $\sigma.s_h \gets \sigma.L.\slot$
    \State reset $\jtargets, \ntargets, \nslots$
  \EndIf
  \State \Return $\sigma$
\EndFunction
\vspace{3pt}

\Function{\finalizeConflict}{$v_1, v_2$} \Comment{Does $v_1$ conflict with $v_2$'s finalize commitment?}
  \State \Return $v_2.\finalizeTarget \neq \bot \;\wedge\; v_1.\height = v_2.\finalizeHeight \;\wedge\; v_1.\target \neq v_2.\finalizeTarget$
\EndFunction
\vspace{3pt}

\Function{\isSlashable}{$v_1, v_2$}
  \State \Return $v_1.\validator = v_2.\validator$
  \Statex \hspace{3em}$\land \;(\finalizeConflict(v_1, v_2) \lor \finalizeConflict(v_2, v_1))$
\EndFunction

\section{Accountable safety}

\begin{lemma}[Height progression]\label{lem:heightprog}
\processHeight increments~$h$ by at most~$1$ per invocation, and each increment is gated by a $\geq 2n/3$ justification or prefix-notarization quorum.
\end{lemma}

\begin{proof}
The only assignments to~$h$ are $h \gets h + 1$ in the justification and notarization branches of \processHeight, each guarded by its $\geq 2n/3$ quorum.
\end{proof}

\begin{lemma}[Checkpoint monotonicity]\label{lem:slotmono}
On any chain, the fields $s$, $h$, $h_j$, $h_n$, $s_h$, $F$, $J$, $N$ are non-decreasing along~$\preceq$, with $F \preceq J \preceq N$ at all times.
$J$ and $h_j$ advance only on justification events, and $h_n, s_h$ only on height-advances, with $s_h$ set to the slot of the advancing block.
Consequently, when $\sigma[Y].h = h$, the ancestors of~$Y$ at state-height~$h$ are $\{B' \preceq Y : B'.\slot \geq \sigma[Y].s_h\}$.
\end{lemma}

\begin{proof}
Reading off Figure~\ref{alg:state-machine}: $s$ only increases (in \processSlot); $h$ only increases, by~$1$, in the height-advance branches of \processHeight, which also set $h_j, h_n \gets h$ on their respective updates and $s_h \gets L.\slot$.
Slots strictly increase along~$\preceq$ (Definition~\ref{def:block}), so $s_h$ is non-decreasing.
$F$ updates only via $F \gets J$, so $F \preceq J$ reduces to $J$-monotonicity.

\emph{Height-$h$ interval.}
Along any chain, $h$ increases monotonically; when $\sigma[Y].h = h$, the value $s_h$ was set at the block that advanced state-height to~$h$.
By strict slot increase along~$\preceq$ and the monotonicity of $(h, s_h)$, the ancestors $B' \preceq Y$ with $\sigma[B'].h = h$ are exactly those with $B'.\slot \geq s_h$.

\emph{Block fields $J$, $N$.}
At a height-advance on chain~$Y$ firing at target~$T$ with pre-state height~$h$, freshness forces $T.\slot \geq s_h$: directly for justification, since every contributing \just vote has target~$T$; for prefix notarization, since the firing~$T$ is some $\ntargets[i]$ value, hence has slot $\geq s_h$.
By the height-$h$ interval claim, $\sigma[T].h = h$, so the previous notarized block $N_{\mathrm{old}}$ with $\sigma[N_{\mathrm{old}}].h \leq h-1$ and $N_{\mathrm{old}} \preceq Y$ satisfies $T \succ N_{\mathrm{old}}$ by state-height monotonicity.
The justification branch sets $J = N = T$ jointly, the notarization branch only $N \gets T$: $N$ strictly advances at every height-advance, $J$ at every justification, and $J \preceq N$ is preserved.
\end{proof}

\begin{lemma}[Any chain past a finalized height contains the finalized block]\label{lem:mainsafety}
Unless $\geq n/3$ validators are slashable: if~$C$ is finalized at height~$h$, then $C \preceq B$ for every block~$B$ with $\sigma[B].h > h$.
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:heightprog}, some $B^* \preceq B$ advanced height from~$h$ to~$h+1$ via a quorum~$Q$ at height~$h$ on~$B^*$'s chain; each $i \in Q$ contributed a fresh vote~$a_i$ at height~$h$ (freshness is automatic for \just votes; prefix-notarization entries in $\ntargets$ are also fresh by Figure~\ref{alg:state-machine}).
Finality of~$C$ at~$h$ gives a quorum~$Q_F$ each signing a vote~$b$ with $b.\finalizeHeight = h$ and $b.\finalizeTarget = C$.

By quorum intersection, $|Q \cap Q_F| \geq n/3$; pick $v \in Q \cap Q_F$ not slashable.
E1 (Definition~\ref{def:slashing}) applied to $v$'s votes $a \in Q$, $b \in Q_F$ with $a.\height = h = b.\finalizeHeight$ forces $a.\target = C$; freshness gives $C = a.\target \preceq B^* \preceq B$.
\end{proof}

\begin{lemma}[Finalized blocks form a chain]\label{lem:finchain}
Unless $\geq n/3$ validators are slashable: any two finalized checkpoints $(C, h), (C', h')$ are compatible, with $C \preceq C'$ whenever $h \leq h'$.
\end{lemma}

\begin{proof}
\emph{Case $h = h'$.}
Finality of~$C$ gives a quorum~$Q_F$ with $\finalizeHeight = h$, $\finalizeTarget = C$.
Finality of~$C'$ requires justification of~$C'$, giving a \just-quorum~$Q'$ at height~$h$ with target~$C'$.
By quorum intersection, $|Q_F \cap Q'| \geq n/3$; pick $v \in Q_F \cap Q'$ not slashable.
E1 (Definition~\ref{def:slashing}) applied to $v$'s votes in~$Q_F, Q'$ forces $C = C'$.

\emph{Case $h < h'$.}
Let~$Z$ be the chain on which~$C'$ was finalized, at the post-state where the finalize-quorum closed; then $\sigma[Z].h \geq h' > h$, $C' \preceq Z$, and $\sigma[C'].h = h'$ (Remark~\ref{rem:fresh-equiv}).
Lemma~\ref{lem:mainsafety} at~$Z$ gives $C \preceq Z$; Remark~\ref{rem:fresh-equiv} on~$C$'s own finalizing chain gives $\sigma[C].h = h$.
Both~$C, C'$ lie on~$Z$, so comparable; with $\sigma[C].h = h < h' = \sigma[C'].h$, state-height monotonicity (Lemma~\ref{lem:slotmono}) forces $C \prec C'$.
\end{proof}

\begin{theorem}[Accountable safety]\label{thm:safety}
Unless $\geq n/3$ validators are slashable, no two conflicting blocks can be finalized.
\end{theorem}

\begin{proof}
Immediate from Lemma~\ref{lem:finchain}: any two finalized blocks are compatible.
\end{proof}

\begin{remark}[Ancestor notarizations]\label{rem:notarization-safety}
Because \just votes bump $\ntargets$, a prefix notarization at height~$h$ on a chain containing a finalized~$C$ may fire at an ancestor $T^* \preceq C$.
No per-chain invariant requires $N \succeq C$; the store (Section~\ref{sec:store}) resolves such cases via lexicographic cert keys.
\end{remark}

\section{Fork-choice store}\label{sec:store}

A node may track multiple chains simultaneously; the \emph{store} is the node-local data structure that mediates which chain the node follows.
It maintains a single \emph{root block} $R$ together with a cached \emph{cert key} $\key_R$ for the event that last set $R$.
Each firing of \processHeight (a \emph{cert event}) is compared against $\key_R$ in lex order; whenever its key strictly dominates, $(R, \key_R)$ is updated to the new event.
The key packages height, a justified-bit, the notarization slot, the block slot, and the block hash, so that a justification strictly dominates a prior notarization-only event at the same block, a higher notarization slot outranks a lower one at the same (height, bit), and a higher-height cert always wins over a lower one.

\begin{definition}[Cert event and cert key]\label{def:cert-event}
When \processHeight fires a height advance on a chain (Figure~\ref{alg:state-machine}), we call the triggering event a \emph{cert event}, characterized by:
\begin{itemize}
  \item the block $X = \sigma.N$ at firing (equal to the justified target~$T$ in the justification branch, and to the prefix-notarization endpoint~$T^*$ in the notarization branch);
  \item the height~$h$ at which the advance fires, equal to the pre-advance $\sigma.h$ and to the post-advance $\sigma.h_n$;
  \item a \emph{bit} $b \in \{0, 1\}$: $b = 1$ for a justification, $b = 0$ for a prefix notarization without accompanying justification;
  \item the \emph{notarization slot} $s_n = \sigma.s_n$: for a bit-$0$ event, the common slot~$s$ shared by the $\geq 2n/3$ firing contributors, i.e., a slot such that $\geq 2n/3$ of the contributors with $\sigma.\ntargets$ covering~$X$ on~$\sigma.L$ satisfy $\sigma.\nslots = s$; for a bit-$1$ event (justification), $s_n = 0$.
\end{itemize}
The associated \emph{cert key} is the lexicographic tuple
\[
  \key(X, h, b, s_n) \;\coloneqq\; \bigl(h,\; b,\; s_n,\; X.\slot,\; \hash(X)\bigr).
\]
\end{definition}

A justification also prefix-notarizes its block (the justification branch sets $(J, N) \gets (T, T)$ jointly), but we classify the combined event as a justification with key $\key(T, h, 1, s_n)$.
A given block may therefore fire a bit-$0$ cert event on one chain and a bit-$1$ cert event on another, with the bit-$1$ key strictly dominating at the same block.

\begin{definition}[Store]\label{def:store}
A node maintains a \emph{store}
\[
  \Sigma \;=\; (\sigma,\; \T,\; F,\; R,\; \key_R),
\]
where:
\begin{itemize}
  \item $\sigma$ is the \emph{state map}, assigning each accepted block its per-chain post-state (Section~1);
  \item $\T$ is the \emph{block tree}, the set of accepted blocks;
  \item $F$ is the \emph{store-finalized block};
  \item $R$ is the \emph{store root}: the block of the highest-key cert event ever observed;
  \item $\key_R$ is the cached cert key of the event that set $R$.
\end{itemize}
The genesis store has $\T = \{\genesis\}$, $F = R = \genesis$, and $\key_R = (0,\, 1,\, 0,\, \genesis.\slot,\, \hash(\genesis))$, treating genesis as a bit-$1$ cert event at height~$0$ with notarization slot~$0$.
\end{definition}

The capital~$\Sigma$ distinguishes the store from the per-chain state~$\sigma$.
We write $\Sigma.\sigma$, $\Sigma.\T$, $\Sigma.F$, $\Sigma.R$, $\Sigma.\key_R$ for field access, and $\Sigma.\key_R.h$, $\Sigma.\key_R.b$ for the height and justification-bit components of the cached key.

The acceptance routine \onBlock (Figure~\ref{alg:store}) admits $B$ only if $B.\parent \in \Sigma.\T$ and $\Sigma.F \preceq B$.
It runs \processSlot once per slot from $\Sigma.\sigma[B.\parent].s + 1$ up to $B.\slot$ (so any height advance enabled by the accumulated tally fires before~$B$'s votes are folded in), then $\processBlock(\cdot, B)$; the resulting post-state becomes $\Sigma.\sigma[B]$.
The post-state then offers two cert-event descriptors to \updateForkChoiceRoot: the justified checkpoint $(\sigma[B].h_j,\, 1,\, \sigma[B].J,\, \sigma[B].s_n)$ and the notarized checkpoint $(\sigma[B].h_n,\, 0,\, \sigma[B].N,\, \sigma[B].s_n)$; the bit-$1$ / bit-$0$ split reflects the distinct events (the justification may have fired at a strictly smaller height than the notarization).
Each call to \updateForkChoiceRoot first filters out any candidate~$C$ with $C \not\succeq \Sigma.F$ and then updates $(\Sigma.R,\, \Sigma.\key_R)$ iff the candidate's cert key strictly exceeds the current $\key_R$; this maintains $R$ incrementally as the lex-max cert event ever observed on a chain extending the current $F$, never recomputed.
Finally, \updateFinalized sets $\Sigma.F \gets \sigma[B].F$ iff $\sigma[B].F \succ \Sigma.F$ and $\sigma[B].F \preceq \Sigma.R$.

The $F$-filter never discards a key-dominant event: Theorem~\ref{thm:fleqr} below shows that $F$ advances only to blocks $\preceq R$, so every cert event whose key could beat $R$ remains on the $F$-chain.

\subsection{Store invariants and safety}

\subsubsection*{Unconditional invariants}

\begin{lemma}[Root-key monotonicity]\label{lem:Rs-key-monotone}
The cached cert key $\Sigma.\key_R$ is non-decreasing over time; in particular, its height component $\Sigma.\key_R.h$ is non-decreasing.
\end{lemma}

\begin{proof}
The cached key changes only inside \updateForkChoiceRoot, whose guard $\newKey > \Sigma.\key_R$ (lex) allows only strict increases; the height coordinate $h$ is the leading component of the key.
\end{proof}

\begin{theorem}[Local irreversibility of finality]\label{thm:finperm}
Once a node sets $\Sigma.F = F$, $\Sigma.F$ descends from~$F$ at all future times.
\end{theorem}

\begin{proof}
$\Sigma.F$ is updated only by \updateFinalized, whose guard $F_B \succ \Sigma.F$ forces strict extension along~$\preceq$.
\end{proof}

\begin{theorem}[$F \preceq R$]\label{thm:fleqr}
The store maintains $F \preceq R$ at all times.
\end{theorem}

\begin{proof}
By induction on store operations.
At genesis, $F = R = \genesis$, so $F \preceq R$.
Assume the invariant before a call to either mutator in Figure~\ref{alg:store}.

\emph{\updateForkChoiceRoot.}
$F$ is untouched, and $R$ either stays or is replaced by a candidate~$C$ that passed the $C \succeq \Sigma.F$ filter; either way $R \succeq F$.

\emph{\updateFinalized.}
$R$ is untouched, and the guard $F_B \preceq \Sigma.R$ ensures $F \gets F_B \preceq R$.
\end{proof}

\begin{remark}[Store-finalized invariant]\label{rem:fs-invariant}
At all times $\Sigma.F$ is either~$\genesis$ or a block that was finalized on some chain; moreover, whenever a justification $(J, h)$ has fired on some chain, the cert event $(h,\, 1,\, J,\, 0)$ was offered to \updateForkChoiceRoot, so $\Sigma.\key_R \geq (h,\, 1,\, 0,\, J.\slot,\, \hash(J))$ for every such~$(J, h)$ with $J \succeq \Sigma.F$.
$\Sigma.F$ only advances via \updateFinalized to $F_B = \sigma[B].F$, and $\sigma[B].F \neq \genesis$ requires the finality branch of \processHeight to have fired on~$B$'s chain, so $F_B$ is a block finalized on that chain.
\end{remark}

\begin{theorem}[Fork-choice consistency]\label{thm:fcconsistency}
Once a node sets $\Sigma.F = F$, \getHead returns a block descending from~$F$ at all future times.
\end{theorem}

\begin{proof}
By Theorems~\ref{thm:finperm} and~\ref{thm:fleqr}, $F \preceq \Sigma.F \preceq \Sigma.R$, and \getHead returns a descendant of~$\Sigma.R$.
\end{proof}

\subsubsection*{Further properties under $f < n/3$}

\begin{lemma}[Checkpoint chain]\label{lem:certchain}
Unless $\geq n/3$ validators are slashable: if~$F$ is finalized at height~$h_F$ and a cert event $(C,\, h,\, b)$ has fired on any chain processed by the node with $h \geq h_F$, then $F$ and~$C$ are compatible, with $F \prec C$ when $h > h_F$.
\end{lemma}

\begin{proof}
If $h_F = 0$ then $F = \genesis$ and compatibility is trivial; assume $h_F \geq 1$.
The cert event fired at the post-state of some processed block~$B$ whose \processSlot call triggered \processHeight with $(\sigma[B].N,\, \sigma[B].h_n) = (C,\, h)$, so $C \preceq B$.
Every $h_n$-update is paired with $\sigma.h \gets \sigma.h + 1$ (Lemma~\ref{lem:slotmono}), hence $\sigma[B].h \geq h + 1 > h_F$; Lemma~\ref{lem:mainsafety} gives $F \preceq B$, and with $C \preceq B$ this yields $F \sim C$.
For $h > h_F$: $\sigma[F].h = h_F < h = \sigma[C].h$ by Remark~\ref{rem:fresh-equiv}, and state-height monotonicity along~$\preceq$ (Lemma~\ref{lem:slotmono}) forces $F \prec C$ on the comparable pair $F \sim C$.
\end{proof}

\begin{lemma}[Upgrade property]\label{lem:upgrade}
Unless $\geq n/3$ validators are slashable: if~$F$ is finalized at height~$h_F$ and some block~$B$ whose post-state has $(F, h_F)$ as its justified checkpoint has been processed by the node, then $R \succeq F$ at all future times.
\end{lemma}

\begin{proof}
Fix the moment when $\onBlock(B)$ offered the descriptor $(h_F,\, 1,\, F,\, 0)$ to \updateForkChoiceRoot, and let $F^0 = \Sigma.F$ at that moment.
Both $F^0$ and $F$ are finalized or genesis (Remark~\ref{rem:fs-invariant}), so Lemma~\ref{lem:finchain} gives $F^0 \preceq F$ or $F \preceq F^0$.

\emph{Case $F \preceq F^0$.}
$\Sigma.R \succeq F^0 \succeq F$ already holds by Theorem~\ref{thm:fleqr}; Theorem~\ref{thm:finperm} preserves $\Sigma.F \succeq F^0$ and Theorem~\ref{thm:fleqr} preserves $\Sigma.R \succeq \Sigma.F$ at all future times, so $\Sigma.R \succeq F$ persists.

\emph{Case $F^0 \preceq F$.}
The filter passes, so after the call $\Sigma.\key_R \geq (h_F,\, 1,\, 0,\, F.\slot,\, \hash(F))$; Lemma~\ref{lem:Rs-key-monotone} preserves this bound thereafter.
Write $h_R \coloneqq \Sigma.\key_R.h$ and $\bit_R \coloneqq \Sigma.\key_R.b$; the bound gives $h_R \geq h_F$.
If $h_R > h_F$, Lemma~\ref{lem:certchain} applied to the cert event $(R,\, h_R,\, \bit_R)$ gives $F \prec R$.
If $h_R = h_F$, the bound forces $\bit_R = 1$ (strictly greater would require $\bit_R > 1$, impossible), so $R$ was justified at height~$h_F$ on some processed chain, giving a \just-quorum~$Q'$ of size $\geq 2n/3$ with target~$R$ at height~$h_F$.
Finality of~$F$ at~$h_F$ gives a quorum~$Q_F$ with $\finalizeHeight = h_F$ and $\finalizeTarget = F$.
By quorum intersection, $|Q' \cap Q_F| \geq n/3$; pick $v \in Q' \cap Q_F$ not slashable.
E1 (Definition~\ref{def:slashing}) applied to $v$'s votes in~$Q', Q_F$ forces $R = F$.
\end{proof}

\begin{theorem}[Local acceptance of finality updates]\label{thm:finlive}
Unless $\geq n/3$ validators are slashable: if a block~$B$ is processed by \onBlock and its post-state has finalized block~$F_B$, then after processing $F \succeq F_B$.
\end{theorem}

\begin{proof}
If $F \succeq F_B$ already, the claim holds.
Otherwise $F$ is genesis or finalized (Remark~\ref{rem:fs-invariant}), so Lemma~\ref{lem:finchain} gives $F \sim F_B$ and hence $F \prec F_B$.
$F_B$ being finalized in~$B$'s post-state requires some ancestor~$B' \preceq B$ to have post-state $(J_{B'},\, h_j^{B'}) = (F_B,\, h_{F_B})$; $B'$ was processed before~$B$ (parent-first), so Lemma~\ref{lem:upgrade} gives $R \succeq F_B$, and the \updateFinalized guard passes.
\end{proof}

\begin{theorem}[Lock-in]\label{thm:lockin}
Unless $\geq n/3$ validators are slashable: if block~$F$ is finalized at height~$h$ on any chain, and some block whose post-state has $(F, h)$ as its justified checkpoint has been processed by the node, then $R \succeq F$ at all future times, and \getHead always returns a descendant of~$F$.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:upgrade} gives $R \succeq F$ directly.
\getHead (descending from~$R$) therefore returns a descendant of~$F$.
\end{proof}

\begin{theorem}[Order independence]\label{thm:orderindep}
Unless $\geq n/3$ validators are slashable: the store state after processing a set of blocks depends only on which blocks have been processed, not on their order.
\end{theorem}

\begin{proof}
Each block's post-state is a deterministic function of itself and its parent's post-state, so $\sigma$ depends only on the processed set.

\emph{$F$.} By Lemma~\ref{lem:finchain}, $\{\sigma[B].F : B \in \T\}$ is a chain with maximum $F_{\max}$, attained at some processed~$B^\ast$.
Remark~\ref{rem:fs-invariant} forces $F$ into this chain, so $F \preceq F_{\max}$; once $B^\ast$ has been processed, Theorem~\ref{thm:finlive} gives $F \succeq F_{\max}$.

\emph{$R$.} The multiset~$D$ of descriptors offered to \updateForkChoiceRoot is a deterministic function of the processed set: each block contributes its post-state's two descriptors exactly once.
Write $K_{\max} \coloneqq (h_{F_{\max}},\, 1,\, 0,\, F_{\max}.\slot,\, \hash(F_{\max}))$.
For any $(h, b, C, s_n) \in D$ with $h \geq h_{F_{\max}}$, Lemma~\ref{lem:certchain} gives $C \succeq F_{\max}$ (strict when $h > h_{F_{\max}}$); at $h = h_{F_{\max}}$, Remark~\ref{rem:fs-invariant} with $F_{\max}$ justified gives $\Sigma.\key_R \geq K_{\max}$, forcing any competitive descriptor to have bit~$1$, hence a \just-quorum~$Q_C$ of size $\geq 2n/3$ targeting~$C$ at height~$h_{F_{\max}}$, while finality of~$F_{\max}$ gives a quorum~$Q_F$ with $\finalizeHeight = h_{F_{\max}}$ and $\finalizeTarget = F_{\max}$; quorum intersection and E1 (as in Lemma~\ref{lem:upgrade}) force $C = F_{\max}$.
Descriptors with $h < h_{F_{\max}}$ have key strictly below $K_{\max}$ on the leading coordinate, so once $\key_R \geq K_{\max}$ holds they are dominated.
Hence $R$'s final value is the lex-max over $\{(h, b, C, s_n) \in D : h \geq h_{F_{\max}}\}$, a deterministic function of the processed set.
\end{proof}

\begin{corollary}[Cross-node agreement]\label{cor:agreement}
Unless $\geq n/3$ validators are slashable: two nodes that have processed the same set of blocks have the same $(\sigma,\, \T,\, F,\, R,\, \key_R)$.
\end{corollary}

\begin{proof}
Immediate from Theorem~\ref{thm:orderindep}.
\end{proof}

\section{Inactivity leak}

This section extends the per-chain state machine of Figure~\ref{alg:state-machine} with a penalty counter and a new transition function \processInactivityPenalties, invoked each slot by an extended \processSlot (Figure~\ref{alg:leak-processslot}); \processBlock, \processHeight, and \processVote are unchanged.
Write $X = \getHead(\Sigma)$ for the \emph{canonical chain}.

Two results follow.
\emph{Tightness} (Theorem~\ref{thm:tightness}) is purely state-logic: on any fixed chain, the total penalty increment over a non-finality period is at least $(T-1) \cdot n/3$, with no hypothesis on synchrony, honesty, or the slashable set.
\emph{Fairness} (Theorem~\ref{thm:fairness}) is local to a single honest~$i$: the vote~$i$ produces per the honest rule (Definition~\ref{def:voting-rule}), once included on any extension of~$i$'s canonical chain still at the same state-height, exempts~$i$ from Layer~1 penalties at that state-height.
Layer-2 exemption is out of scope: it depends on quorum formation on a common target, not on any local property of~$i$.

\subsection{Penalty units}

Augment the per-chain state tuple with one non-negative counter $\sigma.\penalties[1..n]$, initialized to~$0$ at genesis; no other field is added or modified.
Figure~\ref{alg:leak-processslot} replaces \processSlot from Figure~\ref{alg:state-machine} with a version that snapshots the pre-advance state and invokes \processInactivityPenalties after \processHeight.
The store of Section~\ref{sec:store} invokes this extended \processSlot in place of the base one.

\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]

\Function{\processSlot}{$\sigma$} \Comment{extended: tracks penalties}
  \State $\sigma.s \gets \sigma.s + 1$
  \State $\sigma_{\mathrm{pre}} \gets \sigma$ \Comment{snapshot of pre-advance state}
  \State $\sigma \gets \processHeight(\sigma)$
  \State $\sigma \gets \processInactivityPenalties(\sigma_{\mathrm{pre}},\; \sigma)$
  \State \Return $\sigma$

\Function{\processInactivityPenalties}{$\sigma_{\mathrm{pre}},\; \sigma_{\mathrm{post}}$}
  \If{$\sigma_{\mathrm{pre}}.h = \sigma_{\mathrm{post}}.h$} \Comment{Layer 1: stall}
    \For{each $i \in \{1, \ldots, n\}$}
      \If{$\sigma_{\mathrm{post}}.\ntargets[i] = \bot$}
        \State $\sigma_{\mathrm{post}}.\penalties[i] \gets \sigma_{\mathrm{post}}.\penalties[i] + 1$
      \EndIf
    \EndFor
  \EndIf
  \If{$\sigma_{\mathrm{pre}}.J = \sigma_{\mathrm{post}}.J$} \Comment{Layer 2 target: justification did \emph{not} advance}
    \State $T^* \gets \arg\max_{T' \in \{\sigma_{\mathrm{pre}}.\jtargets[i] \,:\, \sigma_{\mathrm{pre}}.\jtargets[i] \neq \bot\}} |\{i : \sigma_{\mathrm{pre}}.\jtargets[i] = T'\}|$ \Comment{$T^* \gets \bot$ if the set is empty; ties broken deterministically}

      \If{$\sigma_{\mathrm{pre}}.\jtargets[i] \neq T^*$ \textbf{or} $\sigma_{\mathrm{pre}}.\jtargets[i] = \bot$}
        \State $\sigma_{\mathrm{post}}.\penalties[i] \gets \sigma_{\mathrm{post}}.\penalties[i] + 1$
      \EndIf
    \EndFor
  \EndIf
  \If{$\sigma_{\mathrm{pre}}.F = \sigma_{\mathrm{post}}.F$ \textbf{and} $\sigma_{\mathrm{pre}}.F \prec \sigma_{\mathrm{pre}}.J$} \Comment{Layer 2 finalize: $F$ did not advance, pending finality}
    \For{each $i \in \{1, \ldots, n\}$}
      \If{$i \notin \sigma_{\mathrm{pre}}.P$}
        \State $\sigma_{\mathrm{post}}.\penalties[i] \gets \sigma_{\mathrm{post}}.\penalties[i] + 1$
      \EndIf
    \EndFor
  \EndIf
  \State \Return $\sigma_{\mathrm{post}}$
\EndFunction

\end{algorithmic}
\end{boxedminipage}
\end{center}
\end{figure}

\paragraph{Reading Figure~\ref{alg:leak-processslot}.}
\processInactivityPenalties checks three independent guards; any subset may fire at a given slot.

\emph{Layer~1 (stall).} Fires when the slot did not advance height.
Every~$i$ with $\sigma_{\mathrm{post}}.\ntargets[i] = \bot$ is bumped, i.e.\ every validator that failed to cast a fresh vote at the current height ($\ntargets$ is not reset at a stall).

\emph{Layer~2 target.} Fires when the justification branch of \processHeight did not fire (else $J$ would strictly advance by Lemma~\ref{lem:slotmono}).
$T^*$ is the plurality target across $\sigma_{\mathrm{pre}}.\jtargets$ (deterministic tiebreak), or $\bot$ if no \just votes have been registered; the second disjunct of the guard bumps non-voters when $T^* = \bot$ (in which case all~$n$ validators are bumped).
Since no $2n/3$-quorum on a single target formed, the plurality count is $< 2n/3$, so $|\{i : \sigma_{\mathrm{pre}}.\jtargets[i] \neq T^* \vee \sigma_{\mathrm{pre}}.\jtargets[i] = \bot\}| > n/3$.

\emph{Layer~2 finalize.} Fires when a pending-finality gap is open at the start of the slot ($\sigma_{\mathrm{pre}}.F \prec \sigma_{\mathrm{pre}}.J$) and $F$ did not advance; every $i \notin \sigma_{\mathrm{pre}}.P$ is bumped.
The finality branch of \processHeight, when it fires, sets $\sigma_{\mathrm{post}}.F = \sigma_{\mathrm{pre}}.J \succ \sigma_{\mathrm{pre}}.F$, so this guard is skipped automatically.

\paragraph{Canonical-freshness.}
A vote~$v$ processed at the current slot on canonical chain~$X$ at state-height $h_c = \sigma_{\mathrm{post}}.h$ bumps $\sigma.\ntargets[v.\validator]$ iff $v$ is \emph{canonical-fresh}:
\[
  v.\height = h_c \;\wedge\; v.\target \preceq X \;\wedge\; v.\target.\slot \geq \sigma_{\mathrm{post}}.s_h,
\]
equivalently $v.\target \preceq X$ with $\sigma[v.\target].h = h_c$ (Remark~\ref{rem:fresh-equiv}).

\subsection{Leak tightness}

\begin{definition}[Non-finality period]\label{def:non-finality-period}
Fix a chain and write $\sigma_{\mathrm{pre}}^{(t)}, \sigma_{\mathrm{post}}^{(t)}$ for its pre/post-\processHeight snapshots at slot~$k$.
A \emph{non-finality period} is a maximal interval $[t_1, t_T]$ of consecutive slots on which $F$ is stuck, i.e., $\sigma_{\mathrm{post}}^{(t)}.F = \sigma_{\mathrm{pre}}^{(t)}.F$ throughout.
Each such slot falls into exactly one category:
\begin{itemize}
  \item \emph{pending-stuck}: $\sigma_{\mathrm{pre}}^{(t)}.F \prec \sigma_{\mathrm{pre}}^{(t)}.J$ (pending-finality gap open at the start);
  \item \emph{stable-stuck}: $\sigma_{\mathrm{pre}}^{(t)}.F = \sigma_{\mathrm{pre}}^{(t)}.J$ and $J$ does not advance at~$t$;
  \item \emph{transition}: $\sigma_{\mathrm{pre}}^{(t)}.F = \sigma_{\mathrm{pre}}^{(t)}.J$ and the justification branch fires at~$t$ (opening a gap by slot's end).
\end{itemize}
\end{definition}

\begin{theorem}[Non-finality period penalty bound]\label{thm:tightness}
Over any non-finality period of length $T \geq 1$ on any chain, the total penalty increment across all validators assessed by \processInactivityPenalties on that chain is at least $(T - 1) \cdot n/3$.
\end{theorem}

\begin{proof}
Fix a non-finality period $[t_1, t_T]$.
We show (i) at most one slot is transition, and (ii) every pending-stuck or stable-stuck slot contributes $> n/3$ to the total penalty increment.

\emph{(i) At most one transition slot.}
After a transition at~$t$, $\sigma_{\mathrm{post}}^{(t)}.J \succ \sigma_{\mathrm{pre}}^{(t)}.F$; since $F$ does not advance in the period and $J$ is monotone across slots, $\sigma_{\mathrm{pre}}^{(t')}.F \prec \sigma_{\mathrm{pre}}^{(t')}.J$ at every later $t' > t$, making every such $t'$ pending-stuck.

\emph{(ii) Per-slot charge $> n/3$.}

\emph{Pending-stuck.} The Layer~2 finalize guard fires at~$t$.
The finality branch of \processHeight did not fire (else $F$ would advance), so $|\sigma_{\mathrm{pre}}^{(t)}.P| < 2n/3$ and $|\{i : i \notin \sigma_{\mathrm{pre}}^{(t)}.P\}| > n/3$.

\emph{Stable-stuck.} The Layer~2 target guard fires at~$t$.
The justification branch did not fire, so the plurality count on $\sigma_{\mathrm{pre}}^{(t)}.\jtargets$ is $< 2n/3$ (counting $T^* = \bot$ as~$0$); hence the bumped set has size $> n/3$.

At most one slot is transition, so at least $T - 1$ slots are pending-stuck or stable-stuck; each contributes $> n/3$ by~(ii), and Layer~1 adds only non-negative charges.
Summing gives total increment $\geq (T-1) \cdot n/3$.
\end{proof}

\subsection{Honest voting rule}

We state the honest rule explicitly, as it underpins the leak fairness proof.

\begin{definition}[Honest voting rule]\label{def:voting-rule}
Let $V$ be the set of votes $i$ has previously cast, and write $\currentSlot$ for the node's protocol time (a slot counter shared across chains).
At each voting opportunity, $i$ constructs its vote by $\constructVote(V,\, \Sigma)$, defined in Figure~\ref{alg:voting-rule}.
The first-vote check in the (R1) branch ensures at most one \just vote per height~$h_c$, and the target-selection clause preserves E1-safety under view shifts.
\end{definition}

\begin{figure}[H]

\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]

\Function{\constructVote}{$V,\; \Sigma$}
  \State $X \gets \getHead(\Sigma)$
  \State $h_c \gets \sigma[X].h$; \;\; $h_j \gets \sigma[X].h_j$; \;\; $J \gets \sigma[X].J$
  \vspace{3pt}
  \State \textbf{-- Target selection --}
  \If{$\exists\, v \in V$ with $v.\finalizeHeight = h_c$ and $v.\finalizeTarget \neq \bot$}
    \State $T_c \gets v.\finalizeTarget$ \Comment{$i$ is locked at $h_c$}
  \Else
    \State $T_c \gets$ any ancestor of $X$ with $\sigma[T].h = h_c$ \label{line:unlocked-target} \Comment{e.g.~$X$ itself}
  \EndIf
  \vspace{3pt}
  \State \textbf{-- Vote kind and finalize fields --}
  \If{$\not \exists v \in V$ s.t. $v.\height = h_c$} \Comment{first vote at $h_c$: cast \just (R1)}
    \If{every $v \in V$ with $v.\height = h_j$ has $v.\target = J$}
      \State $(f_h,\, f_t) \gets (h_j,\, J)$
    \Else
      \State $(f_h,\, f_t) \gets (\bot,\, \bot)$
    \EndIf
    \State \Return $(i,\; h_c,\; \currentSlot,\; T_c,\; \just,\; f_h,\; f_t)$
  \Else \Comment{already voted at $h_c$: cast \notar (R2)}
    \State \Return $(i,\; h_c,\; \currentSlot,\; T_c,\; \notar,\; \bot,\; \bot)$
  \EndIf
\EndFunction

\end{algorithmic}
\end{boxedminipage}
\end{center}
\end{figure}

Only the (R1) branch produces a \just vote and, gated by the first-vote check, fires at most once per~$h_c$, ensuring per-height \just-uniqueness.
The (R2) branch is the \emph{resubmission} fallback: fired at every subsequent voting opportunity while canonical stays at~$h_c$, so when the (R1) \just quorum did not form on canonical (heterogeneous targets across views), the \notar vote's prefix counting may still notarize a common ancestor.

Checking \emph{every} prior vote of~$i$ at~$h_j$ (not just the (R1) one) is required because a (R2) at~$h_j$ may have chosen a target differing from the (R1) target if~$i$'s view shifted; committing $f_t = J$ at height~$h_j$ requires (by E1) that every $h_j$-vote of~$i$ have target~$J$, so in case of a split the conditional fires $(\bot, \bot)$: $i$ misses the finalization at~$h_j$ but remains E1-safe.

\begin{lemma}[Honest E1 safety]\label{lem:honest-e1-safety}
If validator~$i$ follows Definition~\ref{def:voting-rule}, then $i$ never produces a vote-pair $(v_1, v_2)$ with $v_2.\finalizeTarget \neq \bot$, $v_1.\height = v_2.\finalizeHeight$, and $v_1.\target \neq v_2.\finalizeTarget$.
\end{lemma}

\begin{proof}
Write $h_\ast \coloneqq v_2.\finalizeHeight$ and $T_\ast \coloneqq v_2.\finalizeTarget \neq \bot$.
Only (R1) produces finalize-bearing votes, so $v_2$ is an (R1) vote with $(f_h, f_t) = (h_j, J) = (h_\ast, T_\ast)$ read from the canonical state at~$v_2$'s firing.

\emph{Case~A ($v_1$ cast before~$v_2$).}
The (R1) finalize-field conditional at~$v_2$ attached $(h_\ast, T_\ast)$ only because every prior vote of~$i$ at height~$h_\ast$ had target~$T_\ast$; in particular $v_1.\target = T_\ast$.

\emph{Case~B ($v_1$ cast after~$v_2$).}
At $v_1$'s firing, the voting height is $h_c = v_1.\height = h_\ast$, and~$i$ has the prior vote~$v_2$ with $\finalizeHeight = h_\ast = h_c$ and $\finalizeTarget = T_\ast \neq \bot$: $i$ is \emph{locked at $h_c$}, and the target-selection clause of Definition~\ref{def:voting-rule} forces $T_c = T_\ast$ regardless of~$i$'s canonical view at that moment.
Hence $v_1.\target = T_c = T_\ast$.

\emph{Case~C ($v_1 = v_2$).}
Vacuous: $v_2.\height = h_c \neq h_\ast = v_2.\finalizeHeight$ (every height advance sets $h_j \gets h$ before $h \gets h + 1$, so $h_j < h$ in any post-advance state).
\end{proof}

\subsection{Leak fairness}

Fairness is local to a single honest validator~$i$: we fix~$i$, its local store~$\Sigma$, and the canonical chain $X = \getHead(\Sigma)$.
The claim is that the vote~$v$ produced by Definition~\ref{def:voting-rule}, once included on any extension of~$X$ still at the same state-height, exempts~$i$ from Layer~1 penalties at that state-height.
Height advances reset $\jtargets$ and $\ntargets$ (Figure~\ref{alg:state-machine}), so Layer-2 state at the next height is independent of~$i$'s prior votes and no ``lock'' issue propagates.

The proof proceeds via four supporting lemmas: per-height justification uniqueness (Lemma~\ref{lem:just-unique-height}), root-height exclusivity of the notarize-only bit (Lemma~\ref{lem:root-h-notarize-only}), the absence of finalize votes without a matching justification cert event (Lemma~\ref{lem:no-finalize-without-just}), and a canonical-freshness statement for~$i$'s vote (Lemma~\ref{lem:canonical-fresh-honest-vote}).

\begin{lemma}[Per-height justification uniqueness]\label{lem:just-unique-height}
If $f < n/3$ and all honest validators follow Definition~\ref{def:voting-rule}: for any two blocks $C_1, C_2$ justified at the same height~$h$ on any chains processed by the node, $C_1 = C_2$.
\end{lemma}

\begin{proof}
Each justification at height~$h$ on a chain yields a \just-quorum $Q_k$ ($k \in \{1, 2\}$) of size $\geq 2n/3$ signing a \just vote at height~$h$ for $C_k$; quorum intersection gives $|Q_1 \cap Q_2| \geq n/3$.
Since $f < n/3$, some $v \in Q_1 \cap Q_2$ is honest; by Definition~\ref{def:voting-rule}, only (R1) emits a \just vote and it fires at most once per height, so~$v$'s single \just vote at~$h$ has both $C_1$ and $C_2$ as target, forcing $C_1 = C_2$.
\end{proof}

\begin{lemma}[Root-height exclusivity of the notarize-only bit]\label{lem:root-h-notarize-only}
Unless $\geq n/3$ validators are slashable: if $\Sigma.\key_R.b = 0$, then no justification at height $\Sigma.\key_R.h$ has fired on any processed chain.
\end{lemma}

\begin{proof}
Write $h_R \coloneqq \Sigma.\key_R.h$ and suppose some justification at~$h_R$ fired on a processed chain, producing $J_\ast$ justified at~$h_R$.
\onBlock offered the descriptor $(h_R,\, 1,\, J_\ast,\, 0)$ to \updateForkChoiceRoot at that time, with key $(h_R,\, 1,\, 0,\, J_\ast.\slot,\, \hash(J_\ast))$.
Let $F^0 \coloneqq \Sigma.F$ at that moment; $F^0$ is finalized or genesis (Remark~\ref{rem:fs-invariant}) at some height $h_{F^0}$.
When $F^0$ was set, the cert event $(F^0,\, h_{F^0},\, 1,\, 0)$ had already raised the cache height to $\geq h_{F^0}$ (the \updateFinalized guard $F_B \preceq R$ ensures the cache covered it), and Lemma~\ref{lem:Rs-key-monotone} preserves $h_{F^0} \leq h_R$.
We show $F^0 \preceq J_\ast$, so the filter in \updateForkChoiceRoot passes.
If $h_R > h_{F^0}$, Lemma~\ref{lem:certchain} applied to $F^0$ and the cert event at $J_\ast$ gives $F^0 \prec J_\ast$.
If $h_R = h_{F^0}$, finality of $F^0$ at~$h_R$ gives a finalize-quorum $Q_F$ with $\finalizeHeight = h_R$ and $\finalizeTarget = F^0$, and $J_\ast$ being justified at~$h_R$ gives a \just-quorum~$Q'$ targeting~$J_\ast$ at~$h_R$; quorum intersection and E1 (as in Lemma~\ref{lem:upgrade}) force $F^0 = J_\ast$.
Either way $F^0 \preceq J_\ast$, so \updateForkChoiceRoot raises the cache to $\Sigma.\key_R \geq (h_R,\, 1,\, 0,\, J_\ast.\slot,\, \hash(J_\ast))$ at that moment; Lemma~\ref{lem:Rs-key-monotone} preserves this bound.
At the present moment $\Sigma.\key_R.h = h_R$, so lex-comparison with the first coordinate fixed forces $\Sigma.\key_R.b \geq 1$, contradicting $\key_R.b = 0$.
\end{proof}

\begin{lemma}[Finalize vote implies justification]\label{lem:no-finalize-without-just}
If~$i$ has cast a vote~$v$ with $v.\finalizeHeight = h'$ and $v.\finalizeTarget = T_f \neq \bot$, then~$T_f$ was justified at height~$h'$ on some processed chain by the time~$v$ was cast.
\end{lemma}

\begin{proof}
Only (R1) produces $\finalizeHeight \neq \bot$, so~$v$ came from (R1) at an earlier moment with~$i$'s local store~$\Sigma_0$; by the (R1) finalize-field conditional, $(f_h, f_t) = (h_j, J) = (h', T_f)$ read from~$i$'s then-canonical $X_0 = \getHead(\Sigma_0)$, so $\sigma[X_0].h_j = h'$ and $\sigma[X_0].J = T_f$.
A state has $(h_j, J) = (h', T_f)$ only after the justification branch of \processHeight fired on that chain with target~$T_f$ at pre-state height~$h'$, i.e.\ some ancestor $B' \preceq X_0$ triggered justification of~$T_f$ at height~$h'$.
\end{proof}

\begin{lemma}[Canonical-fresh honest vote]\label{lem:canonical-fresh-honest-vote}
Unless $\geq n/3$ validators are slashable, and assuming $f < n/3$ and all honest validators follow Definition~\ref{def:voting-rule}: let~$i$ be honest with local store~$\Sigma$, $X = \getHead(\Sigma)$, and $h_c = \sigma[X].h$.
Let~$v$ be the vote~$i$ produces at the current slot.
Then $v.\target \preceq X$ with $\sigma[v.\target].h = h_c$.
In particular, $v$ is canonical-fresh on every extension $X' \succeq X$ with $\sigma[X'].h = h_c$.
\end{lemma}

\begin{proof}
Write $h_R \coloneqq \Sigma.\key_R.h$ and $\bit_R \coloneqq \Sigma.\key_R.b$.
$X$ descends from~$R$ (Figure~\ref{alg:store}), so $R \preceq X$; the cached cert event fired at height~$h_R$ on a processed chain with $\sigma[R].h = h_R$ (Remark~\ref{rem:fresh-equiv}, genesis trivial).
State-height monotonicity (Lemma~\ref{lem:slotmono}) gives $h_c \geq h_R$.
Every state-height advance on~$X$ fires a cert event whose descriptor is offered to \updateForkChoiceRoot, raising $\key_R.h$ if it exceeds the cached value; hence no advance from $h_R + 1$ to $h_R + 2$ has fired on~$X$, so $h_c \in \{h_R,\, h_R + 1\}$.
By Definition~\ref{def:voting-rule}, $v.\height = h_c$ and $v.\target = T_c$, where $T_c = T \preceq X$ with $\sigma[T].h = h_c$ in the unlocked case, and $T_c = L$ where~$L$ is a lock target from a prior (R1) vote of~$i$ with $\finalizeHeight = h_c$ in the locked case.

\emph{Case $h_c = h_R + 1$.}
A lock at~$h_c$ would, by Lemma~\ref{lem:no-finalize-without-just}, require some~$J_\ast$ justified at~$h_c$ on a processed chain.
The filter-passing argument of Lemma~\ref{lem:root-h-notarize-only} applies (with $h_R$ there replaced by~$h_c$, using $h_{F^0} \leq h_R < h_c$), so \updateForkChoiceRoot raised the cache to $\Sigma.\key_R.h \geq h_c = h_R + 1$ and Lemma~\ref{lem:Rs-key-monotone} preserves this, contradicting $\key_R.h = h_R$.
So~$i$ is unlocked at~$h_c$ and $T_c = T \preceq X$ with $\sigma[T].h = h_c$.

\emph{Case $h_c = h_R$ with $\bit_R = 0$.}
By Lemma~\ref{lem:root-h-notarize-only}, no justification at~$h_c$ has fired on any processed chain; by Lemma~\ref{lem:no-finalize-without-just},~$i$ has not cast a finalize-bearing vote at~$h_c$, so~$i$ is unlocked and $T_c = T \preceq X$ with $\sigma[T].h = h_c$.

\emph{Case $h_c = h_R$ with $\bit_R = 1$.}
If~$i$ is unlocked, $T_c = T \preceq X$ directly.
Otherwise~$i$ is locked at~$h_c$ with target~$L$; Lemma~\ref{lem:no-finalize-without-just} gives $L$ justified at~$h_c$ on some processed chain, and $\bit_R = 1$ gives $R$ justified at~$h_R = h_c$ on some processed chain.
Lemma~\ref{lem:just-unique-height} then gives $L = R \preceq X$, with $\sigma[T_c].h = \sigma[R].h = h_c$.
\end{proof}

\begin{theorem}[Layer~1 leak fairness]\label{thm:fairness}
Unless $\geq n/3$ validators are slashable, and assuming $f < n/3$ and all honest validators follow Definition~\ref{def:voting-rule}: let~$i$ be an honest validator, $\Sigma$ its local store, $X = \getHead(\Sigma)$ its canonical chain, and $h_c = \Sigma.\sigma[X].h$ the current canonical state-height.
Let $v$ be the vote~$i$ produces at the current slot.
Let $\Sigma'$ be any store obtained by processing additional blocks and votes on top of~$\Sigma$ such that $X' = \getHead(\Sigma')$ is an extension of~$X$ with $\Sigma'.\sigma[X'].h = h_c$, and suppose~$v$ is included on~$X'$.
Then for every slot processed on~$X'$ from that point onward while canonical state-height stays at~$h_c$, the Layer~1 increment to $\sigma.\penalties[i]$ on canonical is zero.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:canonical-fresh-honest-vote} gives $T_c \coloneqq v.\target \preceq X \preceq X'$ with $\sigma[T_c].h = h_c$.
The hypothesis $\Sigma'.\sigma[X'].h = h_c$ with Lemma~\ref{lem:slotmono} gives $\Sigma'.\sigma[X'].s_h = \Sigma.\sigma[X].s_h$; Remark~\ref{rem:fresh-equiv} then gives $T_c.\slot \geq \Sigma'.\sigma[X'].s_h$.
So~$v$ is canonical-fresh on~$X'$, and processing~$v$ sets $\ntargets[i] \neq \bot$.
$\ntargets$ resets only at a height advance, which does not fire while state-height stays at~$h_c$, so $\ntargets[i] \neq \bot$ persists and Layer~1 does not bump~$i$.
\end{proof}

The $f < n/3$ hypothesis is used only in Lemma~\ref{lem:canonical-fresh-honest-vote} via Lemma~\ref{lem:just-unique-height}, to collapse the lock target~$L$ onto $R$ in the $h_c = h_R$, $\bit_R = 1$ case.

This section collects the revisions and additions proposed during review.

\subsection{Store root via a single argmax}\label{rev:sec:store-root}

\paragraph{Motivation.}
The revision to \computeRoot is motivated by the development in \Cref{rev:sec:leak-fairness} and \Cref{sec:healing}, specifically \Cref{rev:lem:root-height-cj-exclusive} (``root-height exclusivity''), which states that if $R \in C_N \setminus C_J$ then no $(B, h, s) \in C_J$ has $h = R.h$.
This lemma fails under the iterative walk: if $(A, h, \cdot) \in C_J$ and $(B, h, \cdot) \in C_N \setminus C_J$ with $A \prec B$ at the same state-height, the walk picks~$A$ at its first step (bit tiebreak), then advances to~$B$ as a strict descendant, producing $R = B \in C_N \setminus C_J$ while $A \in C_J$ sits at the same height --- witnessing the negation.
Replacing the walk by a single argmax resolves this: the argmax stops at~$A$ (the $C_J$-bit breaks the tie in $A$'s favour), so $R = A \in C_J$ and the ``$R \in C_N \setminus C_J$'' hypothesis of \Cref{rev:lem:root-height-cj-exclusive} is vacuous against a same-height $C_J$ competitor.
\Cref{rev:lem:root-height-cj-exclusive} underpins \Cref{rev:lem:canonical-fresh-honest-vote} and hence the healing lemmas; the closely related \Cref{rev:lem:Rs-start-anywhere} and \Cref{rev:cor:no-high-cn} likewise rely on $R$ being the argmax itself rather than a descendant.

\paragraph{Revised \texttt{computeRoot}.}
The iterative greedy walk in Figure~\ref{alg:store} is replaced by a single argmax:
\begin{center}
\begin{minipage}{0.9\textwidth}
\begin{algorithmic}[1]
\Function{\computeRoot}{$\Sigma$} \Comment{Single argmax from $\Sigma.F$}
  \State $(X^*, \_, \_) \gets \arg\max_{(X,h,s) \in \Sigma.C_N,\; X \succeq \Sigma.F} \bigl(h,\; \mathbf{1}[(X,h,s) \in \Sigma.C_J],\; s,\; \hash(X)\bigr)$
  \State \Return $X^*$
\EndFunction
\end{algorithmic}
\end{minipage}
\end{center}

\paragraph{Revised proofs of downstream theorems.}
The following proofs are revised to use the single-argmax form. Only the proofs change; the statements of \Cref{thm:fleqr,lem:upgrade,thm:orderindep} are unchanged.

\emph{Revised proof of \Cref{thm:fleqr}.}
Initially $F = R = \genesis$.
\computeRoot returns the block of a tuple with $X \succeq F$, so $R \succeq F$ holds just after \computeRoot.
\updateFinalized sets $F$ only to values $F_B \preceq R$, preserving the invariant.

\emph{Revised proof of \Cref{lem:upgrade}.}
Since $(F, h_F, F.\slot) \in C_J \subseteq C_N$ and $F \succ \Sigma.F$, it is a candidate for \computeRoot's argmax, so the returned tuple has key $\geq (h_F, 1, F.\slot, \hash(F))$ and $h \geq h_F$.
If $h > h_F$, \Cref{lem:certchain} gives $F \prec X$.
If $h = h_F$, the key lower bound forces the $C_J$-bit to be~$1$, i.e.\ $(X, h, s) \in C_J$; the slot/hash argument of the original proof gives $F \preceq X$.
Either way $R = X \succeq F$.

\emph{Revised proof of \Cref{thm:orderindep}.}
Same argument as the original proof for $\sigma$, $C_J$, $C_N$, and $F$; the final step becomes: $R$ is the argmax over $\{(X,h,s) \in C_N : X \succeq F\}$ on the totally ordered key, hence a deterministic function of $(C_N, C_J, F)$.

\subsection{Additional store-level lemmas}\label{rev:sec:store-lemmas}

\begin{lemma}[Monotonicity of the root height]\label{rev:lem:Rs-height-monotone}
$R.h$ is non-decreasing over time.
\end{lemma}

\begin{proof}
$R$ changes only inside \onBlock, which (i) inserts into $C_N, C_J$, (ii) recomputes $R$, (iii) possibly advances $F$ under the guard $F \preceq R$.
Between two calls, $F^{\text{new}} \preceq R^{\text{old}}$ (either unchanged or just-updated under guard), so the old $R$-tuple remains a valid argmax candidate at the next \computeRoot, giving $R^{\text{new}}.h \geq R^{\text{old}}.h$.
\end{proof}

\begin{lemma}[Root independent of the argmax starting point]\label{rev:lem:Rs-start-anywhere}
Unless $\geq n/3$ validators are slashable, the block~$R$ returned by \computeRoot coincides with the block of the argmax over $\{(X,h,s) \in C_N : X \succeq \genesis\}$.
\end{lemma}

\begin{proof}
Let $R^{\genesis}$ be that block; $\{X \succeq F\} \subseteq \{X \succeq \genesis\}$ gives $R^{\genesis}.h \geq R.h \geq h_{F}$.
If $R^{\genesis}.h > h_{F}$, \Cref{lem:certchain} gives $F \prec R^{\genesis}$, so $R^{\genesis} \succeq F$ and $R^{\genesis} = R$ by uniqueness.
If $R^{\genesis}.h = h_{F}$, the $(F, h_{F}, \cdot) \in C_J$ candidate forces the $C_J$-bit of $R^{\genesis}$ to be~$1$; \Cref{lem:finchain}-style quorum intersection with $F$ finalized gives $R^{\genesis} = F$, hence again $R^{\genesis} = R$.
\end{proof}

\begin{corollary}[No $C_N$ record above the root]\label{rev:cor:no-high-cn}
Unless $\geq n/3$ validators are slashable, every $(X, h, s) \in C_N$ satisfies $h \leq R.h$; in particular the same holds for every $(X, h, s) \in C_J \subseteq C_N$.
\end{corollary}

\begin{proof}
By \Cref{rev:lem:Rs-start-anywhere}, $R$'s tuple is the argmax over all of $C_N$; a $C_N$ tuple at height $> R.h$ would strictly dominate on the leading coordinate, contradicting maximality.
\end{proof}

\begin{corollary}[Processed-block height bound]\label{rev:cor:processed-height-bound}
Unless $\geq n/3$ validators are slashable, every processed block~$B$ satisfies $\sigma[B].h \leq R.h + 1$.
\end{corollary}

\begin{proof}
If $\sigma[B].h \geq 1$, \Cref{lem:heightprog} produces a $C_N$ tuple at height $\sigma[B].h - 1$ from the last advance on~$B$'s chain; by \Cref{rev:cor:no-high-cn}, $\sigma[B].h - 1 \leq R.h$.
\end{proof}

\subsection{Leak fairness}\label{rev:sec:leak-fairness}

\begin{lemma}[Unique justified block per height]\label{rev:lem:unique-just-per-height}
If $f < n/3$ and all honest validators follow \Cref{def:voting-rule}, then for any two blocks $C_1, C_2$ justified at the same height~$h$, $C_1 = C_2$.
\end{lemma}

\begin{proof}
Each justification yields a \just-quorum $Q_k$ ($k \in \{1,2\}$) of size $\geq 2n/3$ signing a \just vote at height~$h$ for $C_k$; quorum intersection gives $|Q_1 \cap Q_2| \geq n/3$.
Since $f < n/3$, some $v \in Q_1 \cap Q_2$ is honest; by \Cref{def:voting-rule}, only (R1) emits a \just vote and it fires at most once per height, so~$v$'s single \just vote at~$h$ has both $C_1$ and $C_2$ as target, forcing $C_1 = C_2$.
\end{proof}

\begin{lemma}[Root-height exclusivity]\label{rev:lem:root-height-cj-exclusive}
Unless $\geq n/3$ slashable: if $(R, R.h, \cdot) \in C_N \setminus C_J$ and $(B, h, s) \in C_J$, then $h \neq R.h$.
\end{lemma}

\begin{proof}
Write $h_R \coloneqq R.h$. Since $(F, h_{F}, F.\slot) \in C_J$ is always an argmax candidate, $R \neq F$, so $R \succ F$ and $h_R \geq h_{F}$.
Suppose $(B, h_R, s) \in C_J$ exists.
If $h_R > h_{F}$: \Cref{lem:certchain} gives $F \prec B$, so $B$ is a candidate whose key $(h_R, 1, \ldots)$ beats $R$'s $(h_R, 0, \ldots)$ on the $C_J$-bit, contradicting maximality.
If $h_R = h_{F}$: $B$ is justified at $h_{F}$ and $F$ is finalized at $h_{F}$, so $B = F$ by \Cref{lem:finchain}; then $(F, h_{F}, F.\slot)$ is itself an argmax candidate and wins on the $C_J$-bit, again contradicting maximality.
\end{proof}

\begin{lemma}[No finalize vote without a $C_J$ record]\label{rev:lem:no-finalize-without-cj}
Let~$i$ be a validator with local store~$\Sigma$ and $h$ a height with $(X, h, s) \notin \Sigma.C_J$ for all $X, s$. Then~$i$ has not cast a vote~$v$ with $v.\finalizeHeight = h$.
\end{lemma}

\begin{proof}
Only (R1) produces $\finalizeHeight \neq \bot$, so such a~$v$ came from (R1) at an earlier moment with store $\Sigma_0$, yielding $\sigma[X_0].h_j = h$ on~$i$'s then-canonical $X_0 = \getHead(\Sigma_0)$.
A state has $h_j = h$ only after justification at~$h$ fired on its chain, so the block~$B \preceq X_0$ where it fired has $\sigma[B].J$ at height~$h$ and $\onBlock(B)$ inserts $(\sigma[B].J, h, \cdot) \in \Sigma_0.C_J$; monotonicity of $C_J$ forces this tuple into $\Sigma.C_J$, contradicting the hypothesis.
\end{proof}

\begin{lemma}[Head state-height range]\label{rev:lem:head-height-range}
Let $(R, R.h, \cdot) \in C_N$. Then $\sigma[\getHead(\Sigma)].h \in \{R.h,\, R.h + 1\}$.
\end{lemma}

\begin{proof}
Write $h_R \coloneqq R.h$ and let $Y = \getHead(\Sigma)$, so $Y \succeq R$.
State-height is non-decreasing along $\preceq$ and $\sigma[R].h = h_R$, so $\sigma[Y].h \geq h_R$.
If $\sigma[Y].h \geq h_R + 2$, \Cref{lem:heightprog} produces a second height advance on the chain from $R$ to~$Y$, inserting a $C_N$ tuple at height $h_R + 1$ descending from $R$; its key beats $R$'s on the leading coordinate, contradicting $R$'s maximality.
\end{proof}

\begin{lemma}[Canonical-fresh honest vote]\label{rev:lem:canonical-fresh-honest-vote}
Assume $f < n/3$ and all honest validators follow \Cref{def:voting-rule}.
Let~$i$ be an honest validator, $\Sigma$ its local store, $H = \getHead(\Sigma)$, and $h_c = \sigma[H].h$.
Let~$v$ be the vote~$i$ produces at the current slot.
Then~$v$ is canonical-fresh on every extension~$H'$ of~$H$ satisfying $\sigma[H'].h = h_c$.
\end{lemma}

\begin{proof}
By \Cref{def:voting-rule}, $v.\height = h_c$ and $v.\target = T_c$, where $T_c$ is either the canonical target $T \preceq H$ or $L \neq \bot$ from a prior locked vote at~$h_c$.
It suffices to show $T_c \preceq H$; split on $h_c$ versus $R.h$.

\emph{Case $h_c > R.h$.}
Any $(Z, h_c, s) \in C_N$ would, via \Cref{lem:certchain}, satisfy $Z \succeq F$, making it an argmax candidate with key $(h_c, \ldots)$ beating $R$'s $(R.h, \ldots)$ --- contradiction.
Hence no $C_N$ tuple at $h_c$ exists, a fortiori none in $C_J$; by \Cref{rev:lem:no-finalize-without-cj} no prior~$i$-vote has $\finalizeHeight = h_c$, so the lock clause does not fire and $T_c = T \preceq H$.

\emph{Case $h_c = R.h$ and $(R, h_c, \cdot) \notin C_J$.}
\Cref{rev:lem:root-height-cj-exclusive} rules out any $C_J$ tuple at~$h_c$; \Cref{rev:lem:no-finalize-without-cj} then gives $T_c = T \preceq H$.

\emph{Case $h_c = R.h$ and $(R, h_c, \cdot) \in C_J$.}
If the lock clause does not fire, $T_c = T \preceq H$.
Otherwise~$i$ is locked at~$h_c$ with lock target $L$, and some prior vote of~$i$ had $\finalizeTarget = L$ justified at~$h_c$; \Cref{rev:lem:unique-just-per-height} and the case hypothesis give $L = R$, and $R \preceq H$ (\Cref{thm:fleqr}), so $T_c = L = R \preceq H$.
\end{proof}

\begin{remark}[Weakening $f < n/3$ under a double-justify slashing rule]\label{rev:rem:canonical-fresh-slashable}
The $f < n/3$ hypothesis in \Cref{rev:lem:canonical-fresh-honest-vote} is needed only by the invocation of \Cref{rev:lem:unique-just-per-height}, whose proof uses \Cref{def:voting-rule}'s behavioural constraint (at most one \just vote per height) to close quorum intersection.
If \Cref{def:slashing} is extended with an additional slashing rule declaring any pair of \just votes at the same height with different targets slashable, then the intersection of two \just-quorums at height~$h$ with different targets forces every shared signer to be slashable; assuming fewer than $n/3$ validators are slashable yields one non-slashable signer, whose single \just vote fixes $C_1 = C_2$.
Under this strengthened slashing definition, $f < n/3$ in the hypotheses of \Cref{rev:lem:unique-just-per-height,rev:lem:canonical-fresh-honest-vote} (and downstream lemmas that inherit it) may be replaced by the weaker ``unless $\geq n/3$ validators are slashable.''
\end{remark}

\section{Healing from asynchrony}\label{sec:healing}

This section builds on the revisions in \Cref{sec:roberto-review}.
Throughout, \Cref{def:voting-rule} and \Cref{alg:store} refer respectively to the original definition and to the algorithm figure with the revised \computeRoot of \Cref{rev:sec:store-root}, and the auxiliary lemmas of \Cref{rev:sec:store-lemmas,rev:sec:leak-fairness} are used without further comment.

\paragraph{Modified \getHead.}
For this section we use a variant of \getHead (\Cref{alg:store}) that takes an additional argument~$\Omega$, an auxiliary state of unspecified nature; the returned descendant of $\Sigma.R$ depends on $\Omega$.
Concretely, the function in \Cref{alg:store} is replaced by:
\begin{center}
\begin{minipage}{0.9\textwidth}
\begin{algorithmic}[1]
\Function{\getHead}{$\Sigma,\; \Omega$}
  \State \Return a block in~$\Sigma.\T$ that descends from~$\Sigma.R$, selected by~$\Omega$
\EndFunction
\end{algorithmic}
\end{minipage}
\end{center}

\paragraph{Virtual-block cert events.}
For this section, the store considers candidate cert events not only from on-chain block processing but also from a \emph{virtual extension} of the canonical chain: using the votes in $\Sigma.V$, the following routine hypothesizes a block that carries the applicable subset on top of the current head and offers the resulting cert-event descriptors to \updateForkChoiceRoot.
The virtual block is never inserted into $\Sigma.\T$ and the hypothetical state is discarded; only $(\Sigma.R,\, \Sigma.\key_R)$ may change, via the standard cert-key guard.
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]
\Function{\updateRootFromVotes}{$\Sigma,\; \Omega$}
  \State $L \gets \getHead(\Sigma,\; \Omega)$
  \State $\sigma' \gets \Sigma.\sigma[L]$ \Comment{working copy, not persisted}
  \State $V' \gets \{v \in \Sigma.V : v.\height = \sigma'.h \land v.\target \preceq L \land v.\target.\slot \geq \sigma'.s_h\}$ \Comment{fresh on $\sigma'$}
  \State let $B_V$ be a virtual block with $\parent = L$, $\slot = \sigma'.s + 1$, and votes~$V'$
  \State $\sigma' \gets \processSlot(\sigma')$ \Comment{advances $\sigma'.s$; runs \processHeight}
  \State $\sigma' \gets \processBlock(\sigma',\; B_V)$
  \State $\sigma' \gets \processHeight(\sigma')$ \Comment{fires any cert event enabled by $V'$}
  \State $\Sigma \gets \updateRootInSlot(\Sigma,\; (\sigma'.h_j,\; 1,\; \sigma'.J,\; \sigma'.s_n))$
  \State $\Sigma \gets \updateRootInSlot(\Sigma,\; (\sigma'.h_n,\; 0,\; \sigma'.N,\; \sigma'.s_n))$
  \State \Return $\Sigma$
\EndFunction
\end{algorithmic}
\end{boxedminipage}
\end{center}
\noindent In particular, whenever received-but-not-yet-on-chain votes would justify or notarize some block with a cert key strictly higher than $\key_R$, $R$ advances accordingly.

\paragraph{Root updates during a slot.}
For this section, invocations of \updateForkChoiceRoot{} during a slot --- i.e., by \onBlock{} of \Cref{alg:store} and by \updateRootFromVotes{} above --- use a stricter guard: the cached root advances only when the incoming cert event strictly improves the $(h, s_n)$ pair of $\Sigma.\key_R$. Candidates tied on both coordinates are \emph{not} adopted. Concretely, every in-slot call to \updateForkChoiceRoot{} is replaced by a call to \updateRootInSlot{} below:
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]
\Function{\updateRootInSlot}{$\Sigma,\; (h,\, b,\, C,\, s_n)$} \Comment{in-slot: strict $(h, s_n)$ guard}
  \State $\newKey \gets (h,\; b,\; s_n,\; C.\slot,\; \hash(C))$
  \If{($h > \Sigma.\key_R.h$) \textbf{or} ($h = \Sigma.\key_R.h$ \textbf{and} $s_n > \Sigma.\key_R.s_n$)}
    \If{$\Sigma.F \preceq C$}
      \State $\Sigma.R \gets C$; \;\; $\Sigma.\key_R \gets \newKey$
    \EndIf
  \EndIf
  \State \Return $\Sigma$
\EndFunction
\end{algorithmic}
\end{boxedminipage}
\end{center}

\paragraph{View-merge processing.}
The view-merge protocol (\Cref{app:view-merge}) delivers to each honest validator the syncer's block tree $\comm{\T}$ and message set $\comm{V}$; no other store components are transferred. Upon receipt at $\vote(k)$, the validator folds the newly-delivered blocks through \onBlock{} and then invokes \updateRootFromVotes{} on the enlarged $\Sigma.V$. Formally:
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]
\Function{\onViewMerge}{$\Sigma,\; \Omega,\; \comm{\T},\; \comm{V}$}
  \For{each $B \in \comm{\T} \setminus \Sigma.\T$ processed in any $\preceq$-respecting order}
    \State $\Sigma \gets \onBlock(\Sigma,\; B)$
  \EndFor
  \State $\Sigma.V \gets \Sigma.V \cup \comm{V}$
  \State $\Sigma \gets \updateRootFromVotes(\Sigma,\; \Omega)$
  \State \Return $\Sigma$
\EndFunction
\end{algorithmic}
\end{boxedminipage}
\end{center}
\noindent Because the other store components --- $\sigma,\, F,\, R,\, \key_R$ --- are pure functions of $(\Sigma.\T,\, \Sigma.V)$ under \Cref{alg:store} together with \updateRootInSlot{} and \updateRootFromVotes{}, honest validators that have merged the same $(\comm{\T},\, \comm{V})$ by time $\vote(k)$ obtain matching $\sigma,\, F,\, R,\, \key_R$ locally.

To state the conditions under which the protocol \emph{heals} after a prolonged period of asynchrony, we enrich the system model with slot-based timing and a per-slot view-merge mechanism.
The view-merge mechanism is detailed in \Cref{app:view-merge}; in this section we abstract it via an assumption sufficient for the healing arguments.


\begin{definition}[Slots and voting time]\label{rev:def:slots}
Physical time is partitioned into an infinite sequence of discrete \emph{slots} indexed by $k \in \mathbb{N}$.
Slot~$k$ has a \emph{start time} $\start(k) \in \mathbb{R}_{\geq 0}$ with $\start(k+1) \geq \start(k) + 2\Delta$, and a \emph{voting time} $\vote(k) \coloneqq \start(k) + \Delta$ at which every honest validator~$i$ invokes \Cref{def:voting-rule} on its local store~$\Sigma_i$ to produce at most one slot-$k$ vote.
\end{definition}

\begin{definition}[Syncer]\label{rev:def:syncer}
A publicly known function $\syncer \colon \mathbb{N} \to \{1, \ldots, n\}$ assigns to each slot~$k$ a validator $\syncer(k)$.
During the portion of slot~$k$ preceding $\vote(k)$, $\syncer(k)$ runs the \emph{view-merge protocol} (\Cref{app:view-merge}).
\end{definition}

\begin{assumption}[View-merge guarantee]\label{rev:ass:view-merge}
If $\syncer(k)$ is honest, then there exists a store $\comm{\Sigma}^{\vote(k)}$ such that every honest validator~$i$ has $\Sigma_i = \comm{\Sigma}^{\vote(k)}$ at time $\vote(k)$.
\end{assumption}

By \Cref{thm:orderindep} applied to $\comm{\Sigma}^{\vote(k)}$, if additionally $f < n/3$ and \getHead is deterministic, the components $(R, F, C_J, C_N)$ are common to every honest validator at $\vote(k)$.

\begin{definition}[Target function]\label{rev:def:getTarget}
$\getTarget(\Sigma, \Omega)$ returns the target~$T_c$ as assigned on line~\ref{line:unlocked-target} of \constructVote{} (\Cref{alg:voting-rule}) applied to~$\Sigma$ and computed with respect to~$\Omega$: an arbitrarily chosen ancestor of $\getHead(\Sigma, \Omega)$ at state-height such that $\sigma[\getHead(\Sigma, \Omega)].h = \sigma[\getTarget(\Sigma, \Omega)].h$.
In particular, $\getTarget(\Sigma, \Omega) \preceq \getHead(\Sigma, \Omega)$.
\end{definition}

\paragraph{Modified \constructVote.}
We assume that each store $\Sigma$ carries an additional field $\Sigma.V$, the set of votes the owning validator has cast so far.
The voting rule is correspondingly rewritten to drop its $V$ parameter and to delegate unlocked target selection to \getTarget{}:
\begin{center}
\begin{boxedminipage}{\textwidth}
\begin{algorithmic}[1]
\Function{\constructVote}{$\Sigma,\; \Omega$}
  \State $X \gets \getTarget(\Sigma,\; \Omega)$
  \State $h_c \gets \sigma[X].h$; \;\; $h_j \gets \sigma[X].h_j$; \;\; $J \gets \sigma[X].J$
  \vspace{3pt}
  \State \textbf{-- Target selection --}
  \If{$\exists\, v \in \Sigma.V$ with $v.\finalizeHeight = h_c$ and $v.\finalizeTarget \neq \bot$}
    \State $T_c \gets v.\finalizeTarget$ \Comment{$i$ is locked at $h_c$}
  \Else
    \State $T_c \gets X$
  \EndIf
  \vspace{3pt}
  \State \textbf{-- Vote kind and finalize fields --}
  \If{$\not \exists v \in \Sigma.V$ s.t. $v.\height = h_c$} \Comment{first vote at $h_c$: cast \just (R1)}
    \If{every $v \in \Sigma.V$ with $v.\height = h_j$ has $v.\target = J$}
      \State $(f_h,\, f_t) \gets (h_j,\, J)$
    \Else
      \State $(f_h,\, f_t) \gets (\bot,\, \bot)$
    \EndIf
    \State \Return $(i,\; h_c,\; \currentSlot,\; T_c,\; \just,\; f_h,\; f_t)$
  \Else \Comment{already voted at $h_c$: cast \notar (R2)}
    \State \Return $(i,\; h_c,\; \currentSlot,\; T_c,\; \notar,\; \bot,\; \bot)$
  \EndIf
\EndFunction
\end{algorithmic}
\end{boxedminipage}
\end{center}

\begin{lemma}[Per-slot notarization uniqueness]\label{rev:lem:notar-unique-per-slot}
Assume $f < n/3$ and all honest validators follow \Cref{def:voting-rule}. If blocks $X$ and $Y$ are each notarized at height~$h$ by a quorum whose contributing votes were all cast at a single slot~$s$, then $X \sim Y$.
\end{lemma}

\begin{proof}
Suppose for contradiction $X \not\sim Y$: $X$ and $Y$ lie on conflicting chains $L_X, L_Y$, and no block at state-height~$h$ lies on both chains (state-height is monotone along~$\preceq$, and $L_X, L_Y$ diverge strictly before any block at state-height~$h$).

Let $Q_X, Q_Y$ denote the slot-$s$ contributing quorums, with $|Q_X|, |Q_Y| \geq 2n/3$, and let $h_X, h_Y$ be their honest cardinalities.
By \Cref{def:voting-rule}, an honest validator casts at most one vote per slot; the target of that vote lies on at most one of $L_X, L_Y$ at state-height~$h$. Hence no honest validator is in both $Q_X$ and $Q_Y$, so
\[
  h_X + h_Y \;\leq\; n - f.
\]

At most $f$ adversary validators exist, each possibly equivocating to contribute once to each chain's slot-$s$ quorum; therefore
\[
  |Q_X| \leq h_X + f \quad\text{and}\quad |Q_Y| \leq h_Y + f.
\]
Combined with $|Q_X|, |Q_Y| \geq 2n/3$,
\[
  h_X \geq 2n/3 - f, \quad h_Y \geq 2n/3 - f, \quad\text{so}\quad h_X + h_Y \geq 4n/3 - 2f.
\]

Comparing the two bounds on $h_X + h_Y$ gives $4n/3 - 2f \leq n - f$, i.e.\ $f \geq n/3$, contradicting $f < n/3$.
\end{proof}

\begin{lemma}[Healing: one-slot advance]\label{rev:lem:healing-one-slot}
Assume:
\begin{enumerate}
  \item $f < n/3$ and all honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item for every pair of honest validators $v, v'$, $\getTarget_v^{\vote(k)} \sim \getTarget_{v'}^{\vote(k)}$;
  \item $\sigma[\getTarget_v^{\vote(k)}].h$ is the same for every honest validator~$v$; denote this common value by $\comm{h}$;
  \item no honest validator has cast any vote at height $\geq \comm{h}$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then at $\vote(k+1)$,
\begin{enumerate}
  \item for any honest validator $v$, $R_v^{\vote(k+1)}.h = \comm{h}$ and $R_v^{\vote(k+1)}$ is either justified or notarized for slot $k$;
  \item for any two honest validators $v, v'$, $R_v^{\vote(k+1)} \sim R_{v'}^{\vote(k+1)}$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+1)$.
  \item for any honest validator $v$, no checkpoint $C$ for height $\comm{h}$ with $C \neq R_v^{\vote(k+1)}$ can ever be justified.
\end{enumerate}
\end{lemma}

\begin{proof}
\emph{Step 1: every honest casts a \just vote at height $\comm{h}$ at $\vote(k)$.}
By hypothesis~(4), $v$'s canonical state-height is $h_c = \sigma[\getTarget_v^{\vote(k)}].h = \comm{h}$.
Hypothesis~(5) rules out any earlier honest vote at height $\geq \comm{h}$, so the (R1) branch of \Cref{def:voting-rule} fires at every honest, producing
\[
  v^{\vote(k)}_i \;=\; \bigl(i,\; \comm{h},\; k,\; T_i,\; \just,\; f_h,\; f_t\bigr),
\]
with $T_i = \getTarget_i^{\vote(k)}$ satisfying $\sigma[T_i].h = \comm{h}$.

\emph{Step 2: all honest targets $T_i$ lie on a single chain~$C$.}
Hypothesis~(3) makes the $T_i$'s pairwise $\preceq$-compatible; combined with hypothesis~(4) (all at state-height $\comm{h}$), they lie on a single chain~$C$, linearly ordered under~$\preceq$.

\emph{Step 3: slot-$k$ honest votes and their target blocks reach every honest by $\vote(k+1)$.}
Hypothesis~(2) together with the slot spacing $\vote(k+1) - \vote(k) \geq 2\Delta$ imply that every vote and every block broadcast by an honest validator at or before $\vote(k)$ is delivered to every honest by $\vote(k+1)$.
In particular each $T_i$ is in $\Sigma_{v_i}^{\vote(k)}.\mathcal{T}$ (since $v_i$ voted for $T_i$), hence in every honest's store by $\vote(k+1)$ via store monotonicity.
The virtual-block cert-events rule of \Cref{sec:healing} lets every honest aggregate those received votes into a virtual block extending a canonical chain on~$C$, offering the resulting cert-event descriptors to \updateForkChoiceRoot.

\emph{Step 4: a same-slot cert event on $C$ at state-height $\comm{h}$ fires in every honest's store.}
Fix an honest node~$v$. Its virtual block on chain~$C$ carries all $\geq 2n/3$ honest slot-$k$ \just votes (received by Step~3), together with any dishonest slot-$k$ votes $v$ has received that are fresh on~$C$.
After \processBlock{}, each honest contributor~$i$ satisfies $\sigma.\ntargets[i] = T_i$ and $\sigma.\nslots[i] = k$ (the slot-$k$ vote is the first at height $\comm{h}$, so the slot-gated update fires) and $\sigma.\jtargets[i] = T_i$ (Step~1 votes are \just).
In the revised \processHeight{} of \Cref{alg:state-machine}, the candidate set $\{(B, s) : B \preceq \sigma.L \wedge |\{i : B \preceq \ntargets[i] \preceq \sigma.L \wedge \nslots[i] = s\}| \geq 2n/3\}$ is non-empty at $s = k$: for any $B \in C$ with $B \preceq T_i$ for every honest~$i$, the $\geq 2n/3$ honest slot-$k$ contributors satisfy the count. The $\arg\max$ by~$\preceq$ selects some $(T^\star_v, k)$ with $T^\star_v \in C$ at state-height $\comm{h}$, firing a bit-$0$ cert event with key $(\comm{h},\, 0,\, k,\, T^\star_v.\slot,\, \hash(T^\star_v))$.

If, additionally, $\geq 2n/3$ contributors share a common $\jtargets$ value~$T$, the justification branch of \processHeight{} also fires and produces a bit-$1$ cert event with key $(\comm{h},\, 1,\, 0,\, T.\slot,\, \hash(T))$ at the same height; in that case $T \in C$ since the witnessing $\geq 2n/3 - f$ honest contributions have $\sigma.\jtargets[i] = T_i = T \in C$.

In every case, every cert event at height $\comm{h}$ that fires in $v$'s view has target on~$C$, since $\geq 2n/3$ contributions at height $\comm{h}$ can only come on a chain receiving honest contributions, which target~$C$ (dishonest alone have $\leq f < n/3$).

\emph{Step 5: $R_v$ advances to a cert event at $\comm{h}$; conclusions~(1) and~(2).}
Before any cert event at height $\geq \comm{h}$ fires in $v$'s store, $\Sigma_v.\key_R.h < \comm{h}$: any cert event at height $h^* \geq \comm{h}$ requires a $\geq 2n/3$ quorum of votes at height $h^*$, and hypothesis~(5) rules out honest contributions at such heights before $\vote(k)$, leaving only dishonest ($\leq f < n/3$) --- insufficient. Hence no cert event at height $\geq \comm{h}$ exists globally before $\vote(k)$, so $\Sigma_v^{\vote(k)}.\key_R.h < \comm{h}$ for every honest~$v$.
In $[\vote(k),\, \vote(k+1))$ no honest vote at height $> \comm{h}$ is cast (Step~1 votes are at $\comm{h}$ exactly; no voting event lies in $(\vote(k),\, \vote(k+1))$), so by the same quorum count no cert event at height $> \comm{h}$ fires; hence by \Cref{lem:Rs-key-monotone} the bound $\Sigma_v.\key_R.h < \comm{h}$ persists until the first cert event at height $\comm{h}$ is processed in $v$'s view.
That cert event's key has leading coordinate $\comm{h} > \Sigma_v.\key_R.h$, so \updateForkChoiceRoot{}'s guard passes; $R_v$ advances to the target of the lex-maximum cert event at height $\comm{h}$ processed by~$v$ in $[\vote(k),\, \vote(k+1)]$.

By Step~4, every such cert event is either the slot-$k$ bit-$0$ notarization at some $T^\star_v \in C$ (key $(\comm{h}, 0, k, \ldots)$) or, when honest $\jtargets$ coincide, the bit-$1$ justification at the common $T \in C$ (key $(\comm{h}, 1, 0, \ldots)$). Bit-$1$ strictly dominates bit-$0$ at the same height, so:
\begin{itemize}
  \item If a justification cert at $\comm{h}$ has fired in $v$'s view by $\vote(k+1)$, $R_v^{\vote(k+1)}$ is that justified block (in particular justified at height $\comm{h}$).
  \item Otherwise $R_v^{\vote(k+1)} = T^\star_v$, which is bit-$0$ notarized at height $\comm{h}$ with notarization slot~$k$.
\end{itemize}
No cert event at height $> \comm{h}$ fires in the interval, so $R_v^{\vote(k+1)}.h = \comm{h}$ and $R_v^{\vote(k+1)}$ is either justified or notarized for slot~$k$, proving conclusion~(1).
Every cert event at height $\comm{h}$ in $v$'s view targets a block on~$C$ (Step~4); since $C$ is linearly ordered under~$\preceq$, any two $R_v^{\vote(k+1)}, R_{v'}^{\vote(k+1)}$ are $\preceq$-comparable, proving conclusion~(2).

\emph{Step 6: no honest \just vote at height $> R.h$ before $\vote(k+1)$.}
Hypothesis~(5) rules out honest votes at height $\geq \comm{h}$ before $\vote(k)$; at $\vote(k)$ every honest casts at height $\comm{h} = R_v^{\vote(k+1)}.h$ exactly; no voting event occurs in $(\vote(k),\, \vote(k+1))$.
Hence no honest has cast a \just vote at height $> R.h$ strictly before $\vote(k+1)$, proving conclusion~(3).

\emph{Step 7: no other checkpoint at height $\comm{h}$ can ever be justified; conclusion~(4).}
Any justification at height $\comm{h}$ requires a $\geq 2n/3$ \just-quorum at height $\comm{h}$ with a common target. Honest \just votes at height $\comm{h}$ exist only as the slot-$k$ votes of Step~1 (\Cref{def:voting-rule} fires (R1) at most once per height, hypothesis~(5) excludes pre-$\vote(k)$ honest \just at $\comm{h}$, and there is no honest voting event in $(\vote(k),\, \vote(k+1))$ or beyond at this height). Honest contributions are therefore the fixed multiset $\{T_i : i \text{ honest}\} \subseteq C$, and any future justification quorum at a target~$C'$ requires $\geq 2n/3 - f$ honest with $T_i = C'$.

Suppose for contradiction $C' \neq R_v^{\vote(k+1)}$ at height $\comm{h}$ becomes justified at some time. Then $h^*_{C'} \coloneqq |\{i \text{ honest} : T_i = C'\}| \geq 2n/3 - f$.
\begin{itemize}
  \item If $R_v^{\vote(k+1)}$ is itself justified (at some target~$T \in C$ by Step~5): $h^*_T \geq 2n/3 - f$ as well, so $h^*_T + h^*_{C'} \geq 4n/3 - 2f$. Each honest casts a single \just at height $\comm{h}$, so $h^*_T + h^*_{C'} \leq n - f$, giving $4n/3 - 2f \leq n - f$, i.e.\ $f \geq n/3$, contradicting $f < n/3$.
  \item If $R_v^{\vote(k+1)}$ is bit-$0$ notarized (no justification fired at $\comm{h}$ in $v$'s view): then $h^*_X < 2n/3 - f$ for every $X$ at height $\comm{h}$ (else the justification branch would have fired at slot~$k$ in $v$'s view). In particular $h^*_{C'} < 2n/3 - f$, contradicting the assumed quorum.
\end{itemize}
Either way, no $C' \neq R_v^{\vote(k+1)}$ at height $\comm{h}$ can ever be justified, proving conclusion~(4).
\end{proof}


\begin{lemma}[Two-slot healing from a justified common root]\label{rev:lem:healing-from-justified}
Assume:
\begin{enumerate}
  \item $f < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item for every honest validator~$v$, $R_v^{\vote(k)}$ is the same justified block; denote it by $\comm{R}^{\vote(k)}$;
  \item $\getTarget_v^{\vote(k)} \sim \getTarget_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getTarget_v^{\vote(k)}].h = \comm{R}^{\vote(k)}.h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then at $\vote(k+2)$, every honest validator has:
\begin{enumerate}
  \item $R_v^{\vote(k+2)}.h = \comm{h} + 1$ for every honest validator~$v$;
  \item $R_v^{\vote(k+2)} \sim R_{v'}^{\vote(k+2)}$ for every pair of honest validators~$v, v'$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+2)$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: structural facts at $\vote(k)$ and the slot-$k$ honest votes do not advance $R$.}
By hypothesis~(3), $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is justified in every honest's store; hence $\Sigma_v.\key_R \geq (h,\, 1,\, 0,\, \comm{R}^{\vote(k)}.\slot,\, \hash(\comm{R}^{\vote(k)}))$ at $\vote(k)$ for every honest~$v$.
By hypothesis~(4), the honest $\getTarget_v^{\vote(k)}$ are pairwise compatible and at state-height~$h$; let~$D$ be the chain extending them. Each $\getTarget_v^{\vote(k)} \in D$ at state-height~$h$ (possibly at distinct chain-heights, since state-height is monotone non-decreasing along chain ancestry).
By \Cref{rev:def:getTarget}, each $\getTarget_v^{\vote(k)}$ descends from $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$, so $\comm{R}^{\vote(k)} \in D$ at chain-height~$h$ and $D$ extends~$\comm{R}^{\vote(k)}$ at chain-heights~$\geq h$.
Hypothesis~(5) rules out earlier honest votes at height $>h$, so at $\vote(k)$ every honest casts an (R1) \just vote at height~$h$ with target $T_v = \getTarget_v^{\vote(k)} \in D$.

By synchrony, every slot-$k$ honest \just vote (which also constitutes a slot-$k$ \notar contribution) is delivered to every honest by $\vote(k+1)$. In each honest~$v$'s aggregation, the prefix-notar $\arg\max$-by-$\preceq$ in \processHeight{} produces a candidate $(B^\star_v, k)$ at state-height~$h$ with $B^\star_v \preceq T_i$ for $\geq 2n/3$ honest~$i$, hence $B^\star_v \in D$ at state-height~$h$; this fires a bit-$0$ cert event with key $(h,\, 0,\, k,\, B^\star_v.\slot,\, \hash(B^\star_v))$.
The cached $\Sigma_v.\key_R \geq (h,\, 1,\, 0,\, \ldots)$ strictly dominates the new key on the bit coordinate, so \updateForkChoiceRoot{}'s guard rejects it.
\Cref{rev:cor:no-high-cn} applied per-honest to each $\Sigma_v^{\vote(k)}$ rules out cert events at height $>h$ before $\vote(k)$; the same quorum count extends through $[\vote(k),\, \vote(k+1)]$ (slot-$k$ honest votes are at height~$h$ exactly, and dishonest alone has $\leq f < 2n/3$), and \Cref{lem:Rs-key-monotone} preserves the bit-$1$ bound through any other update.
Hence $R_v^t = \comm{R}^{\vote(k)}$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$.

\emph{Step 2: the canonical state-height reaches $h+1$ by $\vote(k+1)$.}
Pick any honest~$v_\ell$. By Step~1, $\getTarget_{v_\ell}^{\vote(k)} \in D$ at state-height~$h$ on the chain extending $\comm{R}^{\vote(k)}$, and $R_v^t = \comm{R}^{\vote(k)}$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$, so $\getTarget_{v_\ell}^{\vote(k)} \succeq \comm{R}^{\vote(k)} = R_v^t$, discharging the premise of \Cref{rev:prop:honest-vote-inclusion} (whose preamble's compatibility hypothesis is satisfied since all honest $R_v^{\vote(k)}$ coincide).
Its conclusion gives that the chain ending at $\getTarget_v^{\vote(k+1)}$ contains every slot-$k$ honest vote; the slot-$k$ bit-$0$ cert event at $B^\star \in D$ at state-height~$h$ (Step~1) therefore lies on that chain, and its height-advance pushes the state-height to $h+1$, i.e.\ $\sigma[\getTarget_v^{\vote(k+1)}].h = h + 1$ for every honest~$v$.

\emph{Step 3: reduce to the one-slot advance at slot~$k+1$.}
The hypotheses of \Cref{rev:lem:healing-one-slot}, applied with~$k$ replaced by~$k+1$, are all met:
\begin{itemize}
  \item hypotheses~(1) and~(2) are inherited (synchrony at slot~$k+1$: $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$);
  \item hypothesis~(3): \Cref{rev:prop:getTarget-compat}, applied with the same premise as Step~2, gives $\getTarget_v^{\vote(k+1)} \sim \getTarget_{v'}^{\vote(k+1)}$ for every honest pair;
  \item hypothesis~(4): by Step~2, $\sigma[\getTarget_v^{\vote(k+1)}].h = h + 1$ is the common value across honest validators, playing the role of \Cref{rev:lem:healing-one-slot}'s~$\comm{h}$ at slot~$k+1$;
  \item hypothesis~(5) holds because outer~(5) rules out honest votes at height $>h$ before $\vote(k)$, Step~1 places the slot-$k$ honest votes at height~$h$ exactly, and no voting event lies in $(\vote(k),\, \vote(k+1))$; hence no honest has cast at height $\geq h+1$ before $\vote(k+1)$.
\end{itemize}
Invoking \Cref{rev:lem:healing-one-slot} at slot~$k+1$ yields the three stated conclusions at $\vote(k+2)$: $R_v^{\vote(k+2)}.h = h + 1$ for every honest~$v$; $R_v^{\vote(k+2)} \sim R_{v'}^{\vote(k+2)}$ for every pair of honest $v, v'$; and no honest has cast a \just vote at height $> R.h$ strictly before $\vote(k+2)$.
\end{proof}

\begin{lemma}[Two-slot healing from a notarized root, no justification completing]\label{rev:lem:healing-from-notarized-no-just}
Assume:
\begin{enumerate}
  \item $f < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item $\getTarget_v^{\vote(k)} \sim \getTarget_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getTarget_v^{\vote(k)}].h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item for every honest validator~$v$, $R_v^{\vote(k)}$ is a bit-$0$ notarized block at cert-key height $\comm{h}$ with notarization slot $k-1$;
  \item no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$;
  \item no bit-$1$ cert at height $\comm{h}$ has fired in any honest's view at any time in $[\vote(k),\, \vote(k+1)]$.
\end{enumerate}
Then at $\vote(k+2)$, every honest validator has:
\begin{enumerate}
  \item $R_v^{\vote(k+2)}.h = \comm{h} + 1$ for every honest validator~$v$;
  \item $R_v^{\vote(k+2)} \sim R_{v'}^{\vote(k+2)}$ for every pair of honest validators~$v, v'$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+2)$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: structural facts at $\vote(k)$.}
By hypothesis~(3), the honest $\getTarget_v^{\vote(k)}$ are pairwise $\sim$-compatible and all at state-height~$h$; let~$D$ be the chain extending them. Each $\getTarget_v^{\vote(k)} \in D$ at state-height~$h$ (possibly at distinct chain-heights, since state-height is monotone non-decreasing along chain ancestry).
By \Cref{rev:def:getTarget}, each $\getTarget_v^{\vote(k)}$ descends from $R_v^{\vote(k)}$, so $R_v^{\vote(k)} \in D$ as well; consistently, hypothesis~(4) places each $R_v^{\vote(k)}$ at a bit-$0$ slot-$(k-1)$ notarization at cert-key height~$h$, and by \Cref{rev:lem:notar-unique-per-slot} these notarizations are pairwise compatible (also putting the $R_v$ on~$D$). Each honest~$v$ has cached $\Sigma_v.\key_R = (h,\, 0,\, k-1,\, R_v.\slot,\, \hash(R_v))$; the first three coordinates are uniform, the fourth and fifth vary with~$R_v$.
Hypothesis~(5) rules out any earlier honest vote at height $>h$, so at $\vote(k)$ every honest casts an (R1) \just vote at height~$h$ with target $T_v = \getTarget_v^{\vote(k)} \in D$. Let $T^\dagger \coloneqq \max_\preceq \{T_v : v \text{ honest}\}$ on~$D$.

\emph{Step 2: a slot-$k$ same-slot bit-$0$ cert at height~$h$ on~$D$ fires by $\vote(k+1)$ in every honest's view.}
By synchrony, every slot-$k$ honest \just vote (which also constitutes a slot-$k$ \notar contribution) is delivered to every honest by $\vote(k+1)$. In each honest~$v$'s aggregation, the prefix-notar $\arg\max$-by-$\preceq$ in \processHeight{} (\Cref{alg:state-machine}) produces a candidate $(B^\star_v, k)$ at state-height~$h$ with $B^\star_v \preceq T_i$ for $\geq 2n/3$ honest~$i$, hence $B^\star_v \in D$ (the common chain of honest targets) and $B^\star_v \preceq T^\dagger$. The candidate fires a bit-$0$ cert event with key $(h,\, 0,\, k,\, B^\star_v.\slot,\, \hash(B^\star_v))$. The choice $B^\star_v$ may differ across stores (depending on $\sigma_v.L$), but each is on~$D$ at state-height~$h$.

By the same dishonest-only-bound argument (hypothesis~(5) excludes honest at $>h$ before $\vote(k)$; Step~1 places slot-$k$ honest votes at height~$h$ exactly; no voting event in $(\vote(k), \vote(k+1))$; dishonest alone has $\leq f < 2n/3$), no cert event at height $>h$ fires in any honest's view in $[\vote(k),\, \vote(k+1)]$.

\emph{Step 3: every honest's $R_v^t$ stays on~$D$ with $R_v^t \preceq T^\dagger$ throughout $[\vote(k),\, \vote(k+1)]$.}
By Step~1's per-honest cached key $(h,\, 0,\, k-1,\, R_v.\slot,\, \hash(R_v))$ and \Cref{lem:Rs-key-monotone}, every cert event that updates $R_v$ during $[\vote(k),\, \vote(k+1)]$ has key $(h',\, b',\, s',\, \ldots)$ strictly dominating it. The options are:
\begin{itemize}
  \item $h' > h$: ruled out by Step~2.
  \item $h' = h$, $b' = 1$: ruled out by hypothesis~(6).
  \item $h' = h$, $b' = 0$, $s' > k-1$: then $s' \geq k$. Contributions at slot~$\geq k$ do not exist before $\vote(k)$, so the witnessing single-slot quorum at slot~$s'$ at height~$h$ is the slot-$k$ cert from Step~2 (target on~$D$ with target $\preceq T^\dagger$ by the prefix-notar $\arg\max$ argument).
\end{itemize}
Hence every cert updating $R_v$ during the interval is a slot-$k$ bit-$0$ notarization on~$D$ at target $\preceq T^\dagger$. Combined with $R_v^{\vote(k)} \in D$ at state-height~$h$ (Step~1), every $R_v^t$ in the interval lies on~$D$ with $R_v^t \preceq T^\dagger$.

\emph{Step 4: canonical state-height reaches $h+1$ by $\vote(k+1)$.}
Pick $v_\ell$ with $\getTarget_{v_\ell}^{\vote(k)} = T^\dagger$. By Step~3, $T^\dagger \succeq R_v^t$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$, discharging the premise of \Cref{rev:prop:honest-vote-inclusion} (whose preamble's compatibility hypothesis is satisfied since all honest $R_v^{\vote(k)}$ lie on~$D$ by Step~1, hence pairwise compatible). Its conclusion gives that the chain ending at $\getTarget_v^{\vote(k+1)}$ contains every slot-$k$ honest vote; the slot-$k$ cert-event firing block on~$D$ (Step~2) therefore lies on that chain, and its height-advance places $\sigma[\getTarget_v^{\vote(k+1)}].h = h + 1$ for every honest~$v$.

\emph{Step 5: reduce to the one-slot advance at slot~$k+1$.}
The hypotheses of \Cref{rev:lem:healing-one-slot}, applied with~$k$ replaced by~$k+1$, are all met:
\begin{itemize}
  \item hypotheses~(1) and~(2) are inherited (synchrony at slot~$k+1$: $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$);
  \item hypothesis~(3): \Cref{rev:prop:getTarget-compat}, with premise discharged as in Step~4, gives $\getTarget_v^{\vote(k+1)} \sim \getTarget_{v'}^{\vote(k+1)}$ for every honest pair;
  \item hypothesis~(4): $\sigma[\getTarget_v^{\vote(k+1)}].h = h + 1$ is the common value across honest validators (Step~4);
  \item hypothesis~(5): hypothesis~(5) of this lemma rules out honest votes at height $>h$ before $\vote(k)$, Step~1 places slot-$k$ honest votes at height~$h$ exactly, and no voting event lies in $(\vote(k),\, \vote(k+1))$; hence no honest has cast at height $\geq h+1$ before $\vote(k+1)$.
\end{itemize}
Invoking \Cref{rev:lem:healing-one-slot} at slot~$k+1$ yields the three stated conclusions at $\vote(k+2)$.
\end{proof}

\begin{lemma}[Two-slot healing from a notarized common root]\label{rev:lem:healing-from-notarized}
Assume:
\begin{enumerate}
  \item $f < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item $\syncer(k+1)$ is honest; let $\comm{\Sigma}^{\vote(k+1)}$ be the common store at $\vote(k+1)$;
  \item $\getTarget_v^{\vote(k)} \sim \getTarget_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getTarget_v^{\vote(k)}].h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item for every honest validator~$v$, $R_v^{\vote(k)}$ is a bit-$0$ notarized block at cert-key height $\comm{h}$ with notarization slot $k-1$;
  \item no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then at $\vote(k+2)$, every honest validator has:
\begin{enumerate}
  \item $R_v^{\vote(k+2)}.h = \comm{h} + 1$ for every honest validator~$v$;
  \item $R_v^{\vote(k+2)} \sim R_{v'}^{\vote(k+2)}$ for every pair of honest validators~$v, v'$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+2)$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$. By view-merge + honest $\syncer(k+1)$ (hypothesis~(3)), all honest stores at $\vote(k+1)$ equal $\comm{\Sigma}^{\vote(k+1)}$.

\emph{Case 1: a bit-$1$ cert at height~$h$ has fired in $\comm{\Sigma}^{\vote(k+1)}$.}
Then $\comm{R}^{\vote(k+1)}$ is a justified block at height~$h$ in every honest's view at $\vote(k+1)$. The setup matches the hypotheses of \Cref{rev:lem:healing-from-justified} applied at slot~$k+1$: by view-merge, $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is justified in every honest's view, discharging~(3) of from-justified; honest \getTarget{}s at $\vote(k+1)$ coincide (hence pairwise compatible) and remain at state-height~$h$ (the bit-$1$ cert's height coordinate is~$h$ in cert-key, and its height-advance fires only virtually under view-merge), discharging~(4); hypothesis~(6) of this lemma (no honest at $>h$ before $\vote(k)$) plus the slot-$k$ honest votes at height~$h$ exactly rules out honest votes at height $>h$ before $\vote(k+1)$, discharging~(5). Invoking \Cref{rev:lem:healing-from-justified} yields the three stated conclusions at $\vote(k+2)$.

\emph{Case 2: no bit-$1$ cert at height~$h$ in $\comm{\Sigma}^{\vote(k+1)}$.}
By view-merge, no honest's view contains a bit-$1$ cert at height~$h$ at any time in $[\vote(k),\, \vote(k+1)]$ (cert events are persistent within a store, and view-merge unions stores; if any honest had such a cert at any earlier time, it would appear in $\comm{\Sigma}^{\vote(k+1)}$). The hypotheses of \Cref{rev:lem:healing-from-notarized-no-just} are all met: (1)+(2) inherited; (3)+(4)+(5) are this lemma's (4)+(5)+(6); (6) just established. Invoking it yields the three stated conclusions at $\vote(k+2)$.
\end{proof}

\begin{lemma}[Three-slot healing]\label{rev:lem:healing-three-slot}
Assume:
\begin{enumerate}
  \item $f < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item $\syncer(k+1)$ and $\syncer(k+2)$ are honest;
  \item $\getTarget_v^{\vote(k)} \sim \getTarget_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getTarget_v^{\vote(k)}].h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item no honest validator has cast any vote at height $> \comm{h}$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then at $\vote(k+3)$, every honest validator has:
\begin{enumerate}
  \item $R_v^{\vote(k+3)}.h = \comm{h} + 1$ for every honest validator~$v$;
  \item $R_v^{\vote(k+3)} \sim R_{v'}^{\vote(k+3)}$ for every pair of honest validators~$v, v'$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+3)$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: slot-$k$ honest votes form a cert at height~$h$ on a common chain.}
By hypothesis~(4), the honest $\getTarget_v^{\vote(k)}$ are pairwise compatible and at state-height~$h$; let~$C$ be the chain extending them. Each $\getTarget_v^{\vote(k)} \in C$ at state-height~$h$ (possibly at distinct chain-heights, since state-height is monotone non-decreasing along chain ancestry). Hypothesis~(5) rules out any earlier honest vote at height $>h$, so at $\vote(k)$ every honest casts an (R1) \just vote at height~$h$ with target $T_i = \getTarget_i^{\vote(k)} \in C$. The $\geq 2n/3$ honest slot-$k$ \just votes (which also constitute slot-$k$ \notar contributions on~$C$) together with any slot-$k$ dishonest contributions fire a same-slot cert event at height~$h$ on~$C$ (bit-$1$ justification if the $T_i$ all coincide, bit-$0$ notarization at the prefix-notar $\arg\max$-by-$\preceq$ block on~$C$ otherwise).

\emph{Step 2: case split on $\comm{R}^{\vote(k+1)}$.}
Under view-merge + honest $\syncer(k+1)$, honest share $\comm{\Sigma}^{\vote(k+1)}$ at $\vote(k+1)$. By \Cref{rev:cor:no-high-cn} applied to $\comm{\Sigma}^{\vote(k+1)}$, no cert event at height $>h$ exists there: hypothesis~(5) excludes honest contributions at height $>h$ before $\vote(k)$, Step~1 places slot-$k$ honest contributions at height~$h$ exactly, no voting event lies in $(\vote(k),\, \vote(k+1))$, and dishonest alone ($\leq f < n/3$) cannot form a $\geq 2n/3$ quorum.

\emph{Case 1: $\comm{R}^{\vote(k+1)}$ is a justified block at height~$h$.}
The setup at $\vote(k+1)$ satisfies the hypotheses of \Cref{rev:lem:healing-from-justified} applied at slot~$k+1$: synchrony $\start(k+1) - \Delta \geq \GST$ (from outer~(2)); by view-merge + honest $\syncer(k+1)$ (outer~(3)), $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is justified in every honest's view (discharging new~(3)); honest \getTarget{}s at $\vote(k+1)$ coincide (hence pairwise compatible) and sit at state-height~$h$ (the justification's height-advance fires only virtually in $\comm{\Sigma}^{\vote(k+1)}$ and is not persisted to any actual block), discharging~(4); and outer~(5) together with Step~1 rules out any honest vote at height $>h$ before $\vote(k+1)$, discharging~(5). Invoking \Cref{rev:lem:healing-from-justified} yields the three stated conclusions at $\vote(k+3)$.

\emph{Case 2: $\comm{R}^{\vote(k+1)}$ is not a justified checkpoint.}
Then the cached cert at $\vote(k+1)$ is bit-$0$. By the same no-cert-above-$h$ argument its height is $\leq h$; among bit-$0$ certs at height~$h$, the slot-$k$ notarization of Step~1 strictly dominates any slot-$<k$ cert on the $s_n$ coordinate, and contributions at slot $\geq k$ do not exist before $\vote(k)$ for $s_n > k$ to occur. Hence $\comm{R}^{\vote(k+1)}$ is a bit-$0$ notarization at height~$h$ with notarization slot~$k$.

The setup at $\vote(k+1)$ satisfies the hypotheses of \Cref{rev:lem:healing-from-notarized} applied at slot~$k+1$: synchrony; $\syncer(k+2)$ honest (outer~(3)), discharging~(3); by view-merge + honest $\syncer(k+1)$ (outer~(3)), every honest's store at $\vote(k+1)$ coincides with $\comm{\Sigma}^{\vote(k+1)}$, so honest \getTarget{}s at $\vote(k+1)$ are equal (hence pairwise compatible) and at state-height~$h$ (same virtual-only argument as Case~1), discharging~(4); by view-merge, $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is bit-$0$ notarized at height~$h$ with slot $k = (k+1)-1$ in every honest's view, discharging~(5); and outer~(5) with Step~1 rules out honest votes at height $>h$ before $\vote(k+1)$, discharging~(6). Invoking \Cref{rev:lem:healing-from-notarized} yields the three stated conclusions at $\vote(k+3)$.
\end{proof}

\begin{lemma}[Four-slot healing]\label{rev:lem:healing-four-slot}
Assume:
\begin{enumerate}
  \item $f < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item $\syncer(k+1)$, $\syncer(k+2)$ and $\syncer(k+3)$ are honest;
  \item for every honest validator~$v$, $R_v^{\vote(k)}$ is the same justified block; denote it by $\comm{R}^{\vote(k)}$;
  \item $\getTarget_v^{\vote(k)} \sim \getTarget_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$, and $\sigma[\getTarget_v^{\vote(k)}].h = \comm{R}^{\vote(k)}.h$ is the same for every honest~$v$; denote this common value by $\comm{h}$;
  \item no honest validator has cast any vote at height $> \comm{h}+1$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then at $\vote(k+4)$, every honest validator has:
\begin{enumerate}
  \item $R_v^{\vote(k+4)}.h = \comm{h} + 2$ for every honest validator~$v$;
  \item $R_v^{\vote(k+4)} \sim R_{v'}^{\vote(k+4)}$ for every pair of honest validators~$v, v'$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+3)$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: canonical state-height reaches $h+1$ at $\vote(k+1)$, common across honest.}
By hypothesis~(4), $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is justified at height~$h$ in every honest's view; hence $\Sigma_v.\key_R \geq (h, 1, 0, \comm{R}^{\vote(k)}.\slot, \hash(\comm{R}^{\vote(k)}))$ at $\vote(k)$ for every honest~$v$. By hypothesis~(5), the honest $\getTarget_v^{\vote(k)}$ are pairwise compatible and at state-height~$h$; let~$D$ be the chain extending them. Each $\getTarget_v^{\vote(k)} \in D$ at state-height~$h$, descending from $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$, so $\comm{R}^{\vote(k)} \in D$ at chain-height~$h$ and $D$ extends~$\comm{R}^{\vote(k)}$ at chain-heights~$\geq h$. Hypothesis~(6) allows honest votes at heights $\leq h+1$ before $\vote(k)$; in particular, honest may have cast \just votes at height~$h+1$ at earlier slots.

At $\vote(k)$ every honest casts an (R1) \just vote at height~$h$ with target $T_v = \getTarget_v^{\vote(k)} \in D$; the prefix-notar $\arg\max$-by-$\preceq$ in \processHeight{} fires a same-slot slot-$k$ cert at some $B^\star \in D$ at state-height~$h$, with key $(h, 0, k, \ldots)$ or $(h, 1, 0, \ldots)$, which is lex-dominated by (or tied to) the cached bit-$1$ key $(h, 1, 0, \ldots)$; this cert does not strictly advance~$R_v$. In $[\vote(k),\, \vote(k+1)]$, no cert event at height $>h+1$ can form (hypothesis~(6) combined with Step-1 voting + $f<n/3$), so $\Sigma_v.\key_R.h \leq h+1$ throughout. Two sub-cases arise at $\vote(k+1)$; view-merge + honest $\syncer(k+1)$ makes the outcome common across honest.

\emph{Sub-case A: a cert at height $h+1$ has fired in $\comm{\Sigma}^{\vote(k+1)}$.}
This is possible when pre-existing honest \just votes at $h+1$, together with late dishonest contributions, complete a $\geq 2n/3$ quorum. The resulting cert strictly dominates the cached key, so $\comm{R}^{\vote(k+1)}$ is at cert-key height $h+1$. Canonical state-height at $\vote(k+1)$ is then $h+1$ (no cert event at height $>h+1$ can fire under the same bound).

\emph{Sub-case B: no cert at height $h+1$ has fired.}
Then $\comm{R}^{\vote(k+1)} = \comm{R}^{\vote(k)}$, so $R_v^t = \comm{R}^{\vote(k)}$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$. For any honest $v_\ell$, $\getTarget_{v_\ell}^{\vote(k)} \in D$ at state-height~$h$, so $\getTarget_{v_\ell}^{\vote(k)} \succeq \comm{R}^{\vote(k)} = R_v^t$ (by \Cref{rev:def:getTarget}), discharging the premise of \Cref{rev:prop:honest-vote-inclusion}; its conclusion gives that the chain ending at $\getTarget_v^{\vote(k+1)}$ contains every slot-$k$ honest vote (all on~$D$ by Step~1 above), and the slot-$k$ cert event at $B^\star \in D$ lies on that chain, advancing state-height to $h+1$, i.e.\ $\sigma[\getTarget_v^{\vote(k+1)}].h = h + 1$ for every honest~$v$.

Either way, canonical state-height at $\vote(k+1)$ is $h+1$ and common across honest.

\emph{Step 2: apply \Cref{rev:lem:healing-three-slot} at slot~$k+1$.}
The hypotheses of \Cref{rev:lem:healing-three-slot} applied with $k$ replaced by $k+1$ are all met:
\begin{itemize}
  \item (1), (2) inherited (synchrony at slot $k+1$: $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$);
  \item (3): outer~(3) provides $\syncer(k+2), \syncer(k+3)$ honest;
  \item (4): Step~1 gives $\sigma[\getTarget_v^{\vote(k+1)}].h = h + 1$ as the common value across honest validators, and by view-merge + honest $\syncer(k+1)$ (outer~(3)) honest \getTarget{}s at $\vote(k+1)$ coincide, hence pairwise compatible;
  \item (5): outer~(6) rules out honest votes at height $>h+1$ before $\vote(k)$; at $\vote(k)$ every honest casts at height~$h$ (Step~1); no voting event lies in $(\vote(k),\, \vote(k+1))$; hence no honest has cast at height $>h+1$ before $\vote(k+1)$.
\end{itemize}
Invoking \Cref{rev:lem:healing-three-slot} at slot~$k+1$ yields the three stated conclusions at $\vote(k+4)$: $R_v^{\vote(k+4)}.h = (h+1) + 1 = h + 2$ for every honest~$v$; $R_v^{\vote(k+4)} \sim R_{v'}^{\vote(k+4)}$ for every honest pair; and no honest has cast a \just vote at height $>R.h$ strictly before $\vote(k+4)$, which in particular implies the stated conclusion~(3) before $\vote(k+3)$.
\end{proof}

\begin{lemma}[Four-slot healing from a justified common root]\label{rev:lem:healing-four-slot-from-justified}
Assume:
\begin{enumerate}
  \item $f < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item $\syncer(k+1)$, $\syncer(k+2)$ and $\syncer(k+3)$ are honest;
  \item for every honest validator~$v$, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is a justified block, and $\getTarget_v^{\vote(k)} \sim \getTarget_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$; denote $\comm{h} \coloneqq \comm{R}^{\vote(k)}.h$.
\end{enumerate}
Then at $\vote(k+4)$, every honest validator has:
\begin{enumerate}
  \item $R_v^{\vote(k+4)}.h = \comm{h} + 2$ for every honest validator~$v$;
  \item $R_v^{\vote(k+4)} \sim R_{v'}^{\vote(k+4)}$ for every pair of honest validators~$v, v'$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+3)$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h} = \comm{R}^{\vote(k)}.h$.

\emph{Step 1: structural facts at $\vote(k)$.}
By hypothesis~(4), $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is justified at height~$h$ in every honest's view; hence $\Sigma_v.\key_R \geq (h, 1, 0, \comm{R}^{\vote(k)}.\slot, \hash(\comm{R}^{\vote(k)}))$ at $\vote(k)$ for every honest~$v$, and in particular $\Sigma_v.\key_R.h = h$. \Cref{rev:cor:processed-height-bound} applied per-honest to each $\Sigma_v^{\vote(k)}$ bounds every processed block by $\sigma.h \leq R_v^{\vote(k)}.h + 1 = h + 1$, so $\sigma[\getTarget_v^{\vote(k)}].h \in \{h, h+1\}$ for every honest~$v$. By \Cref{lem:Rs-key-monotone}, $\Sigma_v.\key_R.h \leq h$ at every earlier time, and \Cref{rev:cor:processed-height-bound} applied per-honest at every earlier time bounds processed blocks by state-height $\leq h+1$; hence no honest validator has cast any vote at height $> h+1$ at any time strictly before $\vote(k)$.

\emph{Step 2: case split on common state-height at $\vote(k)$.}

\emph{Case A: $\sigma[\getTarget_v^{\vote(k)}].h = h+1$ for every honest~$v$.}
The hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k$ with the property's $\comm{h}$ taken to be $h+1$: hypothesis~(4) (\getTarget pairwise compatible from outer~(4) and common state-height from the case assumption); hypothesis~(5) (no honest at $> h+1$ before $\vote(k)$) is Step~1; and (3) is inherited. \Cref{rev:lem:healing-three-slot} yields $R_v^{\vote(k+3)}.h = h+2$ at $\vote(k+3)$; by \Cref{lem:Rs-key-monotone} and the same height-bound argument applied through $[\vote(k+3),\, \vote(k+4)]$, $R_v^{\vote(k+4)}.h = h+2$.

\emph{Case B: $\sigma[\getTarget_v^{\vote(k)}].h = h$ for every honest~$v$.}
The hypotheses of \Cref{rev:lem:healing-four-slot} apply at slot~$k$: (3) outer~(3) provides $\syncer(k+1), \syncer(k+2), \syncer(k+3)$ honest; (4) by outer~(4), $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ is the same justified block in every honest's view; (5) honest \getTarget{}s at $\vote(k)$ are pairwise compatible (outer~(4)) and at common state-height $= \comm{R}.h$ (case assumption); (6) is Step~1. \Cref{rev:lem:healing-four-slot} yields the three stated conclusions at $\vote(k+4)$ directly.

\emph{Case C: state-heights at $\vote(k)$ are mixed (some honest at $h$, some at $h+1$).}
Let $T^\dagger$ denote the $\preceq$-maximum honest \getTarget{} at $\vote(k)$; by Step~1 and the case assumption, $\sigma[T^\dagger].h = h+1$. We sub-split on $\comm{R}^{\vote(k+1)}.h$ at $\vote(k+1)$ (well-defined and common across honest by view-merge + honest $\syncer(k+1)$).

\emph{Sub-case C.A: $\comm{R}^{\vote(k+1)}.h = h+1$.}
A cert at height $h+1$ has fired in $\comm{\Sigma}^{\vote(k+1)}$. The hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k+1$ with $\comm{h}' = h+1$: canonical state-height at $\vote(k+1)$ is exactly $h+1$ (\Cref{rev:cor:processed-height-bound} bounds it by $\Sigma.\key_R.h + 1 = h+2$, and the $\geq h+1$ direction follows from $\getTarget$ descending from~$R$ at height $h+1$); the no-honest-above bound is inherited from Step~1 plus the slot-$k$ honest votes lying at heights $\leq h+1$. \Cref{rev:lem:healing-three-slot} yields $R_v^{\vote(k+4)}.h = h+2$.

\emph{Sub-case C.B: $\comm{R}^{\vote(k+1)}.h = h$.}
By justification-uniqueness (\Cref{rev:lem:unique-just-per-height}) the only justified block at height~$h$ is $\comm{R}^{\vote(k)}$, so $\comm{R}^{\vote(k+1)} = \comm{R}^{\vote(k)}$, giving $R_v^t = \comm{R}^{\vote(k)}$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$. Picking $v_\ell$ with $\getTarget_{v_\ell}^{\vote(k)} = T^\dagger$ discharges the premise of \Cref{rev:prop:target-dominance}; its conclusion gives $\getTarget_v^{\vote(k+1)} \succeq T^\dagger$ for every honest~$v$, so $\sigma[\getTarget_v^{\vote(k+1)}].h \geq h + 1$. \Cref{rev:cor:processed-height-bound} applied to $\comm{\Sigma}^{\vote(k+1)}$ bounds it by $\Sigma.\key_R.h + 1 = h + 1$, so canonical state-height at $\vote(k+1)$ is exactly $h+1$ and common across honest. Apply \Cref{rev:lem:healing-three-slot} at slot~$k+1$ with $\comm{h}' = h+1$ exactly as in Sub-case~C.A, yielding $R_v^{\vote(k+4)}.h = h+2$.

In every case, $R_v^{\vote(k+4)}.h = h + 2 = \comm{h} + 2$, $R_v^{\vote(k+4)} \sim R_{v'}^{\vote(k+4)}$ for every honest pair, and no honest has cast a \just vote at height $> R.h$ strictly before $\vote(k+3)$ (the latter follows from the corresponding conclusion of the invoked sub-lemma in each case).
\end{proof}

\begin{lemma}[Four-slot healing from a notarized common root]\label{rev:lem:healing-four-slot-from-notarized}
Assume:
\begin{enumerate}
  \item $f < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item $\syncer(k+1)$, $\syncer(k+2)$ and $\syncer(k+3)$ are honest;
  \item $\getTarget_v^{\vote(k)} \sim \getTarget_{v'}^{\vote(k)}$ for every pair of honest validators~$v, v'$;
  \item for every honest validator~$v$, $R_v^{\vote(k)}$ is a bit-$0$ notarized block at cert-key height $\comm{h}$ with notarization slot $k-1$.
\end{enumerate}
Then at $\vote(k+4)$, every honest validator has:
\begin{enumerate}
  \item $R_v^{\vote(k+4)}.h = \comm{h} + 2$ for every honest validator~$v$;
  \item $R_v^{\vote(k+4)} \sim R_{v'}^{\vote(k+4)}$ for every pair of honest validators~$v, v'$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+3)$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: structural facts at $\vote(k)$.}
By hypothesis~(5), each honest~$v$ has cached $\Sigma_v.\key_R = (h, 0, k-1, R_v.\slot, \hash(R_v))$ at $\vote(k)$, so $R_v^{\vote(k)}.h = h$. By \Cref{rev:lem:notar-unique-per-slot}, the slot-$(k-1)$ notarizations at cert-key height~$h$ are pairwise compatible, so the honest $R_v^{\vote(k)}$ all lie on a single chain at state-height~$h$. \Cref{rev:cor:processed-height-bound} applied per-honest to each $\Sigma_v^{\vote(k)}$ bounds every processed block by state-height $\leq R_v^{\vote(k)}.h + 1 = h + 1$, so $\sigma[\getTarget_v^{\vote(k)}].h \in \{h, h+1\}$ for every honest~$v$. Combined with \Cref{lem:Rs-key-monotone} (so $\Sigma_v.\key_R.h \leq h$ at every earlier time) and \Cref{rev:cor:processed-height-bound} applied per-honest at every earlier time, no honest validator has cast any vote at height $> h+1$ at any time strictly before $\vote(k)$.

\emph{Step 2: case split on common state-height at $\vote(k)$.}

\emph{Case A: $\sigma[\getTarget_v^{\vote(k)}].h = h$ for every honest~$v$.}
The hypotheses of \Cref{rev:lem:healing-from-notarized} apply at slot~$k$: (3) $\syncer(k+1)$ honest from outer~(3); (4) honest \getTarget{}s at $\vote(k)$ are pairwise compatible (outer~(4)) and at common state-height $= h$ (case assumption); (5) for every honest~$v$, $R_v^{\vote(k)}$ is bit-$0$ notarized at $h$ with slot $k-1$ (outer~(5)); (6) no honest at $> h$ before $\vote(k)$ (per-honest argument: by the case assumption $\sigma[\getTarget_v^{\vote(k)}].h = h$, the maximum state-height in $\Sigma_v^{\vote(k)}$ is~$h$; by store monotonicity, the maximum state-height in $\Sigma_v^{t'}$ at any earlier time $t' < \vote(k)$ is also $\leq h$, so $\getTarget_v^{t'}$ sits at state-height $\leq h$ and honest~$v$ cast at height $\leq h$ at every earlier voting time). \Cref{rev:lem:healing-from-notarized} yields $R_v^{\vote(k+2)}.h = h+1$ at $\vote(k+2)$, with no honest \just vote at height $>R.h = h+1$ before $\vote(k+2)$.
At $\vote(k+2)$, sub-split on the bit of $\comm{R}^{\vote(k+2)}$:
\begin{itemize}
  \item if $\comm{R}^{\vote(k+2)}$ is justified, apply \Cref{rev:lem:healing-from-justified} at slot~$k+2$;
  \item otherwise $\comm{R}^{\vote(k+2)}$ is bit-$0$ at height~$h+1$; the same-slot quorum that fired it includes the slot-$(k+1)$ honest contributions, so its notarization slot is $k+1 = (k+2)-1$, and \Cref{rev:lem:healing-from-notarized} applies at slot~$k+2$.
\end{itemize}
Either invocation yields $R_v^{\vote(k+4)}.h = (h+1) + 1 = h+2$ at $\vote(k+4)$.

\emph{Case B: $\sigma[\getTarget_v^{\vote(k)}].h = h+1$ for every honest~$v$.}
The hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k$ with $\comm{h}' = h+1$: hypothesis~(4) is the case assumption; hypothesis~(5) (no honest at $> h+1$ before $\vote(k)$) is Step~1; (3) is inherited. \Cref{rev:lem:healing-three-slot} yields $R_v^{\vote(k+3)}.h = h+2$. By \Cref{lem:Rs-key-monotone} and the no-honest-above-$h+2$ bound through $[\vote(k+3),\, \vote(k+4)]$ (no cert event at height $>h+2$ can form: honest contributions at $> h+2$ are absent until $\vote(k+3)$, and slot-$(k+3)$ contributions alone cannot complete a $\geq 2n/3$ same-slot quorum at $> h+2$ without further honest above-$h+2$ activity), $R_v^{\vote(k+4)}.h = h+2$.

\emph{Case C: state-heights mixed at $\vote(k)$.}
Let $T^\dagger := \max_\preceq \{\getTarget_v^{\vote(k)} : v \text{ honest}\}$; by Step~1 and the case assumption, $\sigma[T^\dagger].h = h+1$. Sub-split on $\comm{R}^{\vote(k+1)}.h$ (well-defined and common across honest by view-merge + honest $\syncer(k+1)$).

\emph{Sub-case C.A: $\comm{R}^{\vote(k+1)}.h = h+1$.}
A cert at height $h+1$ has fired in $\comm{\Sigma}^{\vote(k+1)}$. Hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k+1$ with $\comm{h}' = h+1$: canonical state-height at $\vote(k+1)$ is exactly $h+1$ (\Cref{rev:cor:processed-height-bound} bounds it by $h+2$, and the $\geq h+1$ direction follows from $\getTarget$ descending from~$R$); the no-honest-above-$h+1$ bound is inherited from Step~1 plus slot-$k$ honest voting at heights $\leq h+1$. \Cref{rev:lem:healing-three-slot} yields $R_v^{\vote(k+4)}.h = h+2$.

\emph{Sub-case C.B: $\comm{R}^{\vote(k+1)}.h = h$.}
By the same argument as Step~2 of \Cref{rev:lem:healing-from-notarized}, every cert event that updates $R_v$ during $[\vote(k),\, \vote(k+1)]$ is a slot-$k$ notarization with target on the chain of slot-$k$ honest contributions, hence $\preceq T^\dagger$. So $R_v^t \preceq T^\dagger$ for every honest~$v$ and every $t \in [\vote(k),\, \vote(k+1)]$. Picking $v_\ell$ with $\getTarget_{v_\ell}^{\vote(k)} = T^\dagger$ discharges the premise of \Cref{rev:prop:target-dominance}; its conclusion gives $\getTarget_v^{\vote(k+1)} \succeq T^\dagger$ for every honest~$v$, so $\sigma[\getTarget_v^{\vote(k+1)}].h \geq h+1$. \Cref{rev:cor:processed-height-bound} bounds it by $\Sigma.\key_R.h + 1 = h+1$. Apply \Cref{rev:lem:healing-three-slot} at slot~$k+1$ with $\comm{h}' = h+1$ as in Sub-case~C.A, yielding $R_v^{\vote(k+4)}.h = h+2$.

In every case, $R_v^{\vote(k+4)}.h = h + 2 = \comm{h} + 2$, $R_v^{\vote(k+4)} \sim R_{v'}^{\vote(k+4)}$ for every honest pair, and no honest has cast a \just vote at height $> R.h$ strictly before $\vote(k+3)$ (the latter follows from the corresponding conclusion of the invoked sub-lemma in each case).
\end{proof}

\begin{lemma}[Five-slot healing]\label{rev:lem:healing-five-slot}
Assume:
\begin{enumerate}
  \item $f < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item $\syncer(k)$, $\syncer(k+1)$, $\syncer(k+2)$, $\syncer(k+3)$ and $\syncer(k+4)$ are honest;
  \item $\sigma[\getTarget_v^{\vote(k)}].h = \comm{R}^{\vote(k)}.h$ is the same for every honest validator~$v$; denote this common value by $\comm{h}$;
  \item no honest validator has cast any vote at height $> \comm{h}+1$ at any time strictly before $\vote(k)$.
\end{enumerate}
Then at $\vote(k+5)$, every honest validator has:
\begin{enumerate}
  \item $R_v^{\vote(k+5)}.h = \comm{h} + 2$ for every honest validator~$v$;
  \item $R_v^{\vote(k+5)} \sim R_{v'}^{\vote(k+5)}$ for every pair of honest validators~$v, v'$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+3)$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: $\Sigma.\key_R.h \leq h+1$ throughout $[\vote(k),\, \vote(k+1)]$.}
By \Cref{rev:ass:view-merge} and honest $\syncer(k)$, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ for every honest~$v$, with $\Sigma_v.\key_R.h = h$. By hypothesis~(5) and \Cref{lem:Rs-key-monotone}, no honest casts at height $> h+1$ before $\vote(k)$ and the cached key remains $\Sigma_v.\key_R.h \leq h$ at $\vote(k)$. At $\vote(k)$ every honest casts at height~$h$ exactly (hypothesis~(4)), forming slot-$k$ contributions at height~$h$. No cert event at height $> h+1$ can form in $[\vote(k),\, \vote(k+1)]$: any such cert needs a $\geq 2n/3$ quorum of votes at height $> h+1$, and hypothesis~(5) plus Step-1 voting plus $f < n/3$ rule that out.

\emph{Step 2: case split on $\comm{R}^{\vote(k+1)}$.}
View-merge + honest $\syncer(k+1)$ gives common $\comm{\Sigma}^{\vote(k+1)}$. By Step~1 $\comm{R}^{\vote(k+1)}.h \in \{h, h+1\}$; by \Cref{lem:Rs-key-monotone} also $\comm{R}^{\vote(k+1)}.h \geq h$. The cases:
\begin{itemize}
  \item \emph{Case 1a:} $\comm{R}^{\vote(k+1)}$ is a bit-$1$ justification at height~$h$. (Either the cached $\comm{R}^{\vote(k)}$ was already justified at~$h$ and is unchanged, or the slot-$k$ honest \just votes' targets coincided to fire a justification at~$h$.) The hypotheses of \Cref{rev:lem:healing-four-slot-from-justified} apply at slot~$k+1$ with $\comm{h}_{\text{ffj}}' = h$: outer~(3) provides $\syncer(k+2), \syncer(k+3), \syncer(k+4)$ honest, discharging~(3); by view-merge + honest $\syncer(k+1)$ (outer~(3)), $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is the same justified block in every honest's view, and honest \getTarget{}s at $\vote(k+1)$ coincide (hence pairwise compatible), discharging~(4). Invoking it yields $R_v^{\vote(k+5)}.h = h + 2$.
  \item \emph{Case 1b:} $\comm{R}^{\vote(k+1)}$ is a bit-$1$ justification at height $h+1$. (Pre-existing honest \just votes at $h+1$, together with late dishonest contributions in $\comm{\Sigma}^{\vote(k+1)}$, completed a quorum.) The hypotheses of \Cref{rev:lem:healing-from-justified} apply at slot~$k+1$ with $\comm{h}_{\text{fj}}' = h+1$: by view-merge + honest $\syncer(k+1)$ (outer~(3)), $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is justified in every honest's view (discharging new~(3)); canonical state-height at $\vote(k+1)$ is $h+1$ by \Cref{rev:cor:processed-height-bound}, and view-merge gives uniform (hence pairwise compatible) honest \getTarget{}s at $\vote(k+1)$ (discharging~(4)); the no-honest-above-$h+1$ bound holds by Step~1 (discharging~(5)). Invoking it yields $R_v^{\vote(k+3)}.h = h+2$; by \Cref{lem:Rs-key-monotone} and the absence of any cert event at height $> h+2$ in $[\vote(k+3),\, \vote(k+5)]$ (no honest contributions at $>h+2$ exist before $\vote(k+3)$ by the lemma's conclusion~(3) for this sub-case, and slot-$(k+3),(k+4)$ contributions at $h+2$ alone cannot lift the height further), $R_v^{\vote(k+5)}.h = h+2$.
  \item \emph{Case 2:} $\comm{R}^{\vote(k+1)}$ is bit-$0$ notarized at height $h+1$. The hypotheses of \Cref{rev:lem:healing-three-slot} apply at slot~$k+1$ with $\comm{h}_{\text{three}}' = h+1$. Invoking it yields $R_v^{\vote(k+4)}.h = h+2$; the same monotonicity-plus-no-higher-cert argument as in Case~1b extends this to $R_v^{\vote(k+5)}.h = h+2$.
  \item \emph{Case 3:} $\comm{R}^{\vote(k+1)}$ is bit-$0$ notarized at height~$h$. The cached at $\vote(k)$ has $\Sigma.\key_R = (h, b, s, \ldots)$ with $b \in \{0, 1\}$ and $s < k$ (slot-$\geq k$ contributions don't exist before $\vote(k)$); $\comm{R}^{\vote(k+1)}$ being bit-$0$ at height~$h$ rules out the bit-$1$ branch and forces the slot-$k$ notarization (which has slot $k > s$) to dominate, so $\comm{R}^{\vote(k+1)}$ has notarization slot $k = (k+1) - 1$. The hypotheses of \Cref{rev:lem:healing-four-slot-from-notarized} apply at slot~$k+1$: outer~(3) provides $\syncer(k+2), \syncer(k+3), \syncer(k+4)$ honest, discharging~(3); by view-merge + honest $\syncer(k+1)$ (outer~(3)), every honest's store at $\vote(k+1)$ coincides with $\comm{\Sigma}^{\vote(k+1)}$, so honest \getTarget{}s at $\vote(k+1)$ are equal (hence pairwise compatible), discharging~(4); by the same view-merge, $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is bit-$0$ notarized at~$h$ with slot $k = (k+1)-1$ in every honest's view, discharging~(5). Invoking it yields $R_v^{\vote(k+5)}.h = h + 2$.
\end{itemize}

In every case, $R_v^{\vote(k+5)}.h = h + 2 = \comm{h} + 2$, $R_v^{\vote(k+5)} \sim R_{v'}^{\vote(k+5)}$ for every honest pair (the invoked sub-lemma supplies pairwise compatibility, and the propagation argument from $\vote(k+3)$ or $\vote(k+4)$ to $\vote(k+5)$ preserves it via \Cref{lem:Rs-key-monotone}), and no honest has cast a \just vote at height $> R.h$ strictly before $\vote(k+3)$ (each sub-lemma's conclusion~(3) at the appropriate scope covers this).
\end{proof}

\begin{lemma}[Three-slot healing under slot-$k$ dual-height voting]\label{rev:lem:healing-three-slot-dual-vote}
Assume:
\begin{enumerate}
  \item $f < n/3$ and honest validators follow \Cref{def:voting-rule};
  \item $\start(k) - \Delta \geq \GST$;
  \item $\syncer(k), \syncer(k+1), \syncer(k+2)$ are honest;
  \item (\emph{slot-$k$ dual-height voting}) at $\vote(k)$, every honest validator casts \emph{two} votes: one at height $R^{\vote(k)}.h$ and one at height $R^{\vote(k)}.h + 1$ (the latter conditional on a block at state-height $R^{\vote(k)}.h + 1$ existing in its store). Each is of whatever type \Cref{def:voting-rule} would normally produce at the height in question (\just on the first vote at the height, \notar otherwise; with the appropriate $(\finalizeHeight, \finalizeTarget)$ for the (R1) clause). At every voting time other than $\vote(k)$, validators follow \Cref{def:voting-rule} unchanged.
\end{enumerate}
Write $\comm{h} \coloneqq \comm{R}^{\vote(k)}.h$. Then at $\vote(k+3)$, every honest validator has:
\begin{enumerate}
  \item $R_v^{\vote(k+3)}.h \geq \comm{h} + 2$;
  \item $R_v^{\vote(k+3)} \sim R_{v'}^{\vote(k+3)}$ for every pair of honest validators~$v, v'$;
  \item no honest validator has cast a \just vote at height $> R.h$ at any time strictly before $\vote(k+3)$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $h \coloneqq \comm{h}$.

\emph{Step 1: structural facts.}
By \Cref{rev:ass:view-merge} and honest $\syncer(k)$, $R_v^{\vote(k)} = \comm{R}^{\vote(k)}$ and $\Sigma_v^{\vote(k)} = \comm{\Sigma}^{\vote(k)}$ for every honest~$v$. By \Cref{rev:cor:processed-height-bound}, $\sigma[\getTarget_v^{\vote(k)}].h \in \{h, h+1\}$ for every honest~$v$. By \Cref{lem:Rs-key-monotone} and \Cref{rev:cor:processed-height-bound} applied at every earlier time, no honest validator has cast at height $>h+1$ before $\vote(k)$. By \Cref{rev:prop:getTarget-compat} (premise discharged by view-merge), honest \getTarget{}s at $\vote(k)$ are pairwise $\preceq$-compatible; hence both the height-$h$ and height-$(h+1)$ honest targets at $\vote(k)$ lie on a single chain~$C$ through $\comm{R}^{\vote(k)}$.

\emph{Step 2: a same-slot cert at height $h+1$ fires by $\vote(k+1)$ in every honest's view.}
Under the slot-$k$ dual-height extension every honest casts a slot-$k$ vote at height~$h$ \emph{and} a slot-$k$ vote at height $h+1$, both targeting blocks on~$C$. The slot-$k$ contributions at height $h+1$ thus include $\geq n - f > 2n/3$ honest~$\ntargets$ entries with $\nslots = k$ and target on~$C$. In each honest~$v$'s virtual-block aggregation by $\vote(k+1)$, the same-slot $\arg\max$ (\Cref{alg:state-machine}) admits a candidate $(B^\star, k)$ at height $h+1$ on~$C$, firing a cert event with key $(h+1,\, b,\, k,\, \ldots)$. This key strictly dominates the cached $\Sigma_v.\key_R = (h, *, *, \ldots)$ on the leading coordinate, so $R_v$ advances to $B^\star$. By view-merge + honest $\syncer(k+1)$, $\comm{R}^{\vote(k+1)}$ is the resulting cert event at height $h+1$, common across honest.

\emph{Step 3: reduce to a two-slot lemma at slot $k+1$.}
The setup at $\vote(k+1)$ matches the hypotheses of \Cref{rev:lem:healing-from-justified} (if $\comm{R}^{\vote(k+1)}$ is a bit-$1$ justification at $h+1$) or of \Cref{rev:lem:healing-from-notarized} (if $\comm{R}^{\vote(k+1)}$ is a bit-$0$ notarization at $h+1$ with notarization slot~$k$):
\begin{itemize}
  \item synchrony at slot $k+1$: $\start(k+1) - \Delta \geq \start(k) + \Delta \geq \GST$;
  \item by view-merge + honest $\syncer(k+1)$ (outer~(3)), $R_v^{\vote(k+1)} = \comm{R}^{\vote(k+1)}$ is the appropriate justified or notarized block in every honest's view; for the notarized branch, $\syncer(k+2)$ honest is also inherited from outer~(3);
  \item $\sigma[\getTarget_v^{\vote(k+1)}].h = h+1$ common across honest, by view-merge + \Cref{rev:cor:processed-height-bound} bounding state-height by $\comm{R}^{\vote(k+1)}.h + 1 = h+2$ and the $\geq h+1$ direction following from $\getTarget$ descending from $\comm{R}^{\vote(k+1)}$;
  \item no honest at height $>h+1$ before $\vote(k+1)$, by Step~1 plus the slot-$k$ honest contributions lying at heights in $\{h, h+1\}$.
\end{itemize}
Invoking the appropriate sub-lemma at slot~$k+1$ yields $R_v^{\vote(k+3)}.h = (h+1) + 1 = h + 2$ and the pairwise compatibility / no-high-\just clauses.

In all cases the three stated conclusions hold at $\vote(k+3)$.
\end{proof}

\begin{remark}[Why the dual-height extension is needed]\label{rev:rem:dual-vote-needed}
Without the slot-$k$ dual-height extension, the \Cref{rev:cor:processed-height-bound} bound only constrains canonical state-heights to $\{h, h+1\}$, but does not force them to agree across honest. The adversary can split honest \getTarget{}s between the two heights via the per-validator $\Omega$ component of $\getHead(\Sigma, \Omega)$, so the slot-$k$ honest contributions split between two heights:
\[
  h^*_h + h^*_{h+1} \;=\; n - f, \qquad h^*_h, h^*_{h+1} \;\geq\; 0.
\]
A same-slot cert at height $h^* \in \{h, h+1\}$ requires $\geq 2n/3$ contributors at $h^*$. Together with at most $f$ adversarial contributions, the threshold becomes $h^*_{h^*} \geq 2n/3 - f$. Both thresholds met simultaneously requires $h^*_h + h^*_{h+1} \geq 4n/3 - 2f$, i.e., $f \geq n/3$, contradicting $f < n/3$; so at most one height can carry a cert. The adversary can in fact split so that \emph{neither} reaches the threshold (e.g., $h^*_h = h^*_{h+1} = (n-f)/2$, giving honest mass $(n-f)/2 + f = (n+f)/2 < 2n/3$), blocking progress entirely. The split survives view-merge unchanged --- view-merge fixes $\Sigma$ but not the $\Omega$ argument of $\getHead$ --- so neither extra honest syncers nor extra slots resolve it.

The dual-height extension forces every honest to contribute at \emph{both} heights, raising each height's honest count to $n - f > 2n/3$ and guaranteeing a cert at height $h+1$. The extension is restricted to slot~$k$ because exactly one ``alignment slot'' is enough: post-slot-$k$, the cert at $h+1$ has fired, $R$ has advanced, and standard view-merge dynamics realign honest \getTarget{}s for slots $k+1, k+2, \ldots$ without further dual-voting.
\end{remark}



\appendix

\section{View-merge mechanism}\label{app:view-merge}

This appendix details the view-merge mechanism abstracted by \Cref{rev:ass:view-merge} and discharges that assumption under a synchronous-slot hypothesis.
Throughout, $\Delta$ denotes the synchronous network-delivery bound.

\begin{definition}[Slot timing]\label{rev:def:slot-timing}
Each slot~$k$ is anchored at a real-valued slot time $\start(k)$ with $\start(k+1) - \start(k) \geq 2\Delta$ and contains three ordered events:
\begin{itemize}
  \item \emph{Freeze time} $\start(k) - \Delta$: every validator~$i$ records its current local store as its slot-$k$ frozen snapshot $\Sigma^k_{\mathrm{frozen},i}$.
  \item \emph{Syncer time} $\start(k)$: $\syncer(k)$ broadcasts its current (unfrozen) local store to every validator; let $\Sigma^k_{\mathrm{sent}}$ be the broadcast content.
  \item \emph{Voting time} $\start(k) + \Delta$: every honest validator~$i$ invokes \Cref{def:voting-rule} on its slot-$k$ merged view $\Sigma^k_{\mathrm{merge},i} \coloneqq \Sigma^k_{\mathrm{frozen},i} \cup \Sigma^k_{\mathrm{sent}}$.
\end{itemize}
\end{definition}

\begin{definition}[Synchronous slot]\label{rev:def:synchrony-window}
Slot~$k$ is \emph{synchronous} if every message sent during $[\start(k) - 2\Delta, \start(k) + \Delta]$ is delivered to every honest validator within $\Delta$ of its sending time.
\end{definition}

\begin{lemma}[View-merge under an honest synchronous syncer]\label{rev:lem:view-merge}
If slot~$k$ is synchronous and $\syncer(k)$ is honest, then every honest~$i$ has $\Sigma^k_{\mathrm{frozen},i} \subseteq \Sigma^k_{\mathrm{sent}}$ and $\Sigma^k_{\mathrm{merge},i} = \Sigma^k_{\mathrm{sent}}$, discharging \Cref{rev:ass:view-merge} with $\comm{\Sigma}^{\vote(k)} = \Sigma^k_{\mathrm{sent}}$.
\end{lemma}

\begin{proof}
Any block or vote in $\Sigma^k_{\mathrm{frozen},i}$ was received by $\start(k) - \Delta$, hence sent by that time, hence delivered to $\syncer(k)$ by $\start(k)$ under \Cref{rev:def:synchrony-window}, so it belongs to $\Sigma^k_{\mathrm{sent}}$.
The broadcast at $\start(k)$ reaches every honest by $\start(k) + \Delta$, so $\Sigma^k_{\mathrm{merge},i}$ is well-defined and equals $\Sigma^k_{\mathrm{sent}}$.
\end{proof}
